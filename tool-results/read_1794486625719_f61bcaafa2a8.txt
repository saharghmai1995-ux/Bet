   128→   128→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
   129→   129→  const token = await botToken();
   130→   130→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   131→   131→  // v13.2 — سپر توکن ناقص: خطای 404 مبهم تلگرام را با پیام دقیق جایگزین می‌کند
   132→   132→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   133→   133→  try {
   134→   134→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
   135→   135→      method: "POST",
   136→   136→      headers: { "Content-Type": "application/json" },
   137→   137→      body: JSON.stringify(body),
   138→   138→      cache: "no-store",
   139→   139→    });
   140→   140→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   141→   141→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
   142→   142→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   143→   143→  } catch (err) {
   144→   144→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   145→   145→  }
   146→   146→}

   148→   148→async function tgSendPhotoTo(
   149→   149→  chatId: string | null,
   150→   150→  photo: Buffer,
   151→   151→  caption: string,
   152→   152→  filename = "slip-card.png",
   153→   153→  captionless = false
   154→   154→): Promise<{ ok: boolean; detail: string }> {
   155→   155→  const token = await botToken();
   156→   156→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   157→   157→  if (!chatId) return { ok: false, detail: "NO_CHANNEL" };
   158→   158→  // v13.2 — سپر توکن ناقص (بخش hash بدون bot_id) — ریشه باگ «انتشار انجام نمیشه»
   159→   159→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   160→   160→  try {
   161→   161→    const form = new FormData();
   162→   162→    form.append("chat_id", chatId);
   163→   163→    if (!captionless) {
   164→   164→      form.append("caption", caption.slice(0, 1024));
   165→   165→      form.append("parse_mode", "HTML");
   166→   166→    }
   167→   167→    form.append("photo", new Blob([new Uint8Array(photo)], { type: "image/png" }), filename);
   168→   168→    const res = await fetch(`https://api.telegram.org/bot${token}/sendPhoto`, { method: "POST", body: form });
   169→   169→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   170→   170→    if (json.ok) return { ok: true, detail: `photo:${json.result?.message_id ?? "?"}${captionless ? ":broadcast" : ""}` };
   171→   171→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   172→   172→  } catch (err) {
   173→   173→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   174→   174→  }
   175→   175→}

   214→   214→/** پیام HTML فلش‌کارت کامل — تیپ A + تیپ B + داچینگ + SHA-256 + لینک ESPN */
   215→   215→export function buildCardMessage(slip: IssuedSlip): string {
   216→   216→  const kickoff = new Date(slip.kickoff).toLocaleString("fa-IR", { dateStyle: "short", timeStyle: "short" });
   217→   217→  const lines: string[] = [];
   218→   218→  lines.push(`⚽️ <b>فوتبال‌کوانت — فلش‌کارت تیپ مکمل</b>`);
   219→   219→  lines.push(`🏆 <b>${esc(slip.homeTeam)}</b> × <b>${esc(slip.awayTeam)}</b>`);
   220→   220→  lines.push(`📡 ${esc(slip.league)} · ⏰ ${kickoff}`);
   221→   221→  if (slip.archetype) lines.push(`📐 الگو: ${esc(slip.archetype)}`);
   222→   222→  lines.push("");
   223→   223→  lines.push(`🔹 <b>تیپ A</b> (ضریب ${slip.oddsA.toFixed(2)}): ${esc(slip.labelA)}`);
   224→   224→  lines.push(`💵 مبلغ داچینگ A: <b>${slip.stakeA.toFixed(1)}</b> واحد`);
   225→   225→  lines.push("");
   226→   226→  lines.push(`🟢 <b>تیپ B مکمل</b> (ضریب ${slip.oddsB.toFixed(2)}): ${esc(slip.labelB)}`);
   227→   227→  lines.push(`💵 مبلغ داچینگ B: <b>${slip.stakeB.toFixed(1)}</b> واحد`);
   228→   228→  lines.push("");
   229→   229→  // ─── v10.0 — ارزش عددی نسبت به خط مرجع (همیشه عدد دارد) ───
   230→   230→  if (slip.edgeA !== null || slip.edgeB !== null) {
   231→   231→    const isPin = slip.edgeSourceA === "PINNACLE" || slip.edgeSourceB === "PINNACLE";
   232→   232→    const src = isPin ? "خط شارپ پیناکل" : "خط منصفانه بازار";
   233→   233→    const fa = (v: number) => `${v >= 0 ? "+" : ""}${v.toFixed(1)}%`;
   234→   234→    const a = slip.edgeA !== null ? fa(slip.edgeA) : "—";
   235→   235→    const b = slip.edgeB !== null ? fa(slip.edgeB) : "—";
   236→   236→    lines.push(`📈 ارزش عددی نسبت به ${src}: تیپ A <b>${a}</b> · تیپ B <b>${b}</b>`);
   237→   237→    if (isPin) {
   238→   238→      const cmp = (odds: number, pin: number | null): string | null =>
   239→   239→        pin && pin > 1.01 ? `${odds.toFixed(2)} در برابر خط ${pin.toFixed(2)}` : null;
   240→   240→      const pa = slip.edgeSourceA === "PINNACLE" ? cmp(slip.oddsA, slip.pinnacleOddsA) : null;
   241→   241→      const pb = slip.edgeSourceB === "PINNACLE" ? cmp(slip.oddsB, slip.pinnacleOddsB) : null;
   242→   242→      const parts = [pa ? `تیپ A: ${pa}` : null, pb ? `تیپ B: ${pb}` : null].filter((x): x is string => Boolean(x));
   243→   243→      if (parts.length > 0) lines.push(`📌 مقایسه قیمت با خط پیناکل: ${parts.join(" · ")}`);
   244→   244→    }
   245→   245→  }
   246→   246→  // ─── v10.0 — بهترین ساعت ثبت ضریب (Optimal Odds Timing Window) ───
   247→   247→  if (slip.timingWindowA || slip.timingWindowB) {
   248→   248→    const winName = timingWindowLabel(slip.timingWindowA ?? slip.timingWindowB);
   249→   249→    const start = slip.timingStartA ?? slip.timingStartB;
   250→   250→    const end = slip.timingEndA ?? slip.timingEndB;
   251→   251→    const fmt = (d: Date) => new Intl.DateTimeFormat("fa-IR", { hour: "2-digit", minute: "2-digit", timeZone: "Asia/Tehran" }).format(d);
   252→   252→    const range = start && end ? ` | ساعت دقیق پیشنهادی ثبت: ${fmt(new Date(start))} تا ${fmt(new Date(end))} (تهران)` : "";
   253→   253→    lines.push(`⏱️ بهترین زمان ثبت این تیپ: <b>${esc(winName)}</b>${range}`);
   254→   254→  }
   255→   255→  lines.push(`💰 سود تضمینی پوشش: <b>+${slip.guaranteedProfit.toFixed(1)}%</b> (مجموع استیک ۱۰۰ واحد)`);
   256→   256→  lines.push(`🔐 هش راستی‌آزمایی SHA-256:\n<code>${slip.hash.slice(0, 32)}</code>`);
   257→   257→  if (slip.espnUrl) lines.push(`🔎 <a href="${slip.espnUrl}">لینک راستی‌آزمایی رسمی ESPN</a>`);
   258→   258→  lines.push("");
   259→   259→  lines.push(`⚖️ ابزار تحلیل کمّی — مسئولیت استفاده با کاربر است.`);
   260→   260→  return lines.join("\n");
   261→   261→}

   451→   451→async function channelWinRateStats() {
   452→   452→  // v11.0 — فقط کارت‌های واقعی تاییدشده؛ تسویه کاغذی دفتر سایه در وین‌ریت کانال دخالت نمی‌کند
   453→   453→  // v13.2 — فقط کارت‌های واقعی تاییدشده؛ کارت‌های بازخوانی (retroIssued) و دفتر سایه در وین‌ریت کانال دخالت نمی‌کنند
   454→   454→  const settled = await db.issuedSlip.findMany({
   455→   455→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, supervisorApproved: true, retroIssued: false },
   456→   456→    select: { status: true, netProfit: true, stakeA: true, stakeB: true },
   457→   457→  });
   458→   458→  const st = { settled: settled.length, wonA: 0, wonB: 0, doubleWin: 0, middleTrap: 0, netProfit: 0, staked: 0 };
   459→   459→  for (const s of settled) {
   460→   460→    if (s.status === "WON_A") st.wonA += 1;
   461→   461→    else if (s.status === "WON_B") st.wonB += 1;
   462→   462→    else if (s.status === "DOUBLE_WIN") st.doubleWin += 1;
   463→   463→    else if (s.status === "MIDDLE_TRAP") st.middleTrap += 1;
   464→   464→    st.netProfit += s.netProfit ?? 0;
   465→   465→    st.staked += s.stakeA + s.stakeB;
   466→   466→  }
   467→   467→  const winRatePct = st.settled > 0 ? Math.round(((st.settled - st.middleTrap) / st.settled) * 1000) / 10 : 0;
   468→   468→  const netPct = st.staked > 0 ? Math.round((st.netProfit / st.staked) * 1000) / 10 : 0;
   469→   469→  // ─── v11.0 — دفتر سایه: شفافیت counterfactual وتوها (خط جداگانه گزارش) ───
   470→   470→  // v13.5.1 — سایه فقط وتوهای زنده است؛ کارت‌های بازخوانی (retroIssued) داخل سایه شمرده نمی‌شوند
   471→   471→  const shadowRows = await db.issuedSlip.findMany({
   472→   472→    where: { supervisorApproved: false, retroIssued: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   473→   473→    select: { status: true, netProfit: true },
   474→   474→  });
   475→   475→  const shadowWins = shadowRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   476→   476→  const shadowNet = Math.round(shadowRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100;
   477→   477→  const shadowWinRate = shadowRows.length > 0 ? Math.round((shadowWins / shadowRows.length) * 1000) / 10 : null;
   478→   478→  // ─── v13.2 — دفتر بازخوانی: کارت‌های چندروزه‌ای که هرگز پیش‌بازی صادر نشده‌اند (تسویه کاغذی تحلیلی) ───
   479→   479→  // v13.5.1 — بازخوانی با پرچم retroIssued شناسایی می‌شود (مستقل از تایید — کارت بازخوانی هرگز تاییدشده نیست)
   480→   480→  const retroRows = await db.issuedSlip.findMany({
   481→   481→    where: { retroIssued: true, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   482→   482→    select: { status: true, netProfit: true },
   483→   483→  });
   484→   484→  const retroWins = retroRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   485→   485→  return {
   486→   486→    ...st,
   487→   487→    winRatePct,
   488→   488→    netPct,
   489→   489→    netProfit: Math.round(st.netProfit * 100) / 100,
   490→   490→    shadow: { settled: shadowRows.length, wins: shadowWins, winRatePct: shadowWinRate, netProfit: shadowNet },
   491→   491→    retro: {
   492→   492→      settled: retroRows.length,
   493→   493→      wins: retroWins,
   494→   494→      winRatePct: retroRows.length > 0 ? Math.round((retroWins / retroRows.length) * 1000) / 10 : null,
   495→   495→      netProfit: Math.round(retroRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100,
   496→   496→    },
   497→   497→  };
   498→   498→}
