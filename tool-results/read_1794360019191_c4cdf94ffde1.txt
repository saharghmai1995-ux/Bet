   124→   124→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
   125→   125→  const token = await botToken();
   126→   126→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   127→   127→  // v13.2 — سپر توکن ناقص: خطای 404 مبهم تلگرام را با پیام دقیق جایگزین می‌کند
   128→   128→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   129→   129→  try {
   130→   130→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
   131→   131→      method: "POST",
   132→   132→      headers: { "Content-Type": "application/json" },
   133→   133→      body: JSON.stringify(body),
   134→   134→      cache: "no-store",
   135→   135→    });
   136→   136→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   137→   137→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
   138→   138→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   139→   139→  } catch (err) {
   140→   140→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   141→   141→  }
   142→   142→}

   144→   144→async function tgSendPhotoTo(
   145→   145→  chatId: string | null,
   146→   146→  photo: Buffer,
   147→   147→  caption: string,
   148→   148→  filename = "slip-card.png",
   149→   149→  captionless = false
   150→   150→): Promise<{ ok: boolean; detail: string }> {
   151→   151→  const token = await botToken();
   152→   152→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   153→   153→  if (!chatId) return { ok: false, detail: "NO_CHANNEL" };
   154→   154→  // v13.2 — سپر توکن ناقص (بخش hash بدون bot_id) — ریشه باگ «انتشار انجام نمیشه»
   155→   155→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   156→   156→  try {
   157→   157→    const form = new FormData();
   158→   158→    form.append("chat_id", chatId);
   159→   159→    if (!captionless) {
   160→   160→      form.append("caption", caption.slice(0, 1024));
   161→   161→      form.append("parse_mode", "HTML");
   162→   162→    }
   163→   163→    form.append("photo", new Blob([new Uint8Array(photo)], { type: "image/png" }), filename);
   164→   164→    const res = await fetch(`https://api.telegram.org/bot${token}/sendPhoto`, { method: "POST", body: form });
   165→   165→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   166→   166→    if (json.ok) return { ok: true, detail: `photo:${json.result?.message_id ?? "?"}${captionless ? ":broadcast" : ""}` };
   167→   167→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   168→   168→  } catch (err) {
   169→   169→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   170→   170→  }
   171→   171→}

   210→   210→/** پیام HTML فلش‌کارت کامل — تیپ A + تیپ B + داچینگ + SHA-256 + لینک ESPN */
   211→   211→export function buildCardMessage(slip: IssuedSlip): string {
   212→   212→  const kickoff = new Date(slip.kickoff).toLocaleString("fa-IR", { dateStyle: "short", timeStyle: "short" });
   213→   213→  const lines: string[] = [];
   214→   214→  lines.push(`⚽️ <b>فوتبال‌کوانت — فلش‌کارت تیپ مکمل</b>`);
   215→   215→  lines.push(`🏆 <b>${esc(slip.homeTeam)}</b> × <b>${esc(slip.awayTeam)}</b>`);
   216→   216→  lines.push(`📡 ${esc(slip.league)} · ⏰ ${kickoff}`);
   217→   217→  if (slip.archetype) lines.push(`📐 الگو: ${esc(slip.archetype)}`);
   218→   218→  lines.push("");
   219→   219→  lines.push(`🔹 <b>تیپ A</b> (ضریب ${slip.oddsA.toFixed(2)}): ${esc(slip.labelA)}`);
   220→   220→  lines.push(`💵 مبلغ داچینگ A: <b>${slip.stakeA.toFixed(1)}</b> واحد`);
   221→   221→  lines.push("");
   222→   222→  lines.push(`🟢 <b>تیپ B مکمل</b> (ضریب ${slip.oddsB.toFixed(2)}): ${esc(slip.labelB)}`);
   223→   223→  lines.push(`💵 مبلغ داچینگ B: <b>${slip.stakeB.toFixed(1)}</b> واحد`);
   224→   224→  lines.push("");
   225→   225→  // ─── v10.0 — ارزش عددی نسبت به خط مرجع (همیشه عدد دارد) ───
   226→   226→  if (slip.edgeA !== null || slip.edgeB !== null) {
   227→   227→    const isPin = slip.edgeSourceA === "PINNACLE" || slip.edgeSourceB === "PINNACLE";
   228→   228→    const src = isPin ? "خط شارپ پیناکل" : "خط منصفانه بازار";
   229→   229→    const fa = (v: number) => `${v >= 0 ? "+" : ""}${v.toFixed(1)}%`;
   230→   230→    const a = slip.edgeA !== null ? fa(slip.edgeA) : "—";
   231→   231→    const b = slip.edgeB !== null ? fa(slip.edgeB) : "—";
   232→   232→    lines.push(`📈 ارزش عددی نسبت به ${src}: تیپ A <b>${a}</b> · تیپ B <b>${b}</b>`);
   233→   233→    if (isPin) {
   234→   234→      const cmp = (odds: number, pin: number | null): string | null =>
   235→   235→        pin && pin > 1.01 ? `${odds.toFixed(2)} در برابر خط ${pin.toFixed(2)}` : null;
   236→   236→      const pa = slip.edgeSourceA === "PINNACLE" ? cmp(slip.oddsA, slip.pinnacleOddsA) : null;
   237→   237→      const pb = slip.edgeSourceB === "PINNACLE" ? cmp(slip.oddsB, slip.pinnacleOddsB) : null;
   238→   238→      const parts = [pa ? `تیپ A: ${pa}` : null, pb ? `تیپ B: ${pb}` : null].filter((x): x is string => Boolean(x));
   239→   239→      if (parts.length > 0) lines.push(`📌 مقایسه قیمت با خط پیناکل: ${parts.join(" · ")}`);
   240→   240→    }
   241→   241→  }
   242→   242→  // ─── v10.0 — بهترین ساعت ثبت ضریب (Optimal Odds Timing Window) ───
   243→   243→  if (slip.timingWindowA || slip.timingWindowB) {
   244→   244→    const winName = timingWindowLabel(slip.timingWindowA ?? slip.timingWindowB);
   245→   245→    const start = slip.timingStartA ?? slip.timingStartB;
   246→   246→    const end = slip.timingEndA ?? slip.timingEndB;
   247→   247→    const fmt = (d: Date) => new Intl.DateTimeFormat("fa-IR", { hour: "2-digit", minute: "2-digit", timeZone: "Asia/Tehran" }).format(d);
   248→   248→    const range = start && end ? ` | ساعت دقیق پیشنهادی ثبت: ${fmt(new Date(start))} تا ${fmt(new Date(end))} (تهران)` : "";
   249→   249→    lines.push(`⏱️ بهترین زمان ثبت این تیپ: <b>${esc(winName)}</b>${range}`);
   250→   250→  }
   251→   251→  lines.push(`💰 سود تضمینی پوشش: <b>+${slip.guaranteedProfit.toFixed(1)}%</b> (مجموع استیک ۱۰۰ واحد)`);
   252→   252→  lines.push(`🔐 هش راستی‌آزمایی SHA-256:\n<code>${slip.hash.slice(0, 32)}</code>`);
   253→   253→  if (slip.espnUrl) lines.push(`🔎 <a href="${slip.espnUrl}">لینک راستی‌آزمایی رسمی ESPN</a>`);
   254→   254→  lines.push("");
   255→   255→  lines.push(`⚖️ ابزار تحلیل کمّی — مسئولیت استفاده با کاربر است.`);
   256→   256→  return lines.join("\n");
   257→   257→}

   443→   443→async function channelWinRateStats() {
   444→   444→  // v11.0 — فقط کارت‌های واقعی تاییدشده؛ تسویه کاغذی دفتر سایه در وین‌ریت کانال دخالت نمی‌کند
   445→   445→  // v13.2 — فقط کارت‌های واقعی تاییدشده؛ کارت‌های بازخوانی (retroIssued) و دفتر سایه در وین‌ریت کانال دخالت نمی‌کنند
   446→   446→  const settled = await db.issuedSlip.findMany({
   447→   447→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, supervisorApproved: true, retroIssued: false },
   448→   448→    select: { status: true, netProfit: true, stakeA: true, stakeB: true },
   449→   449→  });
   450→   450→  const st = { settled: settled.length, wonA: 0, wonB: 0, doubleWin: 0, middleTrap: 0, netProfit: 0, staked: 0 };
   451→   451→  for (const s of settled) {
   452→   452→    if (s.status === "WON_A") st.wonA += 1;
   453→   453→    else if (s.status === "WON_B") st.wonB += 1;
   454→   454→    else if (s.status === "DOUBLE_WIN") st.doubleWin += 1;
   455→   455→    else if (s.status === "MIDDLE_TRAP") st.middleTrap += 1;
   456→   456→    st.netProfit += s.netProfit ?? 0;
   457→   457→    st.staked += s.stakeA + s.stakeB;
   458→   458→  }
   459→   459→  const winRatePct = st.settled > 0 ? Math.round(((st.settled - st.middleTrap) / st.settled) * 1000) / 10 : 0;
   460→   460→  const netPct = st.staked > 0 ? Math.round((st.netProfit / st.staked) * 1000) / 10 : 0;
   461→   461→  // ─── v11.0 — دفتر سایه: شفافیت counterfactual وتوها (خط جداگانه گزارش) ───
   462→   462→  const shadowRows = await db.issuedSlip.findMany({
   463→   463→    where: { supervisorApproved: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   464→   464→    select: { status: true, netProfit: true },
   465→   465→  });
   466→   466→  const shadowWins = shadowRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   467→   467→  const shadowNet = Math.round(shadowRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100;
   468→   468→  const shadowWinRate = shadowRows.length > 0 ? Math.round((shadowWins / shadowRows.length) * 1000) / 10 : null;
   469→   469→  // ─── v13.2 — دفتر بازخوانی: کارت‌های چندروزه‌ای که هرگز پیش‌بازی صادر نشده‌اند (تسویه کاغذی تحلیلی) ───
   470→   470→  const retroRows = await db.issuedSlip.findMany({
   471→   471→    where: { retroIssued: true, supervisorApproved: true, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   472→   472→    select: { status: true, netProfit: true },
   473→   473→  });
   474→   474→  const retroWins = retroRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   475→   475→  return {
   476→   476→    ...st,
   477→   477→    winRatePct,
   478→   478→    netPct,
   479→   479→    netProfit: Math.round(st.netProfit * 100) / 100,
   480→   480→    shadow: { settled: shadowRows.length, wins: shadowWins, winRatePct: shadowWinRate, netProfit: shadowNet },
   481→   481→    retro: {
   482→   482→      settled: retroRows.length,
   483→   483→      wins: retroWins,
   484→   484→      winRatePct: retroRows.length > 0 ? Math.round((retroWins / retroRows.length) * 1000) / 10 : null,
   485→   485→      netProfit: Math.round(retroRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100,
   486→   486→    },
   487→   487→  };
   488→   488→}
