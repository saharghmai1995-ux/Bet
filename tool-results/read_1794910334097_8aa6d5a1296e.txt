   131→export function assertNoSameGameMultiply(legs: { matchId: string }[], context = "combo"): void {
   132→  const ids = legs.map((l) => String(l.matchId ?? ""));

   134→  const hasDuplicate = ids.some((x, i) => ids.indexOf(x) !== i);

   138→    );

   205→function poissonTail(k: number, lambda: number): number {
   206→  // P(X > k) = 1 − P(X ≤ k)
   207→  let cdf = 0;
   208→  for (let i = 0; i <= k; i++) cdf += poissonPmf(i, lambda);
   209→  return Math.max(0, Math.min(1, 1 - cdf));
   210→}

   212→/** ماتریس نیمه اول با λ×۰.۴۵ (سهم استاندارد گل‌های نیمه اول) + تصحیح دیکسون-کولز */
   213→function buildHtProbs(lambdaHome: number, lambdaAway: number, rho = -0.08): { over05: number; zero: number } {
   214→  const lh = lambdaHome * 0.45;
   215→  const la = lambdaAway * 0.45;
   216→  let zero = 0;
   217→  let any = 0;
   218→  for (let h = 0; h <= 4; h++) {
   219→    for (let a = 0; a <= 4; a++) {
   220→      const p = Math.max(0, poissonPmf(h, lh) * poissonPmf(a, la) * dixonColesTau(h, a, lh, la, rho));
   221→      if (h === 0 && a === 0) zero = p;
   222→      if (h + a >= 1) any += p;
   223→    }
   224→  }
   225→  return { over05: Math.min(1, any), zero: Math.min(1, zero) };
   226→}

   230→function matrixProb(covers: (h: number, a: number) => boolean, matrix: number[][]): number {
   231→  let p = 0;
   232→  for (let h = 0; h < matrix.length; h++) {
   233→    const row = matrix[h] ?? [];
   234→    for (let a = 0; a < row.length; a++) {
   235→      if (covers(h, a)) p += row[a] ?? 0;
   236→    }
   237→  }
   238→  return p;
   239→}

   298→export function matchSeedOf(id: string | null | undefined): number {
   299→  const s = String(id ?? "NO-MATCH");
   300→  let h = 5381;
   301→  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) | 0;

   305→/**
   306→ * v19.0 — ضریب لبه ارزش بر اساس خطای قیمت‌گذاری بازار نسبت به مدل:
   307→ * implied = احتمال ضمنی بازار (۱ ÷ ضریب بازار). اگر بازار مدل را کم‌قیمت کرده باشد
   308→ * (implied < modelProb)، لبه بزرگ‌تر است — بازه مجاز سند: ۱.۱۴ تا ۱.۲۶.
   309→ */
   310→export function edgeBoostForMispricing(modelProb: number, marketImpliedProb: number | null): number {
   311→  if (!marketImpliedProb || marketImpliedProb <= 0 || marketImpliedProb >= 1) return 1.18;
   312→  const mispricing = Math.max(0, Math.min(0.3, modelProb - marketImpliedProb));
   313→  return Math.round(Math.min(1.26, Math.max(1.14, 1.14 + mispricing * 0.5)) * 1000) / 1000;

   465→ * generateOrthogonalComplementarySlips — قلب موتور v16.0.
   466→ * برای هر ۳ الگوی طلایی یک زوج تیپ اتمیک می‌سازد (کاربر بین آن‌ها سوییچ می‌کند)

   468→ * هر تیپ فقط یک مارکت تک‌دکمه‌ای است؛ هیچ ضرب درون‌بازی وجود ندارد.
   469→ */
   470→export function generateOrthogonalComplementarySlips(ctx: OrthoContext): OrthogonalResult {
   471→  const ht = buildHtProbs(ctx.lambdaHome, ctx.lambdaAway);
   472→  const lp: LegProbs = {
   473→    htOver05: ht.over05,
   474→    ht00: ht.zero,
   475→    cornersOver95: poissonTail(9, ctx.lambdaCorners),
   476→    cornersUnder95: 1 - poissonTail(9, ctx.lambdaCorners),
   477→    cardsOver35: poissonTail(3, ctx.lambdaCards),
   478→    cardsOver45: poissonTail(4, ctx.lambdaCards),
   479→    cardsUnder45: 1 - poissonTail(4, ctx.lambdaCards),
   480→  };
   481→
   482→  const archetypes: OrthoPair[] = [];
   483→  for (const def of ARCHETYPES) {
   484→    const slipA = buildAtomicSlip(def.legA(ctx), "A", ctx, lp);
   485→    const slipB = buildAtomicSlip(def.legB(ctx), "B", ctx, lp);
   486→
   487→    const pA = slipA.prob;
   488→    const pB = slipB.prob;
   489→    // احتمال برد همزمان: پایه‌های ماتریسی مشترک سلول
   490→    const bothWin = jointWinProb(slipA.legs, slipB.legs, ctx, lp);
   491→    const aOnly = Math.max(0, pA - bothWin);
   492→    const bOnly = Math.max(0, pB - bothWin);
   493→    const bothLose = Math.max(0, 1 - pA - pB + bothWin);
   494→    // v16.0 — پوشش کل بازی = اجتماع دقیق خانه‌های پوشش‌داده‌شده در ماتریس ۶×۶
   495→    const probUnion = Math.min(1, pA + pB - bothWin);
   496→    const coveragePct = probUnion;
   497→    const impliedSum = 1 / Math.max(1.01, slipA.odds) + 1 / Math.max(1.01, slipB.odds);
   498→
   499→    archetypes.push({
   500→      code: def.code,
   501→      name: def.name,
   502→      short: def.short,
   503→      rationale: def.rationale,
   504→      orthogonalityPct: 100, // خانواده شرط‌های A و B در همه الگوها کاملاً تفکیک‌شده است
   505→      coveragePct,
   506→      overlapPct: bothWin,
   507→      probUnion,
   508→      trueJointLossRiskPct: Math.round(bothLose * 1000) / 10,
   509→      doubleWinProbPct: Math.round(bothWin * 1000) / 10,
   510→      slipA,
   511→      slipB,
   512→      jointStates: { bothWin, aOnly, bOnly, bothLose },
   513→      impliedSum,
   514→      legsA: slipA.legs,
   515→      legsB: slipB.legs,
   516→    });
   517→  }
   518→
   519→  // پیشنهاد رسمی: بالاترین امتیاز انطباق با گیت‌های λ + کمینه‌سازی ریسک باخت همزمان
   520→  let recommended: ArchetypeCode = "G1";
   521→  let bestScore = -Infinity;
   522→  for (const a of archetypes) {
   523→    const def = ARCHETYPES.find((d) => d.code === a.code)!;
   524→    // الگویی که هم انطباق آماری دارد و هم ریسک باخت همزمان معقول
   525→    const s = def.score(ctx) - Math.max(0, a.jointStates.bothLose - 0.105) * 2.0;
   526→    if (s > bestScore) {
   527→      bestScore = s;
   528→      recommended = a.code;
   529→    }
   530→  }
   531→
   532→  return { recommended, archetypes };
   533→}

   535→/** P(برد همزمان A و B) — پایه‌های ماتریسی روی سلول مشترک */
   536→function jointWinProb(legsA: string[], legsB: string[], ctx: OrthoContext, lp: LegProbs): number {
   537→  const offA = legsA.filter(isOffMatrixLeg).map((k) => orthoLegProb(k, ctx, lp));
   538→  const offB = legsB.filter(isOffMatrixLeg).map((k) => orthoLegProb(k, ctx, lp));
   539→  const offProd = [...offA, ...offB].reduce((s, x) => s * x, 1);
   540→  if (offProd <= 1e-6) return 0;
   541→  const matA = legsA.filter((k) => !isOffMatrixLeg(k));
   542→  const matB = legsB.filter((k) => !isOffMatrixLeg(k));
   543→  let p = 0;
   544→  for (let h = 0; h < ctx.matrix.length; h++) {
   545→    const row = ctx.matrix[h] ?? [];
   546→    for (let a = 0; a < row.length; a++) {
   547→      const cell = row[a] ?? 0;
   548→      if (cell <= 0) continue;
   549→      let ok = true;
   550→      for (const k of matA) if (!orthoLegCoversCell(k, h, a)) { ok = false; break; }
   551→      if (!ok) continue;
   552→      for (const k of matB) if (!orthoLegCoversCell(k, h, a)) { ok = false; break; }
   553→      if (ok) p += cell * offProd;
   554→    }
   555→  }
   556→  return p;
   557→}

   565→/**
   566→ * v11.0 — jointShadowMetrics — بازتولید سنجه‌های لحظه تصمیم سوپروایزر از λ و پایه‌های
   567→ * ذخیره‌شده کارت (بدون هیچ عدد ساختگی): ٪ باخت همزمان + ٪ برد طلایی همزمان.
   568→ * دقیقاً همان ریاضیات generateOrthogonalComplementarySlips است تا backfill
   569→ * کارت‌های قدیمی با صدور جدید کاملاً هم‌مقیاس بماند.
   570→ */
   571→export function jointShadowMetrics(
   572→  lambdaHome: number,
   573→  lambdaAway: number,
   574→  lambdaCorners: number,
   575→  lambdaCards: number,
   576→  legsA: string[],
   577→  legsB: string[]
   578→): { jointLossRiskPct: number; doubleWinProbPct: number } {
   579→  const matrix = buildScoreMatrix(lambdaHome || 1.35, lambdaAway || 1.25);
   580→  const ctx: OrthoContext = { lambdaHome, lambdaAway, matrix, lambdaCorners, lambdaCards };
   581→  const ht = buildHtProbs(lambdaHome || 1.35, lambdaAway || 1.25);
   582→  const lp: LegProbs = {
   583→    htOver05: ht.over05,
   584→    ht00: ht.zero,
   585→    cornersOver95: poissonTail(9, ctx.lambdaCorners),
   586→    cornersUnder95: 1 - poissonTail(9, ctx.lambdaCorners),
   587→    cardsOver35: poissonTail(3, ctx.lambdaCards),
   588→    cardsOver45: poissonTail(4, ctx.lambdaCards),
   589→    cardsUnder45: 1 - poissonTail(4, ctx.lambdaCards),
   590→  };
   591→  const pA = atomicLegProb(legsA, ctx, lp);
   592→  const pB = atomicLegProb(legsB, ctx, lp);
   593→  const bothWin = jointWinProb(legsA, legsB, ctx, lp);
   594→  const bothLose = Math.max(0, 1 - pA - pB + bothWin);
   595→  const r1 = (v: number) => Math.round(v * 1000) / 10; // کسر → ٪ با یک رقم اعشار
   596→  return { jointLossRiskPct: r1(bothLose), doubleWinProbPct: r1(bothWin) };
   597→}

   599→/** احتمال یک تیپ (همه پایه‌ها ببرند) — پشتیبانی از کارت‌های قدیمی چندپایه */
   600→function atomicLegProb(legs: string[], ctx: OrthoContext, lp: LegProbs): number {
   601→  const offFactors = legs.filter(isOffMatrixLeg).map((k) => orthoLegProb(k, ctx, lp));
   602→  for (const f of offFactors) if (f <= 1e-6) return 0;
   603→  const offProd = offFactors.reduce((s, x) => s * x, 1);
   604→  const matrixLegs = legs.filter((k) => !isOffMatrixLeg(k));
   605→  let p = 0;
   606→  for (let h = 0; h < ctx.matrix.length; h++) {
   607→    const row = ctx.matrix[h] ?? [];
   608→    for (let a = 0; a < row.length; a++) {
   609→      const cell = row[a] ?? 0;
   610→      if (cell <= 0) continue;
   611→      let ok = true;
   612→      for (const k of matrixLegs) {
   613→        if (!orthoLegCoversCell(k, h, a)) {
   614→          ok = false;
   615→          break;
   616→        }
   617→      }
   618→      if (ok) p += cell * offProd;
   619→    }
   620→  }
   621→  return p;
   622→}

   640→ *   · بال پوششی (تیپ B ~۳.۱۰x): دقیقاً ۳۵ واحد → برد B = ۱۰۸.۵ واحد (اصل پول + ۸.۵٪ سود)
   641→ *   · بال اصلی (تیپ A ~۲.۱۰x): ۶۵ واحد → برد A = ۱۳۶.۵ واحد (+۳۶.۵ سود خالص)
   642→ *   · برد طلایی همزمان (Double-Win در ۱-۰ و ۲-۰): ۱۳۶.۵ + ۱۰۸.۵ = ۲۴۵.۰ → +۱۴۵.۰ واحد در یک مسابقه
   643→ * «سود تضمینی» کارت = کمینه سود دو بال (محافظه‌کارانه و صادقانه).
   644→ */
   645→export function asymmetricAlphaStakes(
   646→  oddsA: number,
   647→  oddsB: number
   648→): { active: boolean; stakeA: number; stakeB: number; profitOnA: number; profitOnB: number; guaranteed: number; profitOnDoubleWin: number; note: string } {
   649→  const total = ASYM_STAKE_A_UNITS + ASYM_STAKE_B_UNITS;
   650→  const stakeA = ASYM_STAKE_A_UNITS;
   651→  const stakeB = ASYM_STAKE_B_UNITS;
   652→  const profitOnA = Math.round((stakeA * oddsA - total) * 10) / 10;
   653→  const profitOnB = Math.round((stakeB * oddsB - total) * 10) / 10;
   654→  const guaranteed = Math.round(Math.min(profitOnA, profitOnB) * 10) / 10;
   655→  const profitOnDoubleWin = Math.round((stakeA * oddsA + stakeB * oddsB - total) * 10) / 10;
   656→  return {
   657→    active: true,
   658→    stakeA,
   659→    stakeB,
   660→    profitOnA,
   661→    profitOnB,
   662→    guaranteed,
   663→    profitOnDoubleWin,
   664→    note: `⚡ تخصیص نامتقارن آلفا — ${stakeA} واحد روی تیپ A در ضریب ${oddsA.toFixed(2)} (برد = +${profitOnA.toFixed(1)}u) + ${stakeB} واحد روی تیپ B در ضریب ${oddsB.toFixed(2)} (برد = ${profitOnB >= 0 ? "+" : ""}${profitOnB.toFixed(1)}u) — برد طلایی همزمان ۱-۰/۲-۰ = جک‌پات +${profitOnDoubleWin.toFixed(1)} واحد`,
   665→  };
   666→}
