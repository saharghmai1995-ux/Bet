   143→/**
   144→ * آنتروپی شانون ماتریس ۶×۶ — شاخص تمرکز یا پراکندگی نتایج.
   145→ * H = −Σ mat[i][j]·ln(mat[i][j]) روی زیرماتریس ۰..۵ × ۰..۵ (نرمال‌شده درون همان بازه).
   146→ */
   147→export function shannonEntropy(matrix: number[][]): number {
   148→  const N = Math.min(6, matrix.length);
   149→  let sum = 0;
   150→  for (let h = 0; h < N; h++) {
   151→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) sum += matrix[h][a];
   152→  }
   153→  if (sum <= 0) return Math.log(36);
   154→  let H = 0;
   155→  for (let h = 0; h < N; h++) {
   156→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   157→      const p = matrix[h][a] / sum;
   158→      if (p > 1e-12) H -= p * Math.log(p);
   159→    }
   160→  }
   161→  return H;
   162→}

   164→/** نسبت عدم تقارن قدرت: Rλ = max(λH, λA) / min(λH, λA) */
   165→export function asymmetryRatio(lambdaHome: number, lambdaAway: number): number {
   166→  const hi = Math.max(lambdaHome, lambdaAway);
   167→  const lo = Math.min(lambdaHome, lambdaAway);
   168→  if (lo <= 0.01) return 9.99;
   169→  return Math.min(9.99, hi / lo);
   170→}

   172→/** مجموع n خانه برتر ماتریس ۶×۶ + فهرست آن‌ها */
   173→export function topCells(matrix: number[][], n = 4): { score: string; p: number }[] {
   174→  const N = Math.min(6, matrix.length);
   175→  const cells: { score: string; p: number }[] = [];
   176→  for (let h = 0; h < N; h++) {
   177→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   178→      cells.push({ score: `${h}-${a}`, p: matrix[h][a] });
   179→    }
   180→  }
   181→  cells.sort((x, y) => y.p - x.p);
   182→  return cells.slice(0, n);
   183→}

   185→export function matrixStructure(matrix: number[][], lambdaHome: number, lambdaAway: number): MatrixStructure {
   186→  const top4 = topCells(matrix, 4);
   187→  return {
   188→    entropy: Math.round(shannonEntropy(matrix) * 1000) / 1000,
   189→    rLambda: Math.round(asymmetryRatio(lambdaHome, lambdaAway) * 100) / 100,
   190→    top4Sum: Math.round(top4.reduce((s, c) => s + c.p, 0) * 1000) / 1000,
   191→    top4,
   192→  };
   193→}

   197→/** جمع احتمال خانه‌های ماتریس ۶×۶ که گزاره را ارضا می‌کنند (نرمال‌شده درون ۶×۶) */
   198→function matrixProb(matrix: number[][], predicate: (h: number, a: number) => boolean): number {
   199→  const N = Math.min(6, matrix.length);
   200→  let inRegion = 0;
   201→  let hit = 0;
   202→  for (let h = 0; h < N; h++) {
   203→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   204→      inRegion += matrix[h][a];
   205→      if (predicate(h, a)) hit += matrix[h][a];
   206→    }
   207→  }
   208→  return inRegion > 0 ? hit / inRegion : 0;
   209→}

   216→export interface NonLinearCandidateInput {
   217→  /** احتمال برد کل استراتژی (یا مجموع پوشش خوشه) */

   220→  effectiveOdds: number;
   221→  /** احتمال برد دوبل همزمان در نتایج 1-0 / 2-0 */
   222→  doubleWinProb: number;
   223→  /** آنتروپی شانون ماتریس 6x6 */
   224→  entropy: number;

   229→/**
   230→ * تابع غیرخطی امتیازدهی سوپروایزر برای انتخاب بهترین Bet-Type در هر مسابقه
   231→ * (Non-Linear Sharpe & Kelly Utility — عیناً مطابق سند ارتقا v17.0)
   232→ */
   233→export function calculateNonLinearSupervisorScore(candidate: NonLinearCandidateInput): {
   234→  ev: number;
   235→  sharpe: number;
   236→  score: number;
   237→} {
   238→  const netReturnOnWin = candidate.effectiveOdds - 1;
   239→  const lossProb = Math.max(0, 1 - candidate.winProb);
   240→  const expectedValue = candidate.winProb * netReturnOnWin + candidate.doubleWinProb * 0.45 - lossProb;
   241→  const variance =
   242→    candidate.winProb * Math.pow(netReturnOnWin - expectedValue, 2) +
   243→    lossProb * Math.pow(-1 - expectedValue, 2);
   244→  const sharpeRatio = expectedValue / Math.sqrt(Math.max(0.01, variance));
   245→
   246→  // جریمه غیرخطی ریسک و پاداش آنتروپی پایین
   247→  const nonLinearSafetyBonus = Math.pow(Math.max(0, candidate.winProb), 2.2) * 100;
   248→  const entropyBonus = Math.max(0, (2.8 - candidate.entropy) * 12);
   249→
   250→  const score = Number(
   251→    ((sharpeRatio * 45 + nonLinearSafetyBonus * 0.55 + entropyBonus) * candidate.leagueReliability).toFixed(1)
   252→  );
   253→  return { ev: expectedValue, sharpe: sharpeRatio, score };
   254→}

   258→/**
   259→ * کاندیدهای 🏦 BANKER — تک‌ستون امن با اطمینان بالا.
   260→ * شرط انتخاب غیرخطی: عدم تقارن قدرت بالا (Rλ ≥ ۱.۹۵ یا احتمال پایه بالای ۸۴٪)
   261→ * اما ضریب برای داچینگ دوگانه پایین است.
   262→ * مارکت‌های رسمی: ۱) DNB/AH 0.0  ۲) برد مستقیم فاوریت  ۳) شانس مجدد 1X/X2 —
   263→ * قوی‌ترین مارکتی که احتمال ماتریس آن ≥ ۸۲٪ است.
   264→ * ضریب هدف ۱.۳۸x تا ۱.۶۸x | استیک مستقیم ۶۰ واحد | پایه اصلی Accumulator.
   265→ */
   266→export function buildBankerCandidates(
   267→  matrix: number[][],
   268→  prices: MarketPrices,
   269→  structure: MatrixStructure
   270→): BetTypeCandidate[] {
   271→  const out: BetTypeCandidate[] = [];
   272→
   273→  const p1X = matrixProb(matrix, (h, a) => h >= a); // شانس مجدد میزبان
   274→  const pX2 = matrixProb(matrix, (h, a) => a >= h); // شانس مجدد میهمان
   275→  const pH = matrixProb(matrix, (h, a) => h > a);
   276→  const pA = matrixProb(matrix, (h, a) => a > h);
   277→
   278→  // ضریب‌های بازار واقعی: DC از خط 1X2، تک‌ستون مستقیم از خود خط
   279→  const dc1xOdds = 1 / (1 / prices.oddsHome + 1 / prices.oddsDraw);
   280→  const dcx2Odds = 1 / (1 / prices.oddsAway + 1 / prices.oddsDraw);
   281→
   282→  const variants: {
   283→    id: string;
   284→    label: string;
   285→    labelEn: string;
   286→    legs: string[];
   287→    prob: number;
   288→    odds: number;
   289→    favSide: "HOME" | "AWAY";
   290→  }[] = [
   291→    // v17.0 — فقط مارکت‌های با تسویه دقیق ماتریسی (DNB به‌دلیل مفهوم بازگشت وجه در مساوی خارج شد)
   292→    { id: "HOME", label: "برد مستقیم فاوریت میزبان (Full-Time Result: 1)", labelEn: "Full-Time Result: Home Win", legs: ["HOME"], prob: pH, odds: prices.oddsHome, favSide: "HOME" },
   293→    { id: "AWAY", label: "برد مستقیم فاوریت میهمان (Full-Time Result: 2)", labelEn: "Full-Time Result: Away Win", legs: ["AWAY"], prob: pA, odds: prices.oddsAway, favSide: "AWAY" },
   294→    { id: "DC_1X", label: "شانس مجدد فاوریت میزبان (Double Chance 1X)", labelEn: "Double Chance 1X", legs: ["DC_1X"], prob: p1X, odds: dc1xOdds, favSide: "HOME" },
   295→    { id: "DC_X2", label: "شانس مجدد فاوریت میهمان (Double Chance X2)", labelEn: "Double Chance X2", legs: ["DC_X2"], prob: pX2, odds: dcx2Odds, favSide: "AWAY" },
   296→  ];
   297→
   298→  for (const v of variants) {
   299→    // شرط غیرخطی: مارکت باید واقعاً «امن» باشد — احتمال ماتریس ≥ ۸۰٪ (الیت ≥ ۸۲٪ سند)
   300→    if (v.prob < 0.80) continue;
   301→    // محدوده منطقی ضریب تک‌ستون امن (سند: ۱.۳۸..۱.۶۸ — در بازارهای واقعی فاوریت قاطع تا ۱.۱۰ نیز معتبر است)
   302→    if (v.odds < 1.08 || v.odds > 2.0) continue;
   303→    const triggered = structure.rLambda >= BANKER_R_LAMBDA_MIN || v.prob >= BANKER_BASE_PROB_MIN;
   304→    if (!triggered) continue;
   305→
   306→    const sc = calculateNonLinearSupervisorScore({
   307→      winProb: v.prob,
   308→      effectiveOdds: v.odds,
   309→      doubleWinProb: 0,
   310→      entropy: structure.entropy,
   311→      leagueReliability: 1,
   312→    });
   313→    out.push({
   314→      key: "BANKER",
   315→      variantId: `BANKER:${v.id}`,
   316→      labelFa: "بنکر امن — تک‌ستون با اطمینان بالا",
   317→      labelEn: "Safe Banker — High-Confidence Single Column",
   318→      icon: "🏦",
   319→      rationale: `عدم تقارن قدرت Rλ = ${structure.rLambda.toFixed(2)} و احتمال ماتریس مارکت ${(v.prob * 100).toFixed(1)}٪ — فاوریت قاطع، بهترین ابزار: تک‌ستون امن با ضریب هدف ${v.odds.toFixed(2)}x`,
   320→      mainLabel: v.label,
   321→      mainLabelEn: v.labelEn,
   322→      mainOdds: r2(v.odds),
   323→      mainProb: r3(v.prob),
   324→      mainLegs: v.legs,
   325→      winProb: r3(v.prob),
   326→      effectiveOdds: r2(v.odds),
   327→      doubleWinProb: 0,
   328→      jointLossProb: r3(1 - v.prob),
   329→      matrixCoveragePct: Math.round(v.prob * 1000) / 10,
   330→      eligible: v.prob >= BANKER_MATRIX_PROB_MIN,
   331→      ...sc,
   332→      stakeMain: 60,
   333→      stakeCover: 0,
   334→      guaranteedProfitUnits: r2(60 * v.odds - 60),
   335→    });
   336→  }
   337→  return out.sort((x, y) => y.score - x.score);
   338→}

   342→/**
   343→ * کاندیدهای 🧩 COMBO — مارکت ترکیبی رسمی تک‌دکمه‌ای بوک‌میکر.
   344→ * شرط انتخاب غیرخطی: همبستگی بین برنده بازی و رژیم گلزنی در ماتریس بالای ۷۴٪.
   345→ * سه رژیم: فاوریت کنترل‌گر (1X & U3.5 + پوشش O3.5)، فاوریت تهاجمی (1 & O1.5 + پوشش U1.5/1-1)،
   346→ * نبرد باز (BTTS Yes & O2.5 + پوشش U2.5) — پوشش بالای ۹۳٪ فضای ماتریس.
   347→ */
   348→export function buildComboCandidates(
   349→  matrix: number[][],
   350→  prices: MarketPrices,
   351→  structure: MatrixStructure
   352→): BetTypeCandidate[] {
   353→  const out: BetTypeCandidate[] = [];
   354→
   355→  interface ComboDef {
   356→    id: string;
   357→    mainLabel: string;
   358→    mainLabelEn: string;
   359→    mainPred: (h: number, a: number) => boolean;
   360→    legKey: string;
   361→    /** پایه مرجع بازار برای استخراج بایاس مدل×بازار */
   362→    biasLeg: "1X" | "HOME" | "O25";
   363→    coverLabel: string;
   364→    coverLabelEn: string;
   365→    coverPred: (h: number, a: number) => boolean;
   366→    coverKey: string;
   367→    regime: string;
   368→  }
   369→
   370→  const defs: ComboDef[] = [
   371→    {
   372→      id: "1X_U35",
   373→      mainLabel: "1X & زیر ۳.۵ گل (برد میزبان یا مساوی و زیر ۳.۵ گل)",
   374→      mainLabelEn: "1X & Under 3.5 Goals",
   375→      mainPred: (h, a) => h >= a && h + a <= 3,
   376→      legKey: "C1XU35",
   377→      biasLeg: "1X",
   378→      coverLabel: "پوشش مکمل: بالای ۳.۵ گل (Over 3.5 Goals)",
   379→      coverLabelEn: "Over 3.5 Goals",
   380→      coverPred: (h, a) => h + a >= 4,
   381→      coverKey: "O35",
   382→      regime: "فاوریت کنترل‌گر",
   383→    },
   384→    {
   385→      id: "HOME_O15",
   386→      mainLabel: "برد میزبان و بالای ۱.۵ گل (Home Win & Over 1.5 Goals)",
   387→      mainLabelEn: "Home Win & Over 1.5 Goals",
   388→      mainPred: (h, a) => h > a && h + a >= 2,
   389→      legKey: "CHOME15",
   390→      biasLeg: "HOME",
   391→      coverLabel: "پوشش مکمل: زیر ۱.۵ گل یا مساوی ۱-۱ (Under 1.5 / Draw 1-1)",
   392→      coverLabelEn: "Under 1.5 Goals / Draw 1-1",
   393→      coverPred: (h, a) => h + a <= 1 || (h === 1 && a === 1),
   394→      coverKey: "U15",
   395→      regime: "فاوریت تهاجمی",
   396→    },
   397→    {
   398→      id: "BTTS_O25",
   399→      mainLabel: "هر دو تیم گل می‌زنند و بالای ۲.۵ گل (BTTS: Yes & Over 2.5)",
   400→      mainLabelEn: "BTTS: Yes & Over 2.5 Goals",
   401→      mainPred: (h, a) => h > 0 && a > 0 && h + a >= 3,
   402→      legKey: "CBTTS25",
   403→      biasLeg: "O25",
   404→      coverLabel: "پوشش مکمل: زیر ۲.۵ گل (Under 2.5 Goals)",
   405→      coverLabelEn: "Under 2.5 Goals",
   406→      coverPred: (h, a) => h + a <= 2,
   407→      coverKey: "U25",
   408→      regime: "نبرد باز",
   409→    },
   410→  ];
   411→
   412→  for (const d of defs) {
   413→    const pMain = matrixProb(matrix, d.mainPred);
   414→    const pCover = matrixProb(matrix, d.coverPred);
   415→    if (pMain < 0.55) continue; // کمبو بی‌کیفیت — فضای ماتریس ناکافی
   416→    // پوشش کل = سناریوهای برد اصلی ∪ پوشش مکمل (سند: بالای ۹۳٪ — کف مهندسی ۸۸٪)
   417→    const pUnion = matrixProb(matrix, (h, a) => d.mainPred(h, a) || d.coverPred(h, a));
   418→    if (pUnion < 0.88) continue;
   419→
   420→    // ═══ v17.0 — قیمت‌گذاری کمبو در برابر بازار واقعی (عدم توافق مدل × بازار) ═══
   421→    // بایاس بازار = احتمال ضمنی بازار پایه مرجع ÷ احتمال ماتریس همان پایه؛
   422→    // اگر بازار پای مرجع را کمتر از مدل ببیند (بایاس < ۱)، کمبو ارزش مثبت پیدا می‌کند.
   423→    const biasOf = (leg: "1X" | "HOME" | "O25"): number => {
   424→      const pMat =
   425→        leg === "1X"
   426→          ? matrixProb(matrix, (h, a) => h >= a)
   427→          : leg === "HOME"
   428→            ? matrixProb(matrix, (h, a) => h > a)
   429→            : matrixProb(matrix, (h, a) => h + a > 2.5);
   430→      const implied =
   431→        leg === "1X"
   432→          ? 1 / (1 / prices.oddsHome + 1 / prices.oddsDraw)
   433→          : leg === "HOME"
   434→            ? 1 / prices.oddsHome
   435→            : 1 / prices.oddsOver25;
   436→      if (!(implied > 0) || !(pMat > 1e-6)) return 1;
   437→      return Math.min(1.6, Math.max(0.5, implied / pMat));
   438→    };
   439→    const bias = biasOf(d.biasLeg);
   440→    const realMainProb = Math.min(0.97, Math.max(0.02, pMain * bias));
   441→    // پوشش‌های U25 قیمت واقعی دارند؛ O35/U15 از ماتریس با بایاس خنثی
   442→    const biasCover = d.coverKey === "U25" ? biasOf("O25") : 1;
   443→    const realCoverProb = Math.min(0.97, Math.max(0.02, pCover * biasCover));
   444→
   445→    // مارکت ترکیبی رسمی تک‌دکمه‌ای — حاشیه بوک‌میکر ~۴٪ روی نمای بازار
   446→    const mainOdds = Math.round((1 / realMainProb) * 0.96 * 100) / 100;
   447→    const coverOdds = Math.round((1 / realCoverProb) * 0.96 * 100) / 100;
   448→    if (mainOdds < 1.28 || mainOdds > 2.20) continue;
   449→
   450→    // داچینگ ۱۰۰ واحدی بین کمبو اصلی و پوشش مکمل
   451→    const implied = 1 / mainOdds + 1 / coverOdds;
   452→    const stakeMain = r2((1 / mainOdds / implied) * 100);
   453→    const stakeCover = r2((1 / coverOdds / implied) * 100);
   454→    const effectiveOdds = r2(1 / implied);
   455→    const winProb = r3(pUnion);
   456→    // برد دوبل: سناریوهایی که هر دو تیپ می‌برند (تلاقی کمبو اصلی × پوشش)
   457→    const pDouble = matrixProb(matrix, (h, a) => d.mainPred(h, a) && d.coverPred(h, a));
   458→
   459→    const sc = calculateNonLinearSupervisorScore({
   460→      winProb,
   461→      effectiveOdds,
   462→      doubleWinProb: pDouble,
   463→      entropy: structure.entropy,
   464→      leagueReliability: 1,
   465→    });
   466→
   467→    out.push({
   468→      key: "COMBO",
   469→      variantId: `COMBO:${d.id}`,
   470→      labelFa: `کمبوی رسمی تک‌دکمه‌ای — رژیم ${d.regime}`,
   471→      labelEn: `Official Combo — ${d.id}`,
   472→      icon: "🧩",
   473→      rationale: `همبستگی برنده بازی × رژیم گلزنی در ماتریس ۶×۶ بالای ۷۴٪ — رژیم ${d.regime}: احتمال کمبو ${(pMain * 100).toFixed(1)}٪ با پوشش مکمل → پوشش کل ${(pUnion * 100).toFixed(1)}٪ فضای ماتریس`,
   474→      mainLabel: d.mainLabel,
   475→      mainLabelEn: d.mainLabelEn,
   476→      mainOdds,
   477→      mainProb: r3(pMain),
   478→      mainLegs: [d.legKey],
   479→      coverLabel: d.coverLabel,
   480→      coverLabelEn: d.coverLabelEn,
   481→      coverOdds,
   482→      coverProb: r3(pCover),
   483→      coverLegs: [d.coverKey],
   484→      winProb,
   485→      effectiveOdds,
   486→      doubleWinProb: r3(pDouble),
   487→      jointLossProb: r3(1 - pUnion),
   488→      matrixCoveragePct: Math.round(pUnion * 1000) / 10,
   489→      eligible: pMain >= COMBO_CORRELATION_MIN,
   490→      ...sc,
   491→      stakeMain,
   492→      stakeCover,
   493→      guaranteedProfitUnits: r2(100 / implied - 100),
   494→      doubleWinProfitUnits: 0,
   495→    });
   496→  }
   497→  return out.sort((x, y) => y.score - x.score);
   498→}
   499→
   500→// ═══════════════════════════════ 🎯 CORRECT SCORE CLUSTER ═══════════════════════════════
   501→
   502→/**
   503→ * کاندید 🎯 CORRECT SCORE CLUSTER — خوشه‌بندی نتیجه دقیق ۳ یا ۴ شانسه.
   504→ * شرط انتخاب غیرخطی: آنتروپی شانون پایین (H ≤ ۲.۴۲) و مجموع احتمال ۴ خانه برتر > ۵۶٪.
   505→ * داچینگ ۱۰۰ واحدی بین نمرات دقیق به نسبت معکوس ضرایب — ضریب موثر خوشه ~۱.۷۵x تا ۲.۱۰x.
   506→ */
   507→export function buildCorrectScoreCandidate(
   508→  matrix: number[][],
   509→  structure: MatrixStructure
   510→): BetTypeCandidate | null {
   511→  if (structure.entropy > CS_ENTROPY_MAX || structure.top4Sum <= CS_TOP4_SUM_MIN) return null;
   512→
   513→  // خوشه ۳ یا ۴ تایی: خانه‌های برتر تا وقتی مجموع ≥ ۶۰٪ شود (حداقل ۳ نتیجه)
   514→  const cells = topCells(matrix, 4);
   515→  let cluster = cells.slice(0, 3);
   516→  for (let i = 3; i < cells.length; i++) {
   517→    if (cluster.reduce((s, c) => s + c.p, 0) < 0.6) cluster = cells.slice(0, i + 1);
   518→  }
   519→
   520→  // ضرایب نتیجه دقیق — خط ماتریس موتور با حاشیه واقعی بوک‌میکر CS (۷٪)
   521→  const items = cluster.map((c) => {
   522→    const odds = Math.min(11.5, Math.round((1 / c.p) * 0.93 * 100) / 100);
   523→    return { score: c.score, odds, prob: c.p };
   524→  });
   525→  // تقسیم ۱۰۰ واحد بودجه به نسبت معکوس ضرایب (Equal-Profit Dutching)
   526→  const implied = items.reduce((s, it) => s + 1 / it.odds, 0);
   527→  const withStakes = items.map((it) => ({
   528→    ...it,
   529→    odds: it.odds,
   530→    stake: Math.round((1 / it.odds / implied) * 1000) / 10,
   531→    profit: Math.round((100 / implied - 100) * 10) / 10,
   532→  }));
   533→  const effectiveOdds = r2(1 / implied);
   534→  const winProb = r3(withStakes.reduce((s, it) => s + it.prob, 0));
   535→  const clusterLegs = withStakes.map((s) => `CS_${s.score}`);
   536→
   537→  const sc = calculateNonLinearSupervisorScore({
   538→    winProb,
   539→    effectiveOdds,
   540→    doubleWinProb: 0,
   541→    entropy: structure.entropy,
   542→    leagueReliability: 1,
   543→  });
   544→
   545→  return {
   546→    key: "CORRECT_SCORE",
   547→    variantId: `CS${withStakes.length}`,
   548→    labelFa: `خوشه ${withStakes.length} تایی نتیجه دقیق (Correct Score Dutching)`,
   549→    labelEn: `${withStakes.length}-Score Correct Score Cluster Dutching`,
   550→    icon: "🎯",
   551→    rationale: `آنتروپی شانون ماتریس H = ${structure.entropy.toFixed(2)} ≤ ۲.۴۲ و مجموع ۴ خانه برتر ${(structure.top4Sum * 100).toFixed(0)}٪ > ۵۶٪ — توزیع نتایج متمرکز؛ خوشه داچینگ نتیجه دقیق با ضریب موثر ${effectiveOdds.toFixed(2)}x`,
   552→    mainLabel: `خوشه ${withStakes.length} نتیجه دقیق: ${withStakes.map((s) => s.score).join(" / ")}`,
   553→    mainLabelEn: `Correct Score Cluster: ${withStakes.map((s) => s.score).join(" / ")}`,
   554→    mainOdds: effectiveOdds,
   555→    mainProb: winProb,
   556→    mainLegs: clusterLegs,
   557→    winProb,
   558→    effectiveOdds,
   559→    doubleWinProb: 0,
   560→    jointLossProb: r3(1 - winProb),
   561→    matrixCoveragePct: Math.round(winProb * 1000) / 10,
   562→    eligible: true, // سازنده فقط با معیارهای سند (آنتروپی ≤ ۲.۴۲ و Top4 > ۵۶٪) کاندید می‌سازد
   563→    ...sc,
   564→    stakeMain: 100,
   565→    stakeCover: 0,
   566→    guaranteedProfitUnits: Math.round((100 / implied - 100) * 10) / 10,
   567→    cluster: withStakes.map((s) => ({
   568→      score: s.score,
   569→      odds: s.odds,
   570→      prob: Math.round(s.prob * 1000) / 1000,
   571→      stake: s.stake,
   572→      profit: s.profit,
   573→    })),
   574→  };
   575→}
   576→
   577→// ═══════════════════════════════ ⚖️ DUAL-HEDGE ═══════════════════════════════
   578→
   579→/**
   580→ * کاندیدهای ⚖️ DUAL-HEDGE — زوج‌تیپ متعامد تک‌دکمه‌ای (۳ جفت طلایی G1/G2/G3).
   581→ * شرط انتخاب غیرخطی: رژیم متعادل-هجومی و پوشش بالای ۹۱٪ با قابلیت برد طلایی همزمان
   582→ * در نتایج 1-0 و 2-0. ضریب ~۲.۰x و ~۳.۰x | شتاب‌دهنده نامتقارن (Principal-Lock Booster).
   583→ * کاندیدها از زوج‌های مونت‌کارلو موتور پرتفوی می‌آیند.
   584→ */
   585→export function buildDualHedgeCandidates(pairs: PortfolioPairDTO[]): BetTypeCandidate[] {
   586→  const out: BetTypeCandidate[] = [];
   587→  for (const p of pairs) {
   588→    const winProb = 1 - p.jointStates.bothLose;
   589→    if (winProb < DUAL_HEDGE_COVERAGE_MIN) continue;
   590→    const implied = 1 / p.totalOddsA + 1 / p.totalOddsB;
   591→    if (implied >= 1) continue; // بدون سود تضمینی
   592→    const effectiveOdds = r2(1 / implied);
   593→    const entropyProxy = 2.45; // رژیم متعادل — آنتروپی ماتریس بیرونی توسط routeBetTypes تزریق می‌شود
   594→
   595→    const sc = calculateNonLinearSupervisorScore({
   596→      winProb: r3(winProb),
   597→      effectiveOdds,
   598→      doubleWinProb: r3(p.jointStates.bothWin),
   599→      entropy: entropyProxy,
   600→      leagueReliability: 1,
   601→    });
   602→
   603→    out.push({
   604→      key: "DUAL_HEDGE",
   605→      variantId: `DUAL:${p.id}`,
   606→      labelFa: `زوج‌تیپ متعامد تک‌دکمه‌ای${p.archetype ? ` — ${p.archetype.short}` : ""}`,
   607→      labelEn: "Dual-Hedge Orthogonal Pair",
   608→      icon: "⚖️",
   609→      rationale: `رژیم متعادل-هجومی — پوشش ${(winProb * 100).toFixed(1)}٪${winProb >= 0.91 ? " (بالای ۹۱٪ سند)" : ""} با برد طلایی همزمان در نتایج 1-0 / 2-0 (P = ${(p.jointStates.bothWin * 100).toFixed(1)}٪) — گیت کیفیت با تابع مطلوبیت غیرخطی انجام می‌شود`,
   610→      mainLabel: p.labelA,
   611→      mainLabelEn: p.marketAEn ?? p.labelA,
   612→      mainOdds: p.totalOddsA,
   613→      mainProb: r3(p.selectionA.modelProb),
   614→      mainLegs: [],
   615→      coverLabel: p.labelB,
   616→      coverLabelEn: p.marketBEn ?? p.labelB,
   617→      coverOdds: p.totalOddsB,
   618→      coverProb: r3(p.selectionB.modelProb),
   619→      coverLegs: [],
   620→      winProb: r3(winProb),
   621→      effectiveOdds,
   622→      doubleWinProb: r3(p.jointStates.bothWin),
   623→      jointLossProb: r3(p.jointStates.bothLose),
   624→      matrixCoveragePct: Math.round(winProb * 1000) / 10,
   625→      eligible: winProb >= 0.9 && p.jointStates.bothLose <= 0.1,
   626→      ...sc,
   627→      stakeMain: p.equalProfit.stakeA,
   628→      stakeCover: p.equalProfit.stakeB,
   629→      guaranteedProfitUnits: p.equalProfit.profit,
   630→      pairId: p.id,
   631→    });
   632→  }
   633→  return out.sort((x, y) => y.score - x.score);
   634→}
   635→
   636→// ═══════════════════════════════ مسیریاب اصلی ═══════════════════════════════
   637→
   638→/**
   639→ * مسیریاب غیرخطی ۵ رژیمی — برای هر مسابقه هر ۴ روش تک‌بازی را با تابع مطلوبیت
   640→ * غیرخطی شارپ/کلِی امتیاز می‌دهد و بهترین Bet-Type را برمی‌گرداند.
   641→ *
   642→ * قانون تایید هوشمند سوپروایزر (v17.0):
   643→ *   · بازی‌های لیگ سمّی (esp.2 / usa.1 / fifa.friendly) → وتو مطلق
   644→ *   · اگر بهترین کاندید EV > +0.08 و Sharpe > 0.32 دارد → تایید با آن Bet-Type
   645→ *   · اگر هیچ روشی شارپ مثبت معنادار ندارد → وتو (بازی واقعاً بی‌کیفیت و پرنوسان)
   646→ */
   647→export function routeBetTypes(input: {
   648→  matrix: number[][];
   649→  lambdaHome: number;
   650→  lambdaAway: number;
   651→  prices: MarketPrices;
   652→  pairs: PortfolioPairDTO[];
   653→  league: string;
   654→  /** ضریب اعتبار لیگ از کالیبراسیون (۰.۸۵..۱.۱۵) */
   655→  leagueReliability?: number;
   656→}): BetTypeRouteResult {
   657→  const structure = matrixStructure(input.matrix, input.lambdaHome, input.lambdaAway);
   658→  const reliability = Math.min(1.15, Math.max(0.85, input.leagueReliability ?? 1));
   659→
   660→  const poison = POISON_LEAGUES.some((p) => input.league.toLowerCase().includes(p.toLowerCase()));
   661→  if (poison) {
   662→    return {
   663→      structure,
   664→      candidates: [],
   665→      winnerVariantId: null,
   666→      winner: null,
   667→      crownPairId: null,
   668→      veto: true,
   669→      vetoReason: `🛡️ گیت v17.0: لیگ سمّی (${input.league}) — نوسان ساختاری بالا و غیرقابل مدل‌سازی؛ همه ۴ روش تک‌بازی وتو می‌شوند — انتقال به دفتر سایه`,
   670→    };
   671→  }
   672→
   673→  // اعمال ضریب اعتبار لیگ روی امتیاز کاندیدها (بازمحاسبه با reliability)
   674→  const candidates: BetTypeCandidate[] = [
   675→    ...buildBankerCandidates(input.matrix, input.prices, structure),
   676→    ...buildComboCandidates(input.matrix, input.prices, structure),
   677→    ...(buildCorrectScoreCandidate(input.matrix, structure) ? [buildCorrectScoreCandidate(input.matrix, structure)!] : []),
   678→    ...buildDualHedgeCandidates(input.pairs),
   679→  ];
   680→  for (const c of candidates) {
   681→    const rescored = calculateNonLinearSupervisorScore({
   682→      winProb: c.winProb,
   683→      effectiveOdds: c.effectiveOdds,
   684→      doubleWinProb: c.doubleWinProb,
   685→      entropy: structure.entropy,
   686→      leagueReliability: reliability,
   687→    });
   688→    c.sharpe = r2(rescored.sharpe);
   689→    c.ev = r3(rescored.ev);
   690→    c.score = rescored.score;
   691→  }
   692→
   693→  // ═══ گیت تایید غیرخطی v17.0 (قانون تایید هوشمند سند) ═══
   694→  //  · کاندیدهای با شارپ واضحاً منفی (EV سوخت) کنار گذاشته می‌شوند
   695→  //  · تایید مبنای سند: معیار اختصاصی هر Bet-Type (eligible)
   696→  //  · تایید ارزش: EV > +۸٪ و Sharpe > ۰.۳۲ (قانون بازبینی مسابقات ریسک ۱۱-۱۳٪ سند)
   697→  const qualifying = candidates.filter(
   698→    (c) => c.sharpe >= -0.05 && (c.eligible || (c.ev > APPROVE_EV_MIN && c.sharpe > APPROVE_SHARPE_MIN))
   699→  );
   700→
   701→  if (candidates.length === 0 || qualifying.length === 0) {
   702→    return {
   703→      structure,
   704→      candidates: [],
   705→      winnerVariantId: null,
   706→      winner: null,
   707→      crownPairId: null,
   708→      veto: true,
   709→      vetoReason: "🛡️ گیت v17.0: هیچ‌یک از ۴ روش تک‌بازی (Banker / Combo / Correct Score / Dual-Hedge) آستانه کیفیت غیرخطی را پاس نکرد — بازی پرنوسان؛ انتقال به دفتر سایه",
   710→    };
   711→  }
   712→
   713→  const best = qualifying.reduce((a, b) => (b.score > a.score ? b : a));
   714→
   715→  // قانون تایید هوشمند: شارپ ضعیف در بهترین روش → بررسی امکان تایید با روش دیگر
   716→  if (best.sharpe < APPROVE_SHARPE_MIN || best.ev < APPROVE_EV_MIN) {
   717→    const eliteEligible = qualifying.filter((c) => c.ev > APPROVE_EV_MIN && c.sharpe > APPROVE_SHARPE_MIN);
   718→    if (eliteEligible.length === 0) {
   719→      // تایید مبنای سند — معیار اختصاصی Bet-Type (پوشش/آنتروپی/همبستگی) برنده است
   720→      return {
   721→        structure,
   722→        candidates,
   723→        winnerVariantId: best.variantId,
   724→        winner: best,
   725→        crownPairId: best.pairId ?? null,
   726→        veto: false,
   727→        vetoReason: null,
   728→      };
   729→    }
   730→    const bestEligible = eliteEligible.reduce((a, b) => (b.score > a.score ? b : a));
   731→    return {
   732→      structure,
   733→      candidates,
   734→      winnerVariantId: bestEligible.variantId,
   735→      winner: bestEligible,
   736→      crownPairId: bestEligible.pairId ?? null,
   737→      veto: false,
   738→      vetoReason: null,
   739→    };
   740→  }
   741→
   742→  return {
   743→    structure,
   744→    candidates,
   745→    winnerVariantId: best.variantId,
   746→    winner: best,
   747→    crownPairId: best.pairId ?? null,
   748→    veto: false,
   749→    vetoReason: null,
   750→  };
   751→}

   753→// ═══════════════════════════════ 🔗 ACCUMULATOR ═══════════════════════════════
   754→
   755→/** یک پای Accumulator — از بنکرها/کمبوهای تاییدشده بهترین بازی‌های روز */
   756→export interface AccumulatorLeg {
   757→  matchId: string;
   758→  matchLabel: string;
   759→  league: string;
   760→  kickoff: string;
   761→  tip: string;
   762→  tipEn: string;
   763→  odds: number;
   764→  betType: BetTypeKey;
   765→}
   766→
   767→export type AccumulatorKind = "TWO_FOLD_BANKER" | "THREE_FOLD_COMBO" | "TIME_STAGGERED";
   768→
   769→export interface AccumulatorSlip {
   770→  id: string;
   771→  kind: AccumulatorKind;
   772→  labelFa: string;
   773→  labelEn: string;
   774→  icon: string;
   775→  legs: AccumulatorLeg[];
   776→  totalOdds: number;
   777→  /** استیک بهینه بر اساس معیار کلِی همزمان (واحد) */
   778→  stake: number;
   779→  hitRatePct: number;
   780→  potentialReturn: number;
   781→  /** فرمول محاسبه مبلغ پوشش زنده پای آخر (Time-Staggered) */
   782→  hedgeNote?: string;
   783→  hedgeTargetLabel?: string;
   784→  hedgeTargetOdds?: number;
   785→  hedgeStake?: number;
   786→  lockedProfitIfHedged?: number;
   787→}
   788→
   789→/**
   790→ * 🔗 سازنده سه برگه میکس زنجیره‌ای از بهترین بنکرها/کمبوهای تاییدشده روز:
   791→ *  ۱) 2-Fold Banker Accumulator — ترکیب ۲ پایه Banker (ضریب کل ۲.۱۰..۲.۵۵، نرخ اصابت > ۸۰٪)
   792→ *  ۲) 3-Fold Pro Combo Accumulator — ترکیب ۳ پایه قوی Banker/Combo (ضریب کل ۳.۶۰..۵.۴۰)
   793→ *  ۳) Time-Staggered Hedged Accumulator — میکس ۳ بازی با ساعت شروع متفاوت +
   794→ *     فرمول محاسبه مبلغ پوشش زنده پای آخر (Cash-Out / Final-Leg Hedge).
   795→ * استیک بهینه با معیار کلِی همزمان (Simultaneous Kelly fraction × ۱۰۰).
   796→ */
   797→export function buildAccumulators(
   798→  winners: { matchId: string; matchLabel: string; league: string; kickoff: string; candidate: BetTypeCandidate }[]
   799→): AccumulatorSlip[] {
   800→  const bankers = winners.filter((w) => w.candidate.key === "BANKER");
   801→  const comboOrBanker = winners.filter((w) => w.candidate.key === "BANKER" || w.candidate.key === "COMBO");
   802→  const out: AccumulatorSlip[] = [];
   803→
   804→  const legOf = (w: (typeof winners)[number]): AccumulatorLeg => ({
   805→    matchId: w.matchId,
   806→    matchLabel: w.matchLabel,
   807→    league: w.league,
   808→    kickoff: w.kickoff,
   809→    // v17.0 — پای زوج‌تیپ: کل پوزیشن داچینگ‌شده با «ضریب موثر» و «پوشش کل» لحاظ می‌شود
   810→    tip: w.candidate.key === "DUAL_HEDGE" ? `${w.candidate.mainLabel} ↔ ${w.candidate.coverLabel ?? ""}` : w.candidate.mainLabel,
   811→    tipEn: w.candidate.key === "DUAL_HEDGE" ? "Dual-Hedge Effective Position" : w.candidate.mainLabelEn,
   812→    odds: w.candidate.key === "DUAL_HEDGE" ? w.candidate.effectiveOdds : w.candidate.mainOdds,
   813→    betType: w.candidate.key,
   814→  });
   815→
   816→  // ۱) 2-Fold Banker Accumulator
   817→  if (bankers.length >= 2) {
   818→    const legs = [legOf(bankers[0]), legOf(bankers[1])];
   819→    const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
   820→    const hit = bankers[0].candidate.mainProb * bankers[1].candidate.mainProb;
   821→    // کلِی همزمان: f* ≈ (p·b − q)/b × نیم‌کِلی محافظه‌کار برای میکس چندپایه
   822→    const b = totalOdds - 1;
   823→    const f = Math.max(0.015, Math.min(0.06, (hit * b - (1 - hit)) / b * 0.5));
   824→    out.push({
   825→      id: "ACC-2F",
   826→      kind: "TWO_FOLD_BANKER",
   827→      labelFa: "میکس زنجیره‌ای ۲ پایه بنکر امن (2-Fold Banker Accumulator)",
   828→      labelEn: "2-Fold Banker Accumulator",
   829→      icon: "🔗",
   830→      legs,
   831→      totalOdds,
   832→      stake: Math.round(f * 1000) / 10,
   833→      hitRatePct: Math.round(hit * 1000) / 10,
   834→      potentialReturn: Math.round(f * totalOdds * 10) / 10,
   835→    });
   836→  } else if (winners.length >= 2) {
   837→    // v17.0 — fallback: قوی‌ترین پایه‌های تاییدشده روز (هر Bet-Type)
   838→    const legs = [legOf(winners[0]), legOf(winners[1])];
   839→    const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
   840→    const legProb = (w: (typeof winners)[number]) => (w.candidate.key === "DUAL_HEDGE" ? w.candidate.winProb : w.candidate.mainProb);
   841→    const hit = legProb(winners[0]) * legProb(winners[1]);
   842→    const b = totalOdds - 1;
   843→    const f = Math.max(0.01, Math.min(0.045, (hit * b - (1 - hit)) / b * 0.5));
   844→    const kinds = [...new Set(winners.slice(0, 2).map((w) => w.candidate.icon + " " + w.candidate.key))].join(" + ");
   845→    out.push({
   846→      id: "ACC-2F",
   847→      kind: "TWO_FOLD_BANKER",
   848→      labelFa: `میکس زنجیره‌ای ۲ پایه تاییدشده روز (${kinds})`,
   849→      labelEn: "2-Fold Smart Accumulator",
   850→      icon: "🔗",
   851→      legs,
   852→      totalOdds,
   853→      stake: Math.round(f * 1000) / 10,
   854→      hitRatePct: Math.round(hit * 1000) / 10,
   855→      potentialReturn: Math.round(f * totalOdds * 10) / 10,
   856→    });
   857→  }
   858→
   859→  // ۲) 3-Fold Pro Combo Accumulator
   860→  if (comboOrBanker.length >= 3) {
   861→    const legs = comboOrBanker.slice(0, 3).map(legOf);
   862→    const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
   863→    const hit = comboOrBanker.slice(0, 3).reduce((s, w) => s * w.candidate.mainProb, 1);
   864→    const b = totalOdds - 1;
   865→    const f = Math.max(0.008, Math.min(0.035, (hit * b - (1 - hit)) / b * 0.5));
   866→    out.push({
   867→      id: "ACC-3F",
   868→      kind: "THREE_FOLD_COMBO",
   869→      labelFa: "میکس حرفه‌ای ۳ پایه بنکر/کمبو (3-Fold Pro Combo Accumulator)",
   870→      labelEn: "3-Fold Pro Combo Accumulator",
   871→      icon: "🔗",
   872→      legs,
   873→      totalOdds,
   874→      stake: Math.round(f * 1000) / 10,
   875→      hitRatePct: Math.round(hit * 1000) / 10,
   876→      potentialReturn: Math.round(f * totalOdds * 10) / 10,
   877→    });
   878→  } else if (winners.length >= 3) {
   879→    const legs = winners.slice(0, 3).map(legOf);
   880→    const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
   881→    const hit = winners.slice(0, 3).reduce((s, w) => s * (w.candidate.key === "DUAL_HEDGE" ? w.candidate.winProb : w.candidate.mainProb), 1);
   882→    const b = totalOdds - 1;
   883→    const f = Math.max(0.006, Math.min(0.025, (hit * b - (1 - hit)) / b * 0.5));
   884→    out.push({
   885→      id: "ACC-3F",
   886→      kind: "THREE_FOLD_COMBO",
   887→      labelFa: "میکس حرفه‌ای ۳ پایه تاییدشده روز (3-Fold Pro Accumulator)",
   888→      labelEn: "3-Fold Pro Accumulator",
   889→      icon: "🔗",
   890→      legs,
   891→      totalOdds,
   892→      stake: Math.round(f * 1000) / 10,
   893→      hitRatePct: Math.round(hit * 1000) / 10,
   894→      potentialReturn: Math.round(f * totalOdds * 10) / 10,
   895→    });
   896→  }
   897→
   898→  // ۳) Time-Staggered Hedged Accumulator — ۳ بازی با ساعت شروع متفاوت
   899→  const staggerPool = comboOrBanker.length >= 3 ? comboOrBanker : winners;
   900→  if (staggerPool.length >= 3) {
   901→    const sorted = [...staggerPool].sort((a, b) => new Date(a.kickoff).getTime() - new Date(b.kickoff).getTime());
   902→    // سه پای با بیشترین فاصله ساعت شروع
   903→    const picked = [sorted[0], sorted[Math.floor(sorted.length / 2)], sorted[sorted.length - 1]];
   904→    const distinctHours = new Set(picked.map((p) => new Date(p.kickoff).getHours())).size;
   905→    if (distinctHours >= 2) {
   906→      const legs = picked.map(legOf);
   907→      const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
   908→      const stake = 30;
   909→      const payoutIfFirstTwoWin = stake * legs[0].odds * legs[1].odds;
   910→      // فرمول پوشش پای آخر: مبلغ شرط مقابل = (پرداخت فعلی − سود قفل‌شده هدف) ÷ ضریب مقابل
   911→      const hedgeTargetOdds = Math.max(1.4, Math.round(1 / (1 - 1 / legs[2].odds) * 100) / 100);
   912→      const lockedProfit = Math.round(payoutIfFirstTwoWin * 0.35 * 10) / 10;
   913→      const hedgeStake = Math.round(((payoutIfFirstTwoWin - lockedProfit) / hedgeTargetOdds) * 10) / 10;
   914→      const hit = picked.reduce((s, w) => s * (w.candidate.key === "DUAL_HEDGE" ? w.candidate.winProb : w.candidate.mainProb), 1);
   915→      out.push({
   916→        id: "ACC-TS",
   917→        kind: "TIME_STAGGERED",
   918→        labelFa: "میکس پله‌زمانی با پوشش زنده پای آخر (Time-Staggered Hedged Accumulator)",
   919→        labelEn: "Time-Staggered Hedged Accumulator",
   920→        icon: "🔗",
   921→        legs,
   922→        totalOdds,
   923→        stake,
   924→        hitRatePct: Math.round(hit * 1000) / 10,
   925→        potentialReturn: Math.round(stake * totalOdds * 10) / 10,
   926→        hedgeNote: `اگر ۲ بازی اول بردند، پیش از شروع بازی سوم: شرط مقابل «${legs[2].tipEn}» به مبلغ ${hedgeStake} واحد با ضریب ~${hedgeTargetOdds.toFixed(2)}x ببندید → سود قفل‌شده ~${lockedProfit} واحد در هر دو سناریو`,
   927→        hedgeTargetLabel: "مقابل تیپ پای آخر (مثلاً X2 یا زیر گل)",
   928→        hedgeTargetOdds,
   929→        hedgeStake,
   930→        lockedProfitIfHedged: lockedProfit,
   931→      });
   932→    }
   933→  }
   934→
   935→  return out;
   936→}
