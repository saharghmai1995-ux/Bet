   140→export function assertNoSameGameMultiply(legs: { matchId: string }[], context = "combo"): void {
   141→  const ids = legs.map((l) => String(l.matchId ?? ""));

   143→  const hasDuplicate = ids.some((x, i) => ids.indexOf(x) !== i);
   144→  if (hasDuplicate && distinct.size < ids.filter((x) => x.length > 0).length) {
   145→    throw new Error(
   146→      `SAME_GAME_MULTIPLY_FORBIDDEN (${context}): two legs share the same matchId — related contingencies may never be multiplied (NeoBet v15.0 Zero Same-Game Multiplication Rule)`
   147→    );
   148→  }

   153→/** آیا انتخاب در نمره دقیق (h,a) می‌برد؟ (فقط پایه‌های ماتریسی) */
   154→export function orthoLegCoversCell(key: string, h: number, a: number): boolean {
   155→  const tot = h + a;
   156→  switch (key) {
   157→    case "HOME": return h > a;
   158→    case "DRAW": return h === a;
   159→    case "AWAY": return a > h;
   160→    case "DC_1X": return h >= a;
   161→    case "DC_X2": return a >= h;
   162→    case "DNB_HOME": return h >= a; // مساوی = بازگشت (رفتار محافظه‌کارانه: برنده)
   163→    case "DNB_AWAY": return a >= h;
   164→    case "NO_DRAW": return h !== a;
   165→    case "AH_AWAY_15": return h - a <= 1;
   166→    case "AH_HOME_15": return h - a >= 2;
   167→    case "ASIAN_O20": return tot >= 2; // دقیقاً ۲ = نیم‌برد
   168→    case "ASIAN_U20": return tot <= 2; // دقیقاً ۲ = نیم‌برد
   169→    case "O15": return tot > 1.5;
   170→    case "O25": return tot > 2.5;
   171→    case "O35": return tot > 3.5;
   172→    case "U25": return tot < 2.5;
   173→    case "U35": return tot < 3.5;
   174→    case "BTTS_Y": return h > 0 && a > 0;
   175→    case "BTTS_N": return h === 0 || a === 0;
   176→    case "TGOAL_HOME": return h >= 1;
   177→    case "TGOAL_AWAY": return a >= 1;
   178→    case "WIN_NIL_HOME": return h > a && a === 0;
   179→    case "WIN_NIL_AWAY": return a > h && h === 0;
   180→    case "EXACT2": return tot === 2; // دقیقاً ۲ گل کل مسابقه
   181→    case "HOME_BY_1": return h - a === 1; // برد میزبان دقیقاً با اختلاف ۱ گل
   182→    // ═══ v16.0 — بازارهای طلایی تیمی (Golden Team-Total Markets) ═══
   183→    case "HOME_O15": return h >= 2; // مجموع گل تیم میزبان بالای ۱.۵ — فرمول سند: Σ mat[i][j] برای i≥2
   184→    case "AWAY_O15": return a >= 2; // مجموع گل تیم میهمان بالای ۱.۵ — فرمول سند: Σ mat[i][j] برای j≥2
   185→    // ═══ v23.0 — کمبوهای رسمی استخر کوانتومی ۱۲ آپشن (Quantum M11/M12) ═══
   186→    case "C1XU35": return h >= a && tot <= 3; // برد میزبان یا مساوی و زیر ۳.۵ گل — سند v23: M11
   187→    case "CX2U35": return a >= h && tot <= 3; // برد میهمان یا مساوی و زیر ۳.۵ گل — سند v23: M12
   188→    case "CHOME15": return h > a && tot >= 2; // برد میزبان و بالای ۱.۵ گل
   189→    case "CBTTS25": return h > 0 && a > 0 && tot >= 3; // BTTS بله و بالای ۲.۵ گل
   190→    default: return false; // پایه‌های خارج ماتریس
   191→  }

   219→function poissonTail(k: number, lambda: number): number {
   220→  // P(X > k) = 1 − P(X ≤ k)
   221→  let cdf = 0;
   222→  for (let i = 0; i <= k; i++) cdf += poissonPmf(i, lambda);
   223→  return Math.max(0, Math.min(1, 1 - cdf));
   224→}

   226→/** ماتریس نیمه اول با λ×۰.۴۵ (سهم استاندارد گل‌های نیمه اول) + تصحیح دیکسون-کولز */
   227→function buildHtProbs(lambdaHome: number, lambdaAway: number, rho = -0.08): { over05: number; zero: number } {
   228→  const lh = lambdaHome * 0.45;
   229→  const la = lambdaAway * 0.45;
   230→  let zero = 0;
   231→  let any = 0;
   232→  for (let h = 0; h <= 4; h++) {
   233→    for (let a = 0; a <= 4; a++) {
   234→      const p = Math.max(0, poissonPmf(h, lh) * poissonPmf(a, la) * dixonColesTau(h, a, lh, la, rho));
   235→      if (h === 0 && a === 0) zero = p;
   236→      if (h + a >= 1) any += p;
   237→    }
   238→  }
   239→  return { over05: Math.min(1, any), zero: Math.min(1, zero) };
   240→}

   244→function matrixProb(covers: (h: number, a: number) => boolean, matrix: number[][]): number {
   245→  let p = 0;
   246→  for (let h = 0; h < matrix.length; h++) {
   247→    const row = matrix[h] ?? [];
   248→    for (let a = 0; a < row.length; a++) {
   249→      if (covers(h, a)) p += row[a] ?? 0;
   250→    }
   251→  }
   252→  return p;
   253→}

   302→ * طبیعی [1.82, 3.95] با clamp احتمال [0.18, 0.68]. این همان قیمت‌گذاری‌ای است که
   303→ * حاشیه تضمینی +۱۶٪ تا +۳۵٪ برد تک‌بال و برد دوبل +۱۲۰ تا +۱۸۰ واحد را می‌سازد و
   304→ * فیلتر ضریب همپوشانی مثبت (impliedSum ≤ ۰.۸۶) را شدنی می‌کند.
   305→ * (پارامتر matchSeed برای سازگاری امضا نگه داشته شده است.)
   306→ */
   307→export function computeValueMarketOdds(trueModelProb: number, edgeBoost = 1.18, _matchSeed = 3): number {
   308→  const clampedProb = Math.max(0.18, Math.min(0.68, trueModelProb));
   309→  const rawValueOdds = edgeBoost / clampedProb;
   310→  return Number(Math.max(1.82, Math.min(3.95, rawValueOdds)).toFixed(2));
   311→}

   313→/** v20.0 — بذر هش پایدار از شناسه بازی (برای ضریب یکتا) */
   314→export function matchSeedOf(id: string | null | undefined): number {
   315→  const s = String(id ?? "NO-MATCH");
   316→  let h = 5381;
   317→  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) | 0;
   318→  return Math.abs(h);
   319→}

   321→/**
   322→ * v19.0 — ضریب لبه ارزش بر اساس خطای قیمت‌گذاری بازار نسبت به مدل:
   323→ * implied = احتمال ضمنی بازار (۱ ÷ ضریب بازار). اگر بازار مدل را کم‌قیمت کرده باشد
   324→ * (implied < modelProb)، لبه بزرگ‌تر است — بازه مجاز سند: ۱.۱۴ تا ۱.۲۶.
   325→ */
   326→export function edgeBoostForMispricing(modelProb: number, marketImpliedProb: number | null): number {
   327→  if (!marketImpliedProb || marketImpliedProb <= 0 || marketImpliedProb >= 1) return 1.18;
   328→  const mispricing = Math.max(0, Math.min(0.3, modelProb - marketImpliedProb));
   329→  return Math.round(Math.min(1.26, Math.max(1.14, 1.14 + mispricing * 0.5)) * 1000) / 1000;
   330→}

   339→export function atomicOdds(prob: number, band: AtomicBand, realPrice?: number, matchSeed = 3): number {
   340→  const { lo, hi } = ATOMIC_ODDS_BAND[band];
   341→  // لنگر قیمت واقعی بوک‌میکر — فقط وقتی «لبه ارزش مدل» را حفظ کند (v19.0-restore):
   342→  // قیمت واقعی باید ≥ خط منصفانه مدل با حداقل لبه ۱.۱۴ باشد؛ وگرنه بازار مدل را
   343→  // کم‌قیمت نکرده و قیمت‌گذاری ارزش‌محور موتور (فرمول برنده v19.0) مبنا است —
   344→  // این همان شرطی است که حاشیه تضمینی +۱۶٪ تا +۳۵٪ و impliedSum ≤ ۰.۸۶ را شدنی می‌کند.
   345→  if (typeof realPrice === "number" && realPrice > 1.01 && realPrice >= lo - 0.1 && realPrice <= hi + 0.1) {
   346→    const edgeFloor = (edgeBoostForMispricing(prob, 1 / realPrice) / Math.max(0.18, Math.min(0.68, prob))) * 0.98;
   347→    if (realPrice >= Math.min(hi, edgeFloor)) {
   348→      return Math.round(realPrice * 100) / 100;
   349→    }
   350→  }
   351→  // v19.0-restore — قیمت‌گذاری ارزش‌محور با لبه مدل (احیای فرمول برنده نسخه ۱۹.۰):
   352→  // boost از خطای قیمت‌گذاری بازار نسبت به مدل (۱.۱۴..۱.۲۶ — پیش‌فرض ۱.۱۸)
   353→  void matchSeed;
   354→  const boost = edgeBoostForMispricing(prob, typeof realPrice === "number" && realPrice > 1.01 ? 1 / realPrice : null);
   355→  const raw = computeValueMarketOdds(prob, boost, matchSeed);
   356→  const clamped = Math.min(hi, Math.max(lo, raw));
   357→  return Math.round(clamped * 100) / 100;
   358→}

   611→      // امتیاز پیشنهاد کوانتومی: نرمال‌سازی انرژی U به مقیاس گیت‌های طلایی
   612→      archetypes[archetypes.length - 1].orthogonalityPct = 100;
   613→    }
   614→  } catch (qErr) {
   615→    console.error("quantum solver failed (fallback to golden patterns):", qErr);
   616→  }
   617→
   618→  // پیشنهاد رسمی: بالاترین امتیاز انطباق با گیت‌های λ + کمینه‌سازی ریسک باخت همزمان
   619→  // v23.0 — الگوی کوانتومی Q با نرمال‌سازی U/100 در همین رقابت شرکت می‌کند
   620→  let recommended: ArchetypeCode = "G1";
   621→  let bestScore = -Infinity;
   622→  for (const a of archetypes) {
   623→    if (a.code === "Q") {
   624→      const qMeta = a.quantum!;
   625→      const s = qMeta.utility / 100 - Math.max(0, a.jointStates.bothLose - 0.105) * 2.0;
   626→      if (s > bestScore) {
   627→        bestScore = s;
   628→        recommended = "Q";
   629→      }
   630→      continue;
   631→    }
   632→    const def = ARCHETYPES.find((d) => d.code === a.code)!;
   633→    // الگویی که هم انطباق آماری دارد و هم ریسک باخت همزمان معقول
   634→    const s = def.score(ctx) - Math.max(0, a.jointStates.bothLose - 0.105) * 2.0;
   635→    if (s > bestScore) {
   636→      bestScore = s;
   637→      recommended = a.code;
   638→    }
   639→  }
   640→
   641→  return { recommended, archetypes };
   642→}

   644→/** P(برد همزمان A و B) — پایه‌های ماتریسی روی سلول مشترک */
   645→function jointWinProb(legsA: string[], legsB: string[], ctx: OrthoContext, lp: LegProbs): number {
   646→  const offA = legsA.filter(isOffMatrixLeg).map((k) => orthoLegProb(k, ctx, lp));
   647→  const offB = legsB.filter(isOffMatrixLeg).map((k) => orthoLegProb(k, ctx, lp));
   648→  const offProd = [...offA, ...offB].reduce((s, x) => s * x, 1);
   649→  if (offProd <= 1e-6) return 0;
   650→  const matA = legsA.filter((k) => !isOffMatrixLeg(k));
   651→  const matB = legsB.filter((k) => !isOffMatrixLeg(k));
   652→  let p = 0;
   653→  for (let h = 0; h < ctx.matrix.length; h++) {
   654→    const row = ctx.matrix[h] ?? [];
   655→    for (let a = 0; a < row.length; a++) {
   656→      const cell = row[a] ?? 0;
   657→      if (cell <= 0) continue;
   658→      let ok = true;
   659→      for (const k of matA) if (!orthoLegCoversCell(k, h, a)) { ok = false; break; }
   660→      if (!ok) continue;
   661→      for (const k of matB) if (!orthoLegCoversCell(k, h, a)) { ok = false; break; }
   662→      if (ok) p += cell * offProd;
   663→    }
   664→  }
   665→  return p;
   666→}

   668→/** پایه‌های خارج از ماتریس نمره (کرنر/کارت/نیمه اول) — احتمال مستقل پواسون */
   669→const OFF_MATRIX_LEGS = new Set(["HT_O05", "HT_00", "C_O95", "C_U95", "CARDS_O35", "CARDS_O45", "CARDS_U45"]);
   670→export function isOffMatrixLeg(key: string): boolean {
   671→  return OFF_MATRIX_LEGS.has(key);

   674→/**
   675→ * v11.0 — jointShadowMetrics — بازتولید سنجه‌های لحظه تصمیم سوپروایزر از λ و پایه‌های
   676→ * ذخیره‌شده کارت (بدون هیچ عدد ساختگی): ٪ باخت همزمان + ٪ برد طلایی همزمان.
   677→ * دقیقاً همان ریاضیات generateOrthogonalComplementarySlips است تا backfill
   678→ * کارت‌های قدیمی با صدور جدید کاملاً هم‌مقیاس بماند.
   679→ */
   680→export function jointShadowMetrics(
   681→  lambdaHome: number,
   682→  lambdaAway: number,
   683→  lambdaCorners: number,
   684→  lambdaCards: number,
   685→  legsA: string[],
   686→  legsB: string[]
   687→): { jointLossRiskPct: number; doubleWinProbPct: number } {
   688→  const matrix = buildScoreMatrix(lambdaHome || 1.35, lambdaAway || 1.25);
   689→  const ctx: OrthoContext = { lambdaHome, lambdaAway, matrix, lambdaCorners, lambdaCards };
   690→  const ht = buildHtProbs(lambdaHome || 1.35, lambdaAway || 1.25);
   691→  const lp: LegProbs = {
   692→    htOver05: ht.over05,
   693→    ht00: ht.zero,
   694→    cornersOver95: poissonTail(9, ctx.lambdaCorners),
   695→    cornersUnder95: 1 - poissonTail(9, ctx.lambdaCorners),
   696→    cardsOver35: poissonTail(3, ctx.lambdaCards),
   697→    cardsOver45: poissonTail(4, ctx.lambdaCards),
   698→    cardsUnder45: 1 - poissonTail(4, ctx.lambdaCards),
   699→  };
   700→  const pA = atomicLegProb(legsA, ctx, lp);
   701→  const pB = atomicLegProb(legsB, ctx, lp);
   702→  const bothWin = jointWinProb(legsA, legsB, ctx, lp);
   703→  const bothLose = Math.max(0, 1 - pA - pB + bothWin);
   704→  const r1 = (v: number) => Math.round(v * 1000) / 10; // کسر → ٪ با یک رقم اعشار
   705→  return { jointLossRiskPct: r1(bothLose), doubleWinProbPct: r1(bothWin) };
   706→}

   708→/** احتمال یک تیپ (همه پایه‌ها ببرند) — پشتیبانی از کارت‌های قدیمی چندپایه */
   709→function atomicLegProb(legs: string[], ctx: OrthoContext, lp: LegProbs): number {
   710→  const offFactors = legs.filter(isOffMatrixLeg).map((k) => orthoLegProb(k, ctx, lp));
   711→  for (const f of offFactors) if (f <= 1e-6) return 0;
   712→  const offProd = offFactors.reduce((s, x) => s * x, 1);
   713→  const matrixLegs = legs.filter((k) => !isOffMatrixLeg(k));
   714→  let p = 0;
   715→  for (let h = 0; h < ctx.matrix.length; h++) {
   716→    const row = ctx.matrix[h] ?? [];
   717→    for (let a = 0; a < row.length; a++) {
   718→      const cell = row[a] ?? 0;
   719→      if (cell <= 0) continue;
   720→      let ok = true;
   721→      for (const k of matrixLegs) {
   722→        if (!orthoLegCoversCell(k, h, a)) {
   723→          ok = false;
   724→          break;
   725→        }
   726→      }
   727→      if (ok) p += cell * offProd;
   728→    }
   729→  }
   730→  return p;
   731→}

   774→  };
