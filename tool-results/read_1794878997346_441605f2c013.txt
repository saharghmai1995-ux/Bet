   197→   197→ * H = −Σ mat[i][j]·ln(mat[i][j]) روی زیرماتریس ۰..۵ × ۰..۵ (نرمال‌شده درون همان بازه).
   198→   198→ */
   199→   199→export function shannonEntropy(matrix: number[][]): number {
   200→   200→  const N = Math.min(6, matrix.length);
   201→   201→  let sum = 0;
   202→   202→  for (let h = 0; h < N; h++) {
   203→   203→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) sum += matrix[h][a];
   204→   204→  }
   205→   205→  if (sum <= 0) return Math.log(36);
   206→   206→  let H = 0;
   207→   207→  for (let h = 0; h < N; h++) {
   208→   208→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   209→   209→      const p = matrix[h][a] / sum;
   210→   210→      if (p > 1e-12) H -= p * Math.log(p);
   211→   211→    }
   212→   212→  }
   213→   213→  return H;
   214→   214→}

   216→   216→/** نسبت عدم تقارن قدرت: Rλ = max(λH, λA) / min(λH, λA) */
   217→   217→export function asymmetryRatio(lambdaHome: number, lambdaAway: number): number {
   218→   218→  const hi = Math.max(lambdaHome, lambdaAway);
   219→   219→  const lo = Math.min(lambdaHome, lambdaAway);
   220→   220→  if (lo <= 0.01) return 9.99;
   221→   221→  return Math.min(9.99, hi / lo);
   222→   222→}

   225→   225→export function topCells(matrix: number[][], n = 4): { score: string; p: number }[] {
   226→   226→  const N = Math.min(6, matrix.length);
   227→   227→  const cells: { score: string; p: number }[] = [];
   228→   228→  for (let h = 0; h < N; h++) {
   229→   229→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   230→   230→      cells.push({ score: `${h}-${a}`, p: matrix[h][a] });
   231→   231→    }
   232→   232→  }
   233→   233→  cells.sort((x, y) => y.p - x.p);
   234→   234→  return cells.slice(0, n);
   235→   235→}

   238→   238→  const top4 = topCells(matrix, 4);

   242→   242→    top4Sum: Math.round(top4.reduce((s, c) => s + c.p, 0) * 1000) / 1000,
   243→   243→    top4,
   244→   244→  };
   245→   245→}

   249→   249→/** جمع احتمال خانه‌های ماتریس ۶×۶ که گزاره را ارضا می‌کنند (نرمال‌شده درون ۶×۶) */
   250→   250→function matrixProb(matrix: number[][], predicate: (h: number, a: number) => boolean): number {
   251→   251→  const N = Math.min(6, matrix.length);
   252→   252→  let inRegion = 0;
   253→   253→  let hit = 0;
   254→   254→  for (let h = 0; h < N; h++) {
   255→   255→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   256→   256→      inRegion += matrix[h][a];
   257→   257→      if (predicate(h, a)) hit += matrix[h][a];
   258→   258→    }
   259→   259→  }
   260→   260→  return inRegion > 0 ? hit / inRegion : 0;
   261→   261→}

   283→   283→ * (Non-Linear Sharpe & Kelly Utility — عیناً مطابق سند ارتقا v17.0)
   284→   284→ */
   285→   285→export function calculateNonLinearSupervisorScore(candidate: NonLinearCandidateInput): {
   286→   286→  ev: number;
   287→   287→  sharpe: number;
   288→   288→  score: number;
   289→   289→} {
   290→   290→  const netReturnOnWin = candidate.effectiveOdds - 1;
   291→   291→  const lossProb = Math.max(0, 1 - candidate.winProb);
   292→   292→  const expectedValue = candidate.winProb * netReturnOnWin + candidate.doubleWinProb * 0.45 - lossProb;
   293→   293→  const variance =
   294→   294→    candidate.winProb * Math.pow(netReturnOnWin - expectedValue, 2) +
   295→   295→    lossProb * Math.pow(-1 - expectedValue, 2);
   296→   296→  const sharpeRatio = expectedValue / Math.sqrt(Math.max(0.01, variance));
   297→   297→
   298→   298→  // جریمه غیرخطی ریسک و پاداش آنتروپی پایین
   299→   299→  const nonLinearSafetyBonus = Math.pow(Math.max(0, candidate.winProb), 2.2) * 100;
   300→   300→  const entropyBonus = Math.max(0, (2.8 - candidate.entropy) * 12);
   301→   301→
   302→   302→  const score = Number(
   303→   303→    ((sharpeRatio * 45 + nonLinearSafetyBonus * 0.55 + entropyBonus) * candidate.leagueReliability).toFixed(1)
   304→   304→  );
   305→   305→  return { ev: expectedValue, sharpe: sharpeRatio, score };
   306→   306→}

   573→   573→ * کاندید 🎯 CORRECT SCORE CLUSTER — خوشه‌بندی نتیجه دقیق ۳ یا ۴ شانسه.
   574→   574→ * شرط انتخاب غیرخطی: آنتروپی شانون پایین (H ≤ ۲.۴۲) و مجموع احتمال ۴ خانه برتر > ۵۶٪.
   575→   575→ * داچینگ ۱۰۰ واحدی بین نمرات دقیق به نسبت معکوس ضرایب — ضریب موثر خوشه ~۱.۷۵x تا ۲.۱۰x.
   576→   576→ */
   577→   577→export function buildCorrectScoreCandidate(
   578→   578→  matrix: number[][],
   579→   579→  structure: MatrixStructure
   580→   580→): BetTypeCandidate | null {
   581→   581→  // v18.0 — آستانه‌های سند v18: آنتروپی ≤ ۲.۴۸ (CS_ENTROPY_MAX_V18) و مجموع ۴ خانه برتر ≥ ۵۴٪ (CS_TOP4_SUM_MIN_V18)
   582→   582→  if (structure.entropy > CS_ENTROPY_MAX_V18 || structure.top4Sum <= CS_TOP4_SUM_MIN_V18) return null;
   583→   583→
   584→   584→  // خوشه ۳ یا ۴ تایی: خانه‌های برتر تا وقتی مجموع ≥ ۶۰٪ شود (حداقل ۳ نتیجه)
   585→   585→  const cells = topCells(matrix, 4);
   586→   586→  let cluster = cells.slice(0, 3);
   587→   587→  for (let i = 3; i < cells.length; i++) {
   588→   588→    if (cluster.reduce((s, c) => s + c.p, 0) < 0.6) cluster = cells.slice(0, i + 1);
   589→   589→  }
   590→   590→
   591→   591→  // ضرایب نتیجه دقیق — v21.0: کالیبراسیون واقعی 1xBet (0.945/prob) — بدون بذر/تورم مصنوعی
   592→   592→  // سقف سلامت ۱۱.۵x برای قیمت‌های بی‌معنا باقی است.
   593→   593→  const items = cluster.map((c) => {
   594→   594→    const odds = Math.min(11.5, computeRealistic1xBetOdds(c.p));
   595→   595→    return { score: c.score, odds, prob: c.p };
   596→   596→  });
   597→   597→  // تقسیم ۱۰۰ واحد بودجه به نسبت معکوس ضرایب (Equal-Profit Dutching)
   598→   598→  const implied = items.reduce((s, it) => s + 1 / it.odds, 0);
   599→   599→  const withStakes = items.map((it) => ({
   600→   600→    ...it,
   601→   601→    odds: it.odds,
   602→   602→    stake: Math.round((1 / it.odds / implied) * 1000) / 10,
   603→   603→    profit: Math.round((100 / implied - 100) * 10) / 10,
   604→   604→  }));
   605→   605→  const effectiveOdds = r2(1 / implied);
   606→   606→  const winProb = r3(withStakes.reduce((s, it) => s + it.prob, 0));
   607→   607→  const clusterLegs = withStakes.map((s) => `CS_${s.score}`);
   608→   608→
   609→   609→  const sc = calculateNonLinearSupervisorScore({
   610→   610→    winProb,
   611→   611→    effectiveOdds,
   612→   612→    doubleWinProb: 0,
   613→   613→    entropy: structure.entropy,
   614→   614→    leagueReliability: 1,
   615→   615→  });
   616→   616→
   617→   617→  return {
   618→   618→    key: "CORRECT_SCORE",
   619→   619→    variantId: `CS${withStakes.length}`,
   620→   620→    labelFa: `خوشه ${withStakes.length} تایی نتیجه دقیق (Correct Score Dutching)`,
   621→   621→    labelEn: `${withStakes.length}-Score Correct Score Cluster Dutching`,
   622→   622→    icon: "🎯",
   623→   623→    rationale: `آنتروپی شانون ماتریس H = ${structure.entropy.toFixed(2)} ≤ ۲.۴۲ و مجموع ۴ خانه برتر ${(structure.top4Sum * 100).toFixed(0)}٪ > ۵۶٪ — توزیع نتایج متمرکز؛ خوشه داچینگ نتیجه دقیق با ضریب موثر ${effectiveOdds.toFixed(2)}x`,
   624→   624→    mainLabel: `خوشه ${withStakes.length} نتیجه دقیق: ${withStakes.map((s) => s.score).join(" / ")}`,
   625→   625→    mainLabelEn: `Correct Score Cluster: ${withStakes.map((s) => s.score).join(" / ")}`,
   626→   626→    mainOdds: effectiveOdds,
   627→   627→    mainProb: winProb,
   628→   628→    mainLegs: clusterLegs,
   629→   629→    winProb,
   630→   630→    effectiveOdds,
   631→   631→    doubleWinProb: 0,
   632→   632→    jointLossProb: r3(1 - winProb),
   633→   633→    matrixCoveragePct: Math.round(winProb * 1000) / 10,
   634→   634→    eligible: true, // سازنده فقط با معیارهای سند v18 (آنتروپی ≤ ۲.۴۸ و Top4 ≥ ۵۴٪) کاندید می‌سازد
   635→   635→    tier: null,
   636→   636→    ...sc,
   637→   637→    stakeMain: 100,
   638→   638→    stakeCover: 0,
   639→   639→    guaranteedProfitUnits: Math.round((100 / implied - 100) * 10) / 10,
   640→   640→    cluster: withStakes.map((s) => ({
   641→   641→      score: s.score,
   642→   642→      odds: s.odds,
   643→   643→      prob: Math.round(s.prob * 1000) / 1000,
   644→   644→      stake: s.stake,
   645→   645→      profit: s.profit,
   646→   646→    })),
   647→   647→  };
   648→   648→}
   649→   649→
   650→   650→// ═══════════════════════════════ ⚖️ DUAL-HEDGE ═══════════════════════════════
   651→   651→
   652→   652→/**
   653→   653→ * کاندیدهای ⚖️ DUAL-HEDGE — زوج‌تیپ متعامد تک‌دکمه‌ای (۳ جفت طلایی G1/G2/G3).
   654→   654→ * شرط انتخاب غیرخطی: رژیم متعادل-هجومی و پوشش بالای ۹۱٪ با قابلیت برد طلایی همزمان
   655→   655→ * در نتایج 1-0 و 2-0. ضریب ~۲.۰x و ~۳.۰x | شتاب‌دهنده نامتقارن (Principal-Lock Booster).
   656→   656→ * کاندیدها از زوج‌های مونت‌کارلو موتور پرتفوی می‌آیند.
   657→   657→ */
   658→   658→export function buildDualHedgeCandidates(pairs: PortfolioPairDTO[]): BetTypeCandidate[] {
   659→   659→  const out: BetTypeCandidate[] = [];
   660→   660→  for (const p of pairs) {
   661→   661→    const winProb = 1 - p.jointStates.bothLose;
   662→   662→    if (winProb < DUAL_HEDGE_COVERAGE_MIN) continue;
   663→   663→    const implied = 1 / p.totalOddsA + 1 / p.totalOddsB;
   664→   664→    if (implied >= 1) continue; // بدون سود تضمینی
   665→   665→    const effectiveOdds = r2(1 / implied);
   666→   666→    const entropyProxy = 2.45; // رژیم متعادل — آنتروپی ماتریس بیرونی توسط routeBetTypes تزریق می‌شود
   667→   667→
   668→   668→    const sc = calculateNonLinearSupervisorScore({
   669→   669→      winProb: r3(winProb),
   670→   670→      effectiveOdds,
   671→   671→      doubleWinProb: r3(p.jointStates.bothWin),
   672→   672→      entropy: entropyProxy,
   673→   673→      leagueReliability: 1,
   674→   674→    });
   675→   675→
   676→   676→    out.push({
   677→   677→      key: "DUAL_HEDGE",
   678→   678→      variantId: `DUAL:${p.id}`,
   679→   679→      labelFa: `زوج‌تیپ متعامد تک‌دکمه‌ای${p.archetype ? ` — ${p.archetype.short}` : ""}`,
   680→   680→      labelEn: "Dual-Hedge Orthogonal Pair",
   681→   681→      icon: "⚖️",
   682→   682→      rationale: `رژیم متعادل-هجومی — پوشش ${(winProb * 100).toFixed(1)}٪${winProb >= 0.91 ? " (بالای ۹۱٪ سند)" : ""} با برد طلایی همزمان در نتایج 1-0 / 2-0 (P = ${(p.jointStates.bothWin * 100).toFixed(1)}٪) — گیت کیفیت با تابع مطلوبیت غیرخطی انجام می‌شود`,
   683→   683→      mainLabel: p.labelA,
   684→   684→      mainLabelEn: p.marketAEn ?? p.labelA,
   685→   685→      mainOdds: p.totalOddsA,
   686→   686→      mainProb: r3(p.selectionA.modelProb),
   687→   687→      mainLegs: [],
   688→   688→      coverLabel: p.labelB,
   689→   689→      coverLabelEn: p.marketBEn ?? p.labelB,
   690→   690→      coverOdds: p.totalOddsB,
   691→   691→      coverProb: r3(p.selectionB.modelProb),
   692→   692→      coverLegs: [],
   693→   693→      winProb: r3(winProb),
   694→   694→      effectiveOdds,
   695→   695→      doubleWinProb: r3(p.jointStates.bothWin),
   696→   696→      jointLossProb: r3(p.jointStates.bothLose),
   697→   697→      matrixCoveragePct: Math.round(winProb * 1000) / 10,
   698→   698→      // v18.0 — واجد شرایط دual: ریسک باخت همزمان ≤ ۱۱.۲٪ (DUAL_HEDGE_RISK_PCT_MAX_V18)
   699→   699→      eligible: winProb >= 0.9 && p.jointStates.bothLose <= DUAL_HEDGE_RISK_PCT_MAX_V18 / 100,
   700→   700→      tier: null,
   701→   701→      ...sc,
   702→   702→      stakeMain: p.equalProfit.stakeA,
   703→   703→      stakeCover: p.equalProfit.stakeB,
   704→   704→      guaranteedProfitUnits: p.equalProfit.profit,
   705→   705→      pairId: p.id,
   706→   706→    });
   707→   707→  }
   708→   708→  return out.sort((x, y) => y.score - x.score);
   709→   709→}
   710→   710→
   711→   711→// ═══════════════════════════════ مسیریاب اصلی ═══════════════════════════════
   712→   712→
   713→   713→/**
   714→   714→ * مسیریاب غیرخطی ۵ رژیمی — برای هر مسابقه هر ۴ روش تک‌بازی را با تابع مطلوبیت
   715→   715→ * غیرخطی شارپ/کلِی امتیاز می‌دهد و بهترین Bet-Type را برمی‌گرداند.
   716→   716→ *
   717→   717→ * قانون تایید هوشمند سوپروایزر (v17.0):
   718→   718→ *   · بازی‌های لیگ سمّی (esp.2 / usa.1 / fifa.friendly) → وتو مطلق
   719→   719→ *   · اگر بهترین کاندید EV > +0.08 و Sharpe > 0.32 دارد → تایید با آن Bet-Type
   720→   720→ *   · اگر هیچ روشی شارپ مثبت معنادار ندارد → وتو (بازی واقعاً بی‌کیفیت و پرنوسان)
   721→   721→ */
   722→   722→export function routeBetTypes(input: {
   723→   723→  matrix: number[][];
   724→   724→  lambdaHome: number;
   725→   725→  lambdaAway: number;
   726→   726→  prices: MarketPrices;
   727→   727→  pairs: PortfolioPairDTO[];
   728→   728→  league: string;
   729→   729→  /** ضریب اعتبار لیگ از کالیبراسیون (۰.۸۵..۱.۱۵) */
   730→   730→  leagueReliability?: number;
   731→   731→  /** v20.0 — بذر هش بازی برای ضریب یکتای هر مسابقه (فرمول سند §۲-۲) */
   732→   732→  matchSeed?: number;
   733→   733→  /**
   734→   734→   * v19.0 — تعداد کارت‌های تسویه‌شده هر رژیم در کارنامه (ورودی از auto-issue):
   735→   735→   * برای «توزیع متوازن ۵ رژیم» — رژیمِ کم‌نمایت‌ترین در استخر طلایی اولویت انتخاب
   736→   736→   * می‌گیرد تا ویترین سودآوری ۵ رژیمی هیچ باکس +0.0u نداشته باشد.
   737→   737→   */
   738→   738→  regimeCounts?: Partial<Record<BetTypeKey, number>>;
   739→   739→}): BetTypeRouteResult {
   740→   740→  const seed = input.matchSeed ?? matchSeedOf(`${input.league}|${input.prices.oddsHome}|${input.prices.oddsAway}|${input.lambdaHome}|${input.lambdaAway}`);
   741→   741→  const structure = matrixStructure(input.matrix, input.lambdaHome, input.lambdaAway);
   742→   742→  const reliability = Math.min(1.15, Math.max(0.85, input.leagueReliability ?? 1));
   743→   743→
   744→   744→  const poison = POISON_LEAGUES.some((p) => input.league.toLowerCase().includes(p.toLowerCase()));
   745→   745→  if (poison) {
   746→   746→    return {
   747→   747→      structure,
   748→   748→      candidates: [],
   749→   749→      winnerVariantId: null,
   750→   750→      winner: null,
   751→   751→      crownPairId: null,
   752→   752→      veto: true,
   753→   753→      vetoReason: `🚫 وتوی قرمز مطلق v18.0: لیگ سمّی (${input.league}) — نوسان ساختاری بالا و غیرقابل مدل‌سازی؛ همه ۴ روش تک‌بازی وتو می‌شوند — انتقال به دفتر سایه (سطح ۳)`,
   754→   754→      tier: null,
   755→   755→    };
   756→   756→  }
   757→   757→
   758→   758→  // اعمال ضریب اعتبار لیگ روی امتیاز کاندیدها (بازمحاسبه با reliability)
   759→   759→  const candidates: BetTypeCandidate[] = [
   760→   760→    ...buildBankerCandidates(input.matrix, input.prices, structure, seed),
   761→   761→    ...buildComboCandidates(input.matrix, input.prices, structure, seed),
   762→   762→    ...(buildCorrectScoreCandidate(input.matrix, structure) ? [buildCorrectScoreCandidate(input.matrix, structure)!] : []),
   763→   763→    ...buildDualHedgeCandidates(input.pairs),
   764→   764→  ];
   765→   765→  for (const c of candidates) {
   766→   766→    const rescored = calculateNonLinearSupervisorScore({
   767→   767→      winProb: c.winProb,
   768→   768→      effectiveOdds: c.effectiveOdds,
   769→   769→      doubleWinProb: c.doubleWinProb,
   770→   770→      entropy: structure.entropy,
   771→   771→      leagueReliability: reliability,
   772→   772→    });
   773→   773→    c.sharpe = r2(rescored.sharpe);
   774→   774→    c.ev = r3(rescored.ev);
   775→   775→    c.score = rescored.score;
   776→   776→  }
   777→   777→
   778→   778→  // ═══ v18.0 — گیت اختصاصی هر رژیم (Regime-Specific Supervisor Gate) ═══
   779→   779→  // وتوی ثابت λ_Fav/λ_Dog نسخه ۱۶ «قبل از مسیریابی» حذف شد؛ حالا رژیمِ برنده
   780→   780→  // گیت اختصاصی خودش را دارد (جدول سند v18.0) + سقف جهانی تاج طلایی:
   781→   781→  // ریسک ≤ ۱۱.۲٪ و شارپ ≥ ۰.۲۴.
   782→   782→  const fav = Math.max(input.lambdaHome, input.lambdaAway);
   783→   783→  const dog = Math.min(input.lambdaHome, input.lambdaAway);
   784→   784→  const riskPctOf = (c: BetTypeCandidate): number => c.jointLossProb * 100;
   785→   785→
   786→   786→  /** گیت اختصاصی رژیم هر کاندید — جدول سند v18.0 */
   787→   787→  const regimeGatePass = (c: BetTypeCandidate): { pass: boolean; why: string } => {
   788→   788→    const riskPct = riskPctOf(c);
   789→   789→    switch (c.key) {
   790→   790→      case "CORRECT_SCORE":
   791→   791→        // مجاز برای λ_Fav ∈ [1.15, 1.85] و λ_Dog ≤ 1.15 | شرط: Top4 ≥ ۵۴٪ و آنتروپی ≤ ۲.۴۸
   792→   792→        if (fav < CS_FAV_LAMBDA_MIN || fav > CS_FAV_LAMBDA_MAX || dog > CS_DOG_LAMBDA_MAX) {
   793→   793→          return { pass: false, why: `λ خارج پنجره خوشه نتیجه دقیق (λ_Fav=${fav.toFixed(2)} ∈ [1.15, 1.85]؟ λ_Dog=${dog.toFixed(2)} ≤ 1.15؟)` };
   794→   794→        }
   795→   795→        if (structure.top4Sum < CS_TOP4_SUM_MIN_V18 || structure.entropy > CS_ENTROPY_MAX_V18) {
   796→   796→          return { pass: false, why: `ساختار ماتریس برای خوشه نتیجه دقیق نامناسب (Top4=${(structure.top4Sum * 100).toFixed(0)}٪ ≥ ۵۴٪؟ H=${structure.entropy.toFixed(2)} ≤ 2.48؟)` };
   797→   797→        }
   798→   798→        return { pass: true, why: "خوشه نتیجه دقیق — بازی کم‌گل تاکتیکی با توزیع متمرکز" };
   799→   799→      case "COMBO":
   800→   800→        // مجاز برای λ_Fav ≥ 1.35 و λ_Dog ≤ 1.25 | شرط: پوشش کمبو ≥ ۷۸٪ و شارپ ≥ ۰.۲۲
   801→   801→        if (fav < COMBO_FAV_LAMBDA_MIN || dog > COMBO_DOG_LAMBDA_MAX) {
   802→   802→          return { pass: false, why: `λ خارج پنجره کمبو (λ_Fav=${fav.toFixed(2)} ≥ 1.35؟ λ_Dog=${dog.toFixed(2)} ≤ 1.25؟)` };
   803→   803→        }
   804→   804→        if (c.winProb < COMBO_COVERAGE_MIN_V18 || c.sharpe < COMBO_SHARPE_MIN_V18) {
   805→   805→          return { pass: false, why: `پوشش/شارپ کمبو کافی نیست (پوشش=${(c.winProb * 100).toFixed(0)}٪ ≥ ۷۸٪؟ شارپ=${c.sharpe.toFixed(2)} ≥ 0.22؟)` };
   806→   806→        }
   807→   807→        return { pass: true, why: "کمبوی رسمی — فاوریت کنترل‌گر با پوشش ≥ ۷۸٪" };
   808→   808→      case "BANKER":
   809→   809→        // مجاز برای rLambda ≥ 1.65 و احتمال پایه ≥ ۸۰٪
   810→   810→        if (structure.rLambda < BANKER_R_LAMBDA_MIN_V18 || c.mainProb < BANKER_BASE_PROB_MIN_V18) {
   811→   811→          return { pass: false, why: `عدم تقارن/احتمال پایه کافی نیست (Rλ=${structure.rLambda.toFixed(2)} ≥ 1.65؟ P=${(c.mainProb * 100).toFixed(0)}٪ ≥ ۸۰٪؟)` };
   812→   812→        }
   813→   813→        return { pass: true, why: "بنکر امن — فاوریت قاطع با عدم تقارن بالا" };
   814→   814→      case "DUAL_HEDGE":
   815→   815→        // شرط: ریسک باخت همزمان ≤ ۱۱.۲٪ و شارپ ≥ ۰.۲۴ (شتاب‌دهنده نامتقارن از گیت سود معاف است)
   816→   816→        if (riskPct > DUAL_HEDGE_RISK_PCT_MAX_V18 || c.sharpe < DUAL_HEDGE_SHARPE_MIN_V18) {
   817→   817→          return { pass: false, why: `ریسک/شارپ زوج‌تیپ کافی نیست (ریسک=${riskPct.toFixed(1)}٪ ≤ ۱۱.۲٪؟ شارپ=${c.sharpe.toFixed(2)} ≥ 0.24؟)` };
   818→   818→        }
   819→   819→        return { pass: true, why: "زوج‌تیپ متعامد — ریسک ≤ ۱۱.۲٪ و شارپ ≥ ۰.۲۴" };
   820→   820→      default:
   821→   821→        return { pass: false, why: "رژیم نامشخص" };
   822→   822→    }
   823→   823→  };
   824→   824→
   825→   825→  const riskCap = (c: BetTypeCandidate): boolean => riskPctOf(c) <= GOLD_RISK_PCT_MAX;
   826→   826→  const eliteValue = (c: BetTypeCandidate): boolean => c.ev > APPROVE_EV_MIN && c.sharpe > APPROVE_SHARPE_MIN;
   827→   827→
   828→   828→  /** 👑 سطح ۱ — تاج طلایی: گیت اختصاصی رژیم + سقف جهانی ریسک + کف شارپ ۰.۲۴ */
   829→   829→  const goldPass = (c: BetTypeCandidate): boolean =>
   830→   830→    riskCap(c) && c.sharpe >= GOLD_SHARPE_MIN && (regimeGatePass(c).pass || eliteValue(c));
   831→   831→
   832→   832→  /** 🥈 سطح ۲ — شکارچی آلفای سایه: winProb ≥ ۸۴٪ + شارپ ≥ ۰.۱۶ + ریسک ≤ ۱۵٪ (لیگ معتبر) */
   833→   833→  const silverPass = (c: BetTypeCandidate): boolean =>
   834→   834→    riskPctOf(c) <= HARD_RED_VETO_RISK_PCT &&
   835→   835→    c.winProb >= SILVER_WIN_PROB_MIN &&
   836→   836→    c.sharpe >= SILVER_SHARPE_MIN;
   837→   837→
   838→   838→  // برچسب‌گذاری تاج همه کاندیدها (شفافیت کامل تصمیم سطح‌بندی‌شده)
   839→   839→  for (const c of candidates) {

   841→   841→  }

   845→   845→
   846→   846→  // 🚫 وتوی قرمز مطلق سطح ۳ — همه کاندیدها ریسک بالای ۱۵٪ دارند → فقط دفتر کاغذی
   847→   847→  if (candidates.length === 0 || (gold.length === 0 && silver.length === 0)) {
   848→   848→    const worstRisk = candidates.length > 0 ? Math.max(...candidates.map(riskPctOf)) : 0;
   849→   849→    const reason =
   850→   850→      candidates.length === 0

   852→   852→        : worstRisk > HARD_RED_VETO_RISK_PCT
   853→   853→          ? `🚫 وتوی قرمز مطلق v18.0: کمترین سناریوی کیفیت ریسک ${worstRisk.toFixed(1)}٪ بالای سقف ۱۵٪ دارد — سطح ۳ (فقط تسویه کاغذی دفتر سایه)`
   854→   854→          : "🛡️ گیت v18.0: هیچ رژیمی گیت اختصاصی خود (جدول سند) و هیچ کاندیدی آستانه تاج نقره‌ای (winProb ≥ ۸۴٪ و شارپ ≥ ۰.۱۶) را پاس نکرد — انتقال به دفتر سایه";
   855→   855→    return {

   859→   859→      winner: null,
   860→   860→      crownPairId: null,

   865→   865→  }
   866→   866→
   867→   867→  // ═══ سطح‌بندی نهایی: تاج طلایی اولویت مطلق دارد؛ در نبود طلایی، بهترین نقره‌ای
   868→   868→  // به‌جای حبس در دفتر سایه با استیک نصف ۵۰ واحدی (Half-Kelly) منتشر می‌شود ═══
   869→   869→  const useSilver = gold.length === 0 && silver.length > 0;
   870→   870→
   871→   871→  // ═══ v19.0 — توزیع متوازن ۵ رژیم (Multi-Regime Balanced Routing — سند جراحی v19.0) ═══
   872→   872→  // «کارت‌های تاییدشده بر اساس ویژگی‌های بازی بین هر ۵ رژیم توزیع شوند تا هر ۵ باکس
   873→   873→  // ویترین دارای کارت تسویه‌شده واقعی و سود مثبت باشد». استخر توزیع = طلایی‌ها +
   874→   874→  // کاندیدهایی که گیت اختصاصی رژیم خود + سقف ریسک + EV مثبت را پاس کرده‌اند (دکترین
   875→   875→  // v18: هر رژیم با گیت خودش قضاوت می‌شود — شارپ فقط معیار سطح تاج است).
   876→   876→  const regimeCounts = input.regimeCounts ?? {};

   878→   878→  // سقف ریسک فقط برای رژیم‌هایی معنا دارد که jointLossProb «باخت همزمان دو تیپ» است؛
   879→   879→  // در 🎯 CORRECT_SCORE این عدد = احتمال وقوع هیچ نمره‌ای (~۵۰٪ — ذات داچینگ نتیجه دقیق)
   880→   880→  // است، نه ریسک ساختاری. کنترل کیفیت این رژیم گیت اختصاصی خودش (آنتروپی/Top4/λ) است.
   881→   881→  const balanceEligible = (c: BetTypeCandidate): boolean =>
   882→   882→    (c.key === "CORRECT_SCORE" || riskCap(c)) && c.ev > APPROVE_EV_MIN && regimeGatePass(c).pass;
   883→   883→  const gateQualified = candidates.filter((c) => c.tier === null && balanceEligible(c));

   885→   885→  const poolRegimes = [...new Set(pool.map((c) => c.key))];
   886→   886→  let best = pool.reduce((a, b) => (b.score > a.score ? b : a));
   887→   887→  if (poolRegimes.length > 1) {

   889→   889→    const minCount = Math.min(...poolRegimes.map(countOf));
   890→   890→    const underRep = poolRegimes.filter((k) => countOf(k) <= minCount + 4 && countOf(k) < countOf(best.key));
   891→   891→    if (underRep.length > 0) {
   892→   892→      // کف امتیاز مقطعی بین رژیم‌ها معنا ندارد (داچینگ نتیجه دقیق ذاتاً واریانس و شارپ
   893→   893→      // کمتری دارد اما گیت اختصاصی رژیم + EV مثبت + سقف ریسک آن را تایید کرده‌اند —
   894→   894→      // دکترین v18: «هر رژیم با گیت خودش قضاوت می‌شود»). بین رژیم‌های کم‌نمایت،
   895→   895→      // بالاترین امتیاز آن رژیم انتخاب می‌شود.
   896→   896→      const eligible = underRep
   897→   897→        .map((k) => pool.filter((c) => c.key === k).reduce((a, b) => (b.score > a.score ? b : a)))
   898→   898→        .sort((a, b) => b.score - a.score);
   899→   899→      if (eligible.length > 0) best = eligible[0];
   900→   900→    }
   901→   901→  }
   902→   902→
   903→   903→  if (useSilver) {
   904→   904→    // 🥈 تاج نقره‌ای — استیک نصف‌شده ۵۰ واحدی به روش Half-Kelly
   905→   905→    const scale = 0.5;
   906→   906→    best.stakeMain = Math.round(best.stakeMain * scale * 10) / 10;
   907→   907→    best.stakeCover = Math.round(best.stakeCover * scale * 10) / 10;
   908→   908→    best.guaranteedProfitUnits = Math.round(best.guaranteedProfitUnits * scale * 10) / 10;

   910→   910→    return {
   911→   911→      structure,
   912→   912→      candidates,

   923→   923→  if (best.sharpe < APPROVE_SHARPE_MIN || best.ev < APPROVE_EV_MIN) {
   924→   924→    const eliteEligible = gold.filter((c) => eliteValue(c));
   925→   925→    if (eliteEligible.length === 0) {
   926→   926→      // تایید مبنای سند — گیت اختصاصی رژیم برنده است
   927→   927→      return {
   928→   928→        structure,
   929→   929→        candidates,
   930→   930→        winnerVariantId: best.variantId,
   931→   931→        winner: best,
   932→   932→        crownPairId: best.pairId ?? null,
   933→   933→        veto: false,
   934→   934→        vetoReason: null,
   935→   935→        tier: "GOLD",
   936→   936→      };
   937→   937→    }
   938→   938→    const bestEligible = eliteEligible.reduce((a, b) => (b.score > a.score ? b : a));
   939→   939→    return {
   940→   940→      structure,
   941→   941→      candidates,
   942→   942→      winnerVariantId: bestEligible.variantId,
   943→   943→      winner: bestEligible,
   944→   944→      crownPairId: bestEligible.pairId ?? null,
   945→   945→      veto: false,
   946→   946→      vetoReason: null,
   947→   947→      tier: "GOLD",
   948→   948→    };
   949→   949→  }
   950→   950→
   951→   951→  return {
   952→   952→    structure,
   953→   953→    candidates,
   954→   954→    winnerVariantId: best.variantId,
   955→   955→    winner: best,
   956→   956→    crownPairId: best.pairId ?? null,
   957→   957→    veto: false,
   958→   958→    vetoReason: null,
   959→   959→    tier: "GOLD",
   960→   960→  };
   961→   961→}

   999→   999→/**

  1012→  1012→  const out: AccumulatorSlip[] = [];

  1014→  1014→  const legOf = (w: (typeof winners)[number]): AccumulatorLeg => ({
  1015→  1015→    matchId: w.matchId,
  1016→  1016→    matchLabel: w.matchLabel,

  1019→  1019→    // v17.0 — پای زوج‌تیپ: کل پوزیشن داچینگ‌شده با «ضریب موثر» و «پوشش کل» لحاظ می‌شود
  1020→  1020→    tip: w.candidate.key === "DUAL_HEDGE" ? `${w.candidate.mainLabel} ↔ ${w.candidate.coverLabel ?? ""}` : w.candidate.mainLabel,
  1021→  1021→    tipEn: w.candidate.key === "DUAL_HEDGE" ? "Dual-Hedge Effective Position" : w.candidate.mainLabelEn,

  1032→  1032→    const b = totalOdds - 1;

  1055→  1055→    out.push({

  1074→  1074→    const b = totalOdds - 1;
  1075→  1075→    const f = Math.max(0.008, Math.min(0.035, (hit * b - (1 - hit)) / b * 0.5));
  1076→  1076→    out.push({

  1078→  1078→      kind: "THREE_FOLD_COMBO",

  1081→  1081→      icon: "🔗",

  1083→  1083→      totalOdds,

  1091→  1091→    const hit = winners.slice(0, 3).reduce((s, w) => s * (w.candidate.key === "DUAL_HEDGE" ? w.candidate.winProb : w.candidate.mainProb), 1);
  1092→  1092→    const b = totalOdds - 1;
  1093→  1093→    const f = Math.max(0.006, Math.min(0.025, (hit * b - (1 - hit)) / b * 0.5));
  1094→  1094→    out.push({
  1095→  1095→      id: "ACC-3F",
  1096→  1096→      kind: "THREE_FOLD_COMBO",
  1097→  1097→      labelFa: "میکس حرفه‌ای ۳ پایه تاییدشده روز (3-Fold Pro Accumulator)",
  1098→  1098→      labelEn: "3-Fold Pro Accumulator",
  1099→  1099→      icon: "🔗",
  1100→  1100→      legs,
  1101→  1101→      totalOdds,
  1102→  1102→      stake: Math.round(f * 1000) / 10,
  1103→  1103→      hitRatePct: Math.round(hit * 1000) / 10,
  1104→  1104→      potentialReturn: Math.round(f * totalOdds * 10) / 10,
  1105→  1105→    });
  1106→  1106→  }
  1107→  1107→
  1108→  1108→  // ۳) Time-Staggered Hedged Accumulator — ۳ بازی با ساعت شروع متفاوت
  1109→  1109→  const staggerPool = comboOrBanker.length >= 3 ? comboOrBanker : winners;
  1110→  1110→  if (staggerPool.length >= 3) {
  1111→  1111→    const sorted = [...staggerPool].sort((a, b) => new Date(a.kickoff).getTime() - new Date(b.kickoff).getTime());
  1112→  1112→    // سه پای با بیشترین فاصله ساعت شروع
  1113→  1113→    const picked = [sorted[0], sorted[Math.floor(sorted.length / 2)], sorted[sorted.length - 1]];
  1114→  1114→    const distinctHours = new Set(picked.map((p) => new Date(p.kickoff).getHours())).size;
  1115→  1115→    if (distinctHours >= 2) {
  1116→  1116→      const legs = picked.map(legOf);
  1117→  1117→      const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
  1118→  1118→      const stake = 30;
  1119→  1119→      const payoutIfFirstTwoWin = stake * legs[0].odds * legs[1].odds;
