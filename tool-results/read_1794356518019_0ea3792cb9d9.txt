    80→    80→      if (lines.length === maxLines) break;
    81→    81→    }
    82→    82→  }
    83→    83→  if (cur && lines.length < maxLines) lines.push(cur);
    84→    84→  if (lines.length === maxLines && words.join(" ") !== lines.join(" ")) {
    85→    85→    let last = lines[maxLines - 1];
    86→    86→    while (textWidth(`${last}…`, fontSize, weight) > maxWidth && last.length > 4) last = last.slice(0, -1);
    87→    87→    lines[maxLines - 1] = `${last}…`;
    88→    88→  }
    89→    89→  return lines.length > 0 ? lines : ["—"];
    90→    90→}

   174→   174→    kind === "DOUBLE_WIN" ? "gold" : kind === "LOSS" ? "rain" : kind === "SNIPER" ? "sniper" : kind === "DAILY_REPORT" ? "aerial" : kind === "WIN" ? "win" : kind === "TEASER" ? "teaser" : "night";
