   129→   164→   164→function settlementProfit(
   130→   165→   165→  status: Exclude<SlipStatus, "PENDING" | "LIVE" | "MANUAL_REVIEW" | "VOID">,
   131→   166→   166→  stakeA: number,
   132→   167→   167→  stakeB: number,
   133→   168→   168→  oddsA: number,
   134→   169→   169→  oddsB: number
   135→   170→   170→): { netProfit: number; profitA: number; profitB: number } {
   136→   171→   171→  const r1 = (v: number) => Math.round(v * 100) / 100;
   137→   172→   172→  switch (status) {
   138→   173→   173→    case "WON_A":
   139→   174→   174→      // 🟢 برد تیپ A: سود خالص = stakeA × oddsA − ۱۰۰
   140→   175→   175→      return { netProfit: r1(stakeA * oddsA - DUTCH_TOTAL), profitA: r1(stakeA * oddsA - DUTCH_TOTAL), profitB: r1(-stakeB) };
   141→   176→   176→    case "WON_B":
   142→   177→   177→      // 🟢 برد تیپ B: سود خالص = stakeB × oddsB − ۱۰۰
   143→   178→   178→      return { netProfit: r1(stakeB * oddsB - DUTCH_TOTAL), profitA: r1(-stakeA), profitB: r1(stakeB * oddsB - DUTCH_TOTAL) };
   144→   179→   179→    case "DOUBLE_WIN":
   145→   180→   180→      // 🏆 برد طلایی هر دو تیپ: جمع سود هر دو
   146→   181→   181→      return {
   147→   182→   182→        netProfit: r1(stakeA * oddsA + stakeB * oddsB - DUTCH_TOTAL),
   148→   183→   183→        profitA: r1(stakeA * oddsA - stakeA),
   149→   184→   184→        profitB: r1(stakeB * oddsB - stakeB),
   150→   185→   185→      };
   151→   186→   186→    case "MIDDLE_TRAP":
   152→   187→   187→      // 🔴 باخت همزمان — ثبت صادقانه
   153→   188→   188→      return { netProfit: r1(-DUTCH_TOTAL), profitA: r1(-stakeA), profitB: r1(-stakeB) };
   154→   189→   189→  }
   155→   190→   190→}

   379→   414→   414→export function equalProfitStakes(oddsA: number, oddsB: number): { stakeA: number; stakeB: number } {
   380→   415→   415→  const invA = 1 / Math.max(1.01, oddsA);
   381→   416→   416→  const invB = 1 / Math.max(1.01, oddsB);
   382→   417→   417→  const sum = invA + invB;
   383→   418→   418→  const r1 = (v: number) => Math.round(v * 10) / 10;
   384→   419→   419→  return { stakeA: r1((DUTCH_TOTAL * invA) / sum), stakeB: r1((DUTCH_TOTAL * invB) / sum) };
   385→   420→   420→}

   388→   423→   423→export function slipHash(input: {
   389→   424→   424→  externalId: string | null;
   390→   425→   425→  homeTeam: string;
   391→   426→   426→  awayTeam: string;
   392→   427→   427→  kickoff: string;
   393→   428→   428→  labelA: string;
   394→   429→   429→  labelB: string;
   395→   430→   430→  oddsA: number;
   396→   431→   431→  oddsB: number;
   397→   432→   432→  stakeA: number;
   398→   433→   433→  stakeB: number;
   399→   434→   434→}): string {
   400→   435→   435→  const canonical = [
   401→   436→   436→    input.externalId ?? "NO-ESPN-ID",
   402→   437→   437→    input.homeTeam,
   403→   438→   438→    input.awayTeam,
   404→   439→   439→    new Date(input.kickoff).toISOString(),
   405→   440→   440→    input.labelA,
   406→   441→   441→    input.labelB,
   407→   442→   442→    input.oddsA.toFixed(2),
   408→   443→   443→    input.oddsB.toFixed(2),
   409→   444→   444→    input.stakeA.toFixed(1),
   410→   445→   445→    input.stakeB.toFixed(1),
   411→   446→   446→  ].join("|");

   413→   448→   448→}
