    56→export async function verifyBotToken(token: string): Promise<{ ok: boolean; username: string | null; detail: string }> {
    57→  try {
    58→    const res = await fetch(`https://api.telegram.org/bot${token.trim()}/getMe`, { cache: "no-store" });
    59→    const json = (await res.json()) as { ok?: boolean; result?: { username?: string }; description?: string };
    60→    if (json.ok && json.result?.username) return { ok: true, username: json.result.username, detail: `@${json.result.username}` };
    61→    return { ok: false, username: null, detail: json.description ?? `HTTP ${res.status}` };
    62→  } catch (err) {
    63→    return { ok: false, username: null, detail: err instanceof Error ? err.message : "network error" };
    64→  }
    65→}

   128→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
   129→  const token = await botToken();
   130→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   131→  // v13.2 — سپر توکن ناقص: خطای 404 مبهم تلگرام را با پیام دقیق جایگزین می‌کند
   132→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   133→  try {
   134→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
   135→      method: "POST",
   136→      headers: { "Content-Type": "application/json" },
   137→      body: JSON.stringify(body),
   138→      cache: "no-store",
   139→    });
   140→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   141→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
   142→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   143→  } catch (err) {
   144→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   145→  }
   146→}

   148→async function tgSendPhotoTo(
   149→  chatId: string | null,
   150→  photo: Buffer,
   151→  caption: string,
   152→  filename = "slip-card.png",
   153→  captionless = false
   154→): Promise<{ ok: boolean; detail: string }> {
   155→  const token = await botToken();
   156→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   157→  if (!chatId) return { ok: false, detail: "NO_CHANNEL" };
   158→  // v13.2 — سپر توکن ناقص (بخش hash بدون bot_id) — ریشه باگ «انتشار انجام نمیشه»
   159→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   160→  try {
   161→    const form = new FormData();
   162→    form.append("chat_id", chatId);
   163→    if (!captionless) {
   164→      form.append("caption", caption.slice(0, 1024));
   165→      form.append("parse_mode", "HTML");
   166→    }
   167→    form.append("photo", new Blob([new Uint8Array(photo)], { type: "image/png" }), filename);
   168→    const res = await fetch(`https://api.telegram.org/bot${token}/sendPhoto`, { method: "POST", body: form });
   169→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   170→    if (json.ok) return { ok: true, detail: `photo:${json.result?.message_id ?? "?"}${captionless ? ":broadcast" : ""}` };
   171→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   172→  } catch (err) {
   173→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   174→  }
   175→}

   214→/** پیام HTML فلش‌کارت کامل — تیپ A + تیپ B + داچینگ + SHA-256 + لینک ESPN */
   215→export function buildCardMessage(slip: IssuedSlip): string {
   216→  const kickoff = new Date(slip.kickoff).toLocaleString("fa-IR", { dateStyle: "short", timeStyle: "short" });
   217→  const lines: string[] = [];
   218→  lines.push(`⚽️ <b>فوتبال‌کوانت — فلش‌کارت تیپ مکمل</b>`);
   219→  lines.push(`🏆 <b>${esc(slip.homeTeam)}</b> × <b>${esc(slip.awayTeam)}</b>`);
   220→  lines.push(`📡 ${esc(slip.league)} · ⏰ ${kickoff}`);
   221→  if (slip.archetype) lines.push(`📐 الگو: ${esc(slip.archetype)}`);
   222→  lines.push("");
   223→  lines.push(`🔹 <b>تیپ A</b> (ضریب ${slip.oddsA.toFixed(2)}): ${esc(slip.labelA)}`);
   224→  lines.push(`💵 مبلغ داچینگ A: <b>${slip.stakeA.toFixed(1)}</b> واحد`);
   225→  lines.push("");
   226→  lines.push(`🟢 <b>تیپ B مکمل</b> (ضریب ${slip.oddsB.toFixed(2)}): ${esc(slip.labelB)}`);
   227→  lines.push(`💵 مبلغ داچینگ B: <b>${slip.stakeB.toFixed(1)}</b> واحد`);
   228→  lines.push("");
   229→  // ─── v10.0 — ارزش عددی نسبت به خط مرجع (همیشه عدد دارد) ───
   230→  if (slip.edgeA !== null || slip.edgeB !== null) {
   231→    const isPin = slip.edgeSourceA === "PINNACLE" || slip.edgeSourceB === "PINNACLE";
   232→    const src = isPin ? "خط شارپ پیناکل" : "خط منصفانه بازار";
   233→    const fa = (v: number) => `${v >= 0 ? "+" : ""}${v.toFixed(1)}%`;
   234→    const a = slip.edgeA !== null ? fa(slip.edgeA) : "—";
   235→    const b = slip.edgeB !== null ? fa(slip.edgeB) : "—";
   236→    lines.push(`📈 ارزش عددی نسبت به ${src}: تیپ A <b>${a}</b> · تیپ B <b>${b}</b>`);
   237→    if (isPin) {
   238→      const cmp = (odds: number, pin: number | null): string | null =>
   239→        pin && pin > 1.01 ? `${odds.toFixed(2)} در برابر خط ${pin.toFixed(2)}` : null;
   240→      const pa = slip.edgeSourceA === "PINNACLE" ? cmp(slip.oddsA, slip.pinnacleOddsA) : null;
   241→      const pb = slip.edgeSourceB === "PINNACLE" ? cmp(slip.oddsB, slip.pinnacleOddsB) : null;
   242→      const parts = [pa ? `تیپ A: ${pa}` : null, pb ? `تیپ B: ${pb}` : null].filter((x): x is string => Boolean(x));
   243→      if (parts.length > 0) lines.push(`📌 مقایسه قیمت با خط پیناکل: ${parts.join(" · ")}`);
   244→    }
   245→  }
   246→  // ─── v10.0 — بهترین ساعت ثبت ضریب (Optimal Odds Timing Window) ───
   247→  if (slip.timingWindowA || slip.timingWindowB) {
   248→    const winName = timingWindowLabel(slip.timingWindowA ?? slip.timingWindowB);
   249→    const start = slip.timingStartA ?? slip.timingStartB;
   250→    const end = slip.timingEndA ?? slip.timingEndB;
   251→    const fmt = (d: Date) => new Intl.DateTimeFormat("fa-IR", { hour: "2-digit", minute: "2-digit", timeZone: "Asia/Tehran" }).format(d);
   252→    const range = start && end ? ` | ساعت دقیق پیشنهادی ثبت: ${fmt(new Date(start))} تا ${fmt(new Date(end))} (تهران)` : "";
   253→    lines.push(`⏱️ بهترین زمان ثبت این تیپ: <b>${esc(winName)}</b>${range}`);
   254→  }
   255→  lines.push(`💰 سود تضمینی پوشش: <b>+${slip.guaranteedProfit.toFixed(1)}%</b> (مجموع استیک ۱۰۰ واحد)`);
   256→  lines.push(`🔐 هش راستی‌آزمایی SHA-256:\n<code>${slip.hash.slice(0, 32)}</code>`);
   257→  if (slip.espnUrl) lines.push(`🔎 <a href="${slip.espnUrl}">لینک راستی‌آزمایی رسمی ESPN</a>`);
   258→  lines.push("");
   259→  lines.push(`⚖️ ابزار تحلیل کمّی — مسئولیت استفاده با کاربر است.`);
   260→  return lines.join("\n");
   261→}

   450→/** آمار وین‌ریت کانال — مستقیم از دیتابیس (بدون وابستگی چرخه‌ای) */
   451→async function channelWinRateStats() {
   452→  // v11.0 — فقط کارت‌های واقعی تاییدشده؛ تسویه کاغذی دفتر سایه در وین‌ریت کانال دخالت نمی‌کند
   453→  // v13.2 — فقط کارت‌های واقعی تاییدشده؛ کارت‌های بازخوانی (retroIssued) و دفتر سایه در وین‌ریت کانال دخالت نمی‌کنند
   454→  const settled = await db.issuedSlip.findMany({
   455→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, supervisorApproved: true, retroIssued: false },
   456→    select: { status: true, netProfit: true, stakeA: true, stakeB: true },
   457→  });
   458→  const st = { settled: settled.length, wonA: 0, wonB: 0, doubleWin: 0, middleTrap: 0, netProfit: 0, staked: 0 };
   459→  for (const s of settled) {
   460→    if (s.status === "WON_A") st.wonA += 1;
   461→    else if (s.status === "WON_B") st.wonB += 1;
   462→    else if (s.status === "DOUBLE_WIN") st.doubleWin += 1;
   463→    else if (s.status === "MIDDLE_TRAP") st.middleTrap += 1;
   464→    st.netProfit += s.netProfit ?? 0;
   465→    st.staked += s.stakeA + s.stakeB;
   466→  }
   467→  const winRatePct = st.settled > 0 ? Math.round(((st.settled - st.middleTrap) / st.settled) * 1000) / 10 : 0;
   468→  const netPct = st.staked > 0 ? Math.round((st.netProfit / st.staked) * 1000) / 10 : 0;
   469→  // ─── v11.0 — دفتر سایه: شفافیت counterfactual وتوها (خط جداگانه گزارش) ───
   470→  // v13.5.1 — سایه فقط وتوهای زنده است؛ کارت‌های بازخوانی (retroIssued) داخل سایه شمرده نمی‌شوند
   471→  const shadowRows = await db.issuedSlip.findMany({
   472→    where: { supervisorApproved: false, retroIssued: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   473→    select: { status: true, netProfit: true },
   474→  });
   475→  const shadowWins = shadowRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   476→  const shadowNet = Math.round(shadowRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100;
   477→  const shadowWinRate = shadowRows.length > 0 ? Math.round((shadowWins / shadowRows.length) * 1000) / 10 : null;
   478→  // ─── v13.2 — دفتر بازخوانی: کارت‌های چندروزه‌ای که هرگز پیش‌بازی صادر نشده‌اند (تسویه کاغذی تحلیلی) ───
   479→  // v13.5.1 — بازخوانی با پرچم retroIssued شناسایی می‌شود (مستقل از تایید — کارت بازخوانی هرگز تاییدشده نیست)
   480→  const retroRows = await db.issuedSlip.findMany({
   481→    where: { retroIssued: true, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   482→    select: { status: true, netProfit: true },
   483→  });
   484→  const retroWins = retroRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   485→  return {
   486→    ...st,
   487→    winRatePct,
   488→    netPct,
   489→    netProfit: Math.round(st.netProfit * 100) / 100,
   490→    shadow: { settled: shadowRows.length, wins: shadowWins, winRatePct: shadowWinRate, netProfit: shadowNet },
   491→    retro: {
   492→      settled: retroRows.length,
   493→      wins: retroWins,
   494→      winRatePct: retroRows.length > 0 ? Math.round((retroWins / retroRows.length) * 1000) / 10 : null,
   495→      netProfit: Math.round(retroRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100,
   496→    },
   497→  };
   498→}

   533→/** صف انتشار بدون ارسال — پیش‌نمایش زنده فید کانال (Live Telegram Channel Simulator) */
   534→export async function previewDispatchQueue(): Promise<{ queue: DispatchQueueItem[]; autoDispatch: boolean; configured: boolean }> {
   535→  // v13.2 — فقط کارت‌های بازی‌های «امروز تهران» به کانال ارسال می‌شوند؛ بازی‌های روزهای آینده

   539→    db.issuedSlip.findMany({
   540→      where: { status: { in: ["PENDING", "LIVE"] }, telegramCard: false, supervisorApproved: true, retroIssued: false, kickoff: { gte: day.start, lt: day.end } },
   541→      orderBy: { kickoff: "asc" },
   542→      take: 8,
   543→    }),
   544→    db.issuedSlip.findMany({
   545→      where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, telegramResult: false, supervisorApproved: true, retroIssued: false },
   546→      orderBy: { settledAt: "asc" },
   547→      take: 8,
   548→    }),

   550→    telegramStatus(),

   554→  for (const s of pendingCards) {

   556→      kind: "CARD",

   558→      slipHash: s.hash,

   561→      status: s.status,
   562→    });

   564→  for (const s of pendingResults) {

   570→      message: buildResultMessage(s),
   571→      status: s.status,
   572→    });
   573→  }
   574→  const st = await channelWinRateStats();
   575→  if (pendingResults.length > 0) {

   584→ *   · پیام نوع دوم (پس از FT): برگه نتیجه + به‌روزرسانی وین‌ریت کل کانال
   585→ * Dedupe سمت سرور با پرچم‌های telegramCard/telegramResult — ارسال تکراری صفر.
   586→ * v9.0 — سقف هر پاس قابل تنظیم است: پس‌زمینه ۳/۳ (حفظ تأخیر سینک) و دکمه فوری ۸/۸.
   587→ */
   588→export async function autoDispatchApprovedSlips(options?: {
   589→  maxCards?: number;
   590→  maxResults?: number;
   591→}): Promise<{
   592→  ok: boolean;
   593→  sentCards: number;
   594→  sentResults: number;
   595→  winRateSent: boolean;
   596→  skippedVetoed: number;
   597→  detail: string;
   598→}> {
   599→  const maxCards = Math.max(1, Math.min(12, options?.maxCards ?? 3));
   600→  const maxResults = Math.max(1, Math.min(12, options?.maxResults ?? 3));
   601→  const [tg, autoOn] = await Promise.all([telegramStatus(), getAutoDispatch()]);
   602→  // v13.5.1 — کارت‌های بازخوانی (retroIssued) وتو نیستند — شمارش وتو فقط برای کارت‌های زنده
   603→  const skippedVetoed = await db.issuedSlip.count({ where: { supervisorApproved: false, retroIssued: false, telegramCard: false } });
   604→
   605→  if (!tg.configured || !tg.channelSet) {
   606→    return { ok: false, sentCards: 0, sentResults: 0, winRateSent: false, skippedVetoed, detail: "TELEGRAM_NOT_CONFIGURED" };
   607→  }
   608→  if (!tg.tokenFormat) {
   609→    return { ok: false, sentCards: 0, sentResults: 0, winRateSent: false, skippedVetoed, detail: "INVALID_TOKEN_FORMAT" };
   610→  }
   611→  if (!autoOn) {
   612→    return { ok: false, sentCards: 0, sentResults: 0, winRateSent: false, skippedVetoed, detail: "AUTO_DISPATCH_OFF" };
   613→  }
   614→
   615→  // v13.2 — «فقط کارت‌های روزانه»: پنجره انتشار کارت = بازی‌های امروز تهران
   616→  const day = tehranDayRange();

   621→  // ─── مرحله ۱: پست پیش از بازی — فقط کارت‌های 👑 تاییدشده سوپروایزر «بازی‌های امروز تهران» ───
   622→  const cards = await db.issuedSlip.findMany({
   623→    where: { status: { in: ["PENDING", "LIVE"] }, telegramCard: false, supervisorApproved: true, retroIssued: false, kickoff: { gte: day.start, lt: day.end } },
   624→    orderBy: { kickoff: "asc" },
   625→    take: maxCards,
   626→  });
   627→  for (const s of cards) {
   628→    try {
   629→      const res = await publishSlipToTelegram(s, "CARD");
   630→      if (res.ok) sentCards += 1;
   631→    } catch (err) {
   632→      console.warn("auto dispatch card failed:", err);
   633→    }
   634→  }
   635→
   636→  // ─── مرحله ۲: پست تسویه‌حساب پس از بازی — برگه نتیجه FT (کارت‌های بازخوانی مستثنا) ───
   637→  const results = await db.issuedSlip.findMany({
   638→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, telegramResult: false, supervisorApproved: true, retroIssued: false },
   639→    orderBy: { settledAt: "asc" },
   640→    take: maxResults,
   641→  });
   642→  for (const s of results) {
   643→    try {
   644→      const ok = await sendSettlementNotification(s);
   645→      if (ok) sentResults += 1;
   646→    } catch (err) {
   647→      console.warn("auto dispatch result failed:", err);
   648→    }
   649→  }
   650→
   651→  // ─── به‌روزرسانی وین‌ریت کل کانال — فقط وقتی نتیجه تازه‌ای مخابره شد ───
   652→  let winRateSent = false;
   653→  if (sentResults > 0) {
   654→    winRateSent = await sendWinRateReport();
   655→  }
   656→
   657→  return {
   658→    ok: sentCards + sentResults > 0,
   659→    sentCards,
   660→    sentResults,
   661→    winRateSent,
   662→    skippedVetoed,
   663→    detail: sentCards + sentResults > 0 ? "dispatched" : "QUEUE_EMPTY",
   664→  };

  1306→export async function sendNightlyPnl(force = false): Promise<boolean> {
  1307→  const token = await botToken();
  1308→  if (!token) return false;
  1309→  const chatId = await publicChannelId();
  1310→  if (!chatId) return false;
  1311→
  1312→  const today = tehranDayKey();
  1313→  if (!force) {
  1314→    const last = await getSetting(KEY_FUNNEL_PNL_DAY);
  1315→    if (last === today) return false;
  1316→  }
  1317→
  1318→  const rows = await db.issuedSlip.findMany({
  1319→    where: { supervisorApproved: true, retroIssued: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, settledAt: { not: null } },
  1320→    orderBy: { settledAt: "desc" },
  1321→    take: 80,
  1322→  });

  1324→
  1325→  const tierNet = (line: string) => todayRows.filter((r) => (r.productLine ?? "SINGLE") === line).reduce((a, r) => a + (r.netProfit ?? 0), 0);
  1326→  const netA = tierNet("SINGLE");
  1327→  const netB = tierNet("COMBO");
  1328→  const netC = tierNet("SNIPER");

  1330→  const settled = todayRows.length;
  1331→  const wins = todayRows.filter((r) => r.status !== "MIDDLE_TRAP").length;
  1332→  const winRate = settled > 0 ? (wins / settled) * 100 : 0;
  1333→

  1335→  const timeEn = (d: Date) => new Intl.DateTimeFormat("en-GB", { hour: "2-digit", minute: "2-digit", hour12: false, timeZone: "Asia/Tehran" }).format(d);

  1338→    dateLabelEn,
  1339→    kpis: [

  1343→      { label: "CHANNEL WIN-RATE", value: `${winRate.toFixed(0)}%`, color: C_cyan },
  1344→    ],

  1349→      profit: r.netProfit ?? 0,

  1361→
  1362→  let res = await tgSendPhotoTo(chatId, photo, caption, `nightly-pnl-${today}.png`);
  1363→  if (!res.ok) {
  1364→    res = await tgCall("sendMessage", { chat_id: chatId, text: caption, parse_mode: "HTML", disable_web_page_preview: true });

  1368→    await logOutbox(null, "FUNNEL_PNL", true, `public:${chatId}`);
  1369→  }
  1370→  return res.ok;
