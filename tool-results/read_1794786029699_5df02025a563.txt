    65→async function enforceDailyBroadcastQuorum(): Promise<number> {
    66→  const shifted = Date.now() + TEHRAN_OFFSET_MS;
    67→  const dayStart = new Date(Math.floor(shifted / 86_400_000) * 86_400_000 - TEHRAN_OFFSET_MS);
    68→  const dayEnd = new Date(dayStart.getTime() + 86_400_000);
    69→
    70→  const approvedToday = await db.issuedSlip.count({
    71→    where: { supervisorApproved: true, retroIssued: false, kickoff: { gte: dayStart, lt: dayEnd } },
    72→  });
    73→  if (approvedToday >= DAILY_QUORUM_MIN) return 0;
    74→
    75→  const need = DAILY_QUORUM_MIN - approvedToday;
    76→  const candidates = await db.issuedSlip.findMany({
    77→    where: {
    78→      supervisorApproved: false,
    79→      retroIssued: false,
    80→      status: "PENDING",
    81→      kickoff: { gte: new Date(Date.now() + QUORUM_MIN_LEAD_MS), lt: dayEnd },
    82→      vetoReason: { startsWith: "🛡️" }, // فقط گیت ریسک/فاوریت/سقف — لیگ سمّی (🚫) هرگز ترویج نمی‌شود
    83→      jointLossRiskPct: { lte: QUORUM_MAX_RISK_PCT },
    84→    },
    85→    orderBy: { supervisorScore: "desc" },
    86→    take: need,
    87→  });
    88→  if (candidates.length === 0) return 0;
    89→
    90→  let promoted = 0;
    91→  for (const c of candidates) {
    92→    try {
    93→      // v18.0 — ترویج به «تاج نقره‌ای سطح ۲» (شکارچی آلفای سایه) نه تاج طلایی قلابی:
    94→      // استیک نصف ۵۰ واحدی Half-Kelly + نشان crownTier=SILVER_SHADOW — صادقانه‌تر و
    95→      // هم‌مبنا با سیستم دو‌سطحی تاج سند v18.0.
    96→      const halfStake = (v: number): number => Math.round(v * 0.5 * 10) / 10;
    97→      await db.issuedSlip.update({
    98→        where: { id: c.id },
    99→        data: {
   100→          supervisorApproved: true,
   101→          crownTier: "SILVER_SHADOW",
   102→          stakeA: halfStake(c.stakeA),
   103→          stakeB: halfStake(c.stakeB),
   104→          guaranteedProfit: Math.max(0, halfStake(c.guaranteedProfit)),
   105→          vetoReason: `🥈 ترویج سطح ۲ v18.0 (شکارچی آلفای سایه): بهترین کاندیدای باقی‌مانده امروز (امتیاز ${(c.supervisorScore ?? 0).toFixed(1)}، ریسک باخت همزمان ${(c.jointLossRiskPct ?? 0).toFixed(1)}٪) با استیک ایمن ۵۰ واحدی Half-Kelly منتشر می‌شود — حفظ تقویم پخش در روز فاقد کارت تراز اول`,
   106→        },
   107→      });
   108→      promoted += 1;
   109→    } catch (err) {
   110→      console.warn("quorum promotion failed:", err);
   111→    }
   112→  }

   117→// رژیم پنجم ویترین (🔗 آکومولاتور ۲-۳ تایی) تا امروز فقط سطح پرتفوی UI بود و کارت

   119→// هر روز یک کارت میکس ۲-۳ پایه واقعی صادر می‌شود (externalId = acc-<dayKey> idempotent،
   120→// احتمال مدل = حاصل‌ضرب احتمال پای‌ها، ضریب کل = حاصل‌ضرب ضرایب) و با فید ESPN
   121→// پای‌به‌پا تسویه می‌شود — تسویه در settleFinishedSlips (شاخه ACCUMULATOR).
   122→const ACC_MIN_LEGS = 2;
   123→const ACC_MAX_LEGS = 3;
   124→const ACC_LEG_LINES = ["BANKER", "COMBO"];
   125→const ACC_LOOKBACK_DAYS = 7;

   127→async function issueDailyAccumulators(cleanSlate: boolean): Promise<number> {
   128→  const nowMs = Date.now();
   129→  const from = new Date(nowMs - ACC_LOOKBACK_DAYS * 86_400_000);
   130→  const eligible = await db.issuedSlip.findMany({
   131→    where: {
   132→      supervisorApproved: true,
   133→      retroIssued: false,
   134→      productLine: { in: ACC_LEG_LINES },
   135→      kickoff: { gte: from },
   136→      status: { in: ["PENDING", "LIVE", "WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] },
   137→    },
   138→    orderBy: { supervisorScore: "desc" },
   139→    take: 60,
   140→  });
   141→  if (eligible.length < ACC_MIN_LEGS) return 0;
   142→
   143→  // گروه‌بندی پای‌ها بر اساس روز تهران کیک‌آف
   144→  const byDay = new Map<string, typeof eligible>();
   145→  for (const s of eligible) {
   146→    if (!s.externalId) continue;
   147→    const dayKey = new Date(new Date(s.kickoff).getTime() + TEHRAN_OFFSET_MS).toISOString().slice(0, 10);
   148→    const arr = byDay.get(dayKey) ?? [];
   149→    arr.push(s);
   150→    byDay.set(dayKey, arr);
   151→  }
   152→
   153→  let created = 0;
   154→  for (const [dayKey, slips] of byDay) {
   155→    const externalId = `acc-${dayKey}`;
   156→    // Idempotency — آکومولاتور هر روز فقط یک‌بار
   157→    const existingAcc = await db.issuedSlip.findFirst({ where: { externalId, autoIssued: true }, select: { id: true } });
   158→    if (existingAcc) continue;
   159→
   160→    const usedExt = new Set<string>();
   161→    const legs: { key: string; matchExternalId: string; matchLabel: string; odds: number; prob: number; label: string }[] = [];
   162→    for (const s of slips) {
   163→      if (legs.length >= ACC_MAX_LEGS) break;
   164→      if (usedExt.has(s.externalId as string)) continue;
   165→      let leg: { key?: string; matchExternalId?: string; matchLabel?: string } | undefined;
   166→      try {
   167→        leg = (JSON.parse(s.legsA || "[]") as { key?: string; matchExternalId?: string; matchLabel?: string }[])[0];
   168→      } catch {
   169→        leg = undefined;
   170→      }
   171→      if (!leg?.key || !leg?.matchExternalId) continue;
   172→      usedExt.add(s.externalId as string);
   173→      legs.push({
   174→        key: leg.key,
   175→        matchExternalId: leg.matchExternalId,
   176→        matchLabel: leg.matchLabel ?? `${s.homeTeam} × ${s.awayTeam}`,
   177→        odds: s.oddsA,
   178→        prob: s.modelProbA ?? 0.5,
   179→        label: s.labelA ?? leg.key,
   180→      });
   181→    }
   182→    if (legs.length < ACC_MIN_LEGS) continue;
   183→
   184→    const totalOdds = Math.round(legs.reduce((a, l) => a * l.odds, 1) * 100) / 100;
   185→    const totalProb = Math.max(0.02, Math.min(0.99, legs.reduce((a, l) => a * Math.max(0.05, Math.min(0.97, l.prob)), 1)));
   186→    const firstKo = [...slips.map((s) => new Date(s.kickoff).getTime())].sort((a, b) => a - b)[0];
   187→    const isFinishedAcc = firstKo < nowMs;
   188→    const item: IssueSlipItemDTO = {
   189→      match: {
   190→        id: externalId,
   191→        externalId,
   192→        league: "Smart Accumulator",
   193→        homeTeam: `ACC ${legs.length}-Leg`,
   194→        awayTeam: "Multi-Match Chain",
   195→        kickoff: new Date(firstKo).toISOString(),
   196→        espnUrl: null,
   197→      },
   198→      labelA: `🔗 ${legs.length}-Leg Smart Accumulator — ${legs.map((l) => l.label.slice(0, 34)).join(" + ")}`,
   199→      labelB: "—",
   200→      legsA: legs.map((l) => ({ key: l.key, matchExternalId: l.matchExternalId, matchLabel: l.matchLabel })),
   201→      legsB: [],
   202→      oddsA: totalOdds,
   203→      oddsB: 1.01,
   204→      stakeA: 100,
   205→      stakeB: 0,
   206→      guaranteedProfit: 0,
   207→      marginPct: 0,
   208→      archetype: "Smart Accumulator — best approved banker/combo chain",
   209→      archetypeCode: "ACC",
   210→      productLine: "ACCUMULATOR",
   211→      modelProbAOverride: totalProb,
   212→      modelProbBOverride: null,
   213→      supervisorApproved: true,
   214→      retroIssued: isFinishedAcc && !cleanSlate,
   215→      crownTier: "GOLD",
   216→      supervisorScore: Math.round((slips.reduce((a, s) => a + (s.supervisorScore ?? 0), 0) / Math.max(1, slips.length)) * 10) / 10,
   217→      vetoReason: null,
   218→    };
   219→    try {
   220→      const issued = await issueSlip(item, { autoIssued: true });
   221→      if (issued.created) {
   222→        // کارت‌های تاریخی میکس هرگز به کانال اعلام نمی‌شوند (تاریخ‌اند، نه خبر تازه)
   223→        if (isFinishedAcc) {
   224→          await db.issuedSlip.update({
   225→            where: { id: issued.slip.id },
   226→            data: { telegramCard: true, telegramResult: true, settleNote: "v19.0 smart accumulator — legs settled on official ESPN FT" },
   227→          });
   228→        }
   229→        created += 1;
   230→        console.log(`[auto-issue] v19.0 accumulator ${dayKey}: ${legs.length} legs @ ${totalOdds.toFixed(2)}x issued`);
   231→      }
   232→    } catch (err) {
   233→      console.warn(`accumulator issuance failed for ${dayKey}:`, err);
   234→    }
   235→  }
   236→  return created;
   237→}
