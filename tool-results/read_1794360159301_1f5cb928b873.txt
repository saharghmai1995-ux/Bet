    30→   489→/** نقطه‌های فرم تیم (W/D/L) — ۵ بازی آخر */
    31→   490→function formDots(form: string | null, cx: number, cy: number): string {
    32→   491→  const seq = String(form ?? "")
    33→   492→    .toUpperCase()
    34→   493→    .replace(/[^WDL]/g, "")
    35→   494→    .slice(-5)
    36→   495→    .split("");
    37→   496→  if (seq.length === 0) return "";
    38→   497→  const gap = 21;
    39→   498→  const x0 = cx - ((seq.length - 1) * gap) / 2;
    40→   499→  return seq
    41→   500→    .map((ch, i) => {
    42→   501→      const color = ch === "W" ? C.emerald : ch === "L" ? C.rose : "#93a7c4";
    43→   502→      return `<circle cx="${(x0 + i * gap).toFixed(1)}" cy="${cy}" r="7.5" fill="${color}" fill-opacity="0.92"/>`;
    44→   503→    })
    45→   504→    .join("");
    46→   505→}

    63→   522→    `<rect x="${sx.toFixed(1)}" y="${y}" width="${sw.toFixed(1)}" height="${h}" fill="${color}" fill-opacity="${fillOp}"/>`;
    64→   523→  return `
    65→   524→  <g>
    66→   525→    ${seg(x, hW, C.cyan, "0.55")}
    67→   526→    ${seg(x + hW, dW, "#93a7c4", "0.4")}
