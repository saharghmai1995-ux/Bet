   106→   106→export async function claimSetting(key: string, value: string): Promise<boolean> {
   107→   107→  const existing = await db.setting.findUnique({ where: { key } }).catch(() => null);
   108→   108→  if (!existing) {
   109→   109→    try {
   110→   110→      await db.setting.create({ data: { key, value } });
   111→   111→      return true;
   112→   112→    } catch {
   113→   113→      return false; // مسیر موازی هم‌زمان ردیف را ساخت
   114→   114→    }
   115→   115→  }
   116→   116→  if (existing.value === value) return false;
   117→   117→  const res = await db.setting.updateMany({ where: { key, value: { not: value } }, data: { value } });
   118→   118→  return res.count === 1;
   119→   119→}

   275→   275→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
   276→   276→  const token = await botToken();
   277→   277→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   278→   278→  // v13.2 — سپر توکن ناقص: خطای 404 مبهم تلگرام را با پیام دقیق جایگزین می‌کند
   279→   279→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   280→   280→  try {
   281→   281→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
   282→   282→      method: "POST",
   283→   283→      headers: { "Content-Type": "application/json" },
   284→   284→      body: JSON.stringify(body),
   285→   285→      cache: "no-store",
   286→   286→    });
   287→   287→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   288→   288→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
   289→   289→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   290→   290→  } catch (err) {
   291→   291→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   292→   292→  }
   293→   293→}

   327→   327→/** v20.0 — اعتبارسنجی سخت PNG: امضای ۸ بایتی + حداقل ۲۰KB */
   328→   328→function isValidPng(buf: Buffer | null | undefined): boolean {
   329→   329→  if (!buf || buf.length < 20480) return false;
   330→   330→  const SIG = [0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a];
   331→   331→  for (let i = 0; i < 8; i++) if (buf[i] !== SIG[i]) return false;
   332→   332→  return true;
   333→   333→}

   335→   335→/**
   336→   336→ * v20.0 — کپشن ایمن عکس: برش فقط در مرز خط + بستن تگ‌های باز + سقف ۱۰۰۰ کاراکتر.
   337→   337→ * خروجی همیشه HTML-معتبر است (تلگرام هرگز parse-error نمی‌گیرد → پیام بی‌تصویر صفر).
   338→   338→ */
   339→   339→export function safePhotoCaption(text: string, limit = 1000): string {
   340→   340→  let s = String(text ?? "");
   341→   341→  if (s.length <= limit) return s;
   342→   342→  // برش در مرز خط — هیچ تگی نصف نمی‌شود
   343→   343→  s = s.slice(0, limit);
   344→   344→  const lastNl = s.lastIndexOf("\n");
   345→   345→  if (lastNl > 200) s = s.slice(0, lastNl);
   346→   346→  // شمارش تگ‌های باز تلگرام و بستن به ترتیب معکوس (HTML تلگرام فقط ۵ تگ پشتیبانی می‌کند)
   347→   347→  const TAGS = ["b", "i", "u", "s", "code", "pre", "a", "blockquote"];
   348→   348→  const stack: string[] = [];
   349→   349→  const re = /<\/?([a-z]+)[^>]*>/gi;
   350→   350→  let m: RegExpExecArray | null;
   351→   351→  while ((m = re.exec(s)) !== null) {
   352→   352→    const tag = m[1].toLowerCase();
   353→   353→    if (!TAGS.includes(tag)) continue;
   354→   354→    if (m[0][1] === "/") {
   355→   355→      const idx = stack.lastIndexOf(tag);
   356→   356→      if (idx >= 0) stack.length = idx;
   357→   357→    } else if (!m[0].endsWith("/>")) {
   358→   358→      stack.push(tag);
   359→   359→    }
   360→   360→  }
   361→   361→  for (const tag of stack.reverse()) s += `</${tag}>`;
   362→   362→  return s;
   363→   363→}

   421→   421→// پیام را دو بار می‌فرستادند. سه سپر:
   422→   422→//   ۱) قفل in-flight سراسری برای هر جریان ارسال (dispatch / funnel / winrate)

   431→   431→/** قفل رشته‌ای funnel — اجرای هم‌زمان دو پست فانل (دو مسیر ورود) را سریال می‌کند */
   432→   432→function withFunnelLock<F extends (...args: never[]) => Promise<unknown>>(fn: F): F {
   433→   433→  return (async (...args: Parameters<F>) => {
   434→   434→    const prev = gBroadcast.__fqFunnelLock ?? Promise.resolve();
   435→   435→    let release!: () => void;
   436→   436→    const gate = new Promise<void>((r) => (release = r));
   437→   437→    gBroadcast.__fqFunnelLock = prev.then(() => gate);
   438→   438→    await prev.catch(() => undefined);
   439→   439→    try {
   440→   440→      return await fn(...(args as Parameters<F>));
   441→   441→    } finally {
   442→   442→      release();
   443→   443→      if (gBroadcast.__fqFunnelLock === gate) gBroadcast.__fqFunnelLock = null;
   444→   444→    }
   445→   445→  }) as F;
   446→   446→}
