    48→interface GradedLeg {
    49→  key: string;
    50→  won: boolean;
    51→  /** بازار قابل تسویه از اسکوربورد نیست (کرنر/کارت/نیمه اول) */
    52→  offBoard: boolean;
    53→  /** بازیِ این شرط هنوز FT نشده */
    54→  notFinal: boolean;
    55→  matchLabel: string;
    56→}

    58→function legCovers(key: string, hg: number, ag: number): boolean {
    59→  return selectionCoversCell(key, hg, ag);
    60→}
    61→
    62→/**
    63→ * ارزیابی همه شرط‌های یک تیپ با نتایج واقعی ESPN.
    64→ * هر شرط ممکن است روی بازی دیگری باشد (میکس دو بازی) → امتیاز همان بازی از نقشه اسکوربورد.
    65→ */
    66→function gradeLegs(
    67→  legs: SettlementLegDTO[],
    68→  scoreOf: (externalId: string) => { hg: number; ag: number; state: string } | undefined,
    69→  fallbackExternalId: string | null
    70→): { graded: GradedLeg[]; allWon: boolean; anyLost: boolean; hasOffBoard: boolean; allFinal: boolean } {
    71→  const graded: GradedLeg[] = [];
    72→  let allWon = true;
    73→  let anyLost = false;
    74→  let hasOffBoard = false;
    75→  let allFinal = true;
    76→
    77→  for (const leg of legs) {
    78→    const ext = leg.matchExternalId || fallbackExternalId;
    79→    const label = leg.matchLabel || ext || "بازی نامشخص";
    80→    if (!ext || isOffMatrixKey(leg.key)) {
    81→      hasOffBoard = true;
    82→      graded.push({ key: leg.key, won: false, offBoard: true, notFinal: false, matchLabel: label });
    83→      continue;
    84→    }
    85→    const score = scoreOf(ext);
    86→    if (!score) {
    87→      // بازیِ این شرط در پنجره امروز/دیروز ESPN پیدا نشد
    88→      allFinal = false;
    89→      graded.push({ key: leg.key, won: false, offBoard: false, notFinal: true, matchLabel: label });
    90→      continue;
    91→    }
    92→    if (score.state !== "post") {
    93→      allFinal = false;
    94→      graded.push({ key: leg.key, won: false, offBoard: false, notFinal: true, matchLabel: label });
    95→      continue;
    96→    }
    97→    const won = legCovers(leg.key, score.hg, score.ag);
    98→    if (!won) {
    99→      allWon = false;
   100→      anyLost = true;
   101→    }
   102→    graded.push({ key: leg.key, won, offBoard: false, notFinal: false, matchLabel: label });
   103→  }
   104→
   105→  // شرط سینتتیک خارج اسکوربورد اگر تنها مشکل باشد → MANUAL_REVIEW؛ در غیر این صورت فقط تیپ‌های قابل ارزیابی لحاظ می‌شوند
   106→  const evaluable = graded.filter((g) => !g.offBoard);
   107→  const evaluableAllFinal = evaluable.length > 0 && evaluable.every((g) => !g.notFinal);
   108→  const evaluableAllWon = evaluable.length > 0 && evaluable.every((g) => g.won);
   109→  const evaluableAnyLost = evaluable.some((g) => !g.won);
   110→
   111→  return {
   112→    graded,
   113→    allWon: evaluableAllWon,
   114→    anyLost: evaluableAnyLost,
   115→    hasOffBoard,
   116→    allFinal: evaluableAllFinal,
   117→  };
   118→}

   133→/** نقشه اسکوربورد ESPN امروز+دیروز — بر اساس externalId و تطبیق نام تیم */
   134→async function loadEspnScoreWindow(): Promise<{ byId: Map<string, EspnScoreInfo>; byName: Map<string, EspnScoreInfo>; count: number }> {
   135→  const events: LiveMatchEvent[] = await fetchEspnEvents(-RESULT_WINDOW_DAYS, 0);
   136→  const byId = new Map<string, EspnScoreInfo>();
   137→  const byName = new Map<string, EspnScoreInfo>();
   138→  for (const ev of events) {
   139→    const info: EspnScoreInfo = {
   140→      hg: ev.homeScore ?? 0,
   141→      ag: ev.awayScore ?? 0,
   142→      state: ev.statusState,
   143→      statusDetail: ev.statusDetail,
   144→      kickoff: ev.kickoff,
   145→      homeTeam: ev.homeTeam,
   146→      awayTeam: ev.awayTeam,
   147→      league: ev.leagueName,
   148→    };
   149→    byId.set(ev.externalId, info);
   150→    // کلید تطبیق نامی — برای کارت‌هایی که شناسه ESPN ندارند
   151→    const nk = `${normTeam(ev.homeTeam)}::${normTeam(ev.awayTeam)}::${new Date(ev.kickoff).toISOString().slice(0, 10)}`;
   152→    if (!byName.has(nk)) byName.set(nk, info);
   153→  }
   154→  return { byId, byName, count: events.length };
   155→}

   163→/** فرمول سود مستند در سند ارتقا — مجموع استیک ۱۰۰ واحد */
   164→function settlementProfit(
   165→  status: Exclude<SlipStatus, "PENDING" | "LIVE" | "MANUAL_REVIEW" | "VOID">,
   166→  stakeA: number,
   167→  stakeB: number,
   168→  oddsA: number,
   169→  oddsB: number
   170→): { netProfit: number; profitA: number; profitB: number } {
   171→  const r1 = (v: number) => Math.round(v * 100) / 100;
   172→  switch (status) {
   173→    case "WON_A":
   174→      // 🟢 برد تیپ A: سود خالص = stakeA × oddsA − ۱۰۰
   175→      return { netProfit: r1(stakeA * oddsA - DUTCH_TOTAL), profitA: r1(stakeA * oddsA - DUTCH_TOTAL), profitB: r1(-stakeB) };
   176→    case "WON_B":
   177→      // 🟢 برد تیپ B: سود خالص = stakeB × oddsB − ۱۰۰
   178→      return { netProfit: r1(stakeB * oddsB - DUTCH_TOTAL), profitA: r1(-stakeA), profitB: r1(stakeB * oddsB - DUTCH_TOTAL) };
   179→    case "DOUBLE_WIN":
   180→      // 🏆 برد طلایی هر دو تیپ: جمع سود هر دو
   181→      return {
   182→        netProfit: r1(stakeA * oddsA + stakeB * oddsB - DUTCH_TOTAL),
   183→        profitA: r1(stakeA * oddsA - stakeA),
   184→        profitB: r1(stakeB * oddsB - stakeB),
   185→      };
   186→    case "MIDDLE_TRAP":
   187→      // 🔴 باخت همزمان — ثبت صادقانه
   188→      return { netProfit: r1(-DUTCH_TOTAL), profitA: r1(-stakeA), profitB: r1(-stakeB) };
   189→  }
   190→}

   193→ * قلب موتور: بررسی همه کارت‌های در انتظار با فید FT واقعی ESPN و تسویه خودکار.

   195→export async function settleFinishedSlips(options?: { notify?: boolean }): Promise<SettlementSyncResultDTO> {
   196→  const t0 = Date.now();
   197→  const notify = options?.notify !== false;
   198→  const settled: SettlementSyncResultDTO["settled"] = [];
   199→  let liveCount = 0;
   200→  let notified = 0;
   201→  let ftFound = 0;
   202→  /** v9.0 — سقف اعلام تصویری هر پاس (تولید تصویر ~۲۰ ثانیه هر پیام — حفظ تأخیر سینک) */
   203→  const NOTIFY_CAP = 4;
   204→  let notifyBudget = NOTIFY_CAP;
   205→
   206→  const [{ byId, byName, count }, pendingSlips] = await Promise.all([
   207→    loadEspnScoreWindow(),
   208→    db.issuedSlip.findMany({
   209→      where: { status: { in: ["PENDING", "LIVE"] } },
   210→      orderBy: { createdAt: "asc" },
   211→    }),
   212→  ]);
   213→
   214→  const scoreOf = (externalId: string) => {
   215→    const direct = byId.get(externalId);
   216→    if (direct) return { hg: direct.hg, ag: direct.ag, state: direct.state };
   217→    return undefined;
   218→  };
   219→
   220→  for (const slip of pendingSlips) {
   221→    // ─── تطبیق کارت با رخداد ESPN: اولویت externalId، سپس نام تیم‌ها + روز کیک‌آف ───
   222→    let info: EspnScoreInfo | undefined = slip.externalId ? byId.get(slip.externalId) : undefined;
   223→    if (!info && slip.externalId) {
   224→      // شناسه ESPN ممکن است پس از به‌روزرسانی رخداد تغییر کرده باشد → تطبیق نامی
   225→      const dayKey = new Date(slip.kickoff).toISOString().slice(0, 10);
   226→      info = byName.get(`${normTeam(slip.homeTeam)}::${normTeam(slip.awayTeam)}::${dayKey}`);
   227→    }
   228→
   229→    // ─── بازی در جریان → برچسب LIVE ───
   230→    if (info && info.state === "in") {
   231→      if (slip.status !== "LIVE") {
   232→        await db.issuedSlip.update({ where: { id: slip.id }, data: { status: "LIVE" } });
   233→      }
   234→      liveCount += 1;
   235→      continue;
   236→    }
   237→
   238→    // ─── مسابقه به تعویق افتاده/لغو شده → VOID شفاف ───
   239→    if (info && /postpon|cancell|suspend|abandon|forfeit/i.test(info.statusDetail)) {
   240→      await db.issuedSlip.update({
   241→        where: { id: slip.id },
   242→        data: {
   243→          status: "VOID",
   244→          settleNote: `ESPN وضعیت «${info.statusDetail}» را ثبت کرد — تسویه انجام نمی‌شود (VOID)`,
   245→        },
   246→      });
   247→      continue;
   248→    }
   249→
   250→    if (info && info.state === "post" && (info.hg > 0 || info.ag > 0 || /ft|full|agg|pens/i.test(info.statusDetail))) {
   251→      ftFound += 1;
   252→      const legsA = safeParseLegs(slip.legsA);
   253→      const legsB = safeParseLegs(slip.legsB);
   254→      const gradedA = gradeLegs(legsA, scoreOf, slip.externalId);
   255→      const gradedB = gradeLegs(legsB, scoreOf, slip.externalId);
   256→
   257→      let status: SlipStatus;
   258→      let note: string;
   259→
   260→      if (!gradedA.allFinal || !gradedB.allFinal) {
   261→        // شرطی روی بازی دومی است که هنوز FT نشده — صبورانه در انتظار می‌مانیم
   262→        if (Date.now() - new Date(slip.kickoff).getTime() > VOID_AFTER_HOURS * 3600_000) {
   263→          status = "VOID";
   264→          note = "بازی مکمل در پنجره ESPN یافت نشد — احتمالاً به تعویق افتاده (VOID)";
   265→        } else {
   266→          continue;
   267→        }
   268→      } else if (gradedA.hasOffBoard || gradedB.hasOffBoard) {
   269→        // بازار کرنر/کارت/نیمه اول از اسکوربورد قابل تسویه نیست — شفاف اعلام می‌شود
   270→        status = "MANUAL_REVIEW";
   271→        note = "شامل بازار خارج اسکوربورد ESPN (کرنر/کارت/نیمه اول) — تسویه دستی لازم است";
   272→      } else {
   273→        const aWon = gradedA.allWon;
   274→        const bWon = gradedB.allWon;
   275→        if (aWon && bWon) {
   276→          status = "DOUBLE_WIN";
   277→          note = "🏆 نتیجه در خانه‌های مشترک هر دو تیپ — برد طلایی (Jackpot)";
   278→        } else if (aWon) {
   279→          status = "WON_A";
   280→          note = "🟢 تیپ A برد — تسویه خودکار با نتیجه واقعی ESPN";
   281→        } else if (bWon) {
   282→          status = "WON_B";
   283→          note = "🟢 تیپ B برد — تسویه خودکار با نتیجه واقعی ESPN";
   284→        } else {
   285→          status = "MIDDLE_TRAP";
   286→          note = "🔴 تله میانی — هر دو تیپ باختند؛ ثبت صادقانه و شفاف";
   287→        }
   288→      }
   289→
   290→      if ((status as string) === "PENDING" || (status as string) === "LIVE") continue;
   291→
   292→      const hg = info.hg;
   293→      const ag = info.ag;
   294→      const prof =
   295→        status === "WON_A" || status === "WON_B" || status === "DOUBLE_WIN" || status === "MIDDLE_TRAP"
   296→          ? settlementProfit(status, slip.stakeA, slip.stakeB, slip.oddsA, slip.oddsB)
   297→          : { netProfit: null, profitA: null, profitB: null };
   298→
   299→      await db.issuedSlip.update({
   300→        where: { id: slip.id },
   301→        data: {
   302→          status,
   303→          homeGoals: hg,
   304→          awayGoals: ag,
   305→          profitA: prof.profitA,
   306→          profitB: prof.profitB,
   307→          netProfit: prof.netProfit,
   308→          settledAt: new Date(),
   309→          settleNote: note,
   310→        },
   311→      });
   312→
   313→      // ─── v10.0 — ژورنال + دفتر دکترین: سیستم از هر تسویه یاد می‌گیرد ───
   314→      try {
   315→        const fresh = await db.issuedSlip.findUnique({ where: { id: slip.id } });
   316→        if (fresh) await recordSettlementLearning(fresh);
   317→      } catch (err) {
   318→        console.warn("doctrine learning failed:", err);
   319→      }
   320→
   321→      settled.push({
   322→        id: slip.id,
   323→        matchLabel: `${slip.homeTeam} ${hg}−${ag} ${slip.awayTeam}`,
   324→        status,
   325→        netProfit: prof.netProfit ?? 0,
   326→      });
   327→
   328→      // ─── اعلام نتیجه در کانال تلگرام VIP (تصویر جام / رختکن بهم‌ریخته) ───
   329→      if (notify && notifyBudget > 0 && status !== "VOID" && status !== "MANUAL_REVIEW" && !slip.telegramResult) {
   330→        try {
   331→          const { sendSettlementNotification } = await import("./telegram-service");
   332→          const updated = await db.issuedSlip.findUnique({ where: { id: slip.id } });
   333→          if (updated) {
   334→            notifyBudget -= 1;
   335→            const ok = await sendSettlementNotification(updated);
   336→            if (ok) notified += 1;
   337→          }
   338→        } catch (err) {
   339→          console.warn("telegram settlement notify failed:", err);
   340→        }
   341→      }
   342→      continue;
   343→    }
   344→
   345→    // ─── رخدادی در ESPN نیست → VOID پس از ۳۶ ساعت (به تعویق/لغو) ───
   346→    if (!info && Date.now() - new Date(slip.kickoff).getTime() > VOID_AFTER_HOURS * 3600_000) {
   347→      await db.issuedSlip.update({
   348→        where: { id: slip.id },
   349→        data: {
   350→          status: "VOID",
   351→          settleNote: "نتیجه در اسکوربورد ESPN (امروز+دیروز) یافت نشد — بازی به تعویق افتاده یا خارج پنجره (VOID)",
   352→        },
   353→      });
   354→    }
   355→  }
   356→
   357→  // ─── ۴.۶) v9.0 — گزارش وین‌ریت کل کانال پس از تسویه تازه (پیام نوع دوم) ───
   358→  if (notify && settled.length > 0) {
   359→    try {
   360→      const { sendWinRateReport } = await import("./telegram-service");
   361→      await sendWinRateReport();
   362→    } catch (err) {
   363→      console.warn("win-rate report failed:", err);
   364→    }
   365→  }
   366→
   367→  // ─── ۴.۶) تلاش مجدد اعلام نتایج ناموفق (مثلاً تصویر 429 یا قطعی تلگرام) — سقف ۵ در هر اجرا ───
   368→  if (notify) {
   369→    try {
   370→      const unnotified = await db.issuedSlip.findMany({
   371→        where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, telegramResult: false, supervisorApproved: true },
   372→        orderBy: { settledAt: "asc" },
   373→        take: 5,
   374→      });
   375→      if (unnotified.length > 0) {
   376→        const { sendSettlementNotification } = await import("./telegram-service");
   377→        for (const s of unnotified) {
   378→          try {
   379→            const ok = await sendSettlementNotification(s);
   380→            if (ok) notified += 1;
   381→          } catch (err) {
   382→            console.warn("settlement notify retry failed:", err);
   383→          }
   384→        }
   385→      }
   386→    } catch (err) {
   387→      console.warn("settlement notify retry pass failed:", err);
   388→    }
   389→  }
   390→
   391→  return {
   392→    success: true,
   393→    checkedEspn: count,
   394→    ftFound,
   395→    settled,
   396→    live: liveCount,
   397→    notified,
   398→    elapsedMs: Date.now() - t0,
   399→  };
   400→}

   413→/** استیک داچینگ سود برابر روی مجموع ۱۰۰ واحد — مبنای فرمول سند ارتقا */
   414→export function equalProfitStakes(oddsA: number, oddsB: number): { stakeA: number; stakeB: number } {
   415→  const invA = 1 / Math.max(1.01, oddsA);
   416→  const invB = 1 / Math.max(1.01, oddsB);
   417→  const sum = invA + invB;
   418→  const r1 = (v: number) => Math.round(v * 10) / 10;
   419→  return { stakeA: r1((DUTCH_TOTAL * invA) / sum), stakeB: r1((DUTCH_TOTAL * invB) / sum) };
   420→}

   422→/** هش SHA-256 سند کارت — قابل راستی‌آزمایی */
   423→export function slipHash(input: {
   424→  externalId: string | null;
   425→  homeTeam: string;
   426→  awayTeam: string;
   427→  kickoff: string;
   428→  labelA: string;
   429→  labelB: string;
   430→  oddsA: number;
   431→  oddsB: number;
   432→  stakeA: number;
   433→  stakeB: number;
   434→}): string {
   435→  const canonical = [
   436→    input.externalId ?? "NO-ESPN-ID",
   437→    input.homeTeam,
   438→    input.awayTeam,
   439→    new Date(input.kickoff).toISOString(),
   440→    input.labelA,
   441→    input.labelB,
   442→    input.oddsA.toFixed(2),
   443→    input.oddsB.toFixed(2),
   444→    input.stakeA.toFixed(1),
   445→    input.stakeB.toFixed(1),
   446→  ].join("|");
   447→  return crypto.createHash("sha256").update(canonical, "utf8").digest("hex");
   448→}

   450→/** خانه‌های پوشش هر تیپ روی ماتریس ۶×۶ (برای کارت تلگرام و نمایش کارنامه) */
   451→function coveredCells(lambdaHome: number, lambdaAway: number, legs: SettlementLegDTO[], externalId: string | null): string[] {
   452→  const matrix = buildScoreMatrix(lambdaHome || 1.35, lambdaAway || 1.25);
   453→  const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   454→  const cells: string[] = [];
   455→  for (let h = 0; h < matrix.length; h++) {
   456→    for (let a = 0; a < matrix[h].length; a++) {
   457→      if (keys.length > 0 && keys.every((k) => legCovers(k, h, a))) cells.push(`${h}-${a}`);
   458→    }
   459→  }
   460→  void externalId;
   461→  return cells.slice(0, 12);
   462→}

   464→// ═══════════ v10.0 — ارزش عددی پیناکل + احتمال مدل + پنجره زمانی ═══════════
   465→
   466→/** احتمال مدل برای یک تیپ از ماتریس امتیاز دیکسون-کولز (جمع خانه‌های پوشش) */
   467→function legProbability(lambdaHome: number, lambdaAway: number, legs: SettlementLegDTO[]): number | null {
   468→  const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   469→  if (keys.length === 0) return null;
   470→  const matrix = buildScoreMatrix(lambdaHome || 1.35, lambdaAway || 1.25);
   471→  let p = 0;
   472→  for (let h = 0; h < matrix.length; h++) {
   473→    for (let a = 0; a < matrix[h].length; a++) {
   474→      if (keys.every((k) => legCovers(k, h, a))) p += matrix[h][a];
   475→    }
   476→  }
   477→  return Math.round(p * 10000) / 10000;
   478→}

   480→/**
   481→ * ارزش عددی نسبت به خط مرجع (سوال مستقیم کارفرما: «چرا عددی ندارد؟»):
   482→ *   edge٪ = (prob مدل × ضریب ثبت − ۱) × ۱۰۰
   483→ * خط مرجع: پیناکل وقتی خط شارپ واقعی موجود است؛ وگرنه خط منصفانه موتور (MARKET_FAIR)
   484→ * خروجی همیشه عدد دارد — «—» فقط برای بازار خارج ماتریس (کرنر/کارت).
   485→ */
   486→function computeEdge(modelProb: number | null, odds: number): number | null {
   487→  if (modelProb === null || modelProb <= 0 || odds <= 1) return null;
   488→  return Math.round((modelProb * odds - 1) * 1000) / 10;
   489→}

   491→/**
   492→ * صدور کارت به کارنامه — Idempotent با هش SHA-256.
   493→ * اگر کارت با همین هش قبلاً صادر شده باشد، همان ردیف بازگردانده می‌شود.
   494→ * v9.0: نگهداری λ ترکیبی نهایی + کد الگو (حلقه کالیبراسیون) + نشان تایید سوپروایزر
   495→ * و پرچم ثبت خودکار سرور (autoIssued) برای خط لوله Zero-Click.
   496→ */
   497→export async function issueSlip(
   498→  item: IssueSlipItemDTO,
   499→  opts?: { autoIssued?: boolean }
   500→): Promise<{ created: boolean; slip: IssuedSlipDTO }> {
   501→  const externalId = item.match.externalId ?? null;
   502→  const stakes =
   503→    item.stakeA && item.stakeB && item.stakeA + item.stakeB > 0
   504→      ? { stakeA: item.stakeA, stakeB: item.stakeB }
   505→      : equalProfitStakes(item.oddsA, item.oddsB);
   506→
   507→  const hash = slipHash({
   508→    externalId,
   509→    homeTeam: item.match.homeTeam,
   510→    awayTeam: item.match.awayTeam,
   511→    kickoff: item.match.kickoff,
   512→    labelA: item.labelA,
   513→    labelB: item.labelB,
   514→    oddsA: item.oddsA,
   515→    oddsB: item.oddsB,
   516→    stakeA: stakes.stakeA,
   517→    stakeB: stakes.stakeB,
   518→  });
   519→
   520→  const existing = await db.issuedSlip.findUnique({ where: { hash } });
   521→  if (existing) return { created: false, slip: mapSlip(existing) };
   522→
   523→  // ─── v10.0 — احتمال مدل، ارزش عددی و پنجره زمانی طلایی ثبت ───
   524→  const lH = item.lambdaHome ?? 1.35;
   525→  const lA = item.lambdaAway ?? 1.25;
   526→  const probA = legProbability(lH, lA, item.legsA ?? []);
   527→  const probB = legProbability(lH, lA, item.legsB ?? []);
   528→
   529→  // ─── v10.0.1 — خط شارپ واقعی پیناکل برای انتخاب مستقیم (پاسخ «چرا عددی ندارد؟») ───
   530→  // ارزش نسبت به خط مرجع:
   531→  //   · PINNACLE موجود → (ضریب ثبت ÷ خط پیناکل − ۱) × ۱۰۰ — مزیت قیمت نسبت به خط شارپ
   532→  //   · در غیر این صورت → (احتمال مدل × ضریب ثبت − ۱) × ۱۰۰ — EV نسبت به خط منصفانه موتور
   533→  const pinPrices = item.pinnaclePrices ?? {};
   534→  const pinPriceOf = (legs: SettlementLegDTO[]): number | null => {
   535→    const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   536→    if (keys.length !== 1) return null; // تیپ ترکیبی چندبازاره — خط مستقیم پیناکل ندارد
   537→    const p = pinPrices[keys[0]];
   538→    return typeof p === "number" && p > 1.01 ? p : null;
   539→  };
   540→  const pinA = pinPriceOf(item.legsA ?? []);
   541→  const pinB = pinPriceOf(item.legsB ?? []);
   542→  const priceAdvantage = (odds: number, pin: number): number => Math.round((odds / pin - 1) * 1000) / 10;
   543→
   544→  const edgeA = pinA !== null ? priceAdvantage(item.oddsA, pinA) : computeEdge(probA, item.oddsA);
   545→  const edgeB = pinB !== null ? priceAdvantage(item.oddsB, pinB) : computeEdge(probB, item.oddsB);
   546→  const edgeSourceA = edgeA !== null ? (pinA !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   547→  const edgeSourceB = edgeB !== null ? (pinB !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   548→  const tipKindA: TipKind | null = edgeA !== null ? deriveTipKind((item.legsA ?? []).map((l) => l.key), item.oddsA) : null;
   549→  const tipKindB: TipKind | null = edgeB !== null ? deriveTipKind((item.legsB ?? []).map((l) => l.key), item.oddsB) : null;
   550→  const koMs = new Date(item.match.kickoff).getTime();
   551→  const winA = tipKindA ? computeTimingWindowStatic(koMs, tipKindA) : null;
   552→  const winB = tipKindB ? computeTimingWindowStatic(koMs, tipKindB) : null;
   553→
   554→  const row = await db.issuedSlip.create({
   555→    data: {
   556→      hash,
   557→      externalId,
   558→      matchId: item.match.id,
   559→      league: item.match.league,
   560→      homeTeam: item.match.homeTeam,
   561→      awayTeam: item.match.awayTeam,
   562→      kickoff: new Date(item.match.kickoff),
   563→      labelA: item.labelA,
   564→      labelB: item.labelB,
   565→      legsA: JSON.stringify(item.legsA ?? []),
   566→      legsB: JSON.stringify(item.legsB ?? []),
   567→      oddsA: item.oddsA,
   568→      oddsB: item.oddsB,
   569→      stakeA: stakes.stakeA,
   570→      stakeB: stakes.stakeB,
   571→      cellsA: JSON.stringify(coveredCells(lH, lA, item.legsA ?? [], externalId)),
   572→      cellsB: JSON.stringify(coveredCells(lH, lA, item.legsB ?? [], externalId)),
   573→      guaranteedProfit: item.guaranteedProfit,
   574→      marginPct: item.marginPct,
   575→      archetype: item.archetype ?? null,
   576→      archetypeCode: item.archetypeCode ?? null,
   577→      lambdaHome: item.lambdaHome ?? null,
   578→      lambdaAway: item.lambdaAway ?? null,
   579→      espnUrl: item.match.espnUrl ?? espnUrlOf(externalId),
   580→      supervisorApproved: item.supervisorApproved !== false,
   581→      autoIssued: opts?.autoIssued === true,
   582→      supervisorScore: item.supervisorScore ?? null,
   583→      vetoReason: item.vetoReason ?? null,
   584→      jointLossRiskPct: item.jointLossRiskPct ?? null,
   585→      doubleWinProbPct: item.doubleWinProbPct ?? null,
   586→      modelProbA: probA,
   587→      modelProbB: probB,
   588→      pinnacleOddsA: pinA,
   589→      pinnacleOddsB: pinB,
   590→      edgeA,
   591→      edgeB,
   592→      edgeSourceA,
   593→      edgeSourceB,
   594→      tipKindA,
   595→      tipKindB,
   596→      timingWindowA: winA?.windowKey ?? null,
   597→      timingWindowB: winB?.windowKey ?? null,
   598→      timingStartA: winA?.startAt ?? null,
   599→      timingEndA: winA?.endAt ?? null,
   600→      timingStartB: winB?.startAt ?? null,
   601→      timingEndB: winB?.endAt ?? null,
   602→      status: "PENDING",
   603→    },
   604→  });
   605→  return { created: true, slip: mapSlip(row) };
   606→}

   608→// ═══════════ v10.0.2 — بازپررسی یادگیری کارت‌های قدیمی (Idempotent Backfill) ═══════════
   609→
   610→/** throttle بازپررسی — حداکثر هر ۳۰ دقیقه یک‌بار (فراخوانی از خط لوله سرور) */
   611→let lastBackfillAt = 0;
   612→const BACKFILL_TTL_MS = 30 * 60_000;
   613→
   614→/**
   615→ * backfillSlipLearning — پرکردن ستون‌های یادگیری کارت‌های صادرشده پیش از v10:
   616→ *   · modelProbA/B و edgeA/B (ارزش عددی نسبت به پیناکل واقعی یا خط منصفانه)
   617→ *   · tipKind + پنجره زمانی طلایی ثبت
   618→ *   · ردیف ژورنال تسویه (recordSettlementLearning — Idempotent با slipId)
   619→ * همه محاسبات از λ ذخیره‌شده کارت بازتولید می‌شود — هیچ عدد ساختگی نیست.
   620→ */
   621→export async function backfillSlipLearning(force = false): Promise<{ scanned: number; updated: number; journalled: number }> {
   622→  if (!force && Date.now() - lastBackfillAt < BACKFILL_TTL_MS) {
   623→    return { scanned: 0, updated: 0, journalled: 0 };
   624→  }
   625→  lastBackfillAt = Date.now();
   626→
   627→  // کارت‌های تسویه‌شده بدون ارزش عددی (نسخه‌های ≤ v9) — سقف هر پاس ۶۰ ردیف
   628→  const targets = await db.issuedSlip.findMany({
   629→    where: {
   630→      status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP", "MANUAL_REVIEW"] },
   631→      OR: [
   632→        { edgeA: null },
   633→        { edgeB: null },
   634→        { timingWindowA: null },
   635→        { timingWindowB: null },
   636→        { jointLossRiskPct: null },
   637→        { doubleWinProbPct: null },
   638→      ],
   639→    },
   640→    orderBy: { kickoff: "desc" },
   641→    take: 60,
   642→  });
   643→  if (targets.length === 0) return { scanned: 0, updated: 0, journalled: 0 };
   644→
   645→  let updated = 0;
   646→  let journalled = 0;
   647→  for (const s of targets) {
   648→    try {
   649→      const legsA: SettlementLegDTO[] = safeParseLegs(s.legsA);
   650→      const legsB: SettlementLegDTO[] = safeParseLegs(s.legsB);
   651→      const lH = s.lambdaHome ?? 1.35;
   652→      const lA = s.lambdaAway ?? 1.25;
   653→      const probA = s.modelProbA ?? legProbability(lH, lA, legsA);
   654→      const probB = s.modelProbB ?? legProbability(lH, lA, legsB);
   655→
   656→      // خط شارپ واقعی پیناکل — از رکورد OddsRecord مرتبط با بازی (وقتی فید موجود باشد)
   657→      let pinPrices: Record<string, number> = {};
   658→      if (s.matchId) {
   659→        const oddsRows = await db.oddsRecord.findMany({ where: { matchId: s.matchId } });
   660→        const pin = oddsRows.find((o) => o.bookmaker === "PINNACLE");
   661→        if (pin) {
   662→          pinPrices = {
   663→            HOME: pin.oddsHome,
   664→            DRAW: pin.oddsDraw,
   665→            AWAY: pin.oddsAway,
   666→            DC_1X: 1 / (1 / pin.oddsHome + 1 / pin.oddsDraw),
   667→            DC_X2: 1 / (1 / pin.oddsAway + 1 / pin.oddsDraw),
   668→            O25: pin.oddsOver25,
   669→            U25: pin.oddsUnder25,
   670→          };
   671→        }
   672→      }
   673→      const pinPriceOf = (legs: SettlementLegDTO[]): number | null => {
   674→        const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   675→        if (keys.length !== 1) return null;
   676→        const p = pinPrices[keys[0]];
   677→        return typeof p === "number" && p > 1.01 ? p : null;
   678→      };
   679→      const priceAdvantage = (odds: number, pin: number): number => Math.round((odds / pin - 1) * 1000) / 10;
   680→
   681→      const pinA = s.pinnacleOddsA ?? pinPriceOf(legsA);
   682→      const pinB = s.pinnacleOddsB ?? pinPriceOf(legsB);
   683→      const edgeA = s.edgeA ?? (pinA !== null ? priceAdvantage(s.oddsA, pinA) : computeEdge(probA, s.oddsA));
   684→      const edgeB = s.edgeB ?? (pinB !== null ? priceAdvantage(s.oddsB, pinB) : computeEdge(probB, s.oddsB));
   685→      const edgeSourceA = edgeA !== null ? (pinA !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   686→      const edgeSourceB = edgeB !== null ? (pinB !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   687→
   688→      const tipKindA = s.tipKindA ?? (edgeA !== null ? deriveTipKind(legsA.map((l) => l.key), s.oddsA) : null);
   689→      const tipKindB = s.tipKindB ?? (edgeB !== null ? deriveTipKind(legsB.map((l) => l.key), s.oddsB) : null);
   690→      const koMs = s.kickoff.getTime();
   691→      const winA = s.timingWindowA ? null : tipKindA ? computeTimingWindowStatic(koMs, tipKindA) : null;
   692→      const winB = s.timingWindowB ? null : tipKindB ? computeTimingWindowStatic(koMs, tipKindB) : null;
   693→
   694→      // ─── v11.0 — سنجه‌های دفتر سایه: بازتولید ٪ باخت همزمان و ٪ برد طلایی از λ ذخیره‌شده ───
   695→      let jointLossRiskPct: number | undefined;
   696→      let doubleWinProbPct: number | undefined;
   697→      if (s.jointLossRiskPct === null || s.doubleWinProbPct === null) {
   698→        const matchRow = s.matchId ? await db.match.findUnique({ where: { id: s.matchId }, include: { referee: true } }) : null;
   699→        if (matchRow) {
   700→          const eng = computeMatchEngine(matchRow);
   701→          const m = jointShadowMetrics(
   702→            lH,
   703→            lA,
   704→            eng.lambdaCorners,
   705→            eng.lambdaCards,
   706→            legsA.map((l) => l.key),
   707→            legsB.map((l) => l.key)
   708→          );
   709→          jointLossRiskPct = s.jointLossRiskPct ?? m.jointLossRiskPct;
   710→          doubleWinProbPct = s.doubleWinProbPct ?? m.doubleWinProbPct;
   711→        }
   712→      }
   713→
   714→      const refreshed = await db.issuedSlip.update({
   715→        where: { id: s.id },
   716→        data: {
   717→          modelProbA: s.modelProbA ?? probA,
   718→          modelProbB: s.modelProbB ?? probB,
   719→          jointLossRiskPct,
   720→          doubleWinProbPct,
   721→          pinnacleOddsA: pinA,
   722→          pinnacleOddsB: pinB,
   723→          edgeA,
   724→          edgeB,
   725→          edgeSourceA,
   726→          edgeSourceB,
   727→          tipKindA: tipKindA ?? undefined,
   728→          tipKindB: tipKindB ?? undefined,
   729→          timingWindowA: s.timingWindowA ?? winA?.windowKey ?? undefined,
   730→          timingWindowB: s.timingWindowB ?? winB?.windowKey ?? undefined,
   731→          timingStartA: s.timingStartA ?? winA?.startAt ?? undefined,
   732→          timingEndA: s.timingEndA ?? winA?.endAt ?? undefined,
   733→          timingStartB: s.timingStartB ?? winB?.startAt ?? undefined,
   734→          timingEndB: s.timingEndB ?? winB?.endAt ?? undefined,
   735→        },
   736→      });
   737→      updated += 1;
   738→
   739→      // ژورنال یادگیری — Idempotent (هر کارت حداکثر یک ردیف)
   740→      const hasJournal = await db.settlementJournal.findFirst({ where: { slipId: s.id }, select: { id: true } });
   741→      if (!hasJournal) {
   742→        await recordSettlementLearning(refreshed);
   743→        journalled += 1;
   744→      }
   745→    } catch (err) {
   746→      console.warn(`backfill failed for slip ${s.id}:`, err instanceof Error ? err.message : err);
   747→    }
   748→  }
   749→  return { scanned: targets.length, updated, journalled };
   750→}

   824→  const slips = rows.map((s) => mapSlip(s));
   825→  // ─── v11.0 — آمار تیتر کارنامه فقط از کارت‌های تایید سوپروایزر؛ وتوها به «دفتر سایه» می‌روند ───
   826→  const approvedSlips = slips.filter((s) => s.supervisorApproved);
   827→  const stats: SettlementStatsDTO = {
   828→    total: approvedSlips.length,
   829→    pending: 0,
   830→    live: 0,
   831→    wonA: 0,
   832→    wonB: 0,
   833→    doubleWin: 0,
   834→    middleTrap: 0,
   835→    manual: 0,
   836→    void: 0,
   837→    settled: 0,
   838→    netProfit: 0,
   839→    netProfitPct: 0,
   840→    staked: 0,
   841→  };
   842→  for (const s of approvedSlips) {
   843→    switch (s.status) {
   844→      case "PENDING": stats.pending += 1; break;
   845→      case "LIVE": stats.live += 1; break;
   846→      case "WON_A": stats.wonA += 1; break;
   847→      case "WON_B": stats.wonB += 1; break;
   848→      case "DOUBLE_WIN": stats.doubleWin += 1; break;
   849→      case "MIDDLE_TRAP": stats.middleTrap += 1; break;
   850→      case "MANUAL_REVIEW": stats.manual += 1; break;
   851→      case "VOID": stats.void += 1; break;
   852→    }
   853→    if (["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"].includes(s.status)) {
   854→      stats.settled += 1;
   855→      stats.netProfit += s.netProfit ?? 0;
   856→      stats.staked += s.stakeA + s.stakeB;
   857→    }
   858→  }
   859→  stats.netProfit = Math.round(stats.netProfit * 100) / 100;
   860→  stats.netProfitPct = stats.staked > 0 ? Math.round((stats.netProfit / stats.staked) * 1000) / 10 : 0;
   861→
   862→  // ─── v11.0 — دفتر سایه: تسویه کاغذی کارت‌های وتوشده (paper P&L — هرگز واقعی نبوده) ───
   863→  const shadowSlips = slips.filter((s) => !s.supervisorApproved);
   864→  const shadowByStatus: Record<string, number> = {};
   865→  let shadowWins = 0;
   866→  let shadowLosses = 0;
   867→  let shadowNet = 0;
   868→  for (const s of shadowSlips) {
   869→    shadowByStatus[s.status] = (shadowByStatus[s.status] ?? 0) + 1;
   870→    if (["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"].includes(s.status)) {
   871→      if (s.status === "MIDDLE_TRAP") shadowLosses += 1;
   872→      else shadowWins += 1;
   873→      shadowNet += s.netProfit ?? 0;
   874→    }
   875→  }
   876→  const shadowSettled = shadowWins + shadowLosses;
   877→  const shadow = {
   878→    total: shadowSlips.length,
   879→    settled: shadowSettled,
   880→    wins: shadowWins,
   881→    losses: shadowLosses,
   882→    winRate: shadowSettled > 0 ? Math.round((shadowWins / shadowSettled) * 1000) / 10 : null,
   883→    netProfit: Math.round(shadowNet * 100) / 100,
   884→    byStatus: shadowByStatus,
   885→  };
   886→
   887→  // ─── v10.0 — ژورنال + دفتر دکترین + وضعیت اتوماسیون سرور ───
   888→  const [journal, doctrine, sched] = await Promise.all([
   889→    listJournal(30).catch(() => []),
   890→    listDoctrine().catch(() => [] as DoctrineSummary[]),
   891→    Promise.resolve(getSchedulerStatus()),
   892→  ]);
   893→
   894→  return {
   895→    success: true,
   896→    slips,
   897→    stats,
   898→    shadow,
   899→    outbox: outboxRows.map((o) => ({
   900→      id: o.id,
   901→      kind: o.kind,
   902→      ok: o.ok,
   903→      detail: o.detail,
   904→      slipHash: o.slip?.hash ?? null,
   905→      createdAt: o.createdAt.toISOString(),
   906→    })),
   907→    telegram: { configured: tgStatus.configured, channelSet: tgStatus.channelSet },
   908→    journal,
   909→    doctrine,
   910→    automation: {
   911→      schedulerActive: sched.active,
   912→      lastTickAt: sched.lastTickAt ? new Date(sched.lastTickAt).toISOString() : null,
   913→      lastDailyReportAt: sched.lastDailyReportAt ? new Date(sched.lastDailyReportAt).toISOString() : null,
   914→      intervalSec: sched.intervalSec,
   915→    },
   916→    unitBasis: { stakePerTicket: 100, unitPctOfBankroll: 1, exampleBankrollUsd: 10 },
   917→  };
   918→}

   920→// ═════════════════════ تسویه خودکار پس‌زمینه (throttle ۶۰ ثانیه) ═════════════════════
   921→
   922→let autoBusy = false;
   923→let autoLastAt = 0;
   924→const AUTO_THROTTLE_MS = 60_000;
   925→
   926→/**
   927→ * فراخوانی سبک از دل فید زنده — به محض FT در ESPN تسویه انجام می‌شود
   928→ * بدون انتظار برای کلیک کاربر. با throttle و قفل برای صفر هزینه اضافه.
   929→ */
   930→export async function autoSettleInBackground(): Promise<void> {
   931→  const pending = await db.issuedSlip.count({ where: { status: { in: ["PENDING", "LIVE"] } } });
   932→  if (pending === 0) return;
   933→  if (autoBusy || Date.now() - autoLastAt < AUTO_THROTTLE_MS) return;
   934→  autoBusy = true;
   935→  autoLastAt = Date.now();
   936→  try {
   937→    await settleFinishedSlips({ notify: true });
   938→  } catch (err) {
   939→    console.warn("auto-settle failed:", err);
   940→  } finally {
   941→    autoBusy = false;
   942→  }
   943→}
