    64→export function cardCopyText(v: FlashcardView): string {
    65→  const p = v.pair;
    66→  const sup = v.supervisor;
    67→  const head = `⚽ فوتبال‌کوانت v16.0 Supervisor Quant — ${v.match.homeTeam} مقابل ${v.match.awayTeam} (${v.match.league})`;
    68→  const crownLine = sup
    69→    ? `👑 تک‌تیپ تاییدشده سوپروایزر — امتیاز مونت‌کارلو: ${sup.score100}/۱۰۰${p.archetype ? ` (${p.archetype.short})` : ""}`
    70→    : p.archetype
    71→      ? `📐 الگو: ${p.archetype.short}`
    72→      : "";
    73→  const acc = p.accelerator?.active ? p.accelerator : null;
    74→  const profitLine = acc
    75→    ? `⚡ شتاب‌دهنده نامتقارن فعال — قفل اصل سرمایه ۱۰۰ واحدی روی تیپ B · برد A: ${acc.profitOnA >= 0 ? "+" : ""}${faNum(acc.profitOnA, 1)} واحد · برد همزمان (جک‌پات): +${faNum(acc.profitOnDoubleWin, 1)} واحد`
    76→    : p.guaranteedMarginPct > 0
    77→      ? `💰 سود تضمینی: +${p.guaranteedMarginPct.toFixed(1)}٪ (+${faNum(p.equalProfit.profit, 1)} واحد سود خالص)`
    78→      : `🛡️ پوشش بیمه سرمایه — بازگشت وجه در سناریوی پایه`;
    79→  return [
    80→    head,
    81→    crownLine,
    82→    `🔹 تیپ A (ضریب ${p.totalOddsA.toFixed(2)}): ${p.labelA} ⬅️ مبلغ شرط: ${faNum(acc ? acc.stakeA : p.equalProfit.stakeA, 1)} واحد`,
    83→    `🟢 تیپ B مکمل (ضریب ${p.totalOddsB.toFixed(2)}): ${p.labelB} ⬅️ مبلغ شرط: ${faNum(acc ? acc.stakeB : p.equalProfit.stakeB, 1)} واحد`,
    84→    profitLine,
    85→    `✅ پوشش سناریوها: ${(p.archetype ? p.archetype.coveragePct * 100 : (1 - p.jointStates.bothLose) * 100).toFixed(0)}٪ | ریسک باخت همزمان: ${(p.jointStates.bothLose * 100).toFixed(1)}%`,
    86→    `⚖️ ${p.verdict}`,
    87→  ]
    88→    .filter(Boolean)
    89→    .join("\n");
    90→}

    96→const TIMING_STATUS_STYLE: Record<TimingStatus, string> = {
    97→  OPEN: "border-emerald-500/40 bg-emerald-500/[0.08] text-emerald-300",
    98→  WAITING: "border-amber-500/40 bg-amber-500/[0.07] text-amber-300",
    99→  LATE: "border-rose-500/40 bg-rose-500/[0.07] text-rose-300",

   103→/** بنر پنجره زمانی طلایی ثبت — با شمارش معکوس زنده (به‌روزرسانی هر ۳۰ ثانیه) */
   104→function TimingBanner({ kickoff, legsA, oddsA, legsB, oddsB }: { kickoff: string; legsA: string[]; oddsA: number; legsB: string[]; oddsB: number }) {
   105→  const [now, setNow] = useState(() => Date.now());
   106→  useEffect(() => {
   107→    const t = setInterval(() => setNow(Date.now()), 30_000);
   108→    return () => clearInterval(t);
   109→  }, []);
   110→  const pair = useMemo(() => computePairTiming(kickoff, legsA, oddsA, legsB, oddsB, now), [kickoff, legsA, oddsA, legsB, oddsB, now]);
   111→
   112→  const rows = pair.same ? [pair.a] : [pair.a, pair.b];
   113→  const sideLabel = (i: number) => (pair.same ? "تیپ A + تیپ B" : i === 0 ? "تیپ A" : "تیپ B");
   114→
   115→  return (
   116→    <div className="space-y-1.5 rounded-xl border border-amber-500/30 bg-gradient-to-l from-amber-500/[0.08] to-transparent p-2.5">
   117→      <p className="flex items-center gap-1 text-[10px] font-bold text-amber-300">
   118→        ⏱️ بهترین زمان ثبت این تیپ (پنجره طلایی ضرایب — v10.0)
   119→      </p>
   120→      {rows.map((w, i) => {
   121→        const style = TIMING_STATUS_STYLE[w.status];
   122→        const clock = new Intl.DateTimeFormat("fa-IR", { hour: "2-digit", minute: "2-digit" });
   123→        return (
   124→          <div key={`${w.windowKey}-${i}`} className={`rounded-lg border px-2 py-1.5 ${style}`}>
   125→            <p className="text-[10.5px] font-bold leading-4">
   126→              {w.meta.icon} {sideLabel(i)}: {w.meta.nameFa}
   127→              <span className="mx-1 font-normal text-foreground/70">|</span>
   128→              ساعت دقیق پیشنهادی ثبت:
   129→              <b className="mx-1 font-num" dir="ltr">{w.suggestedClock}</b>
   130→            </p>
   131→            <p className="mt-0.5 text-[9.5px] leading-3.5 opacity-90">
   132→              {w.statusText}
   133→              {w.status === "OPEN" && w.msUntilClose > 0 && ` — ${faDuration(w.msUntilClose)} تا بستن پنجره`}
   134→              {w.status === "WAITING" && w.msUntilOpen > 0 && ` (${clock.format(new Date(w.startAt))} تا ${clock.format(new Date(w.endAt))})`}
   135→            </p>
   136→          </div>
   137→        );
   138→      })}
   139→      <p className="text-[9px] leading-3 text-muted-foreground">
   140→        منطق علمی: {rows[0].meta.rationale}
   141→      </p>
   142→    </div>
   143→  );
   144→}
