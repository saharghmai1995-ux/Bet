    51→/** عرض تقریبی کاراکتر به واحد em در DejaVu Sans */
    52→function charW(ch: string): number {
    53→  if (" iIljt.,:;!'|".includes(ch)) return 0.32;
    54→  if ("ftr-()[]".includes(ch)) return 0.42;
    55→  if ("mwMW@".includes(ch)) return 0.92;
    56→  if (ch >= "A" && ch <= "Z") return 0.72;
    57→  if (ch >= "0" && ch <= "9") return 0.64;
    58→  if (ch === " ") return 0.34;
    59→  if (ch >= "a" && ch <= "z") return 0.56;
    60→  return 0.62;
    61→}
    62→
    63→export function textWidth(text: string, fontSize: number, weight = 400): number {
    64→  const w = String(text ?? "").split("").reduce((acc, ch) => acc + charW(ch), 0) * fontSize;
    65→  return w * (weight >= 600 ? 1.06 : 1);
    66→}

    68→/** شکستن متن به خطوط با حداکثر عرض پیکسلی */
    69→export function wrapText(text: string, maxWidth: number, fontSize: number, weight = 400, maxLines = 3): string[] {
    70→  const words = String(text ?? "").split(/\s+/).filter(Boolean);
    71→  const lines: string[] = [];
    72→  let cur = "";
    73→  for (const word of words) {
    74→    const attempt = cur ? `${cur} ${word}` : word;
    75→    if (textWidth(attempt, fontSize, weight) <= maxWidth) {
    76→      cur = attempt;
    77→    } else {
    78→      if (cur) lines.push(cur);
    79→      cur = word;
    80→      if (lines.length === maxLines) break;
    81→    }
    82→  }
    83→  if (cur && lines.length < maxLines) lines.push(cur);
    84→  if (lines.length === maxLines && words.join(" ") !== lines.join(" ")) {
    85→    let last = lines[maxLines - 1];
    86→    while (textWidth(`${last}…`, fontSize, weight) > maxWidth && last.length > 4) last = last.slice(0, -1);
    87→    lines[maxLines - 1] = `${last}…`;
    88→  }
    89→  return lines.length > 0 ? lines : ["—"];
    90→}

    92→/** کوچک‌سازی خودکار فونت تا متن در عرض بگنجد */
    93→export function fitSize(text: string, maxWidth: number, baseSize: number, minSize: number, weight = 700): number {
    94→  let size = baseSize;
    95→  while (size > minSize && textWidth(text, size, weight) > maxWidth) size -= 2;
    96→  return size;
    97→}

    99→function esc(s: string): string {
   100→  return String(s ?? "")
   101→    .replace(/&/g, "&amp;")
   102→    .replace(/</g, "&lt;")
   103→    .replace(/>/g, "&gt;")
   104→    .replace(/"/g, "&quot;");
   105→}

   108→function tspans(lines: string[], x: number, y: number, lh: number): string {
   109→  return lines.map((l, i) => `<tspan x="${x}" y="${y + i * lh}">${esc(l)}</tspan>`).join("");
   110→}

   156→/** LCG قطعی — هر پوستر همیشه با همان ذرات رندر می‌شود (Deterministic) */
   157→function seeded(seed: number): () => number {
   158→  let s = (seed >>> 0) || 1;
   159→  return () => {
   160→    s = (Math.imul(s, 1664525) + 1013904223) >>> 0;
   161→    return s / 0xffffffff;
   162→  };
   163→}

   165→/**
   166→ * ═══ v13.0 — موتور خودساز گرافیک استادیوم و موشن (Zero-Upload Procedural Stadium Engine) ═══
   167→ * بدون هیچ فایل خارجی — نورافکن‌های radialGradient، چمن با پرسپکتیو یک‌سوم پایینی،
   168→ * محوطه جریمه و دایره وسط، ذرات بوکه، آتش‌بازی طلایی/زمردی، باران (LOSS)، تور دروازه (WIN)،
   169→ * نمای هوایی چمن (DAILY_REPORT) و تم بنفش/طلایی (SNIPER).
   170→ */
   171→function proceduralStadium(w: number, h: number, kind: StadiumKind, accent: string): string {
   172→  const rnd = seeded(kind.length * 7919 + 41);
   173→  const theme =
   174→    kind === "DOUBLE_WIN" ? "gold" : kind === "LOSS" ? "rain" : kind === "SNIPER" ? "sniper" : kind === "DAILY_REPORT" ? "aerial" : kind === "WIN" ? "win" : kind === "TEASER" ? "teaser" : "night";
   175→
   176→  const skyTop = theme === "gold" ? "#0d0a04" : theme === "rain" ? "#0a0d14" : theme === "sniper" ? "#0d0716" : theme === "aerial" ? "#0c1a10" : "#070d18";
   177→  const skyBottom = theme === "gold" ? "#1a1206" : theme === "rain" ? "#101623" : theme === "sniper" ? "#170b28" : theme === "aerial" ? "#0f2b18" : "#0b1526";
   178→  const pitchTop = theme === "aerial" ? "#1c7a3d" : "#157a38";
   179→  const pitchBottom = theme === "aerial" ? "#0d5426" : "#0a4a20";
   180→
   181→  // ─── چمن با پرسپکتیو (یک‌سوم پایینی؛ در حالت aerial تمام بوم) ───
   182→  const pitchY0 = theme === "aerial" ? 0 : Math.round(h * 0.52);
   183→  const stripes: string[] = [];
   184→  const N = 9;
   185→  for (let i = 0; i < N; i++) {
   186→    const y0 = pitchY0 + ((h - pitchY0) / N) * i;
   187→    const y1 = pitchY0 + ((h - pitchY0) / N) * (i + 1);
   188→    stripes.push(`<rect x="0" y="${y0.toFixed(1)}" width="${w}" height="${(y1 - y0).toFixed(1)}" fill="${i % 2 === 0 ? "rgba(255,255,255,0.045)" : "rgba(0,0,0,0.05)"}"/>`);
   189→  }
   190→
   191→  // ─── خطوط زمین با پرسپکتیو (محوطه جریمه + دایره وسط) ───
   192→  const cx = w / 2;
   193→  const markings = theme === "aerial"
   194→    ? `
   195→  <g stroke="rgba(255,255,255,0.30)" stroke-width="4" fill="none">
   196→    <rect x="${cx - 330}" y="${h * 0.20}" width="660" height="${h * 0.60}" rx="6"/>
   197→    <rect x="${cx - 130}" y="${h * 0.20}" width="260" height="${h * 0.22}"/>
   198→    <circle cx="${cx}" cy="${h * 0.5}" r="120"/>
   199→    <line x1="0" y1="${h * 0.5}" x2="${w}" y2="${h * 0.5}"/>
   200→  </g>`
   201→    : `
   202→  <g stroke="rgba(255,255,255,0.26)" stroke-width="3.5" fill="none">
   203→    <path d="M${cx - 330} ${h - 60} L${cx - 235} ${pitchY0 + 26} L${cx + 235} ${pitchY0 + 26} L${cx + 330} ${h - 60}"/>
   204→    <path d="M${cx - 150} ${h - 60} L${cx - 112} ${pitchY0 + 26} M${cx + 150} ${h - 60} L${cx + 112} ${pitchY0 + 26}"/>
   205→    <ellipse cx="${cx}" cy="${h * 0.80}" rx="190" ry="62"/>
   206→    <path d="M40 ${pitchY0 + 26} L${w - 40} ${pitchY0 + 26}"/>
   207→  </g>`;
   208→
   209→  // ─── نورافکن‌ها (radialGradient چهارگوشه بالایی) ───
   210→  const beam = (x: number, y: number, r: number, op: number) =>
   211→    `<radialGradient id="fl-${x}-${y}" cx="50%" cy="50%" r="50%">
   212→       <stop offset="0%" stop-color="${accent}" stop-opacity="${op}"/>
   213→       <stop offset="45%" stop-color="${accent}" stop-opacity="${op * 0.28}"/>
   214→       <stop offset="100%" stop-color="${accent}" stop-opacity="0"/>
   215→     </radialGradient>
   216→     <circle cx="${x}" cy="${y}" r="${r}" fill="url(#fl-${x}-${y})"/>`;
   217→  const floodlights = `
   218→  <defs>${beam(120, 110, 430, 0.34)}${beam(w - 120, 110, 430, 0.34)}${beam(cx, 40, 520, 0.16)}</defs>
   219→  ${beam(120, 110, 430, 0.34)}${beam(w - 120, 110, 430, 0.34)}${beam(cx, 40, 520, 0.16)}
   220→  <g stroke="rgba(255,255,255,0.30)" stroke-width="3">${[-1, 1].map((s) => `<line x1="${s < 0 ? 120 : w - 120}" y1="96" x2="${s < 0 ? 150 : w - 150}" y2="${pitchY0 + 10}"/>`).join("")}</g>`;
   221→
   222→  // ─── سکوهای تماشاگر + نقاط جمعیت ───
   223→  const crowd = Array.from({ length: 240 }, () => {
   224→    const x = rnd() * w;
   225→    const y = pitchY0 * (0.30 + rnd() * 0.55);
   226→    const o = 0.05 + rnd() * 0.14;
   227→    return `<circle cx="${x.toFixed(0)}" cy="${y.toFixed(0)}" r="${(0.8 + rnd() * 1.6).toFixed(1)}" fill="rgba(255,255,255,${o.toFixed(2)})"/>`;
   228→  }).join("");
   229→
   230→  // ─── ذرات بوکه ───
   231→  const bokehColors = [accent, "#f5c542", "#3ddc97", "#38d6f5"];
   232→  const bokeh = Array.from({ length: 30 }, () => {
   233→    const x = rnd() * w;
   234→    const y = rnd() * h * 0.55;
   235→    const r = 2 + rnd() * 7;
   236→    const c = bokehColors[Math.floor(rnd() * bokehColors.length)];
   237→    return `<circle cx="${x.toFixed(0)}" cy="${y.toFixed(0)}" r="${r.toFixed(1)}" fill="${c}" opacity="${(0.05 + rnd() * 0.16).toFixed(2)}"/>`;
   238→  }).join("");
   239→
   240→  // ─── آتش‌بازی (DOUBLE_WIN / WIN) ───
   241→  const fireworks =
   242→    theme === "gold" || theme === "win"
   243→      ? Array.from({ length: 2 }, (_, k) => {
   244→          const fx = k === 0 ? w * 0.2 : w * 0.8;
   245→          const fy = 150 + k * 60;
   246→          const rays = Array.from({ length: 14 }, (_, i) => {
   247→            const a = (i / 14) * Math.PI * 2 + k;
   248→            const len = 55 + rnd() * 65;
   249→            const x2 = fx + Math.cos(a) * len;
   250→            const y2 = fy + Math.sin(a) * len * 0.9;
   251→            return `<line x1="${fx}" y1="${fy}" x2="${x2.toFixed(0)}" y2="${y2.toFixed(0)}" stroke="${theme === "gold" ? "#f5c542" : "#3ddc97"}" stroke-width="2.6" stroke-linecap="round" opacity="${(0.35 + rnd() * 0.4).toFixed(2)}"/>`;
   252→          }).join("");
   253→          return `<g>${rays}<circle cx="${fx}" cy="${fy}" r="6" fill="${theme === "gold" ? "#f5c542" : "#3ddc97"}" opacity="0.85"/></g>`;
   254→        }).join("")
   255→      : "";
   256→
   257→  // ─── باران (LOSS) ───
   258→  const rain =
   259→    theme === "rain"
   260→      ? Array.from({ length: 90 }, () => {
   261→          const x = rnd() * w;
   262→          const y = rnd() * h;
   263→          const len = 26 + rnd() * 44;
   264→          return `<line x1="${x.toFixed(0)}" y1="${y.toFixed(0)}" x2="${(x - len * 0.22).toFixed(0)}" y2="${(y + len).toFixed(0)}" stroke="rgba(180,205,255,0.20)" stroke-width="1.6" stroke-linecap="round"/>`;
   265→        }).join("")
   266→      : "";
   267→
   268→  // ─── تور دروازه (WIN) ───
   269→  const net =
   270→    theme === "win"
   271→      ? `<g stroke="rgba(255,255,255,0.13)" stroke-width="1.7" fill="none">
   272→      ${Array.from({ length: 13 }, (_, i) => `<line x1="${(90 + i * 75).toFixed(0)}" y1="${(pitchY0 + 8).toFixed(0)}" x2="${(60 + i * 80).toFixed(0)}" y2="${h - 56}"/>`).join("")}
   273→      ${Array.from({ length: 9 }, (_, i) => { const y = pitchY0 + 20 + ((h - 76 - pitchY0) / 9) * i; return `<path d="M85 ${y.toFixed(0)} Q${cx} ${(y - 26).toFixed(0)} ${w - 85} ${y.toFixed(0)}"/>`; }).join("")}
   274→    </g>`
   275→      : "";
   276→
   277→  return `
   278→  <defs>
   279→    <linearGradient id="bg" x1="0" y1="0" x2="0" y2="1">
   280→      <stop offset="0" stop-color="${skyTop}"/>
   281→      <stop offset="${theme === "aerial" ? 30 : 52}" stop-color="${skyBottom}"/>
   282→      <stop offset="1" stop-color="${theme === "aerial" ? skyBottom : pitchBottom}"/>
   283→    </linearGradient>
   284→    <linearGradient id="pitch" x1="0" y1="0" x2="0" y2="1">
   285→      <stop offset="0" stop-color="${pitchTop}"/>
   286→      <stop offset="1" stop-color="${pitchBottom}"/>
   287→    </linearGradient>
   288→  </defs>
   289→  <rect width="${w}" height="${h}" fill="url(#bg)"/>
   290→  ${floodlights}
   291→  ${crowd}
   292→  ${theme === "aerial" ? "" : `<rect x="0" y="${pitchY0}" width="${w}" height="${h - pitchY0}" fill="url(#pitch)" opacity="0.94"/>`}
   293→  ${stripes.join("")}
   294→  ${markings}
   295→  ${net}
   296→  ${bokeh}
   297→  ${fireworks}
   298→  ${rain}
   299→  <rect width="${w}" height="${h}" fill="rgba(2,6,14,0.30)"/>`;
   300→}

   373→function fmtKickoff(iso: string): string {
   374→  try {
   375→    const d = new Date(iso);
   376→    const day = new Intl.DateTimeFormat("en-GB", { weekday: "short", day: "2-digit", month: "short", timeZone: "Asia/Tehran" }).format(d);
   377→    const time = new Intl.DateTimeFormat("en-GB", { hour: "2-digit", minute: "2-digit", hour12: false, timeZone: "Asia/Tehran" }).format(d);
   378→    return `${day} · ${time} Tehran`;
   379→  } catch {
   380→    return String(iso);
   381→  }
   382→}

   386→function crestInitials(name: string): string {
