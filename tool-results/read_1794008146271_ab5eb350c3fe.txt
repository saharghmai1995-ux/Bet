    39→function statusBadge(status: string): { label: string; className: string } {
    40→  switch (status) {
    41→    case "FIT":
    42→      return { label: "آماده", className: "border-emerald-500/40 text-emerald-400" };
    43→    case "INJURED":
    44→      return { label: "مصدوم", className: "border-rose-500/40 text-rose-400" };
    45→    case "SUSPENDED":
    46→      return { label: "محروم", className: "border-rose-500/40 text-rose-400" };
    47→    case "DOUBTFUL":
    48→      return { label: "مشکوک", className: "border-amber-500/40 text-amber-300" };
    49→    default:
    50→      return { label: status, className: "" };
    51→  }
    52→}

    56→/** نودهای زمین از بازیکنان (T-60) — غایبان با نود قرمز چشمک‌زن */
    57→function pitchNodes(topPlayers: LineupShockDTO["topPlayers"], side: "HOME" | "AWAY"): PitchNode[] {
    58→  const HOME_FORMATION: PitchNode[] = [
    59→    { x: 7, y: 50 }, { x: 22, y: 18 }, { x: 20, y: 39 }, { x: 20, y: 61 }, { x: 22, y: 82 },
    60→    { x: 40, y: 30 }, { x: 37, y: 50 }, { x: 40, y: 70 },
    61→    { x: 62, y: 22 }, { x: 66, y: 50 }, { x: 62, y: 78 },
    62→  ];
    63→  const AWAY_FORMATION: PitchNode[] = [
    64→    { x: 93, y: 50 }, { x: 78, y: 18 }, { x: 80, y: 39 }, { x: 80, y: 61 }, { x: 78, y: 82 },
    65→    { x: 60, y: 30 }, { x: 63, y: 50 }, { x: 60, y: 70 },
    66→    { x: 38, y: 22 }, { x: 34, y: 50 }, { x: 38, y: 78 },
    67→  ];
    68→  const players = topPlayers.filter((p) => p.side === side);
    69→  const formation = side === "HOME" ? HOME_FORMATION : AWAY_FORMATION;
    70→  return formation.map((slot, i) => {
    71→    const p = players[i];
    72→    if (!p) return slot;
    73→    const status: PitchNode["status"] =
    74→      p.status === "INJURED" || p.status === "SUSPENDED" ? "MISSING" : p.status === "DOUBTFUL" ? "DOUBTFUL" : "FIT";
    75→    return { ...slot, status, label: `${p.name} (${posLabel(p.position)}) — ${statusBadge(p.status).label}` };
    76→  });
    77→}
