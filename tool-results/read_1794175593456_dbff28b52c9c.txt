    55→function equalProfitStakes(oddsA: number, oddsB: number): { stakeA: number; stakeB: number } {
    56→  const invA = 1 / Math.max(1.01, oddsA);
    57→  const invB = 1 / Math.max(1.01, oddsB);
    58→  const sum = invA + invB;
    59→  const r1 = (v: number) => Math.round(v * 10) / 10;
    60→  return { stakeA: r1((100 * invA) / sum), stakeB: r1((100 * invB) / sum) };
    61→}

   722→  const meta = STATUS_META[slip.status] ?? STATUS_META.PENDING;
   723→  const StatusIcon = meta.icon;
   724→  const score = slip.homeGoals !== null && slip.awayGoals !== null ? `${slip.homeGoals}−${slip.awayGoals}` : null;
   725→  const profitTone = (slip.netProfit ?? 0) > 0 ? "text-emerald-300" : (slip.netProfit ?? 0) < 0 ? "text-rose-300" : "text-muted-foreground";
   726→  const [copied, setCopied] = useState(false);
   727→
   728→  const copyHash = (e: React.MouseEvent) => {
   729→    e.stopPropagation();
   730→    void navigator.clipboard?.writeText(slip.hash);
   731→    setCopied(true);
   732→    setTimeout(() => setCopied(false), 1500);
   733→  };
   734→
   735→  return (
   736→    <TableRow className="border-white/6 align-top">
   737→      <TableCell className="max-w-[190px] py-2.5">
   738→        <p className="truncate text-[11px] font-bold">
   739→          {slip.homeTeam} <span className="text-muted-foreground">×</span> {slip.awayTeam}
   740→        </p>
   741→        <p className="text-[9px] text-muted-foreground">
   742→          {slip.league} · <span dir="ltr">{new Date(slip.kickoff).toLocaleDateString("fa-IR", { month: "short", day: "numeric" })}</span>
   743→          {score && (
   744→            <b className="mr-1 font-num text-foreground" dir="ltr">
   745→              FT {score}
   746→            </b>
   747→          )}
   748→        </p>
   749→        {slip.archetype && <p className="mt-0.5 truncate text-[9px] text-fuchsia-300">📐 {slip.archetype}</p>}
   750→      </TableCell>
   751→
   752→      <TableCell className="py-2.5">
   753→        <Badge variant="outline" className={cn("gap-1 whitespace-nowrap text-[9px]", meta.cls)}>
   754→          <StatusIcon className={cn("size-2.5", slip.status === "LIVE" && "animate-spin")} />
   755→          {meta.label}
   756→        </Badge>
   757→        {slip.settleNote && (
   758→          <p className="mt-1 max-w-[160px] truncate text-[9px] text-muted-foreground" title={slip.settleNote}>
   759→            {slip.settleNote}
   760→          </p>
   761→        )}
   762→      </TableCell>
   763→
   764→      <TableCell className="max-w-[170px] py-2.5">
   765→        <p className="line-clamp-2 text-[10px] leading-4">{slip.labelA}</p>
   766→        <p className="mt-0.5 text-[9px] text-muted-foreground">
   767→          ضریب <b className="font-num text-sky-300" dir="ltr">{slip.oddsA.toFixed(2)}</b> · استیک{" "}
   768→          <b className="font-num" dir="ltr">{faNum(slip.stakeA, 1)}</b>
   769→          {slip.status === "WON_A" || slip.status === "DOUBLE_WIN" ? (
   770→            <b className="mr-1 font-num text-emerald-300" dir="ltr"> ✓</b>
   771→          ) : ["WON_B", "MIDDLE_TRAP"].includes(slip.status) ? (
   772→            <b className="mr-1 font-num text-rose-300" dir="ltr"> ✗</b>
   773→          ) : null}
   774→        </p>
   775→      </TableCell>
   776→
   777→      <TableCell className="max-w-[170px] py-2.5">
   778→        <p className="line-clamp-2 text-[10px] leading-4">{slip.labelB}</p>
   779→        <p className="mt-0.5 text-[9px] text-muted-foreground">
   780→          ضریب <b className="font-num text-emerald-300" dir="ltr">{slip.oddsB.toFixed(2)}</b> · استیک{" "}
   781→          <b className="font-num" dir="ltr">{faNum(slip.stakeB, 1)}</b>
   782→          {slip.status === "WON_B" || slip.status === "DOUBLE_WIN" ? (
   783→            <b className="mr-1 font-num text-emerald-300" dir="ltr"> ✓</b>
   784→          ) : ["WON_A", "MIDDLE_TRAP"].includes(slip.status) ? (
   785→            <b className="mr-1 font-num text-rose-300" dir="ltr"> ✗</b>
   786→          ) : null}
   787→        </p>
   788→      </TableCell>
   789→
   790→      <TableCell className="py-2.5 text-left">
   791→        {slip.netProfit !== null ? (
   792→          <>
   793→            <b className={cn("font-num text-sm", profitTone)} dir="ltr">
   794→              {slip.netProfit >= 0 ? "+" : ""}
   795→              {faNum(slip.netProfit, 1)}
   796→            </b>
   797→            {(slip.status === "WON_A" || slip.status === "WON_B") && (
   798→              <p className="text-[8px] leading-3 text-muted-foreground" dir="ltr">
   799→                {slip.status === "WON_A" ? `${slip.stakeA.toFixed(1)}×${slip.oddsA.toFixed(2)}−100` : `${slip.stakeB.toFixed(1)}×${slip.oddsB.toFixed(2)}−100`}
   800→              </p>
   801→            )}
   802→            {slip.status === "DOUBLE_WIN" && (
   803→              <p className="text-[8px] leading-3 text-amber-300/80">A: {faNum(slip.profitA ?? 0, 1)} + B: {faNum(slip.profitB ?? 0, 1)}</p>
   804→            )}
   805→          </>
   806→        ) : (
   807→          <span className="text-[10px] text-muted-foreground">—</span>
   808→        )}
   809→      </TableCell>
   810→
   811→      <TableCell className="py-2.5">
   812→        <button type="button" className="flex items-center gap-1 font-mono text-[9px] text-cyan-300/90 hover:text-cyan-200" onClick={copyHash} title={`کپی هش کامل: ${slip.hash}`} aria-label="کپی هش SHA-256 کارت">
   813→          <span dir="ltr">{slip.hash.slice(0, 14)}…</span>
   814→          {copied ? <Check className="size-2.5 text-emerald-400" /> : <Copy className="size-2.5" />}
   815→        </button>
   816→        {slip.espnUrl && (
   817→          <a href={slip.espnUrl} target="_blank" rel="noreferrer" className="mt-1 flex items-center gap-0.5 text-[9px] text-cyan-400 hover:underline">
   818→            راستی‌آزمایی ESPN <ExternalLink className="size-2.5" />
   819→          </a>
   820→        )}
   821→      </TableCell>
   822→
   823→      <TableCell className="py-2.5">
   824→        <div className="flex items-center justify-center gap-1" aria-label="وضعیت انتشار تلگرام">
   825→          <TgDot on={slip.telegramCard} title="کارت کامل" icon="📤" />
   826→          <TgDot on={slip.telegramTeaser} title="تیزر VIP (تیپ B سانسور)" icon="🔒" />
   827→          <TgDot on={slip.telegramResult} title="نتیجه پس از FT (جام/رختکن)" icon={slip.status === "MIDDLE_TRAP" ? "🧹" : "🏆"} />
   828→        </div>
   829→      </TableCell>
   830→    </TableRow>
   831→  );
   832→}
