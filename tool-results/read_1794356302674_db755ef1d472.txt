    92→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
    93→  const token = await botToken();
    94→  if (!token) return { ok: false, detail: "NO_TOKEN" };
    95→  try {
    96→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
    97→      method: "POST",
    98→      headers: { "Content-Type": "application/json" },
    99→      body: JSON.stringify(body),
   100→      cache: "no-store",
   101→    });
   102→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   103→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
   104→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   105→  } catch (err) {
   106→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   107→  }
   108→}

   110→async function tgSendPhotoTo(
   111→  chatId: string | null,
   112→  photo: Buffer,
   113→  caption: string,
   114→  filename = "slip-card.png",
   115→  captionless = false
   116→): Promise<{ ok: boolean; detail: string }> {
   117→  const token = await botToken();
   118→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   119→  if (!chatId) return { ok: false, detail: "NO_CHANNEL" };
   120→  try {
   121→    const form = new FormData();
   122→    form.append("chat_id", chatId);
   123→    if (!captionless) {
   124→      form.append("caption", caption.slice(0, 1024));
   125→      form.append("parse_mode", "HTML");
   126→    }
   127→    form.append("photo", new Blob([new Uint8Array(photo)], { type: "image/png" }), filename);
   128→    const res = await fetch(`https://api.telegram.org/bot${token}/sendPhoto`, { method: "POST", body: form });
   129→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   130→    if (json.ok) return { ok: true, detail: `photo:${json.result?.message_id ?? "?"}${captionless ? ":broadcast" : ""}` };
   131→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   132→  } catch (err) {
   133→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   134→  }
   135→}

   174→/** پیام HTML فلش‌کارت کامل — تیپ A + تیپ B + داچینگ + SHA-256 + لینک ESPN */
   175→export function buildCardMessage(slip: IssuedSlip): string {
   176→  const kickoff = new Date(slip.kickoff).toLocaleString("fa-IR", { dateStyle: "short", timeStyle: "short" });
   177→  const lines: string[] = [];
   178→  lines.push(`⚽️ <b>فوتبال‌کوانت — فلش‌کارت تیپ مکمل</b>`);
   179→  lines.push(`🏆 <b>${esc(slip.homeTeam)}</b> × <b>${esc(slip.awayTeam)}</b>`);
   180→  lines.push(`📡 ${esc(slip.league)} · ⏰ ${kickoff}`);
   181→  if (slip.archetype) lines.push(`📐 الگو: ${esc(slip.archetype)}`);
   182→  lines.push("");
   183→  lines.push(`🔹 <b>تیپ A</b> (ضریب ${slip.oddsA.toFixed(2)}): ${esc(slip.labelA)}`);
   184→  lines.push(`💵 مبلغ داچینگ A: <b>${slip.stakeA.toFixed(1)}</b> واحد`);
   185→  lines.push("");
   186→  lines.push(`🟢 <b>تیپ B مکمل</b> (ضریب ${slip.oddsB.toFixed(2)}): ${esc(slip.labelB)}`);
   187→  lines.push(`💵 مبلغ داچینگ B: <b>${slip.stakeB.toFixed(1)}</b> واحد`);
   188→  lines.push("");
   189→  // ─── v10.0 — ارزش عددی نسبت به خط مرجع (همیشه عدد دارد) ───
   190→  if (slip.edgeA !== null || slip.edgeB !== null) {
   191→    const isPin = slip.edgeSourceA === "PINNACLE" || slip.edgeSourceB === "PINNACLE";
   192→    const src = isPin ? "خط شارپ پیناکل" : "خط منصفانه بازار";
   193→    const fa = (v: number) => `${v >= 0 ? "+" : ""}${v.toFixed(1)}%`;
   194→    const a = slip.edgeA !== null ? fa(slip.edgeA) : "—";
   195→    const b = slip.edgeB !== null ? fa(slip.edgeB) : "—";
   196→    lines.push(`📈 ارزش عددی نسبت به ${src}: تیپ A <b>${a}</b> · تیپ B <b>${b}</b>`);
   197→    if (isPin) {
   198→      const cmp = (odds: number, pin: number | null): string | null =>
   199→        pin && pin > 1.01 ? `${odds.toFixed(2)} در برابر خط ${pin.toFixed(2)}` : null;
   200→      const pa = slip.edgeSourceA === "PINNACLE" ? cmp(slip.oddsA, slip.pinnacleOddsA) : null;
   201→      const pb = slip.edgeSourceB === "PINNACLE" ? cmp(slip.oddsB, slip.pinnacleOddsB) : null;
   202→      const parts = [pa ? `تیپ A: ${pa}` : null, pb ? `تیپ B: ${pb}` : null].filter((x): x is string => Boolean(x));
   203→      if (parts.length > 0) lines.push(`📌 مقایسه قیمت با خط پیناکل: ${parts.join(" · ")}`);
   204→    }
   205→  }
   206→  // ─── v10.0 — بهترین ساعت ثبت ضریب (Optimal Odds Timing Window) ───
   207→  if (slip.timingWindowA || slip.timingWindowB) {
   208→    const winName = timingWindowLabel(slip.timingWindowA ?? slip.timingWindowB);
   209→    const start = slip.timingStartA ?? slip.timingStartB;
   210→    const end = slip.timingEndA ?? slip.timingEndB;
   211→    const fmt = (d: Date) => new Intl.DateTimeFormat("fa-IR", { hour: "2-digit", minute: "2-digit", timeZone: "Asia/Tehran" }).format(d);
   212→    const range = start && end ? ` | ساعت دقیق پیشنهادی ثبت: ${fmt(new Date(start))} تا ${fmt(new Date(end))} (تهران)` : "";
   213→    lines.push(`⏱️ بهترین زمان ثبت این تیپ: <b>${esc(winName)}</b>${range}`);
   214→  }
   215→  lines.push(`💰 سود تضمینی پوشش: <b>+${slip.guaranteedProfit.toFixed(1)}%</b> (مجموع استیک ۱۰۰ واحد)`);
   216→  lines.push(`🔐 هش راستی‌آزمایی SHA-256:\n<code>${slip.hash.slice(0, 32)}</code>`);
   217→  if (slip.espnUrl) lines.push(`🔎 <a href="${slip.espnUrl}">لینک راستی‌آزمایی رسمی ESPN</a>`);
   218→  lines.push("");
   219→  lines.push(`⚖️ ابزار تحلیل کمّی — مسئولیت استفاده با کاربر است.`);
   220→  return lines.join("\n");
   221→}

   240→export function buildTeaserMessage(slip: IssuedSlip): string {

   242→  const lines: string[] = [];

   245→  lines.push(`📡 ${esc(slip.league)} · ⏰ ${kickoff}`);
   246→  lines.push("");
   247→  lines.push(`🔹 <b>تیپ A</b> (ضریب ${slip.oddsA.toFixed(2)}): ${esc(slip.labelA)}`);
   248→  lines.push(`💵 مبلغ داچینگ A: <b>${slip.stakeA.toFixed(1)}</b> واحد`);
   249→  lines.push(`💰 سود تضمینی داچینگ: <b>+${slip.guaranteedProfit.toFixed(1)}%</b>`);
   250→  lines.push("");
   251→  lines.push(`🟢 <b>تیپ B مکمل</b> (ضریب ${slip.oddsB.toFixed(2)}): 🔒 <b>[مخصوص اعضای کانال VIP]</b>`);

   262→export function slipResultLabel(status: string): { label: string; emoji: string; win: boolean } {
   263→  switch (status) {
   264→    case "WON_A":
   265→      return { label: "برد تیپ A (Slip A Won)", emoji: "🟢", win: true };
   266→    case "WON_B":
   267→      return { label: "برد تیپ B (Slip B Won)", emoji: "🟢", win: true };
   268→    case "DOUBLE_WIN":
   269→      return { label: "برد طلایی هر دو تیپ — Double-Win Jackpot", emoji: "🏆", win: true };
   270→    case "MIDDLE_TRAP":
   271→      return { label: "تله میانی — باخت همزمان (Middle Trap)", emoji: "🔴", win: false };
   272→    default:
   273→      return { label: status, emoji: "⚪️", win: false };
   274→  }

   277→export function buildResultMessage(slip: IssuedSlip): string {
   278→  const r = slipResultLabel(slip.status);
   279→  const score = slip.homeGoals !== null && slip.awayGoals !== null ? `${slip.homeGoals}−${slip.awayGoals}` : "؟";
   280→  const lines: string[] = [];
   281→  lines.push(`${r.emoji} <b>نتیجه نهایی کارت</b> ${r.emoji}`);
   282→  lines.push(`🏆 <b>${esc(slip.homeTeam)} ${score} ${esc(slip.awayTeam)}</b>`);
   283→  lines.push(`📡 ${esc(slip.league)} · FT (منبع رسمی ESPN)`);
   284→  lines.push("");
   285→  lines.push(`نتیجه: <b>${r.label}</b>`);
   286→  if (slip.status === "DOUBLE_WIN") {
   287→    lines.push(`💰 سود تیپ A: <b>+${(slip.profitA ?? 0).toFixed(1)}</b> · سود تیپ B: <b>+${(slip.profitB ?? 0).toFixed(1)}</b>`);
   288→    lines.push(`🎯 جمع سود خالص (Jackpot): <b>${(slip.netProfit ?? 0) >= 0 ? "+" : ""}${(slip.netProfit ?? 0).toFixed(1)}</b> واحد`);
   289→  } else if (slip.status === "MIDDLE_TRAP") {
   290→    lines.push(`📉 زیان کل: <b>${(slip.netProfit ?? 0).toFixed(1)}</b> واحد — ثبت صادقانه و شفاف`);
   291→  } else {
   292→    lines.push(`💰 سود خالص: <b>${(slip.netProfit ?? 0) >= 0 ? "+" : ""}${(slip.netProfit ?? 0).toFixed(1)}</b> واحد`);
   293→    lines.push(`🧮 فرمول: ${(slip.status === "WON_A" ? `stakeA×oddsA−۱۰۰ = ${slip.stakeA.toFixed(1)}×${slip.oddsA.toFixed(2)}−۱۰۰` : `stakeB×oddsB−۱۰۰ = ${slip.stakeB.toFixed(1)}×${slip.oddsB.toFixed(2)}−۱۰۰`)}`);
   294→  }
   295→  lines.push(`🔐 هش سند: <code>${slip.hash.slice(0, 32)}</code>`);
   296→  if (slip.espnUrl) lines.push(`🔎 <a href="${slip.espnUrl}">راستی‌آزمایی نتیجه در ESPN</a>`);
   297→  return lines.join("\n");
   298→}

   301→
   302→export type PublishMode = "CARD" | "TEASER";
   303→
   304→/**
   305→ * انتشار کارت/تیزر به کانال VIP:
   306→ * تصویر فوتبالی دو تیم + پیام HTML. تیپ B در حالت TEASER سانسور می‌شود.
   307→ * v9.0 — فیلتر سختگیرانه سوپروایزر: بازی وتوشده مطلقاً به تلگرام ارسال نمی‌شود.
   308→ */
   309→export async function publishSlipToTelegram(slip: IssuedSlip, mode: PublishMode): Promise<{ ok: boolean; detail: string; preview: string }> {
   310→  const preview = mode === "CARD" ? buildCardMessage(slip) : buildTeaserMessage(slip);
   311→  if (!slip.supervisorApproved) {
   312→    await logOutbox(slip.id, mode, false, "VETOED_BY_SUPERVISOR");
   313→    return { ok: false, detail: "VETOED_BY_SUPERVISOR", preview };
   314→  }
   315→  const status = await telegramStatus();
   316→  if (!status.configured || !status.channelSet) {
   317→    await logOutbox(slip.id, mode, false, "TELEGRAM_NOT_CONFIGURED");
   318→    return { ok: false, detail: "TELEGRAM_NOT_CONFIGURED", preview };
   319→  }
   320→
   321→  const photo = mode === "TEASER" ? await renderSlipTeaserPNG(slip) : await renderSlipCardPNG(slip);
   322→
   323→  let result: { ok: boolean; detail: string };
   324→  if (photo) {
   325→    result = await tgSendPhoto(photo, preview, `slip-${slip.hash.slice(0, 10)}.png`);
   326→  } else {
   327→    // تصویر تولید نشد — پیام متنی HTML همچنان کامل ارسال می‌شود
   328→    result = await tgCall("sendMessage", { chat_id: await channelId(), text: preview, parse_mode: "HTML", disable_web_page_preview: true });
   329→  }
   330→
   331→  await logOutbox(slip.id, mode, result.ok, result.detail);
   332→  if (result.ok) {
   333→    await db.issuedSlip.update({
   334→      where: { id: slip.id },
   335→      data: mode === "CARD" ? { telegramCard: true } : { telegramTeaser: true },
   336→    });
   337→  }
   338→  return { ...result, preview };
   339→}

   388→/** آمار وین‌ریت کانال — مستقیم از دیتابیس (بدون وابستگی چرخه‌ای) */
   389→async function channelWinRateStats() {
   390→  // v11.0 — فقط کارت‌های واقعی تاییدشده؛ تسویه کاغذی دفتر سایه در وین‌ریت کانال دخالت نمی‌کند
   391→  const settled = await db.issuedSlip.findMany({
   392→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, supervisorApproved: true },
   393→    select: { status: true, netProfit: true, stakeA: true, stakeB: true },
   394→  });
   395→  const st = { settled: settled.length, wonA: 0, wonB: 0, doubleWin: 0, middleTrap: 0, netProfit: 0, staked: 0 };
   396→  for (const s of settled) {
   397→    if (s.status === "WON_A") st.wonA += 1;
   398→    else if (s.status === "WON_B") st.wonB += 1;
   399→    else if (s.status === "DOUBLE_WIN") st.doubleWin += 1;
   400→    else if (s.status === "MIDDLE_TRAP") st.middleTrap += 1;
   401→    st.netProfit += s.netProfit ?? 0;
   402→    st.staked += s.stakeA + s.stakeB;
   403→  }
   404→  const winRatePct = st.settled > 0 ? Math.round(((st.settled - st.middleTrap) / st.settled) * 1000) / 10 : 0;
   405→  const netPct = st.staked > 0 ? Math.round((st.netProfit / st.staked) * 1000) / 10 : 0;
   406→  // ─── v11.0 — دفتر سایه: شفافیت counterfactual وتوها (خط جداگانه گزارش) ───
   407→  const shadowRows = await db.issuedSlip.findMany({
   408→    where: { supervisorApproved: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   409→    select: { status: true, netProfit: true },
   410→  });
   411→  const shadowWins = shadowRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   412→  const shadowNet = Math.round(shadowRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100;
   413→  const shadowWinRate = shadowRows.length > 0 ? Math.round((shadowWins / shadowRows.length) * 1000) / 10 : null;
   414→  return {
   415→    ...st,
   416→    winRatePct,
   417→    netPct,
   418→    netProfit: Math.round(st.netProfit * 100) / 100,
   419→    shadow: { settled: shadowRows.length, wins: shadowWins, winRatePct: shadowWinRate, netProfit: shadowNet },
   420→  };
   421→}

   915→  const token = await botToken();

   917→  const chatId = await publicChannelId();
   918→  if (!chatId) return false;
   919→
   920→  const today = tehranDayKey();

   922→    const last = await getSetting(KEY_FUNNEL_PNL_DAY);
   923→    if (last === today) return false;

   927→    where: { supervisorApproved: true, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, settledAt: { not: null } },
   928→    orderBy: { settledAt: "desc" },
   929→    take: 80,
   930→  });

   932→

   934→  const netA = tierNet("SINGLE");
   935→  const netB = tierNet("COMBO");
   936→  const netC = tierNet("SNIPER");
   937→  const total = netA + netB + netC;
   938→  const settled = todayRows.length;
   939→  const wins = todayRows.filter((r) => r.status !== "MIDDLE_TRAP").length;
   940→  const winRate = settled > 0 ? (wins / settled) * 100 : 0;
   941→
   942→  const dateLabelEn = new Intl.DateTimeFormat("en-GB", { weekday: "long", day: "2-digit", month: "short", year: "numeric", timeZone: "Asia/Tehran" }).format(new Date());

   954→      teams: `${r.homeTeam} vs ${r.awayTeam}`,

   957→      profit: r.netProfit ?? 0,
   958→    })),

   960→    daySettled: settled,
