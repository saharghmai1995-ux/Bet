   178→function downsampleEvenly(arr: number[], maxPts: number): number[] {
   179→  if (arr.length <= maxPts) return arr;
   180→  const out: number[] = [];
   181→  for (let i = 0; i < maxPts - 1; i++) {
   182→    out.push(arr[Math.round((i * (arr.length - 1)) / (maxPts - 1))]);
   183→  }
   184→  out.push(arr[arr.length - 1]);
   185→  return out;
   186→}

   590→function PeriodBarChart({ data }: { data: BankPeriodStat[] }) {
   591→  const W = 560;
   592→  const H = 150;
   593→  const padB = 26;
   594→  const padT = 16;
   595→  const maxAbs = Math.max(1, ...data.map((d) => Math.abs(d.netProfit)));
   596→  const barW = W / Math.max(1, data.length);
   597→  const zeroY = H - padB; // خط صفر پایین نمودار وقتی همه مثبت‌اند
   598→  // خط صفر وسط اگر منفی و مثبت داریم
   599→  const hasNeg = data.some((d) => d.netProfit < 0);
   600→  const hasPos = data.some((d) => d.netProfit > 0);
   601→  const midZero = hasNeg && hasPos;
   602→  const zeroLine = midZero ? (H - padB + padT) / 2 : zeroY;
   603→  const upScale = midZero ? (zeroLine - padT) / maxAbs : (H - padB - padT) / maxAbs;
   604→  const downScale = midZero ? (H - padB - zeroLine) / maxAbs : upScale;
   605→
   606→  return (
   607→    <svg viewBox={`0 0 ${W} ${H}`} className="h-[150px] w-full" role="img" aria-label="نمودار تغییرات بانک در بازه‌های انتخابی">
   608→      <line x1="0" y1={zeroLine} x2={W} y2={zeroLine} stroke="rgba(255,255,255,0.25)" strokeWidth="1" strokeDasharray="3 3" />
   609→      {data.map((d, i) => {
   610→        const v = d.netProfit;
   611→        const h = v >= 0 ? v * upScale : Math.abs(v) * downScale;
   612→        const x = i * barW + barW * 0.18;
   613→        const bw = barW * 0.64;
   614→        const y = v >= 0 ? zeroLine - h : zeroLine;
   615→        const tone = v > 0 ? "#34d399" : v < 0 ? "#fb7185" : "rgba(255,255,255,0.14)";
   616→        const label = d.label.length > 9 ? d.label.slice(0, 9) : d.label;
   617→        return (
   618→          <g key={d.key}>
   619→            <title>{`${d.label}: ${v >= 0 ? "+" : "−"}${Math.abs(v).toFixed(1)}u · ${d.settled} بازی (${d.wins} برد / ${d.losses} باخت) · بانک پایان: ${d.bankAtEnd.toFixed(0)}u`}</title>
   620→            <rect x={x} y={v >= 0 ? zeroLine - h : zeroLine} width={bw} height={Math.max(v === 0 ? 1.5 : 0, h)} rx="2" fill={tone} opacity={0.88} />
   621→            {Math.abs(v) > 0 && (
   622→              <text
   623→                x={x + bw / 2}
   624→                y={v >= 0 ? zeroLine - h - 3 : zeroLine + h + 8}
   625→                textAnchor="middle"
   626→                fontSize="7.5"
   627→                fontWeight="700"
   628→                fill={v > 0 ? "#6ee7b7" : "#fda4af"}
   629→              >
   630→                {v > 0 ? "+" : "−"}{Math.abs(v).toFixed(0)}
   631→              </text>
   632→            )}
   633→            <text x={x + bw / 2} y={H - 8} textAnchor="middle" fontSize="7" fill="rgba(255,255,255,0.55)">
   634→              {label}
   635→            </text>
   636→          </g>
   637→        );
   638→      })}
   639→    </svg>
   640→  );
