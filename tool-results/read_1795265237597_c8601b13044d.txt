    17→/** تبدیل ضریب آمریکایی (+150 / −120) به ضریب اعشاری استاندارد */
    18→export function americanToDecimal(americanOdds: number): number {
    19→  if (!americanOdds || Number.isNaN(americanOdds)) return 0;
    20→  return americanOdds > 0
    21→    ? Number((americanOdds / 100 + 1).toFixed(2))
    22→    : Number((100 / Math.abs(americanOdds) + 1).toFixed(2));
    23→}

    49→const num = (v: unknown): number | null => {
    50→  const n = typeof v === "string" ? parseFloat(v) : typeof v === "number" ? v : NaN;
    51→  return Number.isFinite(n) ? n : null;

    54→/** عبارت American یا Decimal → Decimal */
    55→function toDecimal(v: unknown): number | null {
    56→  if (v === null || v === undefined) return null;
    57→  if (typeof v === "number") return v > 1 ? v : null;
    58→  const s = String(v).trim();
    59→  if (s.startsWith("+") || s.startsWith("−") || s.startsWith("-")) {
    60→    const d = americanToDecimal(Math.abs(parseFloat(s.replace("−", "-"))));
    61→    return d > 1 ? d : null;
    62→  }
    63→  const n = parseFloat(s);
    64→  return Number.isFinite(n) && n > 1 ? n : null;
    65→}

    69→/** externalId با قالب espn-<slug>-<eventId> → دو جزء آن */
    70→export function parseEspnExternalId(externalId: string | null | undefined): { slug: string; eventId: string } | null {
    71→  if (!externalId || !externalId.startsWith("espn-")) return null;
    72→  const parts = externalId.split("-");
    73→  if (parts.length < 3) return null;
    74→  const eventId = parts[parts.length - 1];
    75→  const slug = parts.slice(1, -1).join("-");
    76→  return slug && eventId ? { slug, eventId } : null;
    77→}

    94→/** ضرایب + احتمالات زنده ESPN Core برای یک رویداد (بدون کلید) */
    95→export async function fetchEspnCoreOdds(slug: string, eventId: string): Promise<EspnCoreOdds | null> {
    96→  const base = `https://sports.core.api.espn.com/v2/sports/soccer/leagues/${slug}/events/${eventId}/competitions/${eventId}`;
    97→  const [oddsRes, probRes] = await Promise.all([
    98→    fetchJson(`${base}/odds`).catch(() => null),
    99→    fetchJson(`${base}/probabilities`).catch(() => null),
   100→  ]);
   101→
   102→  let bookmakers: EspnCoreOdds["bookmakers"] = [];
   103→  const oddsData = oddsRes?.data as { items?: Record<string, unknown>[] } | null;
   104→  if (oddsData && Array.isArray(oddsData.items)) {
   105→    bookmakers = oddsData.items.map((item) => {
   106→      const provider = item.provider as { id?: string | number; name?: string } | undefined;
   107→      const ml = item.moneyline as { home?: Record<string, unknown>; away?: Record<string, unknown>; draw?: Record<string, unknown> } | undefined;
   108→      const ou = item.overUnder as { line?: unknown; over?: Record<string, unknown>; under?: Record<string, unknown> } | undefined;
   109→      const pick = (o?: Record<string, unknown>) => toDecimal(o?.american ?? o?.decimal ?? o?.value);
   110→      return {
   111→        id: String(provider?.id ?? ""),
   112→        name: provider?.name ?? ESPN_PROVIDER_NAMES[String(provider?.id ?? "")] ?? `Provider ${provider?.id ?? "?"}`,
   113→        home: pick(ml?.home),
   114→        draw: pick(ml?.draw),
   115→        away: pick(ml?.away),
   116→        ouLine: num(ou?.line),
   117→        over: pick(ou?.over),
   118→        under: pick(ou?.under),
   119→      };
   120→    }).filter((b) => b.home !== null || b.away !== null || b.over !== null);
   121→  }
   122→
   123→  let probabilities: EspnCoreOdds["probabilities"] = null;
   124→  const probData = probRes?.data as { items?: Record<string, unknown>[]; statistics?: Record<string, unknown>[] } | null;
   125→  const probItems = probData?.items ?? probData?.statistics ?? [];
   126→  if (Array.isArray(probItems) && probItems.length > 0) {
   127→    const first = probItems[0] as { homeTeamProbability?: { winPercentage?: unknown }; awayTeamProbability?: { winPercentage?: unknown }; drawPercentage?: unknown; tiePercentage?: unknown };
   128→    const home = num(first?.homeTeamProbability?.winPercentage);
   129→    const away = num(first?.awayTeamProbability?.winPercentage);
   130→    const draw = num(first?.drawPercentage ?? first?.tiePercentage);
   131→    if (home !== null && away !== null) {
   132→      const s = (home + away + (draw ?? 0)) / 100 || 1;
   133→      probabilities = { home: home / 100 / s, away: away / 100 / s, draw: (draw ?? 0) / 100 / s };
   134→    }
   135→  }
   136→
   137→  if (bookmakers.length === 0 && !probabilities) return null;
   138→  return { bookmakers, probabilities };
   139→}

   143→const POLYMARKET_URL = "https://gamma-api.polymarket.com/events?active=true&closed=false&tag_slug=soccer&limit=30";

   145→interface PolyRow { title: string; home: string | null; draw: string | null; away: string | null; question: string }

   147→function normTokens(s: string): string[] {
   148→  return s
   149→    .toLowerCase()
   150→    .replace(/\b(fc|cf|sc|ac|as|afc|cfc|united|city)\b/g, " ")
   151→    .split(/[^a-z0-9]+/)
   152→    .filter((t) => t.length > 2);
   153→}
   154→
   155→function teamMatch(team: string, candidate: string): boolean {
   156→  const tt = normTokens(team);
   157→  const ct = normTokens(candidate);
   158→  if (tt.length === 0 || ct.length === 0) return false;
   159→  return tt.some((t) => ct.includes(t)) || ct.some((c) => tt.includes(c));
   160→}

   162→/** قیمت‌های بدون کارمزد بازار پیش‌بینی Polymarket — با تطبیق فازی نام تیم‌ها */
   163→export async function fetchPolymarketSoccer(homeTeam: string, awayTeam: string): Promise<PolyRow[]> {
   164→  const res = await fetchJson(POLYMARKET_URL).catch(() => null);
   165→  const data = res?.data as { title?: string; markets?: { question?: string; outcomes?: string; outcomePrices?: string }[] }[] | null;
   166→  if (!Array.isArray(data)) return [];
   167→  const rows: PolyRow[] = [];
   168→  for (const ev of data) {
   169→    for (const mk of ev.markets ?? []) {
   170→      let outcomes: string[] = [];
   171→      let prices: string[] = [];
   172→      try { outcomes = JSON.parse(mk.outcomes ?? "[]"); } catch { continue; }
   173→      try { prices = JSON.parse(mk.outcomePrices ?? "[]"); } catch { continue; }
   174→      if (outcomes.length < 2 || prices.length < 2) continue;
   175→      // دوبی/سه‌تایی: نام تیم‌ها معمولا خود outcomes هستند
   176→      let home: string | null = null;
   177→      let draw: string | null = null;
   178→      let away: string | null = null;
   179→      for (let i = 0; i < outcomes.length; i++) {
   180→        const price = parseFloat(prices[i]);
   181→        if (!Number.isFinite(price) || price <= 0.01 || price >= 0.99) continue;
   182→        const odds = (1 / price).toFixed(2);
   183→        if (outcomes[i].toLowerCase() === "draw") draw = odds;
   184→        else if (teamMatch(homeTeam, outcomes[i])) home = odds;
   185→        else if (teamMatch(awayTeam, outcomes[i])) away = odds;
   186→      }
   187→      if (home || away) {
   188→        rows.push({ title: ev.title ?? "", question: mk.question ?? "", home, draw, away });
   189→      }
   190→    }
   191→  }
   192→  return rows;
   193→}

   197→export async function probeTheOddsApi(key: string): Promise<{ ok: boolean; detail: string; remaining: number | null }> {
   198→  if (!key) return { ok: false, detail: "کلید وارد نشده", remaining: null };
   199→  const res = await fetchJson(`https://api.the-odds-api.com/v4/sports/?apiKey=${encodeURIComponent(key)}`).catch(() => null);
   200→  if (!res) return { ok: false, detail: "شبکه در دسترس نیست", remaining: null };
   201→  const remaining = num(res.headers.get("x-requests-remaining"));
   202→  if (res.ok) return { ok: true, detail: remaining !== null ? `فعال — اعتبار باقی‌مانده: ${remaining} درخواست` : "فعال", remaining };
   203→  if (res.status === 401 || res.status === 422) return { ok: false, detail: "کلید نامعتبر یا منقضی", remaining };
   204→  if (res.status === 429) return { ok: false, detail: "سهمیه ماهانه تمام شده", remaining };
   205→  return { ok: false, detail: `خطای HTTP ${res.status}`, remaining };
   206→}

   242→function currentSeason(): string[] {
   243→  const now = new Date();
   244→  const y = now.getFullYear();
   245→  const yy = now.getMonth() >= 6 ? y : y - 1; // از جولای فصل جدید
   246→  const s = `${yy}`.slice(2);
   247→  const s2 = `${yy + 1}`.slice(2);
   248→  return [`${s}${s2}`, `${Number(s) - 1}${s}`];
   249→}

   251→/** پارس CSV فوتبال‌دیتا: خطوط PSH/PSD/PSA/P>2.5/P<2.5 پیناکل */
   252→export async function fetchFootballDataOdds(league = "E0", limit = 12): Promise<{ leagueName: string; season: string; rows: FootballDataRow[] } | null> {
   253→  const meta = FOOTBALL_DATA_LEAGUES[league] ?? FOOTBALL_DATA_LEAGUES.E0;
   254→  for (const season of currentSeason()) {
   255→    const res = await fetchText(`https://www.football-data.co.uk/mmz4281/${season}/${meta.file}.csv`).catch(() => null);
   256→    if (!res || !res.ok || res.text.length < 100) continue;
   257→    const lines = res.text.split(/\r?\n/).filter((l) => l.trim().length > 0);
   258→    if (lines.length < 5) continue;
   259→    // حذف BOM ابتدای فایل CSV
   260→    const header = lines[0].replace(/^\uFEFF/, "").split(",");
   261→    const idx = (name: string) => header.findIndex((h) => h.trim().toUpperCase() === name);
   262→    const iDiv = idx("DIV");
   263→    const iDate = idx("DATE");
   264→    const iHome = idx("HOMETEAM");
   265→    const iAway = idx("AWAYTEAM");
   266→    const iFthg = idx("FTHG");
   267→    const iFtag = idx("FTAG");
   268→    const iPsh = idx("PSH");
   269→    const iPsd = idx("PSD");
   270→    const iPsa = idx("PSA");
   271→    const iOver = idx("P>2.5");
   272→    const iUnder = idx("P<2.5");
   273→    if (iHome < 0 || iAway < 0 || iPsh < 0) continue;
   274→    const rows: FootballDataRow[] = [];
   275→    for (let i = lines.length - 1; i >= 1 && rows.length < limit; i--) {
   276→      const c = lines[i].split(",");
   277→      if (c.length < header.length - 3) continue;
   278→      if (iDiv >= 0 && c[iDiv] !== meta.file) continue;
   279→      // فقط بازی‌هایی که خط پیناکل (PSH) منتشر شده — هدف فید، بک‌تست CLV است
   280→      const pshRaw = iPsh >= 0 ? (c[iPsh] ?? "").trim() : "";
   281→      if (!pshRaw) continue;
   282→      const pickN = (v: string | undefined) => {
   283→        const n = parseFloat((v ?? "").trim());
   284→        return Number.isFinite(n) && n > 1 ? n : null;
   285→      };
   286→      rows.push({
   287→        date: (c[iDate] ?? "").trim(),
   288→        homeTeam: (c[iHome] ?? "").trim(),
   289→        awayTeam: (c[iAway] ?? "").trim(),
   290→        fthg: iFthg >= 0 ? parseInt(c[iFthg]) || null : null,
   291→        ftag: iFtag >= 0 ? parseInt(c[iFtag]) || null : null,
   292→        psh: pickN(c[iPsh]),
   293→        psd: pickN(c[iPsd]),
   294→        psa: pickN(c[iPsa]),
   295→        pOver25: iOver >= 0 ? pickN(c[iOver]) : null,
   296→        pUnder25: iUnder >= 0 ? pickN(c[iUnder]) : null,
   297→      });
   298→    }
   299→    if (rows.length > 0) return { leagueName: meta.name, season, rows };
   300→  }
   301→  return null;
   302→}

   523→/** خط سه‌گانه یک بازار Polymarket (توکن نرمال‌شده دو تیم + ضریب اعشاری) */
   524→export interface PolyMarketLine {
   525→  a: string;
   526→  b: string;
   527→  oddsA: number;
   528→  oddsB: number;
   529→  draw: number | null;
   530→  question: string;
   531→}
   532→
   533→let polyMapCache: { at: number; map: Map<string, PolyMarketLine> } | null = null;
   534→
   535→/**
   536→ * نقشه کش‌شده (۵ دقیقه) همه بازارهای سه‌گانه Polymarket — بدون کلید.
   537→ * کلید هر ردیف = توکن نرمال‌شده نام تیم؛ هر بازار دو بار (برای هر دو تیم) ثبت می‌شود.
   538→ */
   539→export async function fetchPolymarketMap(ttlMs = 5 * 60_000): Promise<Map<string, PolyMarketLine>> {
   540→  if (polyMapCache && Date.now() - polyMapCache.at < ttlMs) return polyMapCache.map;
   541→  const map = new Map<string, PolyMarketLine>();
   542→  const res = await fetchJson(POLYMARKET_URL).catch(() => null);
   543→  const data = res?.data as { title?: string; markets?: { question?: string; outcomes?: string; outcomePrices?: string }[] }[] | null;
   544→  if (Array.isArray(data)) {
   545→    for (const ev of data) {
   546→      for (const mk of ev.markets ?? []) {
   547→        let outcomes: string[] = [];
   548→        let prices: string[] = [];
   549→        try { outcomes = JSON.parse(mk.outcomes ?? "[]"); } catch { continue; }
   550→        try { prices = JSON.parse(mk.outcomePrices ?? "[]"); } catch { continue; }
   551→        if (outcomes.length < 2 || prices.length < 2) continue;
   552→        const dec = outcomes.map((o, i) => {
   553→          const pv = parseFloat(prices[i]);
   554→          return Number.isFinite(pv) && pv > 0.01 && pv < 0.99 ? Number((1 / pv).toFixed(2)) : null;
   555→        });
   556→        const drawIdx = outcomes.findIndex((o) => o.toLowerCase() === "draw");
   557→        const teamIdx = outcomes
   558→          .map((o, i) => (i === drawIdx ? null : i))
   559→          .filter((i): i is number => i !== null && dec[i] !== null);
   560→        if (teamIdx.length < 2) continue;
   561→        const [iA, iB] = teamIdx;
   562→        const line: PolyMarketLine = {
   563→          a: normTokens(outcomes[iA]).join(" "),
   564→          b: normTokens(outcomes[iB]).join(" "),
   565→          oddsA: dec[iA] as number,
   566→          oddsB: dec[iB] as number,
   567→          draw: drawIdx >= 0 && dec[drawIdx] !== null ? (dec[drawIdx] as number) : null,
   568→          question: mk.question ?? "",
   569→        };
   570→        if (!line.a || !line.b || line.oddsA <= 1 || line.oddsB <= 1) continue;
   571→        map.set(line.a, line);
   572→        map.set(line.b, line);
   573→      }
   574→    }
   575→  }
   576→  polyMapCache = { at: Date.now(), map };
   577→  return map;
   578→}

   580→/** تطبیق فازی یک مسابقه ESPN با بازار Polymarket — خروجی: خط اعشاری 1X2 */
   581→export function matchPolymarketLine(
   582→  map: Map<string, PolyMarketLine>,
   583→  homeTeam: string,
   584→  awayTeam: string
   585→): { home: number; draw: number | null; away: number } | null {
   586→  const hTok = normTokens(homeTeam).join(" ");
   587→  const aTok = normTokens(awayTeam).join(" ");
   588→  if (!hTok || !aTok) return null;
   589→  const line = map.get(hTok) ?? map.get(aTok);
   590→  if (!line) return null;
   591→  const fuzzy = (x: string, y: string) => x === y || (x.length > 2 && y.includes(x)) || (y.length > 2 && x.includes(y));
   592→  const sameMarket = (fuzzy(line.a, hTok) && fuzzy(line.b, aTok)) || (fuzzy(line.a, aTok) && fuzzy(line.b, hTok));
   593→  if (!sameMarket) return null;
   594→  const homeIsA = fuzzy(line.a, hTok);
   595→  const home = homeIsA ? line.oddsA : line.oddsB;
   596→  const away = homeIsA ? line.oddsB : line.oddsA;
   597→  if (!(home > 1) || !(away > 1)) return null;
   598→  return { home, draw: line.draw, away };
   599→}
