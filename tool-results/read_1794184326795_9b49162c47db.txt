    59→/** متن آماده کپی تیپ — v8.0 (تک‌تیپ طلایی سوپروایزر + فرمول صحیح سود) */
    60→export function cardCopyText(v: FlashcardView): string {
    61→  const p = v.pair;
    62→  const sup = v.supervisor;
    63→  const head = `⚽ فوتبال‌کوانت v8.0 Supervisor Quant — ${v.match.homeTeam} مقابل ${v.match.awayTeam} (${v.match.league})`;
    64→  const crownLine = sup
    65→    ? `👑 تک‌تیپ تاییدشده سوپروایزر — امتیاز مونت‌کارلو: ${sup.score100}/۱۰۰${p.archetype ? ` (${p.archetype.short})` : ""}`
    66→    : p.archetype
    67→      ? `📐 الگو: ${p.archetype.short}`
    68→      : "";
    69→  const profitLine =
    70→    p.guaranteedMarginPct > 0
    71→      ? `💰 سود تضمینی: +${p.guaranteedMarginPct.toFixed(1)}٪ (+${faNum(p.equalProfit.profit, 1)} واحد سود خالص)`
    72→      : `🛡️ پوشش بیمه سرمایه — بازگشت وجه در سناریوی پایه`;
    73→  return [
    74→    head,
    75→    crownLine,
    76→    `🔹 تیپ A (ضریب ${p.totalOddsA.toFixed(2)}): ${p.labelA} ⬅️ مبلغ شرط: ${faNum(p.equalProfit.stakeA, 1)} واحد`,
    77→    `🟢 تیپ B مکمل (ضریب ${p.totalOddsB.toFixed(2)}): ${p.labelB} ⬅️ مبلغ شرط: ${faNum(p.equalProfit.stakeB, 1)} واحد`,
    78→    profitLine,
    79→    `✅ پوشش سناریوها: ${(p.archetype ? p.archetype.coveragePct * 100 : (1 - p.jointStates.bothLose) * 100).toFixed(0)}٪ | ریسک باخت همزمان: ${(p.jointStates.bothLose * 100).toFixed(1)}%`,
    80→    `⚖️ ${p.verdict}`,
    81→  ]
    82→    .filter(Boolean)
    83→    .join("\n");
    84→}

    88→// ═══ v10.0 — نشانگر بهترین ساعت ثبت ضریب (Optimal Odds Timing Window) ═══
    89→
    90→const TIMING_STATUS_STYLE: Record<TimingStatus, string> = {
    91→  OPEN: "border-emerald-500/40 bg-emerald-500/[0.08] text-emerald-300",
    92→  WAITING: "border-amber-500/40 bg-amber-500/[0.07] text-amber-300",
    93→  LATE: "border-rose-500/40 bg-rose-500/[0.07] text-rose-300",
    94→  KICKED_OFF: "border-white/15 bg-white/[0.03] text-muted-foreground",
    95→};
    96→
    97→/** بنر پنجره زمانی طلایی ثبت — با شمارش معکوس زنده (به‌روزرسانی هر ۳۰ ثانیه) */
    98→function TimingBanner({ kickoff, legsA, oddsA, legsB, oddsB }: { kickoff: string; legsA: string[]; oddsA: number; legsB: string[]; oddsB: number }) {
    99→  const [now, setNow] = useState(() => Date.now());
   100→  useEffect(() => {
   101→    const t = setInterval(() => setNow(Date.now()), 30_000);
   102→    return () => clearInterval(t);
   103→  }, []);
   104→  const pair = useMemo(() => computePairTiming(kickoff, legsA, oddsA, legsB, oddsB, now), [kickoff, legsA, oddsA, legsB, oddsB, now]);
   105→
   106→  const rows = pair.same ? [pair.a] : [pair.a, pair.b];
   107→  const sideLabel = (i: number) => (pair.same ? "تیپ A + تیپ B" : i === 0 ? "تیپ A" : "تیپ B");
   108→
   109→  return (
   110→    <div className="space-y-1.5 rounded-xl border border-amber-500/30 bg-gradient-to-l from-amber-500/[0.08] to-transparent p-2.5">
   111→      <p className="flex items-center gap-1 text-[10px] font-bold text-amber-300">
   112→        ⏱️ بهترین زمان ثبت این تیپ (پنجره طلایی ضرایب — v10.0)
   113→      </p>
   114→      {rows.map((w, i) => {
   115→        const style = TIMING_STATUS_STYLE[w.status];
   116→        const clock = new Intl.DateTimeFormat("fa-IR", { hour: "2-digit", minute: "2-digit" });
   117→        return (
   118→          <div key={`${w.windowKey}-${i}`} className={`rounded-lg border px-2 py-1.5 ${style}`}>
   119→            <p className="text-[10.5px] font-bold leading-4">
   120→              {w.meta.icon} {sideLabel(i)}: {w.meta.nameFa}
   121→              <span className="mx-1 font-normal text-foreground/70">|</span>
   122→              ساعت دقیق پیشنهادی ثبت:
   123→              <b className="mx-1 font-num" dir="ltr">{w.suggestedClock}</b>
   124→            </p>
   125→            <p className="mt-0.5 text-[9.5px] leading-3.5 opacity-90">
   126→              {w.statusText}
   127→              {w.status === "OPEN" && w.msUntilClose > 0 && ` — ${faDuration(w.msUntilClose)} تا بستن پنجره`}
   128→              {w.status === "WAITING" && w.msUntilOpen > 0 && ` (${clock.format(new Date(w.startAt))} تا ${clock.format(new Date(w.endAt))})`}
   129→            </p>
   130→          </div>
   131→        );
   132→      })}
   133→      <p className="text-[9px] leading-3 text-muted-foreground">
   134→        منطق علمی: {rows[0].meta.rationale}
   135→      </p>
   136→    </div>
   137→  );
   138→}
