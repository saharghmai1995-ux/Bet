    73→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
    74→  const token = await botToken();
    75→  if (!token) return { ok: false, detail: "NO_TOKEN" };
    76→  try {
    77→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
    78→      method: "POST",
    79→      headers: { "Content-Type": "application/json" },
    80→      body: JSON.stringify(body),
    81→      cache: "no-store",
    82→    });
    83→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
    84→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
    85→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
    86→  } catch (err) {
    87→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
    88→  }
    89→}

    91→async function tgSendPhoto(
    92→  photo: Buffer,
    93→  caption: string,
    94→  filename = "slip-card.png"
    95→): Promise<{ ok: boolean; detail: string }> {
    96→  const token = await botToken();
    97→  const chatId = await channelId();
    98→  if (!token) return { ok: false, detail: "NO_TOKEN" };
    99→  if (!chatId) return { ok: false, detail: "NO_CHANNEL" };
   100→  try {
   101→    const form = new FormData();
   102→    form.append("chat_id", chatId);
   103→    form.append("caption", caption.slice(0, 1024));
   104→    form.append("parse_mode", "HTML");
   105→    form.append("photo", new Blob([new Uint8Array(photo)], { type: "image/png" }), filename);
   106→    const res = await fetch(`https://api.telegram.org/bot${token}/sendPhoto`, { method: "POST", body: form });
   107→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   108→    if (json.ok) return { ok: true, detail: `photo:${json.result?.message_id ?? "?"}` };
   109→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   110→  } catch (err) {
   111→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   112→  }

   131→/** پیام HTML فلش‌کارت کامل — تیپ A + تیپ B + داچینگ + SHA-256 + لینک ESPN */
   132→export function buildCardMessage(slip: IssuedSlip): string {
   133→  const kickoff = new Date(slip.kickoff).toLocaleString("fa-IR", { dateStyle: "short", timeStyle: "short" });
   134→  const lines: string[] = [];
   135→  lines.push(`⚽️ <b>فوتبال‌کوانت — فلش‌کارت تیپ مکمل</b>`);
   136→  lines.push(`🏆 <b>${esc(slip.homeTeam)}</b> × <b>${esc(slip.awayTeam)}</b>`);
   137→  lines.push(`📡 ${esc(slip.league)} · ⏰ ${kickoff}`);
   138→  if (slip.archetype) lines.push(`📐 الگو: ${esc(slip.archetype)}`);
   139→  lines.push("");
   140→  lines.push(`🔹 <b>تیپ A</b> (ضریب ${slip.oddsA.toFixed(2)}): ${esc(slip.labelA)}`);
   141→  lines.push(`💵 مبلغ داچینگ A: <b>${slip.stakeA.toFixed(1)}</b> واحد`);
   142→  lines.push("");
   143→  lines.push(`🟢 <b>تیپ B مکمل</b> (ضریب ${slip.oddsB.toFixed(2)}): ${esc(slip.labelB)}`);
   144→  lines.push(`💵 مبلغ داچینگ B: <b>${slip.stakeB.toFixed(1)}</b> واحد`);
   145→  lines.push("");
   146→  // ─── v10.0 — ارزش عددی نسبت به خط مرجع (همیشه عدد دارد) ───
   147→  if (slip.edgeA !== null || slip.edgeB !== null) {
   148→    const isPin = slip.edgeSourceA === "PINNACLE" || slip.edgeSourceB === "PINNACLE";
   149→    const src = isPin ? "خط شارپ پیناکل" : "خط منصفانه بازار";
   150→    const fa = (v: number) => `${v >= 0 ? "+" : ""}${v.toFixed(1)}%`;
   151→    const a = slip.edgeA !== null ? fa(slip.edgeA) : "—";
   152→    const b = slip.edgeB !== null ? fa(slip.edgeB) : "—";
   153→    lines.push(`📈 ارزش عددی نسبت به ${src}: تیپ A <b>${a}</b> · تیپ B <b>${b}</b>`);
   154→    if (isPin) {
   155→      const cmp = (odds: number, pin: number | null): string | null =>
   156→        pin && pin > 1.01 ? `${odds.toFixed(2)} در برابر خط ${pin.toFixed(2)}` : null;
   157→      const pa = slip.edgeSourceA === "PINNACLE" ? cmp(slip.oddsA, slip.pinnacleOddsA) : null;
   158→      const pb = slip.edgeSourceB === "PINNACLE" ? cmp(slip.oddsB, slip.pinnacleOddsB) : null;
   159→      const parts = [pa ? `تیپ A: ${pa}` : null, pb ? `تیپ B: ${pb}` : null].filter((x): x is string => Boolean(x));
   160→      if (parts.length > 0) lines.push(`📌 مقایسه قیمت با خط پیناکل: ${parts.join(" · ")}`);
   161→    }
   162→  }
   163→  // ─── v10.0 — بهترین ساعت ثبت ضریب (Optimal Odds Timing Window) ───
   164→  if (slip.timingWindowA || slip.timingWindowB) {
   165→    const winName = timingWindowLabel(slip.timingWindowA ?? slip.timingWindowB);
   166→    const start = slip.timingStartA ?? slip.timingStartB;
   167→    const end = slip.timingEndA ?? slip.timingEndB;
   168→    const fmt = (d: Date) => new Intl.DateTimeFormat("fa-IR", { hour: "2-digit", minute: "2-digit", timeZone: "Asia/Tehran" }).format(d);
   169→    const range = start && end ? ` | ساعت دقیق پیشنهادی ثبت: ${fmt(new Date(start))} تا ${fmt(new Date(end))} (تهران)` : "";
   170→    lines.push(`⏱️ بهترین زمان ثبت این تیپ: <b>${esc(winName)}</b>${range}`);
   171→  }
   172→  lines.push(`💰 سود تضمینی پوشش: <b>+${slip.guaranteedProfit.toFixed(1)}%</b> (مجموع استیک ۱۰۰ واحد)`);
   173→  lines.push(`🔐 هش راستی‌آزمایی SHA-256:\n<code>${slip.hash.slice(0, 32)}</code>`);
   174→  if (slip.espnUrl) lines.push(`🔎 <a href="${slip.espnUrl}">لینک راستی‌آزمایی رسمی ESPN</a>`);
   175→  lines.push("");
   176→  lines.push(`⚖️ ابزار تحلیل کمّی — مسئولیت استفاده با کاربر است.`);
   177→  return lines.join("\n");
   178→}

   259→export type PublishMode = "CARD" | "TEASER";
   260→
   261→/**
   262→ * انتشار کارت/تیزر به کانال VIP:
   263→ * تصویر فوتبالی دو تیم + پیام HTML. تیپ B در حالت TEASER سانسور می‌شود.
   264→ * v9.0 — فیلتر سختگیرانه سوپروایزر: بازی وتوشده مطلقاً به تلگرام ارسال نمی‌شود.
   265→ */
   266→export async function publishSlipToTelegram(slip: IssuedSlip, mode: PublishMode): Promise<{ ok: boolean; detail: string; preview: string }> {
   267→  const preview = mode === "CARD" ? buildCardMessage(slip) : buildTeaserMessage(slip);
   268→  if (!slip.supervisorApproved) {
   269→    await logOutbox(slip.id, mode, false, "VETOED_BY_SUPERVISOR");
   270→    return { ok: false, detail: "VETOED_BY_SUPERVISOR", preview };
   271→  }
   272→  const status = await telegramStatus();
   273→  if (!status.configured || !status.channelSet) {
   274→    await logOutbox(slip.id, mode, false, "TELEGRAM_NOT_CONFIGURED");
   275→    return { ok: false, detail: "TELEGRAM_NOT_CONFIGURED", preview };
   276→  }
   277→
   278→  const photo = mode === "TEASER" ? await renderSlipTeaserPNG(slip) : await renderSlipCardPNG(slip);
   279→
   280→  let result: { ok: boolean; detail: string };
   281→  if (photo) {
   282→    result = await tgSendPhoto(photo, preview, `slip-${slip.hash.slice(0, 10)}.png`);
   283→  } else {
   284→    // تصویر تولید نشد — پیام متنی HTML همچنان کامل ارسال می‌شود
   285→    result = await tgCall("sendMessage", { chat_id: await channelId(), text: preview, parse_mode: "HTML", disable_web_page_preview: true });
   286→  }
   287→
   288→  await logOutbox(slip.id, mode, result.ok, result.detail);
   289→  if (result.ok) {
   290→    await db.issuedSlip.update({
   291→      where: { id: slip.id },
   292→      data: mode === "CARD" ? { telegramCard: true } : { telegramTeaser: true },
   293→    });
   294→  }
   295→  return { ...result, preview };
   296→}

   298→/**
   299→ * اعلام خودکار نتیجه پس از FT:
   300→ * برنده → تصویر جام پیروزی | بازنده → تصویر رختکن بهم‌ریخته
   301→ */
   302→export async function sendSettlementNotification(slip: IssuedSlip): Promise<boolean> {
   303→  // v11.0 — سپر مطلق دفتر سایه: کارت وتوشده سوپروایزر هرگز نتیجه هم به کانال نمی‌رود
   304→  if (!slip.supervisorApproved) return false;
   305→  const status = await telegramStatus();
   306→  if (!status.configured || !status.channelSet) return false;
   307→  if (slip.homeGoals === null || slip.awayGoals === null) return false;
   308→
   309→  const caption = buildResultMessage(slip);
   310→
   311→  // v10.0 — تصویربرداری قطعی: برد طلایی همزمان (Double-Win) طراحی طلایی اختصاصی دارد
   312→  const photo = await renderSlipResultPNG(slip);
   313→
   314→  let ok = false;
   315→  if (photo) {
   316→    ok = (await tgSendPhoto(photo, caption, `${slip.status.toLowerCase()}-${slip.hash.slice(0, 10)}.png`)).ok;
   317→  } else {
   318→    ok = (await tgCall("sendMessage", { chat_id: await channelId(), text: caption, parse_mode: "HTML", disable_web_page_preview: true })).ok;
   319→  }
   320→
   321→  await logOutbox(slip.id, "RESULT", ok, ok ? "sent" : "send-failed");
   322→  if (ok) {
   323→    await db.issuedSlip.update({ where: { id: slip.id }, data: { telegramResult: true } });
   324→  }
   325→  return ok;
   326→}

   345→/** آمار وین‌ریت کانال — مستقیم از دیتابیس (بدون وابستگی چرخه‌ای) */
   346→async function channelWinRateStats() {
   347→  // v11.0 — فقط کارت‌های واقعی تاییدشده؛ تسویه کاغذی دفتر سایه در وین‌ریت کانال دخالت نمی‌کند
   348→  const settled = await db.issuedSlip.findMany({
   349→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, supervisorApproved: true },
   350→    select: { status: true, netProfit: true, stakeA: true, stakeB: true },
   351→  });
   352→  const st = { settled: settled.length, wonA: 0, wonB: 0, doubleWin: 0, middleTrap: 0, netProfit: 0, staked: 0 };
   353→  for (const s of settled) {
   354→    if (s.status === "WON_A") st.wonA += 1;
   355→    else if (s.status === "WON_B") st.wonB += 1;
   356→    else if (s.status === "DOUBLE_WIN") st.doubleWin += 1;
   357→    else if (s.status === "MIDDLE_TRAP") st.middleTrap += 1;
   358→    st.netProfit += s.netProfit ?? 0;
   359→    st.staked += s.stakeA + s.stakeB;
   360→  }
   361→  const winRatePct = st.settled > 0 ? Math.round(((st.settled - st.middleTrap) / st.settled) * 1000) / 10 : 0;
   362→  const netPct = st.staked > 0 ? Math.round((st.netProfit / st.staked) * 1000) / 10 : 0;
   363→  // ─── v11.0 — دفتر سایه: شفافیت counterfactual وتوها (خط جداگانه گزارش) ───
   364→  const shadowRows = await db.issuedSlip.findMany({
   365→    where: { supervisorApproved: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   366→    select: { status: true, netProfit: true },
   367→  });
   368→  const shadowWins = shadowRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   369→  const shadowNet = Math.round(shadowRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100;
   370→  const shadowWinRate = shadowRows.length > 0 ? Math.round((shadowWins / shadowRows.length) * 1000) / 10 : null;
   371→  return {
   372→    ...st,
   373→    winRatePct,
   374→    netPct,
   375→    netProfit: Math.round(st.netProfit * 100) / 100,
   376→    shadow: { settled: shadowRows.length, wins: shadowWins, winRatePct: shadowWinRate, netProfit: shadowNet },
   377→  };
   378→}

   380→/** پیام نوع دوم — گزارش تسویه + به‌روزرسانی وین‌ریت کل کانال */
   381→export function buildWinRateReportMessage(st: {
   382→  settled: number;
   383→  wonA: number;
   384→  wonB: number;
   385→  doubleWin: number;
   386→  middleTrap: number;
   387→  winRatePct: number;
   388→  netProfit: number;
   389→  netPct: number;
   390→}): string {
   391→  const pct = (n: number) => (st.settled > 0 ? `${Math.round((n / st.settled) * 1000) / 10}٪` : "۰٪");
   392→  const lines: string[] = [];
   393→  lines.push(`📊 <b>گزارش وین‌ریت کانال — تسویه ۱۰۰٪ خودکار ESPN</b>`);
   394→  lines.push(`🎯 وین‌ریت زوج‌تیپ‌های تاییدشده: <b>${st.winRatePct}٪</b> (${st.settled} کارت تسویه‌شده)`);
   395→  lines.push(`🟢 برد تیپ A: <b>${st.wonA}</b> (${pct(st.wonA)}) · برد تیپ B: <b>${st.wonB}</b> (${pct(st.wonB)})`);
   396→  lines.push(`🏆 برد طلایی همزمان (Double-Win Jackpot): <b>${st.doubleWin}</b> (${pct(st.doubleWin)})`);
   397→  lines.push(`🔴 تله میانی (ثبت صادقانه): <b>${st.middleTrap}</b> (${pct(st.middleTrap)})`);
   398→  lines.push(`💰 سود خالص تجمعی کانال: <b>${st.netProfit >= 0 ? "+" : ""}${st.netProfit.toFixed(1)}</b> واحد (${st.netPct >= 0 ? "+" : ""}${st.netPct}٪)`);
   399→  lines.push(`🔐 همه نتایج با اسکوربورد رسمی ESPN و هش SHA-256 راستی‌آزمایی می‌شوند.`);
   400→  return lines.join("\n");

   414→export async function previewDispatchQueue(): Promise<{ queue: DispatchQueueItem[]; autoDispatch: boolean; configured: boolean }> {
   415→  const [pendingCards, pendingResults, autoDispatch, tg] = await Promise.all([
   416→    db.issuedSlip.findMany({

   420→    }),
   421→    db.issuedSlip.findMany({
   422→      where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, telegramResult: false, supervisorApproved: true },
   423→      orderBy: { settledAt: "asc" },
   424→      take: 8,
   425→    }),
   426→    getAutoDispatch(),
   427→    telegramStatus(),
   428→  ]);
   429→
   430→  const queue: DispatchQueueItem[] = [];
   431→  for (const s of pendingCards) {
   432→    queue.push({
   433→      kind: "CARD",
   434→      slipId: s.id,
   435→      slipHash: s.hash,

   437→      message: buildCardMessage(s),

   441→  for (const s of pendingResults) {

   443→      kind: "RESULT",
   444→      slipId: s.id,
   445→      slipHash: s.hash,
   446→      matchLabel: `${s.homeTeam} ${s.homeGoals ?? "?"}−${s.awayGoals ?? "?"} ${s.awayTeam}`,
   447→      message: buildResultMessage(s),
   448→      status: s.status,
   449→    });
   450→  }
   451→  const st = await channelWinRateStats();
   452→  if (pendingResults.length > 0) {
   453→    queue.push({ kind: "WINRATE", slipId: null, slipHash: null, matchLabel: "گزارش وین‌ریت کانال", message: buildWinRateReportMessage(st) });
   454→  }
   455→  return { queue, autoDispatch, configured: tg.configured };

   458→/**
   459→ * صف ارسال خودکار دو مرحله‌ای (Auto-Dispatch Queue):
   460→ *   · پیام نوع اول (پیش از بازی): فقط کارت‌های تاییدشده سوپروایزر (وتوشده‌ها اسکیپ)

   465→export async function autoDispatchApprovedSlips(options?: {
   466→  maxCards?: number;
   467→  maxResults?: number;
   468→}): Promise<{
   469→  ok: boolean;
   470→  sentCards: number;
   471→  sentResults: number;
   472→  winRateSent: boolean;
   473→  skippedVetoed: number;
   474→  detail: string;
   475→}> {
   476→  const maxCards = Math.max(1, Math.min(12, options?.maxCards ?? 3));
   477→  const maxResults = Math.max(1, Math.min(12, options?.maxResults ?? 3));
   478→  const [tg, autoOn] = await Promise.all([telegramStatus(), getAutoDispatch()]);
   479→  const skippedVetoed = await db.issuedSlip.count({ where: { supervisorApproved: false, telegramCard: false } });
   480→
   481→  if (!tg.configured || !tg.channelSet) {
   482→    return { ok: false, sentCards: 0, sentResults: 0, winRateSent: false, skippedVetoed, detail: "TELEGRAM_NOT_CONFIGURED" };
   483→  }
   484→  if (!autoOn) {
   485→    return { ok: false, sentCards: 0, sentResults: 0, winRateSent: false, skippedVetoed, detail: "AUTO_DISPATCH_OFF" };
   486→  }
   487→
   488→  let sentCards = 0;
   489→  let sentResults = 0;
   490→
   491→  // ─── مرحله ۱: پست پیش از بازی — فقط کارت‌های 👑 تاییدشده سوپروایزر ───
   492→  const cards = await db.issuedSlip.findMany({
   493→    where: { status: { in: ["PENDING", "LIVE"] }, telegramCard: false, supervisorApproved: true },
   494→    orderBy: { kickoff: "asc" },
   495→    take: maxCards,
   496→  });
   497→  for (const s of cards) {
   498→    try {
   499→      const res = await publishSlipToTelegram(s, "CARD");
   500→      if (res.ok) sentCards += 1;
   501→    } catch (err) {
   502→      console.warn("auto dispatch card failed:", err);
   503→    }
   504→  }
   505→
   506→  // ─── مرحله ۲: پست تسویه‌حساب پس از بازی — برگه نتیجه FT ───
   507→  const results = await db.issuedSlip.findMany({
   508→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, telegramResult: false, supervisorApproved: true },
   509→    orderBy: { settledAt: "asc" },
   510→    take: maxResults,
   511→  });
   512→  for (const s of results) {
   513→    try {
   514→      const ok = await sendSettlementNotification(s);
   515→      if (ok) sentResults += 1;
   516→    } catch (err) {
   517→      console.warn("auto dispatch result failed:", err);
   518→    }
   519→  }
   520→
   521→  // ─── به‌روزرسانی وین‌ریت کل کانال — فقط وقتی نتیجه تازه‌ای مخابره شد ───

   523→  if (sentResults > 0) {

   528→    ok: sentCards + sentResults > 0,
   529→    sentCards,
   530→    sentResults,
   531→    winRateSent,

   533→    detail: sentCards + sentResults > 0 ? "dispatched" : "QUEUE_EMPTY",
   534→  };
