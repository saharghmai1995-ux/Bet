    43→async function fetchEspnCoreOddsCached(slug: string, eventId: string): Promise<EspnCoreOdds | null> {
    44→  const key = `${slug}-${eventId}`;
    45→  const hit = espnCoreCache.get(key);

    48→  if (espnCoreCache.size > 400) espnCoreCache.clear();
    49→  espnCoreCache.set(key, { at: Date.now(), core });
    50→  return core;

    53→/** بهترین خط 1X2 بین بوک‌میکرهای رسمی ESPN Core (Bet365/Unibet/William Hill/...) */
    54→function bestEspnCoreLine(core: EspnCoreOdds | null): { provider: string; home: number; draw: number; away: number } | null {
    55→  if (!core) return null;
    56→  let best: { provider: string; home: number; draw: number; away: number } | null = null;
    57→  const impliedSum = (l: { home: number; draw: number; away: number }) => 1 / l.home + 1 / l.draw + 1 / l.away;
    58→  for (const b of core.bookmakers) {
    59→    if (!b.home || !b.draw || !b.away) continue;
    60→    if (b.home <= 1.01 || b.draw <= 1.01 || b.away <= 1.01) continue;
    61→    const line = { provider: b.name, home: b.home, draw: b.draw, away: b.away };
    62→    if (!best || impliedSum(line) < impliedSum(best)) best = line; // کم‌حاشیه‌ترین خط
    63→  }
    64→  return best;
    65→}

   113→const marketSolverCache = new Map<string, MarketImpliedResult | null>();
   114→const MARKET_SOLVER_CACHE_MAX = 600;
   115→
   116→export function marketLambdasFor(odds: { oddsHome: number; oddsDraw: number; oddsAway: number; oddsOver25: number; oddsUnder25: number }): { home: number | null; away: number | null; fitError: number | null } {
   117→  const key = [odds.oddsHome, odds.oddsDraw, odds.oddsAway, odds.oddsOver25, odds.oddsUnder25].join("|");
   118→  if (!marketSolverCache.has(key)) {
   119→    if (marketSolverCache.size > MARKET_SOLVER_CACHE_MAX) marketSolverCache.clear();
   120→    marketSolverCache.set(
   121→      key,
   122→      solveMarketImpliedLambdas({
   123→        oddsHome: odds.oddsHome,
   124→        oddsDraw: odds.oddsDraw,
   125→        oddsAway: odds.oddsAway,
   126→        oddsOver25: odds.oddsOver25,
   127→        oddsUnder25: odds.oddsUnder25,
   128→      })
   129→    );
   130→  }
   131→  const solved = marketSolverCache.get(key) ?? null;
   132→  return solved
   133→    ? { home: solved.lambdaHome, away: solved.lambdaAway, fitError: solved.fitError }
   134→    : { home: null, away: null, fitError: null };
   135→}

   139→/** ضریب فرم از ۵ بازی اخیر واقعی (WWDLW) — بازه ۰.۷۸ تا ۱.۲۶ */
   140→function formFactor(form: string): number {
   141→  if (!form || form === "N/A") return 1.0;
   142→  const pts = [...form].reduce((a, ch) => a + (ch === "W" ? 3 : ch === "D" ? 1 : 0), 0);
   143→  return 0.78 + (pts / (form.length * 3)) * 0.48;
   144→}

   146→/** ضریب فصل از رکورد واقعی (مثل 3-4-2) — بازه ۰.۸۲ تا ۱.۱۸ */
   147→function seasonFactor(record: string | null): number {
   148→  const pct = recordToPointsPct(record);
   149→  if (pct === null) return 1.0;
   150→  return 0.82 + pct * 0.36;
   151→}

   154→function eloFromRecord(record: string | null): number {
   155→  const pct = recordToPointsPct(record);

   157→  return Math.round(Math.max(1400, Math.min(2100, 1500 + (pct - 0.5) * 800)));

   168→interface DerivedStrength {
   169→  lambdaHome: number;
   170→  lambdaAway: number;
   171→  homeAttack: number;
   172→  awayAttack: number;
   173→  homeDefense: number;

   183→  // v21.0 — مسیر سوم GLOBAL_ELO_FALLBACK (جدول Elo جهانی وقتی جدول رده‌بندی غایب است)
   184→  statsSource: "ESPN_STANDINGS" | "RECORD_FALLBACK" | "GLOBAL_ELO_FALLBACK";

   196→/**
   197→ * v6.0 — محاسبه امید گل (λ) از جدول رسمی رده‌بندی فصل جاری ESPN (Standings API بدون کلید)
   198→ * مسیر اولویت‌دار (قابل راستی‌آزمایی ۱۰۰٪):
   199→ *   α_i = GF_i / (GP_i × LeagueTeamAvg)   ← قدرت تهاجمی واقعی از گل زده فصل
   200→ *   β_i = GA_i / (GP_i × LeagueTeamAvg)   ← ضعف دفاعی واقعی از گل خورده فصل
   201→ *   λ_Home = α_Home × β_Away × LeagueTeamAvg × 1.12 (فرمول استاندارد دیکسون-کولز)
   202→ *   λ_Away = α_Away × β_Home × LeagueTeamAvg
   203→ * مسیر جایگزین شفاف (فقط وقتی جدول در دسترس نیست): رکورد واقعی W-D-L + فرم ۵ بازی.
   204→ * تعدیل جوی: بارش > ۲mm یا باد > ۳۰km/h → λ × ۰.۹۲ (−۸٪)
   205→ * v9.0 — export شد: مشترک با خط لوله ثبت خودکار بازی‌های گذشته (auto-issue)
   206→ */
   207→export function deriveStrength(ev: LiveMatchEvent, weather: LiveWeather | null, st: LeagueStandings | null): DerivedStrength {
   208→  const adverse = weather?.adverse ?? false;
   209→  const wAdj = adverse ? 0.92 : 1.0;
   210→
   211→  const homeEntry = findStandingsEntry(st, ev.homeTeamId, ev.homeTeam);
   212→  const awayEntry = findStandingsEntry(st, ev.awayTeamId, ev.awayTeam);
   213→  const lta = st && st.leagueTeamAvg > 0.5 ? st.leagueTeamAvg : null;
   214→
   215→  if (lta && homeEntry && awayEntry && homeEntry.gp >= 3 && awayEntry.gp >= 3) {
   216→    // ═══ v6.0 — مسیر واقعی: α/β از جدول رسمی فصل ═══
   217→    const alphaH = homeEntry.gf / (homeEntry.gp * lta);
   218→    const betaH = homeEntry.ga / (homeEntry.gp * lta);
   219→    const alphaA = awayEntry.gf / (awayEntry.gp * lta);
   220→    const betaA = awayEntry.ga / (awayEntry.gp * lta);
   221→
   222→    let lambdaHome = alphaH * betaA * lta * 1.12 * wAdj;
   223→    let lambdaAway = alphaA * betaH * lta * wAdj;
   224→    lambdaHome = clamp(lambdaHome, 0.3, 3.8);
   225→    lambdaAway = clamp(lambdaAway, 0.25, 3.6);
   226→
   227→    const homeElo = eloFromRecord(ev.homeRecord);
   228→    const awayElo = eloFromRecord(ev.awayRecord);
   229→
   230→    const homePower = clamp(alphaH, 0.5, 2.0);
   231→    const awayPower = clamp(alphaA, 0.5, 2.0);
   232→
   233→    return {
   234→      lambdaHome: r2(lambdaHome),
   235→      lambdaAway: r2(lambdaAway),
   236→      // قدرت‌های واقعی دیکسون-کولز — λ = α × β × leagueTeamAvg × homeAdv
   237→      homeAttack: r2(clamp(alphaH, 0.35, 2.3)),
   238→      awayAttack: r2(clamp(alphaA, 0.3, 2.2)),
   239→      homeDefense: r2(clamp(betaH, 0.45, 1.9)),
   240→      awayDefense: r2(clamp(betaA, 0.45, 1.9)),
   241→      homeElo,
   242→      awayElo,
   243→      homePpda: r2(clamp(12 - (homePower - 1) * 7, 8.5, 15)),
   244→      awayPpda: r2(clamp(12.4 - (awayPower - 1) * 7, 8.5, 15.5)),
   245→      homeFieldTilt: Math.round(clamp(50 + (homeElo - 1500) / 12, 42, 63)),
   246→      awayFieldTilt: Math.round(clamp(50 + (awayElo - 1500) / 12, 40, 62)),
   247→      weatherAdjusted: adverse,
   248→      statsSource: "ESPN_STANDINGS",
   249→      homeRank: homeEntry.rank || null,
   250→      homeGF: homeEntry.gf,
   251→      homeGA: homeEntry.ga,
   252→      homeGP: homeEntry.gp,
   253→      awayRank: awayEntry.rank || null,
   254→      awayGF: awayEntry.gf,
   255→      awayGA: awayEntry.ga,
   256→      awayGP: awayEntry.gp,
   257→      leagueTeamAvg: r2(lta),
   258→    };
   259→  }
   260→
   261→  // ═══ v21.0 — مسیر Elo جهانی: پایان λ کور ۱.۰۰–۰.۹۲ برای بازی‌های ملی ═══
   262→  // وقتی جدول رده‌بندی ESPN در دسترس نیست، امید گل از «اختلاف واقعی Elo دو تیم»
   263→  // (جدول GLOBAL_TEAM_ELO_AND_XG_RATINGS — ۱۲۰ تیم ملی + باشگاه‌های برتر) مشتق می‌شود.
   264→  // ژاپن × پاراگوئه دیگر با قرقیزستان × ویتنام عدد یکسان نمی‌گیرد!
   265→  const eloSolved = lambdaFromEloRatings(ev.homeTeam, ev.awayTeam, null, null);
   266→  if (eloSolved) {
   267→    // اولویت Elo جدول جهانی؛ تیم ناشناخته → Elo رکورد واقعی ESPN (۱۵۰۰ = متوسط)
   268→    const eloHome = lookupTeamRating(ev.homeTeam)?.elo ?? eloFromRecord(ev.homeRecord);
   269→    const eloAway = lookupTeamRating(ev.awayTeam)?.elo ?? eloFromRecord(ev.awayRecord);
   270→
   271→    const lambdaHome = clamp(eloSolved.lambdaHome * wAdj, 0.3, 3.8);
   272→    const lambdaAway = clamp(eloSolved.lambdaAway * wAdj, 0.25, 3.6);
   273→
   274→    const homeAttack = clamp(lambdaHome / (1.35 * 1.12), 0.35, 2.3);
   275→    const awayAttack = clamp(lambdaAway / 1.35, 0.3, 2.2);
   276→
   277→    const homeFieldTilt = Math.round(clamp(50 + (eloHome - 1500) / 12, 42, 63));
   278→    const awayFieldTilt = Math.round(clamp(50 + (eloAway - 1500) / 12, 40, 62));
   279→    const homePpda = r2(clamp(12 - (lambdaHome / 1.35 - 1) * 7, 8.5, 15));
   280→    const awayPpda = r2(clamp(12.4 - (lambdaAway / 1.35 - 1) * 7, 8.5, 15.5));
   281→
   282→    return {
   283→      lambdaHome: r2(lambdaHome),
   284→      lambdaAway: r2(lambdaAway),
   285→      homeAttack: r2(homeAttack),
   286→      awayAttack: r2(awayAttack),
   287→      homeDefense: 1.0,
   288→      awayDefense: 1.0,
   289→      homeElo: eloHome,
   290→      awayElo: eloAway,
   291→      homePpda,
   292→      awayPpda,
   293→      homeFieldTilt,
   294→      awayFieldTilt,
   295→      weatherAdjusted: adverse,
   296→      statsSource: "GLOBAL_ELO_FALLBACK",
   297→      homeRank: null,
   298→      homeGF: null,
   299→      homeGA: null,
   300→      homeGP: null,
   301→      awayRank: null,
   302→      awayGF: null,
   303→      awayGA: null,
   304→      awayGP: null,
   305→      leagueTeamAvg: null,
   306→    };
   307→  }
   308→
   309→  // ═══ مسیر شفاف جایگزین — فقط رکورد/فرم واقعی ESPN (جدول رده‌بندی و جدول Elo هم در دسترس نیستند) ═══
   310→  // ⚠️ v21.0: اگر λ نهایی این مسیر در بازه کور ۰.۹۰–۱.۰۵ بماند و خط بازار هم غایب باشد،
   311→  // گیت «No Blind Card» سوپروایزر این بازی را وتو می‌کند (کارت کور صادر نمی‌شود).
   312→  const homeStr = 0.55 * formFactor(ev.homeForm) + 0.45 * seasonFactor(ev.homeRecord);
   313→  const awayStr = 0.55 * formFactor(ev.awayForm) + 0.45 * seasonFactor(ev.awayRecord);
   314→
   315→  let lambdaHome = ev.avgGoals * 0.55 * homeStr * wAdj;
   316→  let lambdaAway = ev.avgGoals * 0.45 * awayStr * wAdj;
   317→
   318→  // همگرایی به محدوده معقول پواسون
   319→  lambdaHome = clamp(lambdaHome, 0.3, 3.8);
   320→  lambdaAway = clamp(lambdaAway, 0.25, 3.6);
   321→
   322→  // استخراج ضرایب حمله/دفاع سازگار با موتور دیکسون-کولز:
   323→  // λh = homeAttack × awayDefense × leagueAvg × 1.12 → دفاع‌ها = ۱، حمله = λ / ثابت
   324→  const homeAttack = clamp(lambdaHome / (1.35 * 1.12), 0.35, 2.3);
   325→  const awayAttack = clamp(lambdaAway / 1.35, 0.3, 2.2);
   326→
   327→  const homeElo = eloFromRecord(ev.homeRecord);
   328→  const awayElo = eloFromRecord(ev.awayRecord);
   329→
   330→  const homeFieldTilt = Math.round(clamp(50 + (homeElo - 1500) / 12, 42, 63));
   331→  const awayFieldTilt = Math.round(clamp(50 + (awayElo - 1500) / 12, 40, 62));
   332→  const homePpda = r2(clamp(12 - (homeStr - 1) * 7, 8.5, 15));
   333→  const awayPpda = r2(clamp(12.4 - (awayStr - 1) * 7, 8.5, 15.5));
   334→
   335→  return {
   336→    lambdaHome: r2(lambdaHome),
   337→    lambdaAway: r2(lambdaAway),
   338→    homeAttack: r2(homeAttack),
   339→    awayAttack: r2(awayAttack),
   340→    homeDefense: 1.0,
   341→    awayDefense: 1.0,
   342→    homeElo,
   343→    awayElo,
   344→    homePpda,
   345→    awayPpda,
   346→    homeFieldTilt,
   347→    awayFieldTilt,
   348→    weatherAdjusted: adverse,
   349→    statsSource: "RECORD_FALLBACK",
   350→    homeRank: null,
   351→    homeGF: null,
   352→    homeGA: null,
   353→    homeGP: null,
   354→    awayRank: null,
   355→    awayGF: null,
   356→    awayGA: null,
   357→    awayGP: null,
   358→    leagueTeamAvg: null,
   359→  };
   360→}

   362→function weatherString(w: LiveWeather | null): string {
   363→  if (!w) return "نامشخص - دیتای جوی در دسترس نیست";
   364→  const cond = w.precipMm > 2 ? "باران" : w.precipMm > 0.3 ? "نیمه‌بارانی" : w.windKmh > 30 ? "باد شدید" : "پاک";
   365→  return `${cond} - ${Math.round(w.tempC)}° · باد ${Math.round(w.windKmh)}km/h${w.precipMm > 0.3 ? ` · بارش ${w.precipMm.toFixed(1)}mm` : ""}`;
   366→}

   368→/** احتمالات مدل دیکسون-کولز از روی استحاله مشتق‌شده (برای ضرایب fallback و T-60) */
   369→export function modelProbs(d: DerivedStrength) {
   370→  return runDixonColes({
   371→    homeAttack: d.homeAttack,
   372→    awayAttack: d.awayAttack,
   373→    homeDefense: d.homeDefense,
   374→    awayDefense: d.awayDefense,
   375→    leagueAvg: d.leagueTeamAvg ?? 1.35,
   376→    homeAdvantage: 1.12,
   377→    rho: -0.08,
   378→    eloDiff: d.homeElo - d.awayElo,
   379→    eloWeight: 0.35,
   380→    homeAbsenceImpact: 0,
   381→    awayAbsenceImpact: 0,
   382→    timeDecayXi: 0.0065,
   383→    daysAgo: 21,

   389→function matchOddsEvent(ev: LiveMatchEvent, oddsEvents: OddsApiEvent[]): OddsApiEvent | null {
   390→  const direct = `${normTeam(ev.homeTeam)}::${normTeam(ev.awayTeam)}`;
   391→  const kick = Date.parse(ev.kickoff);
   392→  let best: OddsApiEvent | null = null;
   393→  let bestDist = Infinity;
   394→  for (const oe of oddsEvents) {
   395→    const same = oe.matchKey === direct || `${normTeam(oe.homeTeam)}::${normTeam(oe.awayTeam)}` === `${normTeam(ev.awayTeam)}::${normTeam(ev.homeTeam)}`;
   396→    if (!same) continue;
   397→    const dist = Math.abs(Date.parse(oe.commenceTime) - kick);
   398→    if (dist < bestDist) {
   399→      bestDist = dist;
   400→      best = oe;
   401→    }
   402→  }
   403→  // تطبیق فقط برای بازی‌های نزدیک (±۲۴h) — جلوگیری از جفت‌شدن بازی‌های دور
   404→  return best && bestDist < 24 * 3600 * 1000 ? best : null;
   405→}

   415→  /** تا ۳ ردیف بوک‌میکر با کلیدهای کنونیکال (Pinnacle = کم‌حاشیه‌ترین) */
   416→  books: { bookmaker: "PINNACLE" | "SOFT_A" | "SOFT_B" | "POLYMARKET"; bookmakerLabel: string; home: number; draw: number; away: number; over25: number | null; under25: number | null }[];

   419→/**
   420→ * v22.0 — ترکیب ضرایب زنده بوک‌میکرها با بازار آزاد Polymarket (Zero-Vig):
   421→ * خط Polymarket به‌عنوان ردیف بوک‌میکر مستقل اضافه می‌شود و بهترین قیمت هر نتیجه
   422→ * بین بوک‌میکرهای رسمی + بازار آزاد انتخاب می‌گردد؛ در نبود بوک‌میکر رسمی،
   423→ * خط کامل سه‌گانه Polymarket تنها منبع واقعی است (source=POLYMARKET).
   424→ */
   425→function resolveOdds(
   426→  ev: LiveMatchEvent,
   427→  oddsEv: OddsApiEvent | null,
   428→  probs: ReturnType<typeof modelProbs>,
   429→  poly?: { home: number; draw: number | null; away: number } | null
   430→): ResolvedOdds {
   431→  // ─── ۱) ضرایب زنده The Odds API — بهترین قیمت هر انتخاب + سه بوک نماینده ───
   432→  if (oddsEv && oddsEv.books.length > 0) {
   433→    const booksSorted = [...oddsEv.books].sort(
   434→      (a, b) => 1 / a.line.home + 1 / a.line.draw + 1 / a.line.away - (1 / b.line.home + 1 / b.line.draw + 1 / b.line.away)
   435→    );
   436→    const canon = ["PINNACLE", "SOFT_A", "SOFT_B"] as const;
   437→    const books: ResolvedOdds["books"] = booksSorted.slice(0, 3).map((b, i) => ({
   438→      bookmaker: canon[i],
   439→      bookmakerLabel: `${b.label}${i === 0 ? " (Sharp)" : ""}`,
   440→      home: b.line.home,
   441→      draw: b.line.draw,
   442→      away: b.line.away,
   443→      over25: b.line.over25,
   444→      under25: b.line.under25,
   445→    }));
   446→    if (poly && poly.home > 1 && poly.away > 1) {
   447→      books.push({
   448→        bookmaker: "POLYMARKET",
   449→        bookmakerLabel: "Polymarket (Zero-Vig)",
   450→        home: poly.home,
   451→        draw: poly.draw ?? fairOdds(probs.pDraw, 1.06),
   452→        away: poly.away,
   453→        over25: null,
   454→        under25: null,
   455→      });
   456→    }
   457→    const best = (sel: (b: ResolvedOdds["books"][number]) => number | null): number => {
   458→      const vals = books.map(sel).filter((v): v is number => v !== null && v >= 1.01);
   459→      return vals.length > 0 ? Math.max(...vals) : 0;
   460→    };
   461→    const oddsHome = best((b) => b.home);
   462→    const oddsDraw = best((b) => b.draw);
   463→    const oddsAway = best((b) => b.away);
   464→    let oddsOver25 = best((b) => b.over25);
   465→    let oddsUnder25 = best((b) => b.under25);
   466→    if (oddsOver25 <= 1.01 || oddsUnder25 <= 1.01) {
   467→      oddsOver25 = fairOdds(probs.pOver25, 1.06);
   468→      oddsUnder25 = fairOdds(probs.pUnder25, 1.06);
   469→    }
   470→    return { oddsHome, oddsDraw, oddsAway, oddsOver25, oddsUnder25, odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06), source: "ODDS_API", books };
   471→  }
   472→
   473→  // ─── ۲) خط پایه ESPN (معمولاً DraftKings) + ترکیب Polymarket ───
   474→  if (ev.espnOdds && ev.espnOdds.home && ev.espnOdds.draw && ev.espnOdds.away) {
   475→    const o = ev.espnOdds;
   476→    const h = o.home as number;
   477→    const d2 = o.draw as number;
   478→    const a2 = o.away as number;
   479→    const books: ResolvedOdds["books"] = [
   480→      {
   481→        bookmaker: "SOFT_A",
   482→        bookmakerLabel: `${o.provider} (ESPN)`,
   483→        home: h,
   484→        draw: d2,
   485→        away: a2,
   486→        over25: o.over25 ?? fairOdds(probs.pOver25, 1.06),
   487→        under25: o.under25 ?? fairOdds(probs.pUnder25, 1.06),
   488→      },
   489→    ];
   490→    if (poly && poly.home > 1 && poly.away > 1) {
   491→      books.push({
   492→        bookmaker: "POLYMARKET",
   493→        bookmakerLabel: "Polymarket (Zero-Vig)",
   494→        home: poly.home,
   495→        draw: poly.draw ?? d2,
   496→        away: poly.away,
   497→        over25: o.over25 ?? fairOdds(probs.pOver25, 1.06),
   498→        under25: o.under25 ?? fairOdds(probs.pUnder25, 1.06),
   499→      });
   500→    }
   501→    const best = (sel: (b: ResolvedOdds["books"][number]) => number): number => Math.max(...books.map(sel));
   502→    return {
   503→      oddsHome: best((b) => b.home),
   504→      oddsDraw: best((b) => b.draw),
   505→      oddsAway: best((b) => b.away),
   506→      oddsOver25: books[0].over25 ?? fairOdds(probs.pOver25, 1.06),
   507→      oddsUnder25: books[0].under25 ?? fairOdds(probs.pUnder25, 1.06),
   508→      odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06),
   509→      source: "ESPN_DRAFTKINGS",
   510→      books,
   511→    };
   512→  }
   513→
   514→  // ─── ۲.۵) v22.0 — بازار آزاد Polymarket (Zero-Vig) — تنها منبع واقعی بدون بوک‌میکر رسمی ───
   515→  if (poly && poly.home > 1 && poly.away > 1 && poly.draw && poly.draw > 1) {
   516→    return {
   517→      oddsHome: poly.home,
   518→      oddsDraw: poly.draw,
   519→      oddsAway: poly.away,
   520→      oddsOver25: fairOdds(probs.pOver25, 1.06),
   521→      oddsUnder25: fairOdds(probs.pUnder25, 1.06),
   522→      odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06),
   523→      source: "POLYMARKET",
   524→      books: [
   525→        {
   526→          bookmaker: "POLYMARKET",
   527→          bookmakerLabel: "Polymarket (Zero-Vig)",
   528→          home: poly.home,
   529→          draw: poly.draw,
   530→          away: poly.away,
   531→          over25: null,
   532→          under25: null,
   533→        },
   534→      ],
   535→    };
   536→  }
   537→
   538→  // ─── ۳) Fallback مدل — ضرایب منصفانه با حاشیه ۶٪ ───
   539→  return {
   540→    oddsHome: fairOdds(probs.pHome, 1.06),
   541→    oddsDraw: fairOdds(probs.pDraw, 1.06),
   542→    oddsAway: fairOdds(probs.pAway, 1.06),
   543→    oddsOver25: fairOdds(probs.pOver25, 1.06),
   544→    oddsUnder25: fairOdds(probs.pUnder25, 1.06),
   545→    odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06),
   546→    source: "ENGINE",
   547→    books: [],
   548→  };
   549→}

   553→async function weatherFor(city: string | null): Promise<LiveWeather | null> {
   554→  if (!city) return null;
   555→  const coords = cityCoords(city);
   556→  if (!coords) return null;
   557→  const hit = cache.weather.get(city);
   558→  if (hit && Date.now() - hit.at < WEATHER_TTL_MS) return hit.w;
   559→  const w = await fetchWeather(coords[0], coords[1], city);
   560→  cache.weather.set(city, { w, at: Date.now() });
   561→  return w;
   562→}

   564→async function poolWeather(cities: (string | null)[], limit = 3): Promise<Map<string, LiveWeather | null>> {
   565→  const unique = [...new Set(cities.filter((c): c is string => Boolean(c)))];
   566→  const out = new Map<string, LiveWeather | null>();
   567→  const queue = [...unique];
   568→  const workers = Array.from({ length: Math.min(limit, queue.length) }, async () => {
   569→    for (;;) {
   570→      const c = queue.shift();
   571→      if (c === undefined) return;
   572→      out.set(c, await weatherFor(c));
   573→    }
   574→  });
   575→  await Promise.all(workers);
   576→  return out;
   577→}
