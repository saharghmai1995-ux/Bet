    57→/** متن آماده کپی تیپ — v8.0 (تک‌تیپ طلایی سوپروایزر + فرمول صحیح سود) */
    58→export function cardCopyText(v: FlashcardView): string {
    59→  const p = v.pair;
    60→  const sup = v.supervisor;
    61→  const head = `⚽ فوتبال‌کوانت v8.0 Supervisor Quant — ${v.match.homeTeam} مقابل ${v.match.awayTeam} (${v.match.league})`;
    62→  const crownLine = sup
    63→    ? `👑 تک‌تیپ تاییدشده سوپروایزر — امتیاز مونت‌کارلو: ${sup.score100}/۱۰۰${p.archetype ? ` (${p.archetype.short})` : ""}`
    64→    : p.archetype
    65→      ? `📐 الگو: ${p.archetype.short}`
    66→      : "";
    67→  const profitLine =
    68→    p.guaranteedMarginPct > 0
    69→      ? `💰 سود تضمینی: +${p.guaranteedMarginPct.toFixed(1)}٪ (+${faNum(p.equalProfit.profit, 1)} واحد سود خالص)`
    70→      : `🛡️ پوشش بیمه سرمایه — بازگشت وجه در سناریوی پایه`;
    71→  return [
    72→    head,
    73→    crownLine,
    74→    `🔹 تیپ A (ضریب ${p.totalOddsA.toFixed(2)}): ${p.labelA} ⬅️ مبلغ شرط: ${faNum(p.equalProfit.stakeA, 1)} واحد`,
    75→    `🟢 تیپ B مکمل (ضریب ${p.totalOddsB.toFixed(2)}): ${p.labelB} ⬅️ مبلغ شرط: ${faNum(p.equalProfit.stakeB, 1)} واحد`,
    76→    profitLine,
    77→    `✅ پوشش سناریوها: ${(p.archetype ? p.archetype.coveragePct * 100 : (1 - p.jointStates.bothLose) * 100).toFixed(0)}٪ | ریسک باخت همزمان: ${(p.jointStates.bothLose * 100).toFixed(1)}%`,
    78→    `⚖️ ${p.verdict}`,
    79→  ]
    80→    .filter(Boolean)
    81→    .join("\n");
    82→}
