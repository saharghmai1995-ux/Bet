    46→interface GradedLeg {
    47→  key: string;
    48→  won: boolean;
    49→  /** بازار قابل تسویه از اسکوربورد نیست (کرنر/کارت/نیمه اول) */
    50→  offBoard: boolean;

    52→  notFinal: boolean;

    56→function legCovers(key: string, hg: number, ag: number): boolean {
    57→  return selectionCoversCell(key, hg, ag);

    60→/**
    61→ * ارزیابی همه شرط‌های یک تیپ با نتایج واقعی ESPN.
    62→ * هر شرط ممکن است روی بازی دیگری باشد (میکس دو بازی) → امتیاز همان بازی از نقشه اسکوربورد.
    63→ */
    64→function gradeLegs(
    65→  legs: SettlementLegDTO[],
    66→  scoreOf: (externalId: string) => { hg: number; ag: number; state: string } | undefined,
    67→  fallbackExternalId: string | null
    68→): { graded: GradedLeg[]; allWon: boolean; anyLost: boolean; hasOffBoard: boolean; allFinal: boolean } {
    69→  const graded: GradedLeg[] = [];
    70→  let allWon = true;
    71→  let anyLost = false;
    72→  let hasOffBoard = false;
    73→  let allFinal = true;
    74→
    75→  for (const leg of legs) {
    76→    const ext = leg.matchExternalId || fallbackExternalId;
    77→    const label = leg.matchLabel || ext || "بازی نامشخص";
    78→    if (!ext || isOffMatrixKey(leg.key)) {
    79→      hasOffBoard = true;
    80→      graded.push({ key: leg.key, won: false, offBoard: true, notFinal: false, matchLabel: label });
    81→      continue;
    82→    }
    83→    const score = scoreOf(ext);
    84→    if (!score) {
    85→      // بازیِ این شرط در پنجره امروز/دیروز ESPN پیدا نشد
    86→      allFinal = false;
    87→      graded.push({ key: leg.key, won: false, offBoard: false, notFinal: true, matchLabel: label });
    88→      continue;
    89→    }
    90→    if (score.state !== "post") {
    91→      allFinal = false;
    92→      graded.push({ key: leg.key, won: false, offBoard: false, notFinal: true, matchLabel: label });
    93→      continue;
    94→    }
    95→    const won = legCovers(leg.key, score.hg, score.ag);
    96→    if (!won) {
    97→      allWon = false;
    98→      anyLost = true;
    99→    }
   100→    graded.push({ key: leg.key, won, offBoard: false, notFinal: false, matchLabel: label });
   101→  }
   102→
   103→  // شرط سینتتیک خارج اسکوربورد اگر تنها مشکل باشد → MANUAL_REVIEW؛ در غیر این صورت فقط تیپ‌های قابل ارزیابی لحاظ می‌شوند
   104→  const evaluable = graded.filter((g) => !g.offBoard);
   105→  const evaluableAllFinal = evaluable.length > 0 && evaluable.every((g) => !g.notFinal);
   106→  const evaluableAllWon = evaluable.length > 0 && evaluable.every((g) => g.won);
   107→  const evaluableAnyLost = evaluable.some((g) => !g.won);
   108→
   109→  return {
   110→    graded,
   111→    allWon: evaluableAllWon,
   112→    anyLost: evaluableAnyLost,
   113→    hasOffBoard,
   114→    allFinal: evaluableAllFinal,
   115→  };
   116→}

   131→/** نقشه اسکوربورد ESPN امروز+دیروز — بر اساس externalId و تطبیق نام تیم */
   132→async function loadEspnScoreWindow(): Promise<{ byId: Map<string, EspnScoreInfo>; byName: Map<string, EspnScoreInfo>; count: number }> {
   133→  const events: LiveMatchEvent[] = await fetchEspnEvents(-RESULT_WINDOW_DAYS, 0);
   134→  const byId = new Map<string, EspnScoreInfo>();
   135→  const byName = new Map<string, EspnScoreInfo>();
   136→  for (const ev of events) {
   137→    const info: EspnScoreInfo = {
   138→      hg: ev.homeScore ?? 0,
   139→      ag: ev.awayScore ?? 0,
   140→      state: ev.statusState,
   141→      statusDetail: ev.statusDetail,
   142→      kickoff: ev.kickoff,
   143→      homeTeam: ev.homeTeam,
   144→      awayTeam: ev.awayTeam,
   145→      league: ev.leagueName,
   146→    };
   147→    byId.set(ev.externalId, info);
   148→    // کلید تطبیق نامی — برای کارت‌هایی که شناسه ESPN ندارند
   149→    const nk = `${normTeam(ev.homeTeam)}::${normTeam(ev.awayTeam)}::${new Date(ev.kickoff).toISOString().slice(0, 10)}`;
   150→    if (!byName.has(nk)) byName.set(nk, info);
   151→  }
   152→  return { byId, byName, count: events.length };
   153→}

   161→/** فرمول سود مستند در سند ارتقا — مجموع استیک ۱۰۰ واحد */
   162→function settlementProfit(
   163→  status: Exclude<SlipStatus, "PENDING" | "LIVE" | "MANUAL_REVIEW" | "VOID">,
   164→  stakeA: number,
   165→  stakeB: number,
   166→  oddsA: number,
   167→  oddsB: number
   168→): { netProfit: number; profitA: number; profitB: number } {
   169→  const r1 = (v: number) => Math.round(v * 100) / 100;
   170→  switch (status) {
   171→    case "WON_A":
   172→      // 🟢 برد تیپ A: سود خالص = stakeA × oddsA − ۱۰۰
   173→      return { netProfit: r1(stakeA * oddsA - DUTCH_TOTAL), profitA: r1(stakeA * oddsA - DUTCH_TOTAL), profitB: r1(-stakeB) };
   174→    case "WON_B":
   175→      // 🟢 برد تیپ B: سود خالص = stakeB × oddsB − ۱۰۰
   176→      return { netProfit: r1(stakeB * oddsB - DUTCH_TOTAL), profitA: r1(-stakeA), profitB: r1(stakeB * oddsB - DUTCH_TOTAL) };
   177→    case "DOUBLE_WIN":
   178→      // 🏆 برد طلایی هر دو تیپ: جمع سود هر دو
   179→      return {
   180→        netProfit: r1(stakeA * oddsA + stakeB * oddsB - DUTCH_TOTAL),
   181→        profitA: r1(stakeA * oddsA - stakeA),
   182→        profitB: r1(stakeB * oddsB - stakeB),
   183→      };
   184→    case "MIDDLE_TRAP":
   185→      // 🔴 باخت همزمان — ثبت صادقانه
   186→      return { netProfit: r1(-DUTCH_TOTAL), profitA: r1(-stakeA), profitB: r1(-stakeB) };
   187→  }
   188→}

   193→export async function settleFinishedSlips(options?: { notify?: boolean }): Promise<SettlementSyncResultDTO> {
   194→  const t0 = Date.now();
   195→  const notify = options?.notify !== false;
   196→  const settled: SettlementSyncResultDTO["settled"] = [];
   197→  let liveCount = 0;
   198→  let notified = 0;
   199→  let ftFound = 0;
   200→  /** v9.0 — سقف اعلام تصویری هر پاس (تولید تصویر ~۲۰ ثانیه هر پیام — حفظ تأخیر سینک) */
   201→  const NOTIFY_CAP = 4;
   202→  let notifyBudget = NOTIFY_CAP;
   203→
   204→  const [{ byId, byName, count }, pendingSlips] = await Promise.all([
   205→    loadEspnScoreWindow(),
   206→    db.issuedSlip.findMany({
   207→      where: { status: { in: ["PENDING", "LIVE"] } },
   208→      orderBy: { createdAt: "asc" },
   209→    }),
   210→  ]);
   211→
   212→  const scoreOf = (externalId: string) => {
   213→    const direct = byId.get(externalId);
   214→    if (direct) return { hg: direct.hg, ag: direct.ag, state: direct.state };
   215→    return undefined;
   216→  };
   217→
   218→  for (const slip of pendingSlips) {
   219→    // ─── تطبیق کارت با رخداد ESPN: اولویت externalId، سپس نام تیم‌ها + روز کیک‌آف ───
   220→    let info: EspnScoreInfo | undefined = slip.externalId ? byId.get(slip.externalId) : undefined;
   221→    if (!info && slip.externalId) {
   222→      // شناسه ESPN ممکن است پس از به‌روزرسانی رخداد تغییر کرده باشد → تطبیق نامی
   223→      const dayKey = new Date(slip.kickoff).toISOString().slice(0, 10);
   224→      info = byName.get(`${normTeam(slip.homeTeam)}::${normTeam(slip.awayTeam)}::${dayKey}`);
   225→    }
   226→
   227→    // ─── بازی در جریان → برچسب LIVE ───
   228→    if (info && info.state === "in") {
   229→      if (slip.status !== "LIVE") {
   230→        await db.issuedSlip.update({ where: { id: slip.id }, data: { status: "LIVE" } });
   231→      }
   232→      liveCount += 1;
   233→      continue;
   234→    }
   235→
   236→    // ─── مسابقه به تعویق افتاده/لغو شده → VOID شفاف ───
   237→    if (info && /postpon|cancell|suspend|abandon|forfeit/i.test(info.statusDetail)) {
   238→      await db.issuedSlip.update({
   239→        where: { id: slip.id },
   240→        data: {
   241→          status: "VOID",
   242→          settleNote: `ESPN وضعیت «${info.statusDetail}» را ثبت کرد — تسویه انجام نمی‌شود (VOID)`,
   243→        },
   244→      });
   245→      continue;
   246→    }
   247→
   248→    if (info && info.state === "post" && (info.hg > 0 || info.ag > 0 || /ft|full|agg|pens/i.test(info.statusDetail))) {
   249→      ftFound += 1;
   250→      const legsA = safeParseLegs(slip.legsA);
   251→      const legsB = safeParseLegs(slip.legsB);
   252→      const gradedA = gradeLegs(legsA, scoreOf, slip.externalId);
   253→      const gradedB = gradeLegs(legsB, scoreOf, slip.externalId);
   254→
   255→      let status: SlipStatus;
   256→      let note: string;
   257→
   258→      if (!gradedA.allFinal || !gradedB.allFinal) {
   259→        // شرطی روی بازی دومی است که هنوز FT نشده — صبورانه در انتظار می‌مانیم
   260→        if (Date.now() - new Date(slip.kickoff).getTime() > VOID_AFTER_HOURS * 3600_000) {
   261→          status = "VOID";
   262→          note = "بازی مکمل در پنجره ESPN یافت نشد — احتمالاً به تعویق افتاده (VOID)";
   263→        } else {
   264→          continue;
   265→        }
   266→      } else if (gradedA.hasOffBoard || gradedB.hasOffBoard) {
   267→        // بازار کرنر/کارت/نیمه اول از اسکوربورد قابل تسویه نیست — شفاف اعلام می‌شود
   268→        status = "MANUAL_REVIEW";
   269→        note = "شامل بازار خارج اسکوربورد ESPN (کرنر/کارت/نیمه اول) — تسویه دستی لازم است";
   270→      } else {
   271→        const aWon = gradedA.allWon;
   272→        const bWon = gradedB.allWon;
   273→        if (aWon && bWon) {
   274→          status = "DOUBLE_WIN";
   275→          note = "🏆 نتیجه در خانه‌های مشترک هر دو تیپ — برد طلایی (Jackpot)";
   276→        } else if (aWon) {
   277→          status = "WON_A";
   278→          note = "🟢 تیپ A برد — تسویه خودکار با نتیجه واقعی ESPN";
   279→        } else if (bWon) {
   280→          status = "WON_B";
   281→          note = "🟢 تیپ B برد — تسویه خودکار با نتیجه واقعی ESPN";
   282→        } else {
   283→          status = "MIDDLE_TRAP";
   284→          note = "🔴 تله میانی — هر دو تیپ باختند؛ ثبت صادقانه و شفاف";
   285→        }
   286→      }
   287→
   288→      if ((status as string) === "PENDING" || (status as string) === "LIVE") continue;
   289→
   290→      const hg = info.hg;
   291→      const ag = info.ag;
   292→      const prof =
   293→        status === "WON_A" || status === "WON_B" || status === "DOUBLE_WIN" || status === "MIDDLE_TRAP"
   294→          ? settlementProfit(status, slip.stakeA, slip.stakeB, slip.oddsA, slip.oddsB)
   295→          : { netProfit: null, profitA: null, profitB: null };
   296→
   297→      await db.issuedSlip.update({
   298→        where: { id: slip.id },
   299→        data: {
   300→          status,
   301→          homeGoals: hg,
   302→          awayGoals: ag,
   303→          profitA: prof.profitA,
   304→          profitB: prof.profitB,
   305→          netProfit: prof.netProfit,
   306→          settledAt: new Date(),
   307→          settleNote: note,
   308→        },
   309→      });
   310→
   311→      // ─── v10.0 — ژورنال + دفتر دکترین: سیستم از هر تسویه یاد می‌گیرد ───
   312→      try {
   313→        const fresh = await db.issuedSlip.findUnique({ where: { id: slip.id } });
   314→        if (fresh) await recordSettlementLearning(fresh);
   315→      } catch (err) {
   316→        console.warn("doctrine learning failed:", err);
   317→      }
   318→
   319→      settled.push({
   320→        id: slip.id,
   321→        matchLabel: `${slip.homeTeam} ${hg}−${ag} ${slip.awayTeam}`,
   322→        status,
   323→        netProfit: prof.netProfit ?? 0,
   324→      });
   325→
   326→      // ─── اعلام نتیجه در کانال تلگرام VIP (تصویر جام / رختکن بهم‌ریخته) ───
   327→      if (notify && notifyBudget > 0 && status !== "VOID" && status !== "MANUAL_REVIEW" && !slip.telegramResult) {
   328→        try {
   329→          const { sendSettlementNotification } = await import("./telegram-service");
   330→          const updated = await db.issuedSlip.findUnique({ where: { id: slip.id } });
   331→          if (updated) {
   332→            notifyBudget -= 1;
   333→            const ok = await sendSettlementNotification(updated);
   334→            if (ok) notified += 1;
   335→          }
   336→        } catch (err) {
   337→          console.warn("telegram settlement notify failed:", err);
   338→        }
   339→      }
   340→      continue;
   341→    }
   342→
   343→    // ─── رخدادی در ESPN نیست → VOID پس از ۳۶ ساعت (به تعویق/لغو) ───
   344→    if (!info && Date.now() - new Date(slip.kickoff).getTime() > VOID_AFTER_HOURS * 3600_000) {
   345→      await db.issuedSlip.update({
   346→        where: { id: slip.id },
   347→        data: {
   348→          status: "VOID",
   349→          settleNote: "نتیجه در اسکوربورد ESPN (امروز+دیروز) یافت نشد — بازی به تعویق افتاده یا خارج پنجره (VOID)",
   350→        },
   351→      });
   352→    }
   353→  }
   354→
   355→  // ─── ۴.۶) v9.0 — گزارش وین‌ریت کل کانال پس از تسویه تازه (پیام نوع دوم) ───
   356→  if (notify && settled.length > 0) {
   357→    try {
   358→      const { sendWinRateReport } = await import("./telegram-service");
   359→      await sendWinRateReport();
   360→    } catch (err) {
   361→      console.warn("win-rate report failed:", err);
   362→    }
   363→  }
   364→
   365→  // ─── ۴.۶) تلاش مجدد اعلام نتایج ناموفق (مثلاً تصویر 429 یا قطعی تلگرام) — سقف ۵ در هر اجرا ───
   366→  if (notify) {
   367→    try {
   368→      const unnotified = await db.issuedSlip.findMany({
   369→        where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, telegramResult: false },
   370→        orderBy: { settledAt: "asc" },
   371→        take: 5,
   372→      });
   373→      if (unnotified.length > 0) {
   374→        const { sendSettlementNotification } = await import("./telegram-service");
   375→        for (const s of unnotified) {
   376→          try {
   377→            const ok = await sendSettlementNotification(s);
   378→            if (ok) notified += 1;
   379→          } catch (err) {
   380→            console.warn("settlement notify retry failed:", err);
   381→          }
   382→        }
   383→      }
   384→    } catch (err) {
   385→      console.warn("settlement notify retry pass failed:", err);
   386→    }
   387→  }
   388→
   389→  return {
   390→    success: true,
   391→    checkedEspn: count,
   392→    ftFound,
   393→    settled,
   394→    live: liveCount,
   395→    notified,
   396→    elapsedMs: Date.now() - t0,
   397→  };
   398→}

   411→/** استیک داچینگ سود برابر روی مجموع ۱۰۰ واحد — مبنای فرمول سند ارتقا */
   412→export function equalProfitStakes(oddsA: number, oddsB: number): { stakeA: number; stakeB: number } {
   413→  const invA = 1 / Math.max(1.01, oddsA);
   414→  const invB = 1 / Math.max(1.01, oddsB);
   415→  const sum = invA + invB;
   416→  const r1 = (v: number) => Math.round(v * 10) / 10;
   417→  return { stakeA: r1((DUTCH_TOTAL * invA) / sum), stakeB: r1((DUTCH_TOTAL * invB) / sum) };
   418→}

   420→/** هش SHA-256 سند کارت — قابل راستی‌آزمایی */
   421→export function slipHash(input: {
   422→  externalId: string | null;
   423→  homeTeam: string;
   424→  awayTeam: string;
   425→  kickoff: string;
   426→  labelA: string;
   427→  labelB: string;
   428→  oddsA: number;
   429→  oddsB: number;
   430→  stakeA: number;
   431→  stakeB: number;
   432→}): string {
   433→  const canonical = [
   434→    input.externalId ?? "NO-ESPN-ID",
   435→    input.homeTeam,
   436→    input.awayTeam,
   437→    new Date(input.kickoff).toISOString(),
   438→    input.labelA,
   439→    input.labelB,
   440→    input.oddsA.toFixed(2),
   441→    input.oddsB.toFixed(2),
   442→    input.stakeA.toFixed(1),
   443→    input.stakeB.toFixed(1),
   444→  ].join("|");
   445→  return crypto.createHash("sha256").update(canonical, "utf8").digest("hex");
   446→}

   448→/** خانه‌های پوشش هر تیپ روی ماتریس ۶×۶ (برای کارت تلگرام و نمایش کارنامه) */
   449→function coveredCells(lambdaHome: number, lambdaAway: number, legs: SettlementLegDTO[], externalId: string | null): string[] {
   450→  const matrix = buildScoreMatrix(lambdaHome || 1.35, lambdaAway || 1.25);
   451→  const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   452→  const cells: string[] = [];
   453→  for (let h = 0; h < matrix.length; h++) {
   454→    for (let a = 0; a < matrix[h].length; a++) {
   455→      if (keys.length > 0 && keys.every((k) => legCovers(k, h, a))) cells.push(`${h}-${a}`);
   456→    }
   457→  }
   458→  void externalId;
   459→  return cells.slice(0, 12);
   460→}

   464→/** احتمال مدل برای یک تیپ از ماتریس امتیاز دیکسون-کولز (جمع خانه‌های پوشش) */
   465→function legProbability(lambdaHome: number, lambdaAway: number, legs: SettlementLegDTO[]): number | null {
   466→  const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   467→  if (keys.length === 0) return null;
   468→  const matrix = buildScoreMatrix(lambdaHome || 1.35, lambdaAway || 1.25);
   469→  let p = 0;
   470→  for (let h = 0; h < matrix.length; h++) {
   471→    for (let a = 0; a < matrix[h].length; a++) {
   472→      if (keys.every((k) => legCovers(k, h, a))) p += matrix[h][a];
   473→    }
   474→  }
   475→  return Math.round(p * 10000) / 10000;
   476→}

   478→/**
   479→ * ارزش عددی نسبت به خط مرجع (سوال مستقیم کارفرما: «چرا عددی ندارد؟»):
   480→ *   edge٪ = (prob مدل × ضریب ثبت − ۱) × ۱۰۰
   481→ * خط مرجع: پیناکل وقتی خط شارپ واقعی موجود است؛ وگرنه خط منصفانه موتور (MARKET_FAIR)
   482→ * خروجی همیشه عدد دارد — «—» فقط برای بازار خارج ماتریس (کرنر/کارت).
   483→ */
   484→function computeEdge(modelProb: number | null, odds: number): number | null {
   485→  if (modelProb === null || modelProb <= 0 || odds <= 1) return null;
   486→  return Math.round((modelProb * odds - 1) * 1000) / 10;
   487→}

   490→ * صدور کارت به کارنامه — Idempotent با هش SHA-256.
   491→ * اگر کارت با همین هش قبلاً صادر شده باشد، همان ردیف بازگردانده می‌شود.
   492→ * v9.0: نگهداری λ ترکیبی نهایی + کد الگو (حلقه کالیبراسیون) + نشان تایید سوپروایزر
   493→ * و پرچم ثبت خودکار سرور (autoIssued) برای خط لوله Zero-Click.
   494→ */
   495→export async function issueSlip(
   496→  item: IssueSlipItemDTO,
   497→  opts?: { autoIssued?: boolean }
   498→): Promise<{ created: boolean; slip: IssuedSlipDTO }> {
   499→  const externalId = item.match.externalId ?? null;
   500→  const stakes =
   501→    item.stakeA && item.stakeB && item.stakeA + item.stakeB > 0
   502→      ? { stakeA: item.stakeA, stakeB: item.stakeB }
   503→      : equalProfitStakes(item.oddsA, item.oddsB);
   504→
   505→  const hash = slipHash({
   506→    externalId,
   507→    homeTeam: item.match.homeTeam,
   508→    awayTeam: item.match.awayTeam,
   509→    kickoff: item.match.kickoff,
   510→    labelA: item.labelA,
   511→    labelB: item.labelB,
   512→    oddsA: item.oddsA,
   513→    oddsB: item.oddsB,
   514→    stakeA: stakes.stakeA,
   515→    stakeB: stakes.stakeB,
   516→  });
   517→
   518→  const existing = await db.issuedSlip.findUnique({ where: { hash } });
   519→  if (existing) return { created: false, slip: mapSlip(existing) };
   520→
   521→  // ─── v10.0 — احتمال مدل، ارزش عددی و پنجره زمانی طلایی ثبت ───
   522→  const lH = item.lambdaHome ?? 1.35;
   523→  const lA = item.lambdaAway ?? 1.25;
   524→  const probA = legProbability(lH, lA, item.legsA ?? []);
   525→  const probB = legProbability(lH, lA, item.legsB ?? []);
   526→  const edgeA = computeEdge(probA, item.oddsA);
   527→  const edgeB = computeEdge(probB, item.oddsB);
   528→  const edgeSourceA = edgeA !== null ? "MARKET_FAIR" : null;
   529→  const edgeSourceB = edgeB !== null ? "MARKET_FAIR" : null;
   530→  const tipKindA: TipKind | null = edgeA !== null ? deriveTipKind((item.legsA ?? []).map((l) => l.key), item.oddsA) : null;
   531→  const tipKindB: TipKind | null = edgeB !== null ? deriveTipKind((item.legsB ?? []).map((l) => l.key), item.oddsB) : null;
   532→  const koMs = new Date(item.match.kickoff).getTime();
   533→  const winA = tipKindA ? computeTimingWindowStatic(koMs, tipKindA) : null;
   534→  const winB = tipKindB ? computeTimingWindowStatic(koMs, tipKindB) : null;
   535→
   536→  const row = await db.issuedSlip.create({
   537→    data: {
   538→      hash,
   539→      externalId,
   540→      matchId: item.match.id,
   541→      league: item.match.league,
   542→      homeTeam: item.match.homeTeam,
   543→      awayTeam: item.match.awayTeam,
   544→      kickoff: new Date(item.match.kickoff),
   545→      labelA: item.labelA,
   546→      labelB: item.labelB,
   547→      legsA: JSON.stringify(item.legsA ?? []),
   548→      legsB: JSON.stringify(item.legsB ?? []),
   549→      oddsA: item.oddsA,
   550→      oddsB: item.oddsB,
   551→      stakeA: stakes.stakeA,
   552→      stakeB: stakes.stakeB,
   553→      cellsA: JSON.stringify(coveredCells(lH, lA, item.legsA ?? [], externalId)),
   554→      cellsB: JSON.stringify(coveredCells(lH, lA, item.legsB ?? [], externalId)),
   555→      guaranteedProfit: item.guaranteedProfit,
   556→      marginPct: item.marginPct,
   557→      archetype: item.archetype ?? null,
   558→      archetypeCode: item.archetypeCode ?? null,
   559→      lambdaHome: item.lambdaHome ?? null,
   560→      lambdaAway: item.lambdaAway ?? null,
   561→      espnUrl: item.match.espnUrl ?? espnUrlOf(externalId),
   562→      supervisorApproved: item.supervisorApproved !== false,
   563→      autoIssued: opts?.autoIssued === true,
   564→      modelProbA: probA,
   565→      modelProbB: probB,
   566→      edgeA,
   567→      edgeB,
   568→      edgeSourceA,
   569→      edgeSourceB,
   570→      tipKindA,
   571→      tipKindB,
   572→      timingWindowA: winA?.windowKey ?? null,
   573→      timingWindowB: winB?.windowKey ?? null,
   574→      timingStartA: winA?.startAt ?? null,
   575→      timingEndA: winA?.endAt ?? null,
   576→      timingStartB: winB?.startAt ?? null,
   577→      timingEndB: winB?.endAt ?? null,
   578→      status: "PENDING",
   579→    },
   580→  });
   581→  return { created: true, slip: mapSlip(row) };
   582→}

   645→export async function getLedger(): Promise<LedgerDTO> {
   646→  const [rows, outboxRows, tgStatus] = await Promise.all([
   647→    db.issuedSlip.findMany({ orderBy: [{ status: "asc" }, { kickoff: "desc" }], take: 120 }),
   648→    db.telegramOutbox.findMany({ orderBy: { createdAt: "desc" }, take: 25, include: { slip: { select: { hash: true } } } }),
   649→    import("./telegram-service").then((m) => m.telegramStatus()),
   650→  ]);
   651→
   652→  const slips = rows.map((s) => mapSlip(s));
   653→  const stats: SettlementStatsDTO = {
   654→    total: slips.length,
   655→    pending: 0,
   656→    live: 0,
   657→    wonA: 0,
   658→    wonB: 0,
   659→    doubleWin: 0,
   660→    middleTrap: 0,
   661→    manual: 0,
   662→    void: 0,
   663→    settled: 0,
   664→    netProfit: 0,
   665→    netProfitPct: 0,
   666→    staked: 0,
   667→  };
   668→  for (const s of slips) {
   669→    switch (s.status) {
   670→      case "PENDING": stats.pending += 1; break;
   671→      case "LIVE": stats.live += 1; break;
   672→      case "WON_A": stats.wonA += 1; break;
   673→      case "WON_B": stats.wonB += 1; break;
   674→      case "DOUBLE_WIN": stats.doubleWin += 1; break;
   675→      case "MIDDLE_TRAP": stats.middleTrap += 1; break;
   676→      case "MANUAL_REVIEW": stats.manual += 1; break;
   677→      case "VOID": stats.void += 1; break;
   678→    }
   679→    if (["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"].includes(s.status)) {
   680→      stats.settled += 1;
   681→      stats.netProfit += s.netProfit ?? 0;
   682→      stats.staked += s.stakeA + s.stakeB;
   683→    }
   684→  }
   685→  stats.netProfit = Math.round(stats.netProfit * 100) / 100;
   686→  stats.netProfitPct = stats.staked > 0 ? Math.round((stats.netProfit / stats.staked) * 1000) / 10 : 0;
   687→
   688→  // ─── v10.0 — ژورنال + دفتر دکترین + وضعیت اتوماسیون سرور ───
   689→  const [journal, doctrine, sched] = await Promise.all([
   690→    listJournal(30).catch(() => []),
   691→    listDoctrine().catch(() => [] as DoctrineSummary[]),
   692→    Promise.resolve(getSchedulerStatus()),
   693→  ]);
   694→
   695→  return {
   696→    success: true,
   697→    slips,
   698→    stats,
   699→    outbox: outboxRows.map((o) => ({
   700→      id: o.id,
   701→      kind: o.kind,
   702→      ok: o.ok,
   703→      detail: o.detail,
   704→      slipHash: o.slip?.hash ?? null,
   705→      createdAt: o.createdAt.toISOString(),
   706→    })),
   707→    telegram: { configured: tgStatus.configured, channelSet: tgStatus.channelSet },
   708→    journal,
   709→    doctrine,
   710→    automation: {
   711→      schedulerActive: sched.active,
   712→      lastTickAt: sched.lastTickAt ? new Date(sched.lastTickAt).toISOString() : null,
   713→      lastDailyReportAt: sched.lastDailyReportAt ? new Date(sched.lastDailyReportAt).toISOString() : null,
   714→      intervalSec: sched.intervalSec,
   715→    },
   716→    unitBasis: { stakePerTicket: 100, unitPctOfBankroll: 1, exampleBankrollUsd: 10 },
   717→  };
   718→}

   721→
   722→let autoBusy = false;
   723→let autoLastAt = 0;
   724→const AUTO_THROTTLE_MS = 60_000;
   725→
   726→/**
   727→ * فراخوانی سبک از دل فید زنده — به محض FT در ESPN تسویه انجام می‌شود
   728→ * بدون انتظار برای کلیک کاربر. با throttle و قفل برای صفر هزینه اضافه.
   729→ */
   730→export async function autoSettleInBackground(): Promise<void> {
   731→  const pending = await db.issuedSlip.count({ where: { status: { in: ["PENDING", "LIVE"] } } });
   732→  if (pending === 0) return;
   733→  if (autoBusy || Date.now() - autoLastAt < AUTO_THROTTLE_MS) return;
   734→  autoBusy = true;
   735→  autoLastAt = Date.now();
   736→  try {
   737→    await settleFinishedSlips({ notify: true });
   738→  } catch (err) {
   739→    console.warn("auto-settle failed:", err);
   740→  } finally {
   741→    autoBusy = false;
   742→  }
   743→}
