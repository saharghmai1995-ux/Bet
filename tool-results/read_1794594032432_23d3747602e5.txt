   113→function downsampleEvenly(arr: number[], maxPts: number): number[] {
   114→  if (arr.length <= maxPts) return arr;
   115→  const out: number[] = [];
   116→  for (let i = 0; i < maxPts - 1; i++) {
   117→    out.push(arr[Math.round((i * (arr.length - 1)) / (maxPts - 1))]);
   118→  }
   119→  out.push(arr[arr.length - 1]);
   120→  return out;
   121→}

   418→ * v18.0 — نبض کارنامه (Ledger Pulse):
   419→ * پاسخ صادقانه و مهندسی‌شده به «چرا عملکرد/وین‌ریت/منحنی تغییری نکرده؟»:
   420→ *   ۱) آمار رسمی فقط از کارت‌های تاییدشده سوپروایزر ساخته می‌شود (پول واقعی).
   421→ *   ۲) اگر گیت v16 همه کارت‌های تازه را وتو کند، منحنی تخت می‌ماند — اینجا عددی
   422→ *      شفاف می‌شود (وتوها، کارت در انتظار FT، آخرین تسویه رسمی).
   423→ *   ۳) دفتر سایه (تسویه کاغذی وتوها) کنارش نمایش داده می‌شود تا همیشه عدد زنده باشد.
   424→ *   ۴) خود ماژول هر ۹۰ ثانیه رفرش می‌شود تا تسویه‌های تازه فوری ظاهر شوند.
   425→ */
   426→function LedgerPulse({ ledger, savedByVeto }: { ledger: LedgerDTO | null; savedByVeto: number }) {
   427→  const pulse = ledger?.pulse;
   428→  const shadow = ledger?.shadow;
   429→  if (!pulse) return null;
   430→
   431→  const rel = (iso: string | null) => {
   432→    if (!iso) return null;
   433→    const diffMs = Date.now() - new Date(iso).getTime();
   434→    const mins = Math.round(diffMs / 60_000);
   435→    if (mins < 1) return "همین حالا";
   436→    if (mins < 60) return `${faNum(mins, 0)} دقیقه پیش`;
   437→    const hours = Math.round(mins / 60);
   438→    if (hours < 24) return `${faNum(hours, 0)} ساعت پیش`;
   439→    const days = Math.round(hours / 24);
   440→    return `${faNum(days, 0)} روز پیش`;
   441→  };
   442→  const lastSettled = rel(pulse.lastSettledAt);
   443→  const frozen = pulse.settled7d === 0;
   444→
   445→  const chip = (label: string, value: string, tone: string) => (
   446→    <span className="flex items-center gap-1.5 rounded-lg border border-white/10 bg-black/25 px-2 py-1 text-[10px]" title={label}>
   447→      <span className="text-muted-foreground">{label}</span>
   448→      <b className={cn("font-num", tone)} dir="ltr">{value}</b>
   449→    </span>
   450→  );
   451→
   452→  return (
   453→    <div className="space-y-2 border-b border-white/8 bg-gradient-to-l from-cyan-500/[0.05] to-transparent px-4 py-2.5" role="status" aria-label="نبض کارنامه">
   454→      <div className="flex flex-wrap items-center gap-1.5">
   455→        <span className="flex items-center gap-1 text-[10px] font-bold text-cyan-300">
   456→          <Activity className="size-3.5" aria-hidden /> نبض کارنامه:
   457→        </span>
   458→        {chip("آخرین تسویه رسمی", lastSettled ?? "—", "text-foreground/90")}
   459→        {chip("تسویه ۷ روز اخیر", faNum(pulse.settled7d, 0), pulse.settled7d > 0 ? "text-emerald-300" : "text-amber-300")}
   460→        {chip("در انتظار FT", faNum(pulse.pendingApproved, 0), "text-cyan-300")}
   461→        {chip("وتوی سوپروایزر", faNum(pulse.vetoedTotal, 0), "text-rose-300")}
   462→        {chip("دفتر سایه (کاغذی)", `${shadow ? faNum(shadow.settled, 0) : faNum(0, 0)} کارت · ${shadow?.winRatePct != null ? faNum(shadow.winRatePct, 1) + "٪" : "—"} · ${faNum(shadow?.netProfit ?? 0, 1)}u`, "text-muted-foreground")}
   463→        {savedByVeto > 0 && chip("سرمایه نجات‌یافته", `+${faNum(savedByVeto, 1)}u`, "text-amber-300")}
   464→      </div>
   465→      {frozen ? (
   466→        <p className="rounded-lg border border-amber-500/35 bg-amber-500/[0.08] px-2 py-1.5 text-[10px] leading-4 text-amber-200">
   467→          <b>وضعیت فعلی: کتاب رسمی در انتظار کارت تازه است.</b> آمار و منحنی فقط از کارت‌های «تایید سوپروایزر با پول واقعی» ساخته می‌شوند؛ {faNum(pulse.vetoedTotal, 0)} کاندیدای پرریسک توسط گیت سخت v16.0 (ریسک همزمان &gt; ۱۰.۵٪ / لیگ‌های سمّی / فاوریت بدون برتری واضح) به دفتر سایه رفت‌اند — به‌محض تایید اولین کارت امروز، منحنی و وین‌ریت همین‌جا به‌روز می‌شود (رفرش خودکار هر ۹۰ ثانیه).
   468→        </p>
   469→      ) : (
   470→        <p className="text-[10px] leading-4 text-muted-foreground">
   471→          تسویه‌های رسمی در جریان‌اند — آمار این ماژول هر ۹۰ ثانیه خودکار رفرش می‌شود؛ منحنی انتهایش دقیقاً برابر «سود خالص مجموع» است.
   472→        </p>
   473→      )}
   474→    </div>
   475→  );
   476→}
