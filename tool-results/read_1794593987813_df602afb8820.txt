    73→export async function verifyBotToken(token: string): Promise<{ ok: boolean; username: string | null; detail: string }> {
    74→  try {
    75→    const res = await fetch(`https://api.telegram.org/bot${token.trim()}/getMe`, { cache: "no-store" });
    76→    const json = (await res.json()) as { ok?: boolean; result?: { username?: string }; description?: string };
    77→    if (json.ok && json.result?.username) return { ok: true, username: json.result.username, detail: `@${json.result.username}` };
    78→    return { ok: false, username: null, detail: json.description ?? `HTTP ${res.status}` };
    79→  } catch (err) {
    80→    return { ok: false, username: null, detail: err instanceof Error ? err.message : "network error" };
    81→  }

   145→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
   146→  const token = await botToken();
   147→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   148→  // v13.2 — سپر توکن ناقص: خطای 404 مبهم تلگرام را با پیام دقیق جایگزین می‌کند
   149→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   150→  try {
   151→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
   152→      method: "POST",
   153→      headers: { "Content-Type": "application/json" },
   154→      body: JSON.stringify(body),
   155→      cache: "no-store",
   156→    });
   157→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   158→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
   159→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   160→  } catch (err) {
   161→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   162→  }
   163→}

   165→async function tgSendPhotoTo(
   166→  chatId: string | null,
   167→  photo: Buffer,
   168→  caption: string,
   169→  filename = "slip-card.png",
   170→  captionless = false
   171→): Promise<{ ok: boolean; detail: string }> {
   172→  const token = await botToken();
   173→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   174→  if (!chatId) return { ok: false, detail: "NO_CHANNEL" };
   175→  // v13.2 — سپر توکن ناقص (بخش hash بدون bot_id) — ریشه باگ «انتشار انجام نمیشه»
   176→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   177→  try {
   178→    const form = new FormData();
   179→    form.append("chat_id", chatId);
   180→    if (!captionless) {
   181→      form.append("caption", caption.slice(0, 1024));
   182→      form.append("parse_mode", "HTML");
   183→    }
   184→    form.append("photo", new Blob([new Uint8Array(photo)], { type: "image/png" }), filename);
   185→    const res = await fetch(`https://api.telegram.org/bot${token}/sendPhoto`, { method: "POST", body: form });
   186→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   187→    if (json.ok) return { ok: true, detail: `photo:${json.result?.message_id ?? "?"}${captionless ? ":broadcast" : ""}` };
   188→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   189→  } catch (err) {
   190→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   191→  }
   192→}

   231→/** پیام HTML فلش‌کارت کامل — تیپ A + تیپ B + داچینگ + SHA-256 + لینک ESPN */
   232→export function buildCardMessage(slip: IssuedSlip): string {
   233→  const kickoff = new Date(slip.kickoff).toLocaleString("fa-IR", { dateStyle: "short", timeStyle: "short" });
   234→  // v15.0 — نام رسمی بین‌المللی مارکت هر تیپ (Institutional Execution Metadata)
   235→  const legKeys = (json: string): string[] => {
   236→    try {
   237→      const arr = JSON.parse(json);
   238→      return Array.isArray(arr) ? arr.map((l: { key?: string }) => String(l?.key ?? l)) : [];
   239→    } catch {
   240→      return [];
   241→    }
   242→  };
   243→  const marketAEn = marketEnLabel(legKeys(slip.legsA)[0] ?? "");
   244→  const marketBEn = marketEnLabel(legKeys(slip.legsB)[0] ?? "");
   245→  const lines: string[] = [];
   246→  lines.push(`⚽️ <b>فوتبال‌کوانت — فلش‌کارت الگوی طلایی اتمیک v16.0</b>`);
   247→  lines.push(`🏆 <b>${esc(slip.homeTeam)}</b> × <b>${esc(slip.awayTeam)}</b>`);
   248→  lines.push(`📡 ${esc(slip.league)} · ⏰ ${kickoff}`);
   249→  if (slip.archetype) lines.push(`📐 الگو: ${esc(slip.archetype)}`);
   250→  lines.push("");
   251→  lines.push(`🔹 <b>تیپ A</b> (ضریب ${slip.oddsA.toFixed(2)}): ${esc(slip.labelA)}`);
   252→  lines.push(`🏷 Market (EN): <i>${esc(marketAEn)}</i>`);
   253→  lines.push(`💵 مبلغ داچینگ A: <b>${slip.stakeA.toFixed(1)}</b> واحد`);
   254→  lines.push("");
   255→  lines.push(`🟢 <b>تیپ B مکمل</b> (ضریب ${slip.oddsB.toFixed(2)}): ${esc(slip.labelB)}`);
   256→  lines.push(`🏷 Market (EN): <i>${esc(marketBEn)}</i>`);
   257→  lines.push(`💵 مبلغ داچینگ B: <b>${slip.stakeB.toFixed(1)}</b> واحد`);
   258→  lines.push("");
   259→  lines.push(`${esc(SETTLEMENT_RULE_FA)}`);
   260→  // ─── v10.0 — ارزش عددی نسبت به خط مرجع (همیشه عدد دارد) ───
   261→  if (slip.edgeA !== null || slip.edgeB !== null) {
   262→    const isPin = slip.edgeSourceA === "PINNACLE" || slip.edgeSourceB === "PINNACLE";
   263→    const src = isPin ? "خط شارپ پیناکل" : "خط منصفانه بازار";
   264→    const fa = (v: number) => `${v >= 0 ? "+" : ""}${v.toFixed(1)}%`;
   265→    const a = slip.edgeA !== null ? fa(slip.edgeA) : "—";
   266→    const b = slip.edgeB !== null ? fa(slip.edgeB) : "—";
   267→    lines.push(`📈 ارزش عددی نسبت به ${src}: تیپ A <b>${a}</b> · تیپ B <b>${b}</b>`);
   268→    if (isPin) {
   269→      const cmp = (odds: number, pin: number | null): string | null =>
   270→        pin && pin > 1.01 ? `${odds.toFixed(2)} در برابر خط ${pin.toFixed(2)}` : null;
   271→      const pa = slip.edgeSourceA === "PINNACLE" ? cmp(slip.oddsA, slip.pinnacleOddsA) : null;
   272→      const pb = slip.edgeSourceB === "PINNACLE" ? cmp(slip.oddsB, slip.pinnacleOddsB) : null;
   273→      const parts = [pa ? `تیپ A: ${pa}` : null, pb ? `تیپ B: ${pb}` : null].filter((x): x is string => Boolean(x));
   274→      if (parts.length > 0) lines.push(`📌 مقایسه قیمت با خط پیناکل: ${parts.join(" · ")}`);
   275→    }
   276→  }
   277→  // ─── v10.0 — بهترین ساعت ثبت ضریب (Optimal Odds Timing Window) ───
   278→  if (slip.timingWindowA || slip.timingWindowB) {
   279→    const winName = timingWindowLabel(slip.timingWindowA ?? slip.timingWindowB);
   280→    const start = slip.timingStartA ?? slip.timingStartB;
   281→    const end = slip.timingEndA ?? slip.timingEndB;
   282→    const fmt = (d: Date) => new Intl.DateTimeFormat("fa-IR", { hour: "2-digit", minute: "2-digit", timeZone: "Asia/Tehran" }).format(d);
   283→    const range = start && end ? ` | ساعت دقیق پیشنهادی ثبت: ${fmt(new Date(start))} تا ${fmt(new Date(end))} (تهران)` : "";
   284→    lines.push(`⏱️ بهترین زمان ثبت این تیپ: <b>${esc(winName)}</b>${range}`);
   285→  }
   286→  lines.push(`💰 سود تضمینی پوشش: <b>+${slip.guaranteedProfit.toFixed(1)}%</b> (مجموع استیک ۱۰۰ واحد)`);
   287→  lines.push(`🔐 هش راستی‌آزمایی SHA-256:\n<code>${slip.hash.slice(0, 32)}</code>`);
   288→  if (slip.espnUrl) lines.push(`🔎 <a href="${slip.espnUrl}">لینک راستی‌آزمایی رسمی ESPN</a>`);
   289→  lines.push("");
   290→  lines.push(`⚖️ ابزار تحلیل کمّی — مسئولیت استفاده با کاربر است.`);
   291→  return lines.join("\n");
   292→}

   348→export function buildResultMessage(slip: IssuedSlip): string {

   368→  return lines.join("\n");

   481→/** آمار وین‌ریت کانال — مستقیم از دیتابیس (بدون وابستگی چرخه‌ای) */
   482→async function channelWinRateStats() {
   483→  // v11.0 — فقط کارت‌های واقعی تاییدشده؛ تسویه کاغذی دفتر سایه در وین‌ریت کانال دخالت نمی‌کند
   484→  // v13.2 — فقط کارت‌های واقعی تاییدشده؛ کارت‌های بازخوانی (retroIssued) و دفتر سایه در وین‌ریت کانال دخالت نمی‌کنند
   485→  const settled = await db.issuedSlip.findMany({
   486→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, supervisorApproved: true, retroIssued: false },
   487→    select: { status: true, netProfit: true, stakeA: true, stakeB: true },
   488→  });
   489→  const st = { settled: settled.length, wonA: 0, wonB: 0, doubleWin: 0, middleTrap: 0, netProfit: 0, staked: 0 };
   490→  for (const s of settled) {
   491→    if (s.status === "WON_A") st.wonA += 1;
   492→    else if (s.status === "WON_B") st.wonB += 1;
   493→    else if (s.status === "DOUBLE_WIN") st.doubleWin += 1;
   494→    else if (s.status === "MIDDLE_TRAP") st.middleTrap += 1;
   495→    st.netProfit += s.netProfit ?? 0;
   496→    st.staked += s.stakeA + s.stakeB;
   497→  }
   498→  const winRatePct = st.settled > 0 ? Math.round(((st.settled - st.middleTrap) / st.settled) * 1000) / 10 : 0;
   499→  const netPct = st.staked > 0 ? Math.round((st.netProfit / st.staked) * 1000) / 10 : 0;
   500→  // ─── v11.0 — دفتر سایه: شفافیت counterfactual وتوها (خط جداگانه گزارش) ───
   501→  // v13.5.1 — سایه فقط وتوهای زنده است؛ کارت‌های بازخوانی (retroIssued) داخل سایه شمرده نمی‌شوند
   502→  const shadowRows = await db.issuedSlip.findMany({
   503→    where: { supervisorApproved: false, retroIssued: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   504→    select: { status: true, netProfit: true },
   505→  });
   506→  const shadowWins = shadowRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   507→  const shadowNet = Math.round(shadowRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100;
   508→  const shadowWinRate = shadowRows.length > 0 ? Math.round((shadowWins / shadowRows.length) * 1000) / 10 : null;
   509→  // ─── v13.2 — دفتر بازخوانی: کارت‌های چندروزه‌ای که هرگز پیش‌بازی صادر نشده‌اند (تسویه کاغذی تحلیلی) ───
   510→  // v13.5.1 — بازخوانی با پرچم retroIssued شناسایی می‌شود (مستقل از تایید — کارت بازخوانی هرگز تاییدشده نیست)
   511→  const retroRows = await db.issuedSlip.findMany({
   512→    where: { retroIssued: true, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   513→    select: { status: true, netProfit: true },
   514→  });
   515→  const retroWins = retroRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   516→  return {
   517→    ...st,
   518→    winRatePct,
   519→    netPct,
   520→    netProfit: Math.round(st.netProfit * 100) / 100,
   521→    shadow: { settled: shadowRows.length, wins: shadowWins, winRatePct: shadowWinRate, netProfit: shadowNet },
   522→    retro: {
   523→      settled: retroRows.length,
   524→      wins: retroWins,
   525→      winRatePct: retroRows.length > 0 ? Math.round((retroWins / retroRows.length) * 1000) / 10 : null,
   526→      netProfit: Math.round(retroRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100,
   527→    },
   528→  };
   529→}
