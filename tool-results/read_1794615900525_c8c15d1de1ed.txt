    97→/**
    98→ * v18.4 — claim اتمی کلید Setting برای dedupe بین‌پردازه‌ای (Atomic Dedupe Claim).
    99→ * الگوی قدیمی «check → send → write» در دو اجرای هم‌زمان (زمان‌بند + مسیر دستی یا دو
   100→ * پردازه) هر دو استمپ کهنه می‌خواند و همان پست را دوبار می‌فرستد. حالا claim قبل از
   101→ * ارسال انجام می‌شود؛ فقط یک فراخوان برنده است (SQLite تک‌نویسنده).
   102→ * خروجی false = دیگری قبلاً مالک شده — بدون ارسال.
   103→ */
   104→export async function claimSetting(key: string, value: string): Promise<boolean> {
   105→  const existing = await db.setting.findUnique({ where: { key } }).catch(() => null);
   106→  if (!existing) {
   107→    try {
   108→      await db.setting.create({ data: { key, value } });
   109→      return true;
   110→    } catch {
   111→      return false; // مسیر موازی هم‌زمان ردیف را ساخت
   112→    }
   113→  }
   114→  if (existing.value === value) return false;
   115→  const res = await db.setting.updateMany({ where: { key, value: { not: value } }, data: { value } });
   116→  return res.count === 1;
   117→}

   128→// ═══ v19.0 — ادعای اتمی Broadcast (Insert-First Claim) ═══
   129→// ریشه ریشه‌ای باگ «ارسال پیام‌های تکراری»: الگوی claim قبلی (خواندن Setting سپس
   130→// update شرطی) در هم‌پوشانی ری‌استارت‌های dev / دو پردازه موازی، توالی خواندن‌ها

   132→// قبل از ارسال، یک ردیف TelegramOutbox با dedupeKey یکتا INSERT می‌شود.
   133→//   · موفق بودن INSERT  = مالکیت ارسال با ماست (اتمیک در سطح SQLite — هر تعداد

   236→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
   237→  const token = await botToken();
   238→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   239→  // v13.2 — سپر توکن ناقص: خطای 404 مبهم تلگرام را با پیام دقیق جایگزین می‌کند
   240→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   241→  try {
   242→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
   243→      method: "POST",
   244→      headers: { "Content-Type": "application/json" },
   245→      body: JSON.stringify(body),
   246→      cache: "no-store",
   247→    });
   248→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   249→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
   250→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   251→  } catch (err) {
   252→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   253→  }
   254→}

   256→async function tgSendPhotoTo(
   257→  chatId: string | null,
   258→  photo: Buffer,
   259→  caption: string,
   260→  filename = "slip-card.png",
   261→  captionless = false
   262→): Promise<{ ok: boolean; detail: string }> {
   263→  const token = await botToken();
   264→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   265→  if (!chatId) return { ok: false, detail: "NO_CHANNEL" };
   266→  // v13.2 — سپر توکن ناقص (بخش hash بدون bot_id) — ریشه باگ «انتشار انجام نمیشه»
   267→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   268→  try {
   269→    const form = new FormData();
   270→    form.append("chat_id", chatId);
   271→    if (!captionless) {
   272→      form.append("caption", caption.slice(0, 1024));
   273→      form.append("parse_mode", "HTML");
   274→    }
   275→    form.append("photo", new Blob([new Uint8Array(photo)], { type: "image/png" }), filename);
   276→    const res = await fetch(`https://api.telegram.org/bot${token}/sendPhoto`, { method: "POST", body: form });
   277→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   278→    if (json.ok) return { ok: true, detail: `photo:${json.result?.message_id ?? "?"}${captionless ? ":broadcast" : ""}` };
   279→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   280→  } catch (err) {
   281→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   282→  }
   283→}

   316→// ═══ v18.4 — ضد تکرار ارسال (Anti-Duplicate Broadcast Infrastructure) ═══
   317→// ریشه باگ «ارسال پیام‌های تکراری به کانال عمومی»: چهار مسیر ورود مستقل
   318→// (زمان‌بند سرور، runSync فید زنده، POST autodispatch، POST config dispatchNow)
   319→// هم‌زمان صف انتشار را اجرا می‌کردند؛ هر دو اسنپ‌شات stale می‌خواندند و همان
   320→// پیام را دو بار می‌فرستادند. سه سپر:
   321→//   ۱) قفل in-flight سراسری برای هر جریان ارسال (dispatch / funnel / winrate)
   322→//   ۲) claim اتمی پرچم‌های telegramCard/telegramResult قبل از ارسال (updateMany شرطی)
   323→//   ۳) آزادسازی claim در شکست ارسال تا ریتیپ بعدی از دست نرود
   324→
   325→const gBroadcast = globalThis as unknown as {
   326→  __fqDispatchInFlight?: boolean;
   327→  __fqFunnelLock?: Promise<unknown> | null;
   328→};
   329→
   330→/** قفل رشته‌ای funnel — اجرای هم‌زمان دو پست فانل (دو مسیر ورود) را سریال می‌کند */
   331→function withFunnelLock<F extends (...args: never[]) => Promise<unknown>>(fn: F): F {
   332→  return (async (...args: Parameters<F>) => {
   333→    const prev = gBroadcast.__fqFunnelLock ?? Promise.resolve();
   334→    let release!: () => void;
   335→    const gate = new Promise<void>((r) => (release = r));
   336→    gBroadcast.__fqFunnelLock = prev.then(() => gate);
   337→    await prev.catch(() => undefined);
   338→    try {
   339→      return await fn(...(args as Parameters<F>));
   340→    } finally {
   341→      release();
   342→      if (gBroadcast.__fqFunnelLock === gate) gBroadcast.__fqFunnelLock = null;
   343→    }
   344→  }) as F;
   345→}

   349→ * خروجی true = مالکیت ارسال با ماست؛ false = مسیر موازی قبلاً مالک شده (بدون ارسال).

   699→/** آمار وین‌ریت کانال — مستقیم از دیتابیس (بدون وابستگی چرخه‌ای) */
   700→async function channelWinRateStats() {
   701→  // v11.0 — فقط کارت‌های واقعی تاییدشده؛ تسویه کاغذی دفتر سایه در وین‌ریت کانال دخالت نمی‌کند
   702→  // v13.2 — فقط کارت‌های واقعی تاییدشده؛ کارت‌های بازخوانی (retroIssued) و دفتر سایه در وین‌ریت کانال دخالت نمی‌کنند
   703→  const settled = await db.issuedSlip.findMany({
   704→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, supervisorApproved: true, retroIssued: false },
   705→    select: { status: true, netProfit: true, stakeA: true, stakeB: true },
   706→  });
   707→  const st = { settled: settled.length, wonA: 0, wonB: 0, doubleWin: 0, middleTrap: 0, netProfit: 0, staked: 0 };
   708→  for (const s of settled) {
   709→    if (s.status === "WON_A") st.wonA += 1;
   710→    else if (s.status === "WON_B") st.wonB += 1;
   711→    else if (s.status === "DOUBLE_WIN") st.doubleWin += 1;
   712→    else if (s.status === "MIDDLE_TRAP") st.middleTrap += 1;
   713→    st.netProfit += s.netProfit ?? 0;
   714→    st.staked += s.stakeA + s.stakeB;
   715→  }
   716→  const winRatePct = st.settled > 0 ? Math.round(((st.settled - st.middleTrap) / st.settled) * 1000) / 10 : 0;
   717→  const netPct = st.staked > 0 ? Math.round((st.netProfit / st.staked) * 1000) / 10 : 0;
   718→  // ─── v11.0 — دفتر سایه: شفافیت counterfactual وتوها (خط جداگانه گزارش) ───
   719→  // v13.5.1 — سایه فقط وتوهای زنده است؛ کارت‌های بازخوانی (retroIssued) داخل سایه شمرده نمی‌شوند
   720→  const shadowRows = await db.issuedSlip.findMany({
   721→    where: { supervisorApproved: false, retroIssued: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   722→    select: { status: true, netProfit: true },
   723→  });
   724→  const shadowWins = shadowRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   725→  const shadowNet = Math.round(shadowRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100;
   726→  const shadowWinRate = shadowRows.length > 0 ? Math.round((shadowWins / shadowRows.length) * 1000) / 10 : null;
   727→  // ─── v13.2 — دفتر بازخوانی: کارت‌های چندروزه‌ای که هرگز پیش‌بازی صادر نشده‌اند (تسویه کاغذی تحلیلی) ───
   728→  // v13.5.1 — بازخوانی با پرچم retroIssued شناسایی می‌شود (مستقل از تایید — کارت بازخوانی هرگز تاییدشده نیست)
   729→  const retroRows = await db.issuedSlip.findMany({
   730→    where: { retroIssued: true, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   731→    select: { status: true, netProfit: true },
   732→  });
   733→  const retroWins = retroRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   734→  // ─── v18.0 — 🥈 حصاد آلفای سایه: کارت‌های تاج نقره‌ای سطح ۲ (استیک ۵۰u Half-Kelly) ───
   735→  const silverRows = await db.issuedSlip.findMany({
   736→    where: { crownTier: "SILVER_SHADOW", retroIssued: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   737→    select: { status: true, netProfit: true },
   738→  });
   739→  const silverWins = silverRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   740→  return {
   741→    ...st,
   742→    winRatePct,
   743→    netPct,
   744→    netProfit: Math.round(st.netProfit * 100) / 100,
   745→    shadow: { settled: shadowRows.length, wins: shadowWins, winRatePct: shadowWinRate, netProfit: shadowNet },
   746→    silver: {
   747→      settled: silverRows.length,
   748→      wins: silverWins,
   749→      winRatePct: silverRows.length > 0 ? Math.round((silverWins / silverRows.length) * 1000) / 10 : null,
   750→      netProfit: Math.round(silverRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100,
   751→    },
   752→    retro: {
   753→      settled: retroRows.length,
   754→      wins: retroWins,
   755→      winRatePct: retroRows.length > 0 ? Math.round((retroWins / retroRows.length) * 1000) / 10 : null,
   756→      netProfit: Math.round(retroRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100,
   757→    },
   758→  };
   759→}

   856→    // ارسال تکراری — چون در پنجره ۱۵ دقیقه‌ای فقط یک مسیر می‌تواند ادعا کند).

   934→async function autoDispatchApprovedSlipsInner(options?: {

   938→  ok: boolean;
   939→  sentCards: number;
   940→  sentResults: number;
   941→  winRateSent: boolean;
   942→  skippedVetoed: number;
   943→  detail: string;
   944→}> {

   946→  const maxResults = Math.max(1, Math.min(12, options?.maxResults ?? 3));

   948→  await repairPrematureCardFlags();

   950→  // v13.5.1 — کارت‌های بازخوانی (retroIssued) وتو نیستند — شمارش وتو فقط برای کارت‌های زنده
   951→  const skippedVetoed = await db.issuedSlip.count({ where: { supervisorApproved: false, retroIssued: false, telegramCard: false } });
   952→
   953→  if (!tg.configured || !tg.channelSet) {
   954→    return { ok: false, sentCards: 0, sentResults: 0, winRateSent: false, skippedVetoed, detail: "TELEGRAM_NOT_CONFIGURED" };
   955→  }
   956→  if (!tg.tokenFormat) {
   957→    return { ok: false, sentCards: 0, sentResults: 0, winRateSent: false, skippedVetoed, detail: "INVALID_TOKEN_FORMAT" };
   958→  }
   959→  if (!autoOn) {
   960→    return { ok: false, sentCards: 0, sentResults: 0, winRateSent: false, skippedVetoed, detail: "AUTO_DISPATCH_OFF" };
   961→  }
   962→
   963→  // v13.2 — «فقط کارت‌های روزانه»: پنجره انتشار کارت = بازی‌های امروز تهران
   964→  const day = tehranDayRange();
   965→
   966→  let sentCards = 0;
   967→  let sentResults = 0;
   968→
   969→  // ─── مرحله ۱: پست پیش از بازی — فقط کارت‌های 👑 تاییدشده سوپروایزر «بازی‌های امروز تهران» ───
   970→  const cards = await db.issuedSlip.findMany({
   971→    where: { status: { in: ["PENDING", "LIVE"] }, telegramCard: false, supervisorApproved: true, retroIssued: false, kickoff: { gte: day.start, lt: day.end } },
   972→    orderBy: { kickoff: "asc" },
   973→    take: maxCards,
   974→  });
   975→  for (const s of cards) {
   976→    try {
