    63→    63→  const crownLine = sup

    69→    69→    p.guaranteedMarginPct > 0

    71→    71→      : `🛡️ پوشش بیمه سرمایه — بازگشت وجه در سناریوی پایه`;

    77→    77→    profitLine,
