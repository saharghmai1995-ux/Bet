    78→/** ضریب فرم از ۵ بازی اخیر واقعی (WWDLW) — بازه ۰.۷۸ تا ۱.۲۶ */
    79→function formFactor(form: string): number {
    80→  if (!form || form === "N/A") return 1.0;
    81→  const pts = [...form].reduce((a, ch) => a + (ch === "W" ? 3 : ch === "D" ? 1 : 0), 0);
    82→  return 0.78 + (pts / (form.length * 3)) * 0.48;
    83→}
    84→
    85→/** ضریب فصل از رکورد واقعی (مثل 3-4-2) — بازه ۰.۸۲ تا ۱.۱۸ */
    86→function seasonFactor(record: string | null): number {
    87→  const pct = recordToPointsPct(record);
    88→  if (pct === null) return 1.0;
    89→  return 0.82 + pct * 0.36;
    90→}

    92→/** الو تقریبی از درصد امتیاز فصل — ۱۵۰۰ = میانگین */
    93→function eloFromRecord(record: string | null): number {
    94→  const pct = recordToPointsPct(record);
    95→  if (pct === null) return 1500;
    96→  return Math.round(Math.max(1400, Math.min(2100, 1500 + (pct - 0.5) * 800)));

   134→/**
   135→ * v6.0 — محاسبه امید گل (λ) از جدول رسمی رده‌بندی فصل جاری ESPN (Standings API بدون کلید)
   136→ * مسیر اولویت‌دار (قابل راستی‌آزمایی ۱۰۰٪):
   137→ *   α_i = GF_i / (GP_i × LeagueTeamAvg)   ← قدرت تهاجمی واقعی از گل زده فصل
   138→ *   β_i = GA_i / (GP_i × LeagueTeamAvg)   ← ضعف دفاعی واقعی از گل خورده فصل
   139→ *   λ_Home = α_Home × β_Away × LeagueTeamAvg × 1.12 (فرمول استاندارد دیکسون-کولز)
   140→ *   λ_Away = α_Away × β_Home × LeagueTeamAvg
   141→ * مسیر جایگزین شفاف (فقط وقتی جدول در دسترس نیست): رکورد واقعی W-D-L + فرم ۵ بازی.
   142→ * تعدیل جوی: بارش > ۲mm یا باد > ۳۰km/h → λ × ۰.۹۲ (−۸٪)
   143→ */
   144→function deriveStrength(ev: LiveMatchEvent, weather: LiveWeather | null, st: LeagueStandings | null): DerivedStrength {
   145→  const adverse = weather?.adverse ?? false;
   146→  const wAdj = adverse ? 0.92 : 1.0;
   147→
   148→  const homeEntry = findStandingsEntry(st, ev.homeTeamId, ev.homeTeam);
   149→  const awayEntry = findStandingsEntry(st, ev.awayTeamId, ev.awayTeam);
   150→  const lta = st && st.leagueTeamAvg > 0.5 ? st.leagueTeamAvg : null;
   151→
   152→  if (lta && homeEntry && awayEntry && homeEntry.gp >= 3 && awayEntry.gp >= 3) {
   153→    // ═══ v6.0 — مسیر واقعی: α/β از جدول رسمی فصل ═══
   154→    const alphaH = homeEntry.gf / (homeEntry.gp * lta);
   155→    const betaH = homeEntry.ga / (homeEntry.gp * lta);
   156→    const alphaA = awayEntry.gf / (awayEntry.gp * lta);
   157→    const betaA = awayEntry.ga / (awayEntry.gp * lta);
   158→
   159→    let lambdaHome = alphaH * betaA * lta * 1.12 * wAdj;
   160→    let lambdaAway = alphaA * betaH * lta * wAdj;
   161→    lambdaHome = clamp(lambdaHome, 0.3, 3.8);
   162→    lambdaAway = clamp(lambdaAway, 0.25, 3.6);
   163→
   164→    const homeElo = eloFromRecord(ev.homeRecord);
   165→    const awayElo = eloFromRecord(ev.awayRecord);
   166→
   167→    const homePower = clamp(alphaH, 0.5, 2.0);
   168→    const awayPower = clamp(alphaA, 0.5, 2.0);
   169→
   170→    return {
   171→      lambdaHome: r2(lambdaHome),
   172→      lambdaAway: r2(lambdaAway),
   173→      // قدرت‌های واقعی دیکسون-کولز — λ = α × β × leagueTeamAvg × homeAdv
   174→      homeAttack: r2(clamp(alphaH, 0.35, 2.3)),
   175→      awayAttack: r2(clamp(alphaA, 0.3, 2.2)),
   176→      homeDefense: r2(clamp(betaH, 0.45, 1.9)),
   177→      awayDefense: r2(clamp(betaA, 0.45, 1.9)),
   178→      homeElo,
   179→      awayElo,
   180→      homePpda: r2(clamp(12 - (homePower - 1) * 7, 8.5, 15)),
   181→      awayPpda: r2(clamp(12.4 - (awayPower - 1) * 7, 8.5, 15.5)),
   182→      homeFieldTilt: Math.round(clamp(50 + (homeElo - 1500) / 12, 42, 63)),
   183→      awayFieldTilt: Math.round(clamp(50 + (awayElo - 1500) / 12, 40, 62)),
   184→      weatherAdjusted: adverse,
   185→      statsSource: "ESPN_STANDINGS",
   186→      homeRank: homeEntry.rank || null,
   187→      homeGF: homeEntry.gf,
   188→      homeGA: homeEntry.ga,
   189→      homeGP: homeEntry.gp,
   190→      awayRank: awayEntry.rank || null,
   191→      awayGF: awayEntry.gf,
   192→      awayGA: awayEntry.ga,
   193→      awayGP: awayEntry.gp,
   194→      leagueTeamAvg: r2(lta),
   195→    };
   196→  }
   197→
   198→  // ═══ مسیر شفاف جایگزین — فقط رکورد/فرم واقعی ESPN (جدول رده‌بندی در دسترس نیست) ═══
   199→  const homeStr = 0.55 * formFactor(ev.homeForm) + 0.45 * seasonFactor(ev.homeRecord);
   200→  const awayStr = 0.55 * formFactor(ev.awayForm) + 0.45 * seasonFactor(ev.awayRecord);
   201→
   202→  let lambdaHome = ev.avgGoals * 0.55 * homeStr * wAdj;
   203→  let lambdaAway = ev.avgGoals * 0.45 * awayStr * wAdj;
   204→
   205→  // همگرایی به محدوده معقول پواسون
   206→  lambdaHome = clamp(lambdaHome, 0.3, 3.8);
   207→  lambdaAway = clamp(lambdaAway, 0.25, 3.6);
   208→
   209→  // استخراج ضرایب حمله/دفاع سازگار با موتور دیکسون-کولز:
   210→  // λh = homeAttack × awayDefense × leagueAvg × 1.12 → دفاع‌ها = ۱، حمله = λ / ثابت
   211→  const homeAttack = clamp(lambdaHome / (1.35 * 1.12), 0.35, 2.3);
   212→  const awayAttack = clamp(lambdaAway / 1.35, 0.3, 2.2);
   213→
   214→  const homeElo = eloFromRecord(ev.homeRecord);
   215→  const awayElo = eloFromRecord(ev.awayRecord);
   216→
   217→  const homeFieldTilt = Math.round(clamp(50 + (homeElo - 1500) / 12, 42, 63));
   218→  const awayFieldTilt = Math.round(clamp(50 + (awayElo - 1500) / 12, 40, 62));
   219→  const homePpda = r2(clamp(12 - (homeStr - 1) * 7, 8.5, 15));
   220→  const awayPpda = r2(clamp(12.4 - (awayStr - 1) * 7, 8.5, 15.5));
   221→
   222→  return {
   223→    lambdaHome: r2(lambdaHome),
   224→    lambdaAway: r2(lambdaAway),
   225→    homeAttack: r2(homeAttack),
   226→    awayAttack: r2(awayAttack),
   227→    homeDefense: 1.0,
   228→    awayDefense: 1.0,
   229→    homeElo,
   230→    awayElo,
   231→    homePpda,
   232→    awayPpda,
   233→    homeFieldTilt,
   234→    awayFieldTilt,
   235→    weatherAdjusted: adverse,
   236→    statsSource: "RECORD_FALLBACK",
   237→    homeRank: null,
   238→    homeGF: null,
   239→    homeGA: null,
   240→    homeGP: null,
   241→    awayRank: null,
   242→    awayGF: null,
   243→    awayGA: null,
   244→    awayGP: null,
   245→    leagueTeamAvg: null,
   246→  };
   247→}

   256→function modelProbs(d: DerivedStrength) {
   257→  return runDixonColes({
   258→    homeAttack: d.homeAttack,
   259→    awayAttack: d.awayAttack,
   260→    homeDefense: d.homeDefense,
   261→    awayDefense: d.awayDefense,
   262→    leagueAvg: d.leagueTeamAvg ?? 1.35,
   263→    homeAdvantage: 1.12,
   264→    rho: -0.08,
   265→    eloDiff: d.homeElo - d.awayElo,
   266→    eloWeight: 0.35,
   267→    homeAbsenceImpact: 0,
   268→    awayAbsenceImpact: 0,
   269→    timeDecayXi: 0.0065,
   270→    daysAgo: 21,
   271→  });
   272→}

   276→function matchOddsEvent(ev: LiveMatchEvent, oddsEvents: OddsApiEvent[]): OddsApiEvent | null {
   277→  const direct = `${normTeam(ev.homeTeam)}::${normTeam(ev.awayTeam)}`;
   278→  const kick = Date.parse(ev.kickoff);
   279→  let best: OddsApiEvent | null = null;
   280→  let bestDist = Infinity;
   281→  for (const oe of oddsEvents) {
   282→    const same = oe.matchKey === direct || `${normTeam(oe.homeTeam)}::${normTeam(oe.awayTeam)}` === `${normTeam(ev.awayTeam)}::${normTeam(ev.homeTeam)}`;
   283→    if (!same) continue;
   284→    const dist = Math.abs(Date.parse(oe.commenceTime) - kick);
   285→    if (dist < bestDist) {
   286→      bestDist = dist;
   287→      best = oe;
   288→    }
   289→  }
   290→  // تطبیق فقط برای بازی‌های نزدیک (±۲۴h) — جلوگیری از جفت‌شدن بازی‌های دور
   291→  return best && bestDist < 24 * 3600 * 1000 ? best : null;
   292→}

   294→interface ResolvedOdds {
   295→  oddsHome: number;
   296→  oddsDraw: number;
   297→  oddsAway: number;
   298→  oddsOver25: number;
   299→  oddsUnder25: number;
   300→  odds1x: number;
   301→  source: "ENGINE" | "ESPN_DRAFTKINGS" | "ODDS_API";
   302→  /** تا ۳ ردیف بوک‌میکر با کلیدهای کنونیکال (Pinnacle = کم‌حاشیه‌ترین) */
   303→  books: { bookmaker: "PINNACLE" | "SOFT_A" | "SOFT_B"; bookmakerLabel: string; home: number; draw: number; away: number; over25: number | null; under25: number | null }[];
   304→}

   306→function resolveOdds(ev: LiveMatchEvent, oddsEv: OddsApiEvent | null, probs: ReturnType<typeof modelProbs>): ResolvedOdds {
   307→  // ─── ۱) ضرایب زنده The Odds API — بهترین قیمت هر انتخاب + سه بوک نماینده ───
   308→  if (oddsEv && oddsEv.books.length > 0) {
   309→    const booksSorted = [...oddsEv.books].sort(
   310→      (a, b) => 1 / a.line.home + 1 / a.line.draw + 1 / a.line.away - (1 / b.line.home + 1 / b.line.draw + 1 / b.line.away)
   311→    );
   312→    const canon = ["PINNACLE", "SOFT_A", "SOFT_B"] as const;
   313→    const books = booksSorted.slice(0, 3).map((b, i) => ({
   314→      bookmaker: canon[i],
   315→      bookmakerLabel: `${b.label}${i === 0 ? " (Sharp)" : ""}`,
   316→      home: b.line.home,
   317→      draw: b.line.draw,
   318→      away: b.line.away,
   319→      over25: b.line.over25,
   320→      under25: b.line.under25,
   321→    }));
   322→    const best = (sel: (b: ResolvedOdds["books"][number]) => number | null): number => {
   323→      const vals = books.map(sel).filter((v): v is number => v !== null && v >= 1.01);
   324→      return vals.length > 0 ? Math.max(...vals) : 0;
   325→    };
   326→    const oddsHome = best((b) => b.home);
   327→    const oddsDraw = best((b) => b.draw);
   328→    const oddsAway = best((b) => b.away);
   329→    let oddsOver25 = best((b) => b.over25);
   330→    let oddsUnder25 = best((b) => b.under25);
   331→    if (oddsOver25 <= 1.01 || oddsUnder25 <= 1.01) {
   332→      oddsOver25 = fairOdds(probs.pOver25, 1.06);
   333→      oddsUnder25 = fairOdds(probs.pUnder25, 1.06);
   334→    }
   335→    return { oddsHome, oddsDraw, oddsAway, oddsOver25, oddsUnder25, odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06), source: "ODDS_API", books };
   336→  }
   337→
   338→  // ─── ۲) خط پایه ESPN (معمولاً DraftKings) ───
   339→  if (ev.espnOdds && ev.espnOdds.home && ev.espnOdds.draw && ev.espnOdds.away) {
   340→    const o = ev.espnOdds;
   341→    const h = o.home as number;
   342→    const d2 = o.draw as number;
   343→    const a2 = o.away as number;
   344→    return {
   345→      oddsHome: h,
   346→      oddsDraw: d2,
   347→      oddsAway: a2,
   348→      oddsOver25: o.over25 ?? fairOdds(probs.pOver25, 1.06),
   349→      oddsUnder25: o.under25 ?? fairOdds(probs.pUnder25, 1.06),
   350→      odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06),
   351→      source: "ESPN_DRAFTKINGS",
   352→      books: [
   353→        {
   354→          bookmaker: "SOFT_A",
   355→          bookmakerLabel: `${o.provider} (ESPN)`,
   356→          home: h,
   357→          draw: d2,
   358→          away: a2,
   359→          over25: o.over25 ?? fairOdds(probs.pOver25, 1.06),
   360→          under25: o.under25 ?? fairOdds(probs.pUnder25, 1.06),
   361→        },
   362→      ],
   363→    };
   364→  }
   365→
   366→  // ─── ۳) Fallback مدل — ضرایب منصفانه با حاشیه ۶٪ ───
   367→  return {
   368→    oddsHome: fairOdds(probs.pHome, 1.06),
   369→    oddsDraw: fairOdds(probs.pDraw, 1.06),
   370→    oddsAway: fairOdds(probs.pAway, 1.06),
   371→    oddsOver25: fairOdds(probs.pOver25, 1.06),
   372→    oddsUnder25: fairOdds(probs.pUnder25, 1.06),
   373→    odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06),
   374→    source: "ENGINE",
   375→    books: [],
   376→  };
   377→}

   381→async function weatherFor(city: string | null): Promise<LiveWeather | null> {
   382→  if (!city) return null;
   383→  const coords = cityCoords(city);
   384→  if (!coords) return null;
   385→  const hit = cache.weather.get(city);
   386→  if (hit && Date.now() - hit.at < WEATHER_TTL_MS) return hit.w;
   387→  const w = await fetchWeather(coords[0], coords[1], city);
   388→  cache.weather.set(city, { w, at: Date.now() });
   389→  return w;
   390→}

   392→async function poolWeather(cities: (string | null)[], limit = 3): Promise<Map<string, LiveWeather | null>> {
   393→  const unique = [...new Set(cities.filter((c): c is string => Boolean(c)))];
   394→  const out = new Map<string, LiveWeather | null>();
   395→  const queue = [...unique];
   396→  const workers = Array.from({ length: Math.min(limit, queue.length) }, async () => {
   397→    for (;;) {
   398→      const c = queue.shift();
   399→      if (c === undefined) return;
   400→      out.set(c, await weatherFor(c));
   401→    }
   402→  });
   403→  await Promise.all(workers);
   404→  return out;
   405→}

   417→async function logCalibrationFor(matchId: string, label: string, league: string, hg: number, ag: number): Promise<void> {
   418→  try {
   419→    const dup = await db.calibrationLog.findFirst({ where: { matchLabel: label } });
   420→    if (dup) return;
   421→    const analysis = await db.agentAnalysis.findFirst({
   422→      where: { matchId },
   423→      orderBy: { createdAt: "desc" },
   424→    });
   425→    if (!analysis) return;
   426→
   427→    const cands: { key: string; p: number }[] = [
   428→      { key: "HOME", p: analysis.pHome },
   429→      { key: "DRAW", p: analysis.pDraw },
   430→      { key: "AWAY", p: analysis.pAway },
   431→      { key: "O25", p: analysis.pOver25 },
   432→    ];
   433→    cands.sort((a, b) => b.p - a.p);
   434→    const pick = cands[0];
   435→    const actual = pick.key === "HOME" ? hg > ag : pick.key === "DRAW" ? hg === ag : pick.key === "AWAY" ? ag > hg : hg + ag > 2.5;
   436→    await db.calibrationLog.create({
   437→      data: {
   438→        matchLabel: label,
   439→        league,
   440→        pick: PICK_LABELS[pick.key],
   441→        predictedProb: pick.p,
   442→        outcome: actual ? "WIN" : "LOSS",
   443→        brier: (pick.p - (actual ? 1 : 0)) ** 2,
   444→        lesson: null,
   445→      },
   446→    });
   447→  } catch (err) {
   448→    console.warn("calibration log failed:", err);
   449→  }
   450→}
