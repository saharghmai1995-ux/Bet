    36→function statusBadge(status: string): { label: string; className: string } {
    37→  switch (status) {
    38→    case "FIT":
    39→      return { label: "آماده", className: "border-emerald-500/40 text-emerald-400" };
    40→    case "INJURED":
    41→      return { label: "مصدوم", className: "border-rose-500/40 text-rose-400" };
    42→    case "SUSPENDED":
    43→      return { label: "محروم", className: "border-rose-500/40 text-rose-400" };
    44→    case "DOUBTFUL":
    45→      return { label: "مشکوک", className: "border-amber-500/40 text-amber-300" };
    46→    default:
    47→      return { label: status, className: "" };
    48→  }

    53→/** نودهای زمین از بازیکنان (T-60) — غایبان با نود قرمز چشمک‌زن */
    54→function pitchNodes(topPlayers: LineupShockDTO["topPlayers"], side: "HOME" | "AWAY"): PitchNode[] {
    55→  const HOME_FORMATION: PitchNode[] = [
    56→    { x: 7, y: 50 }, { x: 22, y: 18 }, { x: 20, y: 39 }, { x: 20, y: 61 }, { x: 22, y: 82 },
    57→    { x: 40, y: 30 }, { x: 37, y: 50 }, { x: 40, y: 70 },
    58→    { x: 62, y: 22 }, { x: 66, y: 50 }, { x: 62, y: 78 },
    59→  ];
    60→  const AWAY_FORMATION: PitchNode[] = [
    61→    { x: 93, y: 50 }, { x: 78, y: 18 }, { x: 80, y: 39 }, { x: 80, y: 61 }, { x: 78, y: 82 },
    62→    { x: 60, y: 30 }, { x: 63, y: 50 }, { x: 60, y: 70 },
    63→    { x: 38, y: 22 }, { x: 34, y: 50 }, { x: 38, y: 78 },
    64→  ];
    65→  const players = topPlayers.filter((p) => p.side === side);
    66→  const formation = side === "HOME" ? HOME_FORMATION : AWAY_FORMATION;
    67→  return formation.map((slot, i) => {
    68→    const p = players[i];
    69→    if (!p) return slot;
    70→    const status: PitchNode["status"] =
    71→      p.status === "INJURED" || p.status === "SUSPENDED" ? "MISSING" : p.status === "DOUBTFUL" ? "DOUBTFUL" : "FIT";
    72→    return { ...slot, status, label: `${p.name} (${posLabel(p.position)}) — ${statusBadge(p.status).label}` };
    73→  });
    74→}

   308→function LiveHedgeCard() {

   311→  const setActiveMatchId = useQuantStore((s) => s.setActiveMatchId);

   315→  const [matchId, setMatchId] = useState("");
   316→  const [selection, setSelection] = useState<"HOME" | "OVER25" | "AWAY">("HOME");
   317→  const [odds, setOdds] = useState("2.10");
   318→  const [stake, setStake] = useState("100");
   319→  const [minute, setMinute] = useState(25);
   320→  const [running, setRunning] = useState(false);
   321→  const [pendingEvent, setPendingEvent] = useState<string | null>(null);
   322→  const [result, setResult] = useState<LiveHedgeDTO | null>(null);
   323→
   324→  useEffect(() => {
   325→    if (activeMatchId) setMatchId(activeMatchId);

   328→  const hasBase = Boolean(matchId && analyses[matchId]);

   363→        toast.error("ارتباط با سرور برقرار نشد");

   366→        setPendingEvent(null);

   392→          <Shield className="size-4 text-emerald-400" />

   404→        <Select
   405→          value={matchId}
