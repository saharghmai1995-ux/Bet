   656→ * خط مرجع: پیناکل وقتی خط شارپ واقعی موجود است؛ وگرنه خط منصفانه موتور (MARKET_FAIR)
   657→ * خروجی همیشه عدد دارد — «—» فقط برای بازار خارج ماتریس (کرنر/کارت).
   658→ */
   659→function computeEdge(modelProb: number | null, odds: number): number | null {
   660→  if (modelProb === null || modelProb <= 0 || odds <= 1) return null;
   661→  return Math.round((modelProb * odds - 1) * 1000) / 10;
   662→}

   664→/**
   665→ * صدور کارت به کارنامه — Idempotent با هش SHA-256.
   666→ * اگر کارت با همین هش قبلاً صادر شده باشد، همان ردیف بازگردانده می‌شود.
   667→ * v9.0: نگهداری λ ترکیبی نهایی + کد الگو (حلقه کالیبراسیون) + نشان تایید سوپروایزر
   668→ * و پرچم ثبت خودکار سرور (autoIssued) برای خط لوله Zero-Click.
   669→ */
   670→export async function issueSlip(
   671→  item: IssueSlipItemDTO,
   672→  opts?: { autoIssued?: boolean }
   673→): Promise<{ created: boolean; slip: IssuedSlipDTO }> {
   674→  const externalId = item.match.externalId ?? null;
   675→  // v19.0 — استیک صریح حتی وقتی stakeB=0 است (کارت‌های تک‌ستون BANKER و آکومولاتور):
   676→  // اگر فراخواننده هر دو استیک را صریح داده، دیگر به داچینگ سود-برابر برنگرد.
   677→  const stakes =
   678→    typeof item.stakeA === "number" && typeof item.stakeB === "number" && item.stakeA + item.stakeB > 0
   679→      ? { stakeA: item.stakeA, stakeB: item.stakeB }
   680→      : equalProfitStakes(item.oddsA, item.oddsB);
   681→
   682→  const hash = slipHash({
   683→    externalId,
   684→    homeTeam: item.match.homeTeam,
   685→    awayTeam: item.match.awayTeam,
   686→    kickoff: item.match.kickoff,
   687→    labelA: item.labelA,
   688→    labelB: item.labelB,
   689→    oddsA: item.oddsA,
   690→    oddsB: item.oddsB,
   691→    stakeA: stakes.stakeA,
   692→    stakeB: stakes.stakeB,
   693→  });
   694→
   695→  const existing = await db.issuedSlip.findUnique({ where: { hash } });
   696→  if (existing) return { created: false, slip: mapSlip(existing) };
   697→
   698→  // ─── v10.0 — احتمال مدل، ارزش عددی و پنجره زمانی طلایی ثبت ───
   699→  // v19.0 — کارت‌های میکس چندبازی (ACCUMULATOR) احتمال مدل فراخلب دارند:
   700→  // حاصل‌ضرب احتمال هر پای مستقل — ماتریس تک‌بازی برای میکس چندبازی معنا ندارد.
   701→  const lH = item.lambdaHome ?? 1.35;
   702→  const lA = item.lambdaAway ?? 1.25;
   703→  const probA = item.modelProbAOverride != null ? item.modelProbAOverride : legProbability(lH, lA, item.legsA ?? []);
   704→  const probB = item.modelProbBOverride != null ? item.modelProbBOverride : legProbability(lH, lA, item.legsB ?? []);
   705→
   706→  // ─── v10.0.1 — خط شارپ واقعی پیناکل برای انتخاب مستقیم (پاسخ «چرا عددی ندارد؟») ───
   707→  // ارزش نسبت به خط مرجع:
   708→  //   · PINNACLE موجود → (ضریب ثبت ÷ خط پیناکل − ۱) × ۱۰۰ — مزیت قیمت نسبت به خط شارپ
   709→  //   · در غیر این صورت → (احتمال مدل × ضریب ثبت − ۱) × ۱۰۰ — EV نسبت به خط منصفانه موتور
   710→  const pinPrices = item.pinnaclePrices ?? {};
   711→  const pinPriceOf = (legs: SettlementLegDTO[]): number | null => {
   712→    const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   713→    if (keys.length !== 1) return null; // تیپ ترکیبی چندبازاره — خط مستقیم پیناکل ندارد
   714→    const p = pinPrices[keys[0]];
   715→    return typeof p === "number" && p > 1.01 ? p : null;
   716→  };
   717→  const pinA = pinPriceOf(item.legsA ?? []);
   718→  const pinB = pinPriceOf(item.legsB ?? []);
   719→  const priceAdvantage = (odds: number, pin: number): number => Math.round((odds / pin - 1) * 1000) / 10;
   720→
   721→  const edgeA = pinA !== null ? priceAdvantage(item.oddsA, pinA) : computeEdge(probA, item.oddsA);
   722→  const edgeB = pinB !== null ? priceAdvantage(item.oddsB, pinB) : computeEdge(probB, item.oddsB);
   723→  const edgeSourceA = edgeA !== null ? (pinA !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   724→  const edgeSourceB = edgeB !== null ? (pinB !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   725→  const tipKindA: TipKind | null = edgeA !== null ? deriveTipKind((item.legsA ?? []).map((l) => l.key), item.oddsA) : null;
   726→  const tipKindB: TipKind | null = edgeB !== null ? deriveTipKind((item.legsB ?? []).map((l) => l.key), item.oddsB) : null;
   727→  const koMs = new Date(item.match.kickoff).getTime();
   728→  const winA = tipKindA ? computeTimingWindowStatic(koMs, tipKindA) : null;
   729→  const winB = tipKindB ? computeTimingWindowStatic(koMs, tipKindB) : null;
   730→
   731→  const row = await db.issuedSlip.create({
   732→    data: {
   733→      hash,
   734→      externalId,
   735→      matchId: item.match.id,
   736→      league: item.match.league,
   737→      homeTeam: item.match.homeTeam,
   738→      awayTeam: item.match.awayTeam,
   739→      kickoff: new Date(item.match.kickoff),
   740→      labelA: item.labelA,
   741→      labelB: item.labelB,
   742→      legsA: JSON.stringify(item.legsA ?? []),
   743→      legsB: JSON.stringify(item.legsB ?? []),
   744→      oddsA: item.oddsA,
   745→      oddsB: item.oddsB,
   746→      stakeA: stakes.stakeA,
   747→      stakeB: stakes.stakeB,
   748→      cellsA: JSON.stringify(coveredCells(lH, lA, item.legsA ?? [], externalId)),
   749→      cellsB: JSON.stringify(coveredCells(lH, lA, item.legsB ?? [], externalId)),
   750→      guaranteedProfit: item.guaranteedProfit,
   751→      marginPct: item.marginPct,
   752→      archetype: item.archetype ?? null,
   753→      archetypeCode: item.archetypeCode ?? null,
   754→      lambdaHome: item.lambdaHome ?? null,
   755→      lambdaAway: item.lambdaAway ?? null,
   756→      espnUrl: item.match.espnUrl ?? espnUrlOf(externalId),
   757→      supervisorApproved: item.supervisorApproved !== false,
   758→      autoIssued: opts?.autoIssued === true,
   759→      retroIssued: item.retroIssued === true,
   760→      // v18.0 — سطح تاج کارت (GOLD / SILVER_SHADOW) برای حصاد آلفای دفتر سایه
   761→      crownTier: item.crownTier ?? null,
   762→      supervisorScore: item.supervisorScore ?? null,
   763→      vetoReason: item.vetoReason ?? null,
   764→      jointLossRiskPct: item.jointLossRiskPct ?? null,
   765→      doubleWinProbPct: item.doubleWinProbPct ?? null,
   766→      modelProbA: probA,
   767→      modelProbB: probB,
   768→      pinnacleOddsA: pinA,
   769→      pinnacleOddsB: pinB,
   770→      edgeA,
   771→      edgeB,
   772→      edgeSourceA,
   773→      edgeSourceB,
   774→      tipKindA,
   775→      tipKindB,
   776→      timingWindowA: winA?.windowKey ?? null,
   777→      timingWindowB: winB?.windowKey ?? null,
   778→      timingStartA: winA?.startAt ?? null,
   779→      timingEndA: winA?.endAt ?? null,
   780→      timingStartB: winB?.startAt ?? null,
   781→      timingEndB: winB?.endAt ?? null,
   782→      // v17.0 — خط تولید Bet-Type برای افراز مجزای کارنامه (پیش‌فرض: SINGLE زوج‌تیپ)
   783→      productLine: item.productLine ?? "SINGLE",
   784→      status: "PENDING",
   785→    },
   786→  });
   787→  return { created: true, slip: mapSlip(row) };
   788→}
   789→
   790→// ═══════════ v10.0.2 — بازپررسی یادگیری کارت‌های قدیمی (Idempotent Backfill) ═══════════
   791→
   792→/** throttle بازپررسی — حداکثر هر ۳۰ دقیقه یک‌بار (فراخوانی از خط لوله سرور) */
   793→let lastBackfillAt = 0;
   794→const BACKFILL_TTL_MS = 30 * 60_000;
   795→
   796→/**
   797→ * backfillSlipLearning — پرکردن ستون‌های یادگیری کارت‌های صادرشده پیش از v10:
   798→ *   · modelProbA/B و edgeA/B (ارزش عددی نسبت به پیناکل واقعی یا خط منصفانه)
   799→ *   · tipKind + پنجره زمانی طلایی ثبت
   800→ *   · ردیف ژورنال تسویه (recordSettlementLearning — Idempotent با slipId)
   801→ * همه محاسبات از λ ذخیره‌شده کارت بازتولید می‌شود — هیچ عدد ساختگی نیست.
   802→ */
   803→export async function backfillSlipLearning(force = false): Promise<{ scanned: number; updated: number; journalled: number }> {
   804→  if (!force && Date.now() - lastBackfillAt < BACKFILL_TTL_MS) {
   805→    return { scanned: 0, updated: 0, journalled: 0 };
   806→  }
   807→  lastBackfillAt = Date.now();
   808→
   809→  // کارت‌های تسویه‌شده بدون ارزش عددی (نسخه‌های ≤ v9) — سقف هر پاس ۶۰ ردیف
   810→  const targets = await db.issuedSlip.findMany({
   811→    where: {
   812→      status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP", "MANUAL_REVIEW"] },
   813→      OR: [
   814→        { edgeA: null },
   815→        { edgeB: null },
   816→        { timingWindowA: null },
   817→        { timingWindowB: null },
   818→        { jointLossRiskPct: null },
   819→        { doubleWinProbPct: null },
   820→      ],
   821→    },
   822→    orderBy: { kickoff: "desc" },
   823→    take: 60,
   824→  });
   825→  if (targets.length === 0) return { scanned: 0, updated: 0, journalled: 0 };
   826→
   827→  let updated = 0;
   828→  let journalled = 0;
   829→  for (const s of targets) {
   830→    try {
   831→      const legsA: SettlementLegDTO[] = safeParseLegs(s.legsA);
   832→      const legsB: SettlementLegDTO[] = safeParseLegs(s.legsB);
   833→      const lH = s.lambdaHome ?? 1.35;
   834→      const lA = s.lambdaAway ?? 1.25;
   835→      const probA = s.modelProbA ?? legProbability(lH, lA, legsA);
   836→      const probB = s.modelProbB ?? legProbability(lH, lA, legsB);
   837→
   838→      // خط شارپ واقعی پیناکل — از رکورد OddsRecord مرتبط با بازی (وقتی فید موجود باشد)
   839→      let pinPrices: Record<string, number> = {};
   840→      if (s.matchId) {
   841→        const oddsRows = await db.oddsRecord.findMany({ where: { matchId: s.matchId } });
   842→        const pin = oddsRows.find((o) => o.bookmaker === "PINNACLE");
   843→        if (pin) {
   844→          pinPrices = {
   845→            HOME: pin.oddsHome,
   846→            DRAW: pin.oddsDraw,
   847→            AWAY: pin.oddsAway,
   848→            DC_1X: 1 / (1 / pin.oddsHome + 1 / pin.oddsDraw),
   849→            DC_X2: 1 / (1 / pin.oddsAway + 1 / pin.oddsDraw),
   850→            O25: pin.oddsOver25,
   851→            U25: pin.oddsUnder25,
   852→          };
   853→        }
   854→      }
   855→      const pinPriceOf = (legs: SettlementLegDTO[]): number | null => {
   856→        const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   857→        if (keys.length !== 1) return null;
   858→        const p = pinPrices[keys[0]];
   859→        return typeof p === "number" && p > 1.01 ? p : null;
   860→      };
   861→      const priceAdvantage = (odds: number, pin: number): number => Math.round((odds / pin - 1) * 1000) / 10;
   862→
   863→      const pinA = s.pinnacleOddsA ?? pinPriceOf(legsA);
   864→      const pinB = s.pinnacleOddsB ?? pinPriceOf(legsB);
   865→      const edgeA = s.edgeA ?? (pinA !== null ? priceAdvantage(s.oddsA, pinA) : computeEdge(probA, s.oddsA));
   866→      const edgeB = s.edgeB ?? (pinB !== null ? priceAdvantage(s.oddsB, pinB) : computeEdge(probB, s.oddsB));
   867→      const edgeSourceA = edgeA !== null ? (pinA !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   868→      const edgeSourceB = edgeB !== null ? (pinB !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   869→
   870→      const tipKindA = (s.tipKindA as TipKind | null) ?? (edgeA !== null ? deriveTipKind(legsA.map((l) => l.key), s.oddsA) : null);
   871→      const tipKindB = (s.tipKindB as TipKind | null) ?? (edgeB !== null ? deriveTipKind(legsB.map((l) => l.key), s.oddsB) : null);
   872→      const koMs = s.kickoff.getTime();
   873→      const winA = s.timingWindowA ? null : tipKindA ? computeTimingWindowStatic(koMs, tipKindA) : null;
   874→      const winB = s.timingWindowB ? null : tipKindB ? computeTimingWindowStatic(koMs, tipKindB) : null;
   875→
   876→      // ─── v11.0 — سنجه‌های دفتر سایه: بازتولید ٪ باخت همزمان و ٪ برد طلایی از λ ذخیره‌شده ───
   877→      let jointLossRiskPct: number | undefined;
   878→      let doubleWinProbPct: number | undefined;
   879→      if (s.jointLossRiskPct === null || s.doubleWinProbPct === null) {
   880→        const matchRow = s.matchId ? await db.match.findUnique({ where: { id: s.matchId }, include: { referee: true } }) : null;
   881→        if (matchRow) {
   882→          const eng = computeMatchEngine(matchRow);
   883→          const m = jointShadowMetrics(
   884→            lH,
   885→            lA,
   886→            eng.lambdaCorners,
   887→            eng.lambdaCards,
   888→            legsA.map((l) => l.key),
   889→            legsB.map((l) => l.key)
   890→          );
   891→          jointLossRiskPct = s.jointLossRiskPct ?? m.jointLossRiskPct;
   892→          doubleWinProbPct = s.doubleWinProbPct ?? m.doubleWinProbPct;
   893→        }
   894→      }
   895→
   896→      const refreshed = await db.issuedSlip.update({
   897→        where: { id: s.id },
   898→        data: {
   899→          modelProbA: s.modelProbA ?? probA,
   900→          modelProbB: s.modelProbB ?? probB,
   901→          jointLossRiskPct,
   902→          doubleWinProbPct,
   903→          pinnacleOddsA: pinA,
   904→          pinnacleOddsB: pinB,
   905→          edgeA,
   906→          edgeB,
   907→          edgeSourceA,
   908→          edgeSourceB,
   909→          tipKindA: tipKindA ?? undefined,
   910→          tipKindB: tipKindB ?? undefined,
   911→          timingWindowA: s.timingWindowA ?? winA?.windowKey ?? undefined,
   912→          timingWindowB: s.timingWindowB ?? winB?.windowKey ?? undefined,
   913→          timingStartA: s.timingStartA ?? winA?.startAt ?? undefined,
   914→          timingEndA: s.timingEndA ?? winA?.endAt ?? undefined,
   915→          timingStartB: s.timingStartB ?? winB?.startAt ?? undefined,
   916→          timingEndB: s.timingEndB ?? winB?.endAt ?? undefined,
   917→        },
   918→      });
   919→      updated += 1;
   920→
   921→      // ژورنال یادگیری — Idempotent (هر کارت حداکثر یک ردیف)
   922→      const hasJournal = await db.settlementJournal.findFirst({ where: { slipId: s.id }, select: { id: true } });
   923→      if (!hasJournal) {
   924→        await recordSettlementLearning(refreshed);
   925→        journalled += 1;
   926→      }
   927→    } catch (err) {
   928→      console.warn(`backfill failed for slip ${s.id}:`, err instanceof Error ? err.message : err);
   929→    }
   930→  }
   931→  return { scanned: targets.length, updated, journalled };
   932→}

  1003→export async function getLedger(): Promise<LedgerDTO> {
  1004→  // v13.5.1 — رفع باگ محاسباتی منحنی بانک: آمار تیتر (سود خالص مجموع) از «تمام تاریخ» محاسبه
  1005→  // می‌شود، نه از ۱۲۰ ردیف آخر؛ وگرنه با منحنی بانک (تمام تاریخ) نمی‌خواند.
  1006→  // ردیف‌های DTO برای نمایش به ۱۲۰ ردیف آخر محدود می‌شوند (سقف نمایش، نه سقف محاسبه).
  1007→  const [allRows, outboxRows, tgStatus] = await Promise.all([
  1008→    db.issuedSlip.findMany({ orderBy: [{ status: "asc" }, { kickoff: "desc" }] }),
  1009→    db.telegramOutbox.findMany({ orderBy: { createdAt: "desc" }, take: 25, include: { slip: { select: { hash: true } } } }),
  1010→    import("./telegram-service").then((m) => m.telegramStatus()),
  1011→  ]);
  1012→
  1013→  const slips = allRows.map((s) => mapSlip(s));
  1014→  const slipsView = slips.slice(0, 120); // سقف نمایش جدول کارنامه
  1015→  // ─── v11.0 — آمار تیتر کارنامه فقط از کارت‌های تایید سوپروایزر؛ وتوها به «دفتر سایه» می‌روند ───
  1016→  // v13.2 — کارت‌های بازخوانی (retroIssued — بازی FT پیش از صدور) از آمار اصلی حذف و به دسته ه می‌روند
  1017→  // v18.0 — کارت‌های تاج نقره‌ای (crownTier=SILVER_SHADOW) از کتاب طلایی جدا شده‌اند
  1018→  // (افراز مجزای Two-Tier Crown — استیک ۵۰u Half-Kelly) و در silverHarvest شمرده می‌شوند.
  1019→  const goldSlips = slips.filter((s) => s.supervisorApproved && !s.retroIssued && s.crownTier !== "SILVER_SHADOW");

  1118→      if (s.status === "MIDDLE_TRAP") retroLosses += 1;
