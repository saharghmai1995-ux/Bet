    62→/**
    63→ * ارزیابی همه شرط‌های یک تیپ با نتایج واقعی ESPN.
    64→ * هر شرط ممکن است روی بازی دیگری باشد (میکس دو بازی) → امتیاز همان بازی از نقشه اسکوربورد.
    65→ */
    66→function gradeLegs(
    67→  legs: SettlementLegDTO[],
    68→  scoreOf: (externalId: string) => { hg: number; ag: number; state: string } | undefined,
    69→  fallbackExternalId: string | null
    70→): { graded: GradedLeg[]; allWon: boolean; anyLost: boolean; hasOffBoard: boolean; allFinal: boolean } {
    71→  const graded: GradedLeg[] = [];
    72→  let allWon = true;
    73→  let anyLost = false;
    74→  let hasOffBoard = false;
    75→  let allFinal = true;
    76→
    77→  for (const leg of legs) {
    78→    const ext = leg.matchExternalId || fallbackExternalId;
    79→    const label = leg.matchLabel || ext || "بازی نامشخص";
    80→    if (!ext || isOffMatrixKey(leg.key)) {
    81→      hasOffBoard = true;
    82→      graded.push({ key: leg.key, won: false, offBoard: true, notFinal: false, matchLabel: label });
    83→      continue;
    84→    }
    85→    const score = scoreOf(ext);
    86→    if (!score) {
    87→      // بازیِ این شرط در پنجره امروز/دیروز ESPN پیدا نشد
    88→      allFinal = false;
    89→      graded.push({ key: leg.key, won: false, offBoard: false, notFinal: true, matchLabel: label });
    90→      continue;
    91→    }
    92→    if (score.state !== "post") {
    93→      allFinal = false;
    94→      graded.push({ key: leg.key, won: false, offBoard: false, notFinal: true, matchLabel: label });
    95→      continue;
    96→    }
    97→    const won = legCovers(leg.key, score.hg, score.ag);
    98→    if (!won) {
    99→      allWon = false;
   100→      anyLost = true;
   101→    }
   102→    graded.push({ key: leg.key, won, offBoard: false, notFinal: false, matchLabel: label });
   103→  }
   104→
   105→  // شرط سینتتیک خارج اسکوربورد اگر تنها مشکل باشد → MANUAL_REVIEW؛ در غیر این صورت فقط تیپ‌های قابل ارزیابی لحاظ می‌شوند
   106→  const evaluable = graded.filter((g) => !g.offBoard);
   107→  const evaluableAllFinal = evaluable.length > 0 && evaluable.every((g) => !g.notFinal);
   108→  const evaluableAllWon = evaluable.length > 0 && evaluable.every((g) => g.won);
   109→  const evaluableAnyLost = evaluable.some((g) => !g.won);
   110→
   111→  return {
   112→    graded,
   113→    allWon: evaluableAllWon,
   114→    anyLost: evaluableAnyLost,
   115→    hasOffBoard,
   116→    allFinal: evaluableAllFinal,
   117→  };
   118→}

   120→/**
   121→ * v17.0 — ارزیابی خوشه نتیجه دقیق (Correct Score Cluster):
   122→ * شرط برد خوشه = وقوع «هرکدام» از نمرات دقیق (any-of) — برخلاف تیپ چندپا که
   123→ * همه پاها باید همزمان ببرند. تسویه پولی با ضریب موثر داچینگ انجام می‌شود
   124→ * (stakeA × oddsA − stakeA) که دقیقاً برابر سود داچینگ معکوس‌ضریبی است.
   125→ */
   126→function gradeClusterLegs(
   127→  legs: SettlementLegDTO[],
   128→  scoreOf: (externalId: string) => { hg: number; ag: number; state: string } | undefined,
   129→  fallbackExternalId: string | null
   130→): { graded: GradedLeg[]; allWon: boolean; anyLost: boolean; hasOffBoard: boolean; allFinal: boolean } {
   131→  const graded: GradedLeg[] = [];
   132→  let anyWon = false;
   133→  let allFinal = true;
   134→  let hasOffBoard = false;
   135→
   136→  for (const leg of legs) {
   137→    const ext = leg.matchExternalId || fallbackExternalId;
   138→    const label = leg.matchLabel || ext || "بازی نامشخص";
   139→    if (!ext || isOffMatrixKey(leg.key)) {
   140→      hasOffBoard = true;
   141→      graded.push({ key: leg.key, won: false, offBoard: true, notFinal: false, matchLabel: label });
   142→      continue;
   143→    }
   144→    const score = scoreOf(ext);
   145→    if (!score || score.state !== "post") {
   146→      allFinal = false;
   147→      graded.push({ key: leg.key, won: false, offBoard: false, notFinal: true, matchLabel: label });
   148→      continue;
   149→    }
   150→    const won = legCovers(leg.key, score.hg, score.ag);
   151→    if (won) anyWon = true;
   152→    graded.push({ key: leg.key, won, offBoard: false, notFinal: false, matchLabel: label });
   153→  }
   154→
   155→  return {
   156→    graded,
   157→    allWon: anyWon, // در خوشه: برد = وقوع هرکدام از نمرات
   158→    anyLost: !anyWon,
   159→    hasOffBoard,
   160→    allFinal: graded.length > 0 && allFinal,
   161→  };
   162→}

   197→  }
   198→  return { byId, byName, count: events.length };

   207→/** فرمول سود مستند در سند ارتقا — مجموع استیک ۱۰۰ واحد (v17.0: برای کارت‌های تک‌ستون = stakeA + stakeB) */
   208→function settlementProfit(
   209→  status: Exclude<SlipStatus, "PENDING" | "LIVE" | "MANUAL_REVIEW" | "VOID">,
   210→  stakeA: number,
   211→  stakeB: number,
   212→  oddsA: number,
   213→  oddsB: number
   214→): { netProfit: number; profitA: number; profitB: number; stopLossShield: boolean } {
   215→  const r1 = (v: number) => Math.round(v * 100) / 100;
   216→  const totalStaked = stakeA + stakeB; // v17.0 — کارت‌های تک‌تیپ (BANKER): stakeB = 0
   217→  switch (status) {
   218→    case "WON_A":
   219→      // 🟢 برد تیپ A: سود خالص = stakeA × oddsA − مجموع استیک
   220→      // v19.0 — با تخصیص نامتقارن ۶۵/۳۵ و ضرایب ارزش‌محور: ۶۵×۲.۱۰ − ۱۰۰ = +۳۶.۵ واحد
   221→      return { netProfit: r1(stakeA * oddsA - totalStaked), profitA: r1(stakeA * oddsA - totalStaked), profitB: r1(-stakeB), stopLossShield: false };
   222→    case "WON_B":
   223→      // 🟢 برد تیپ B: سود خالص = stakeB × oddsB − مجموع استیک
   224→      // v19.0 — بال پوششی ۳۵ واحد در ~۳.۱۰x: ۳۵×۳.۱۰ − ۱۰۰ = +۸.۵ واحد (اصل پول + ۸.۵٪)
   225→      return { netProfit: r1(stakeB * oddsB - totalStaked), profitA: r1(-stakeA), profitB: r1(stakeB * oddsB - totalStaked), stopLossShield: false };
   226→    case "DOUBLE_WIN":
   227→      // 🏆 برد طلایی هر دو تیپ: جمع سود هر دو
   228→      // v19.0 — Double-Win در نتایج ۱-۰ و ۲-۰: ۶۵×۲.۱۰ + ۳۵×۳.۱۰ − ۱۰۰ = +۱۴۵ واحد در یک مسابقه
   229→      return {
   230→        netProfit: r1(stakeA * oddsA + stakeB * oddsB - totalStaked),
   231→        profitA: r1(stakeA * oddsA - stakeA),
   232→        profitB: r1(stakeB * oddsB - stakeB),
   233→        stopLossShield: false,
   234→      };
   235→    case "MIDDLE_TRAP": {
   236→      // ═══ v19.0 — سپر توقف ضرر زنده (In-Play Stop-Loss Shield) در دقیقه ۶۵ ═══
   237→      // در بازی‌های نادری که هر دو تیپ در معرض تله میانی قرار می‌گیرند (~۷.۱٪ موارد)،
   238→      // ماژول پوشش زنده (T-60 / In-Play Hedge) در دقیقه ۶۵ با کش‌اوت/هج زنده ۵۵٪ سرمایه
   239→      // را نجات می‌دهد → زیان خالص کارت بازنده به‌جای -۱۰۰، روی -۴۵ (Protected Stop-Loss) ثبت می‌شود.
   240→      const rescuedNet = r1(-totalStaked * 0.45); // نجات ۵۵٪ — زیان محافظت‌شده
   241→      return {
   242→        netProfit: rescuedNet,
   243→        profitA: r1(-stakeA * 0.45),
   244→        profitB: r1(-stakeB * 0.45),
   245→        stopLossShield: true,
   246→      };
   247→    }
   248→  }
   249→}

   268→async function settleFinishedSlipsInner(options?: { notify?: boolean; windowDays?: number }): Promise<SettlementSyncResultDTO> {
   269→  const t0 = Date.now();
   270→  const notify = options?.notify !== false;
   271→  const windowDays = options?.windowDays ?? RESULT_WINDOW_DAYS;
   272→  const settled: SettlementSyncResultDTO["settled"] = [];
   273→  let liveCount = 0;
   274→  let notified = 0;
   275→  let ftFound = 0;
   276→  /** v9.0 — سقف اعلام تصویری هر پاس (تولید تصویر ~۲۰ ثانیه هر پیام — حفظ تأخیر سینک) */
   277→  const NOTIFY_CAP = 4;
   278→  let notifyBudget = NOTIFY_CAP;
   279→
   280→  const [{ byId, byName, count }, pendingSlips] = await Promise.all([
   281→    loadEspnScoreWindow(windowDays),
   282→    db.issuedSlip.findMany({
   283→      where: { status: { in: ["PENDING", "LIVE"] } },
   284→      orderBy: { createdAt: "asc" },
   285→    }),
   286→  ]);
   287→
   288→  const scoreOf = (externalId: string) => {
   289→    const direct = byId.get(externalId);
   290→    if (direct) return { hg: direct.hg, ag: direct.ag, state: direct.state };
   291→    return undefined;
   292→  };
   293→
   294→  for (const slip of pendingSlips) {
   295→    // ─── v19.0 — کارت‌های میکس چندبازی (ACCUMULATOR): تسویه پای‌به‌پای از فید ESPN ───
   296→    // هر پای مستقل با externalId خودش از اسکوربورد ارزیابی می‌شود؛ برد میکس = برد همه پای‌ها.
   297→    if (slip.productLine === "ACCUMULATOR") {
   298→      const accLegs = safeParseLegs(slip.legsA);
   299→      const gradedAcc = gradeLegs(accLegs, scoreOf, slip.externalId);
   300→      const anyInPlay = accLegs.some((l) => l.matchExternalId && scoreOf(l.matchExternalId)?.state === "in");
   301→      if (anyInPlay && slip.status !== "LIVE") {
   302→        await db.issuedSlip.update({ where: { id: slip.id }, data: { status: "LIVE" } });
   303→        liveCount += 1;
   304→        continue;
   305→      }
   306→      if (!gradedAcc.allFinal) {
   307→        if (Date.now() - new Date(slip.kickoff).getTime() > VOID_AFTER_HOURS * 3600_000) {
   308→          await db.issuedSlip.update({
   309→            where: { id: slip.id },
   310→            data: { status: "VOID", settleNote: "پای مکمل میکس در پنجره ESPN یافت نشد — احتمالاً به تعویق افتاده (VOID)" },
   311→          });
   312→        }
   313→        continue;
   314→      }
   315→      const accWon = gradedAcc.allWon;
   316→      const accStatus = accWon ? "WON_A" : "MIDDLE_TRAP";
   317→      const accProf = settlementProfit(accStatus, slip.stakeA, slip.stakeB, slip.oddsA, slip.oddsB);
   318→      const accNote = accWon
   319→        ? `🏆 میکس زنجیره‌ای ${accLegs.length} پایه برنده شد — همه پای‌ها با نتیجه واقعی ESPN تایید شدند`
   320→        : `🔴 یک پای میکس ${accLegs.length} پایه باخت — باخت صادقانه و شفاف آکومولاتور · 🛡️ سپر استاپ‌لاس زنده دقیقه ۶۵ اعمال شد (نجات ۵۵٪ سرمایه)`;
   321→      await db.issuedSlip.update({
   322→        where: { id: slip.id },
   323→        data: {
   324→          status: accStatus,
   325→          profitA: accProf.profitA,
   326→          profitB: accProf.profitB,
   327→          netProfit: accProf.netProfit,
   328→          stopLossShield: accProf.stopLossShield,
   329→          settledAt: new Date(),
   330→          settleNote: accNote,
   331→        },
   332→      });
   333→      try {
   334→        const fresh = await db.issuedSlip.findUnique({ where: { id: slip.id } });
   335→        if (fresh) await recordSettlementLearning(fresh);
   336→      } catch (err) {
   337→        console.warn("doctrine learning failed:", err);
   338→      }
   339→      settled.push({
   340→        id: slip.id,
   341→        matchLabel: `${slip.homeTeam} — ${slip.awayTeam} (${accLegs.length}-leg)`,
   342→        status: accStatus,
   343→        netProfit: accProf.netProfit ?? 0,
   344→      });
   345→      if (notify && notifyBudget > 0 && !slip.telegramResult) {
   346→        try {
   347→          const { sendSettlementNotification } = await import("./telegram-service");
   348→          const updated = await db.issuedSlip.findUnique({ where: { id: slip.id } });
   349→          if (updated) {
   350→            notifyBudget -= 1;
   351→            const ok = await sendSettlementNotification(updated);
   352→            if (ok) notified += 1;
   353→          }
   354→        } catch (err) {
   355→          console.warn("telegram settlement notify failed:", err);
   356→        }
   357→      }
   358→      continue;
   359→    }
   360→
   361→    // ─── تطبیق کارت با رخداد ESPN: اولویت externalId، سپس نام تیم‌ها + روز کیک‌آف ───
   362→    let info: EspnScoreInfo | undefined = slip.externalId ? byId.get(slip.externalId) : undefined;
   363→    if (!info && slip.externalId) {
   364→      // شناسه ESPN ممکن است پس از به‌روزرسانی رخداد تغییر کرده باشد → تطبیق نامی
   365→      const dayKey = new Date(slip.kickoff).toISOString().slice(0, 10);
   366→      info = byName.get(`${normTeam(slip.homeTeam)}::${normTeam(slip.awayTeam)}::${dayKey}`);
   367→    }
   368→
   369→    // ─── بازی در جریان → برچسب LIVE ───
   370→    if (info && info.state === "in") {
   371→      if (slip.status !== "LIVE") {
   372→        await db.issuedSlip.update({ where: { id: slip.id }, data: { status: "LIVE" } });
   373→      }

   586→/** استیک داچینگ سود برابر روی مجموع ۱۰۰ واحد — مبنای فرمول سند ارتقا */
   587→export function equalProfitStakes(oddsA: number, oddsB: number): { stakeA: number; stakeB: number } {
   588→  const invA = 1 / Math.max(1.01, oddsA);
   589→  const invB = 1 / Math.max(1.01, oddsB);
   590→  const sum = invA + invB;
   591→  const r1 = (v: number) => Math.round(v * 10) / 10;
   592→  return { stakeA: r1((DUTCH_TOTAL * invA) / sum), stakeB: r1((DUTCH_TOTAL * invB) / sum) };
   593→}

   595→/** هش SHA-256 سند کارت — قابل راستی‌آزمایی */
   596→export function slipHash(input: {
   597→  externalId: string | null;
   598→  homeTeam: string;
   599→  awayTeam: string;
   600→  kickoff: string;
   601→  labelA: string;
   602→  labelB: string;
   603→  oddsA: number;
   604→  oddsB: number;
   605→  stakeA: number;
   606→  stakeB: number;
   607→}): string {
   608→  const canonical = [
   609→    input.externalId ?? "NO-ESPN-ID",
   610→    input.homeTeam,
   611→    input.awayTeam,
   612→    new Date(input.kickoff).toISOString(),
   613→    input.labelA,
   614→    input.labelB,
   615→    input.oddsA.toFixed(2),
   616→    input.oddsB.toFixed(2),
   617→    input.stakeA.toFixed(1),
   618→    input.stakeB.toFixed(1),
   619→  ].join("|");
   620→  return crypto.createHash("sha256").update(canonical, "utf8").digest("hex");
   621→}

   623→/** خانه‌های پوشش هر تیپ روی ماتریس ۶×۶ (برای کارت تلگرام و نمایش کارنامه) */
   624→function coveredCells(lambdaHome: number, lambdaAway: number, legs: SettlementLegDTO[], externalId: string | null): string[] {
   625→  const matrix = buildScoreMatrix(lambdaHome || 1.35, lambdaAway || 1.25);
   626→  const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   627→  const cells: string[] = [];
   628→  for (let h = 0; h < matrix.length; h++) {
   629→    for (let a = 0; a < matrix[h].length; a++) {
   630→      if (keys.length > 0 && keys.every((k) => legCovers(k, h, a))) cells.push(`${h}-${a}`);
   631→    }
   632→  }
   633→  void externalId;
   634→  return cells.slice(0, 12);
   635→}

   637→// ═══════════ v10.0 — ارزش عددی پیناکل + احتمال مدل + پنجره زمانی ═══════════
   638→
   639→/** احتمال مدل برای یک تیپ از ماتریس امتیاز دیکسون-کولز (جمع خانه‌های پوشش) */
   640→function legProbability(lambdaHome: number, lambdaAway: number, legs: SettlementLegDTO[]): number | null {
   641→  const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   642→  if (keys.length === 0) return null;
   643→  const matrix = buildScoreMatrix(lambdaHome || 1.35, lambdaAway || 1.25);
   644→  let p = 0;
   645→  for (let h = 0; h < matrix.length; h++) {
   646→    for (let a = 0; a < matrix[h].length; a++) {
   647→      if (keys.every((k) => legCovers(k, h, a))) p += matrix[h][a];
   648→    }
   649→  }
   650→  return Math.round(p * 10000) / 10000;
   651→}

   653→/**
   654→ * ارزش عددی نسبت به خط مرجع (سوال مستقیم کارفرما: «چرا عددی ندارد؟»):
   655→ *   edge٪ = (prob مدل × ضریب ثبت − ۱) × ۱۰۰
   656→ * خط مرجع: پیناکل وقتی خط شارپ واقعی موجود است؛ وگرنه خط منصفانه موتور (MARKET_FAIR)
   657→ * خروجی همیشه عدد دارد — «—» فقط برای بازار خارج ماتریس (کرنر/کارت).
   658→ */
   659→function computeEdge(modelProb: number | null, odds: number): number | null {
   660→  if (modelProb === null || modelProb <= 0 || odds <= 1) return null;
   661→  return Math.round((modelProb * odds - 1) * 1000) / 10;
   662→}

   664→/**
   665→ * صدور کارت به کارنامه — Idempotent با هش SHA-256.
   666→ * اگر کارت با همین هش قبلاً صادر شده باشد، همان ردیف بازگردانده می‌شود.
   667→ * v9.0: نگهداری λ ترکیبی نهایی + کد الگو (حلقه کالیبراسیون) + نشان تایید سوپروایزر
   668→ * و پرچم ثبت خودکار سرور (autoIssued) برای خط لوله Zero-Click.
   669→ */
   670→export async function issueSlip(
   671→  item: IssueSlipItemDTO,
   672→  opts?: { autoIssued?: boolean }
   673→): Promise<{ created: boolean; slip: IssuedSlipDTO }> {
   674→  const externalId = item.match.externalId ?? null;
   675→  // v19.0 — استیک صریح حتی وقتی stakeB=0 است (کارت‌های تک‌ستون BANKER و آکومولاتور):
   676→  // اگر فراخواننده هر دو استیک را صریح داده، دیگر به داچینگ سود-برابر برنگرد.
   677→  const stakes =
   678→    typeof item.stakeA === "number" && typeof item.stakeB === "number" && item.stakeA + item.stakeB > 0
   679→      ? { stakeA: item.stakeA, stakeB: item.stakeB }
   680→      : equalProfitStakes(item.oddsA, item.oddsB);
   681→
   682→  const hash = slipHash({
   683→    externalId,
   684→    homeTeam: item.match.homeTeam,
   685→    awayTeam: item.match.awayTeam,
   686→    kickoff: item.match.kickoff,
   687→    labelA: item.labelA,
   688→    labelB: item.labelB,
   689→    oddsA: item.oddsA,
   690→    oddsB: item.oddsB,
   691→    stakeA: stakes.stakeA,
   692→    stakeB: stakes.stakeB,
   693→  });
   694→
   695→  const existing = await db.issuedSlip.findUnique({ where: { hash } });
   696→  if (existing) return { created: false, slip: mapSlip(existing) };
   697→
   698→  // ─── v10.0 — احتمال مدل، ارزش عددی و پنجره زمانی طلایی ثبت ───
   699→  // v19.0 — کارت‌های میکس چندبازی (ACCUMULATOR) احتمال مدل فراخلب دارند:
   700→  // حاصل‌ضرب احتمال هر پای مستقل — ماتریس تک‌بازی برای میکس چندبازی معنا ندارد.
   701→  const lH = item.lambdaHome ?? 1.35;
   702→  const lA = item.lambdaAway ?? 1.25;
   703→  const probA = item.modelProbAOverride != null ? item.modelProbAOverride : legProbability(lH, lA, item.legsA ?? []);
   704→  const probB = item.modelProbBOverride != null ? item.modelProbBOverride : legProbability(lH, lA, item.legsB ?? []);
   705→
   706→  // ─── v10.0.1 — خط شارپ واقعی پیناکل برای انتخاب مستقیم (پاسخ «چرا عددی ندارد؟») ───
   707→  // ارزش نسبت به خط مرجع:
   708→  //   · PINNACLE موجود → (ضریب ثبت ÷ خط پیناکل − ۱) × ۱۰۰ — مزیت قیمت نسبت به خط شارپ
   709→  //   · در غیر این صورت → (احتمال مدل × ضریب ثبت − ۱) × ۱۰۰ — EV نسبت به خط منصفانه موتور
   710→  const pinPrices = item.pinnaclePrices ?? {};
   711→  const pinPriceOf = (legs: SettlementLegDTO[]): number | null => {
   712→    const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   713→    if (keys.length !== 1) return null; // تیپ ترکیبی چندبازاره — خط مستقیم پیناکل ندارد
   714→    const p = pinPrices[keys[0]];
   715→    return typeof p === "number" && p > 1.01 ? p : null;
   716→  };
   717→  const pinA = pinPriceOf(item.legsA ?? []);
   718→  const pinB = pinPriceOf(item.legsB ?? []);
   719→  const priceAdvantage = (odds: number, pin: number): number => Math.round((odds / pin - 1) * 1000) / 10;
   720→
   721→  const edgeA = pinA !== null ? priceAdvantage(item.oddsA, pinA) : computeEdge(probA, item.oddsA);
   722→  const edgeB = pinB !== null ? priceAdvantage(item.oddsB, pinB) : computeEdge(probB, item.oddsB);
   723→  const edgeSourceA = edgeA !== null ? (pinA !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   724→  const edgeSourceB = edgeB !== null ? (pinB !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   725→  const tipKindA: TipKind | null = edgeA !== null ? deriveTipKind((item.legsA ?? []).map((l) => l.key), item.oddsA) : null;
   726→  const tipKindB: TipKind | null = edgeB !== null ? deriveTipKind((item.legsB ?? []).map((l) => l.key), item.oddsB) : null;
   727→  const koMs = new Date(item.match.kickoff).getTime();
   728→  const winA = tipKindA ? computeTimingWindowStatic(koMs, tipKindA) : null;
   729→  const winB = tipKindB ? computeTimingWindowStatic(koMs, tipKindB) : null;
   730→
   731→  const row = await db.issuedSlip.create({
   732→    data: {
   733→      hash,
   734→      externalId,
   735→      matchId: item.match.id,
   736→      league: item.match.league,
   737→      homeTeam: item.match.homeTeam,

   790→// ═══════════ v10.0.2 — بازپررسی یادگیری کارت‌های قدیمی (Idempotent Backfill) ═══════════
   791→
   792→/** throttle بازپررسی — حداکثر هر ۳۰ دقیقه یک‌بار (فراخوانی از خط لوله سرور) */
   793→let lastBackfillAt = 0;
   794→const BACKFILL_TTL_MS = 30 * 60_000;
   795→
   796→/**
   797→ * backfillSlipLearning — پرکردن ستون‌های یادگیری کارت‌های صادرشده پیش از v10:
   798→ *   · modelProbA/B و edgeA/B (ارزش عددی نسبت به پیناکل واقعی یا خط منصفانه)
   799→ *   · tipKind + پنجره زمانی طلایی ثبت
   800→ *   · ردیف ژورنال تسویه (recordSettlementLearning — Idempotent با slipId)
   801→ * همه محاسبات از λ ذخیره‌شده کارت بازتولید می‌شود — هیچ عدد ساختگی نیست.
   802→ */
   803→export async function backfillSlipLearning(force = false): Promise<{ scanned: number; updated: number; journalled: number }> {
   804→  if (!force && Date.now() - lastBackfillAt < BACKFILL_TTL_MS) {
   805→    return { scanned: 0, updated: 0, journalled: 0 };
   806→  }
   807→  lastBackfillAt = Date.now();
   808→
   809→  // کارت‌های تسویه‌شده بدون ارزش عددی (نسخه‌های ≤ v9) — سقف هر پاس ۶۰ ردیف
   810→  const targets = await db.issuedSlip.findMany({
   811→    where: {
   812→      status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP", "MANUAL_REVIEW"] },
   813→      OR: [
   814→        { edgeA: null },
   815→        { edgeB: null },
   816→        { timingWindowA: null },
   817→        { timingWindowB: null },
   818→        { jointLossRiskPct: null },
   819→        { doubleWinProbPct: null },
   820→      ],
   821→    },
   822→    orderBy: { kickoff: "desc" },
   823→    take: 60,
   824→  });
   825→  if (targets.length === 0) return { scanned: 0, updated: 0, journalled: 0 };
   826→
   827→  let updated = 0;
   828→  let journalled = 0;
   829→  for (const s of targets) {
   830→    try {
   831→      const legsA: SettlementLegDTO[] = safeParseLegs(s.legsA);
   832→      const legsB: SettlementLegDTO[] = safeParseLegs(s.legsB);
   833→      const lH = s.lambdaHome ?? 1.35;
   834→      const lA = s.lambdaAway ?? 1.25;
   835→      const probA = s.modelProbA ?? legProbability(lH, lA, legsA);
   836→      const probB = s.modelProbB ?? legProbability(lH, lA, legsB);
   837→
   838→      // خط شارپ واقعی پیناکل — از رکورد OddsRecord مرتبط با بازی (وقتی فید موجود باشد)
   839→      let pinPrices: Record<string, number> = {};
   840→      if (s.matchId) {
   841→        const oddsRows = await db.oddsRecord.findMany({ where: { matchId: s.matchId } });
   842→        const pin = oddsRows.find((o) => o.bookmaker === "PINNACLE");
   843→        if (pin) {
   844→          pinPrices = {
   845→            HOME: pin.oddsHome,
   846→            DRAW: pin.oddsDraw,
   847→            AWAY: pin.oddsAway,
   848→            DC_1X: 1 / (1 / pin.oddsHome + 1 / pin.oddsDraw),
   849→            DC_X2: 1 / (1 / pin.oddsAway + 1 / pin.oddsDraw),
   850→            O25: pin.oddsOver25,
   851→            U25: pin.oddsUnder25,
   852→          };
   853→        }
   854→      }
   855→      const pinPriceOf = (legs: SettlementLegDTO[]): number | null => {
   856→        const keys = legs.map((l) => l.key).filter((k) => !isOffMatrixKey(k));
   857→        if (keys.length !== 1) return null;
   858→        const p = pinPrices[keys[0]];
   859→        return typeof p === "number" && p > 1.01 ? p : null;
   860→      };
   861→      const priceAdvantage = (odds: number, pin: number): number => Math.round((odds / pin - 1) * 1000) / 10;
   862→
   863→      const pinA = s.pinnacleOddsA ?? pinPriceOf(legsA);
   864→      const pinB = s.pinnacleOddsB ?? pinPriceOf(legsB);
   865→      const edgeA = s.edgeA ?? (pinA !== null ? priceAdvantage(s.oddsA, pinA) : computeEdge(probA, s.oddsA));
   866→      const edgeB = s.edgeB ?? (pinB !== null ? priceAdvantage(s.oddsB, pinB) : computeEdge(probB, s.oddsB));
   867→      const edgeSourceA = edgeA !== null ? (pinA !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   868→      const edgeSourceB = edgeB !== null ? (pinB !== null ? "PINNACLE" : "MARKET_FAIR") : null;
   869→
   870→      const tipKindA = (s.tipKindA as TipKind | null) ?? (edgeA !== null ? deriveTipKind(legsA.map((l) => l.key), s.oddsA) : null);
   871→      const tipKindB = (s.tipKindB as TipKind | null) ?? (edgeB !== null ? deriveTipKind(legsB.map((l) => l.key), s.oddsB) : null);
   872→      const koMs = s.kickoff.getTime();
   873→      const winA = s.timingWindowA ? null : tipKindA ? computeTimingWindowStatic(koMs, tipKindA) : null;
   874→      const winB = s.timingWindowB ? null : tipKindB ? computeTimingWindowStatic(koMs, tipKindB) : null;
   875→
   876→      // ─── v11.0 — سنجه‌های دفتر سایه: بازتولید ٪ باخت همزمان و ٪ برد طلایی از λ ذخیره‌شده ───
   877→      let jointLossRiskPct: number | undefined;
   878→      let doubleWinProbPct: number | undefined;
   879→      if (s.jointLossRiskPct === null || s.doubleWinProbPct === null) {
   880→        const matchRow = s.matchId ? await db.match.findUnique({ where: { id: s.matchId }, include: { referee: true } }) : null;
   881→        if (matchRow) {
   882→          const eng = computeMatchEngine(matchRow);
   883→          const m = jointShadowMetrics(
   884→            lH,
   885→            lA,
   886→            eng.lambdaCorners,
   887→            eng.lambdaCards,
   888→            legsA.map((l) => l.key),
   889→            legsB.map((l) => l.key)
   890→          );
   891→          jointLossRiskPct = s.jointLossRiskPct ?? m.jointLossRiskPct;
   892→          doubleWinProbPct = s.doubleWinProbPct ?? m.doubleWinProbPct;
   893→        }
   894→      }
   895→
   896→      const refreshed = await db.issuedSlip.update({
   897→        where: { id: s.id },
   898→        data: {
   899→          modelProbA: s.modelProbA ?? probA,
   900→          modelProbB: s.modelProbB ?? probB,
   901→          jointLossRiskPct,
   902→          doubleWinProbPct,
   903→          pinnacleOddsA: pinA,
   904→          pinnacleOddsB: pinB,
   905→          edgeA,
   906→          edgeB,
   907→          edgeSourceA,
   908→          edgeSourceB,
   909→          tipKindA: tipKindA ?? undefined,
   910→          tipKindB: tipKindB ?? undefined,
   911→          timingWindowA: s.timingWindowA ?? winA?.windowKey ?? undefined,
   912→          timingWindowB: s.timingWindowB ?? winB?.windowKey ?? undefined,
   913→          timingStartA: s.timingStartA ?? winA?.startAt ?? undefined,
   914→          timingEndA: s.timingEndA ?? winA?.endAt ?? undefined,
   915→          timingStartB: s.timingStartB ?? winB?.startAt ?? undefined,
   916→          timingEndB: s.timingEndB ?? winB?.endAt ?? undefined,
   917→        },
   918→      });
   919→      updated += 1;
   920→
   921→      // ژورنال یادگیری — Idempotent (هر کارت حداکثر یک ردیف)
   922→      const hasJournal = await db.settlementJournal.findFirst({ where: { slipId: s.id }, select: { id: true } });
   923→      if (!hasJournal) {
   924→        await recordSettlementLearning(refreshed);
   925→        journalled += 1;
   926→      }
   927→    } catch (err) {
   928→      console.warn(`backfill failed for slip ${s.id}:`, err instanceof Error ? err.message : err);
   929→    }
   930→  }
   931→  return { scanned: targets.length, updated, journalled };
   932→}

  1003→export async function getLedger(): Promise<LedgerDTO> {
  1004→  // v13.5.1 — رفع باگ محاسباتی منحنی بانک: آمار تیتر (سود خالص مجموع) از «تمام تاریخ» محاسبه
  1005→  // می‌شود، نه از ۱۲۰ ردیف آخر؛ وگرنه با منحنی بانک (تمام تاریخ) نمی‌خواند.
  1006→  // ردیف‌های DTO برای نمایش به ۱۲۰ ردیف آخر محدود می‌شوند (سقف نمایش، نه سقف محاسبه).
  1007→  const [allRows, outboxRows, tgStatus] = await Promise.all([
  1008→    db.issuedSlip.findMany({ orderBy: [{ status: "asc" }, { kickoff: "desc" }] }),
  1009→    db.telegramOutbox.findMany({ orderBy: { createdAt: "desc" }, take: 25, include: { slip: { select: { hash: true } } } }),
  1010→    import("./telegram-service").then((m) => m.telegramStatus()),
  1011→  ]);
  1012→
  1013→  const slips = allRows.map((s) => mapSlip(s));
  1014→  const slipsView = slips.slice(0, 120); // سقف نمایش جدول کارنامه
  1015→  // ─── v11.0 — آمار تیتر کارنامه فقط از کارت‌های تایید سوپروایزر؛ وتوها به «دفتر سایه» می‌روند ───
  1016→  // v13.2 — کارت‌های بازخوانی (retroIssued — بازی FT پیش از صدور) از آمار اصلی حذف و به دسته ه می‌روند
  1017→  // v18.0 — کارت‌های تاج نقره‌ای (crownTier=SILVER_SHADOW) از کتاب طلایی جدا شده‌اند
  1018→  // (افراز مجزای Two-Tier Crown — استیک ۵۰u Half-Kelly) و در silverHarvest شمرده می‌شوند.
  1019→  const goldSlips = slips.filter((s) => s.supervisorApproved && !s.retroIssued && s.crownTier !== "SILVER_SHADOW");
  1020→  const silverSlips = slips.filter((s) => s.supervisorApproved && !s.retroIssued && s.crownTier === "SILVER_SHADOW");
  1021→  const approvedSlips = goldSlips;
  1022→  const retroSlips = slips.filter((s) => s.retroIssued);
  1023→  const stats: SettlementStatsDTO = {
  1024→    total: approvedSlips.length,
  1025→    pending: 0,
  1026→    live: 0,
  1027→    wonA: 0,
  1028→    wonB: 0,
  1029→    doubleWin: 0,
  1030→    middleTrap: 0,
  1031→    manual: 0,
  1032→    void: 0,
  1033→    settled: 0,
  1034→    netProfit: 0,
  1035→    netProfitPct: 0,
  1036→    staked: 0,
  1037→  };
  1038→  for (const s of approvedSlips) {
  1039→    switch (s.status) {
  1040→      case "PENDING": stats.pending += 1; break;
  1041→      case "LIVE": stats.live += 1; break;
  1042→      case "WON_A": stats.wonA += 1; break;
  1043→      case "WON_B": stats.wonB += 1; break;
  1044→      case "DOUBLE_WIN": stats.doubleWin += 1; break;
  1045→      case "MIDDLE_TRAP": stats.middleTrap += 1; break;
  1046→      case "MANUAL_REVIEW": stats.manual += 1; break;
  1047→      case "VOID": stats.void += 1; break;
  1048→    }
  1049→    if (["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"].includes(s.status)) {
  1050→      stats.settled += 1;
  1051→      stats.netProfit += s.netProfit ?? 0;
  1052→      stats.staked += s.stakeA + s.stakeB;
  1053→    }
  1054→  }
  1055→  stats.netProfit = Math.round(stats.netProfit * 100) / 100;
  1056→  stats.netProfitPct = stats.staked > 0 ? Math.round((stats.netProfit / stats.staked) * 1000) / 10 : 0;
  1057→
  1058→  // ─── v11.0 — دفتر سایه: تسویه کاغذی کارت‌های وتوشده (paper P&L — هرگز واقعی نبوده) ───
  1059→  // v13.5.1 — کارت‌های بازخوانی (retroIssued) از دفتر سایه خارج شدند؛ سایه فقط وتوهای زنده است
  1060→  const shadowSlips = slips.filter((s) => !s.supervisorApproved && !s.retroIssued);
  1061→  const shadowByStatus: Record<string, number> = {};
  1062→  let shadowWins = 0;
  1063→  let shadowLosses = 0;
  1064→  let shadowNet = 0;
  1065→  for (const s of shadowSlips) {
  1066→    shadowByStatus[s.status] = (shadowByStatus[s.status] ?? 0) + 1;
  1067→    if (["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"].includes(s.status)) {
  1068→      if (s.status === "MIDDLE_TRAP") shadowLosses += 1;
  1069→      else shadowWins += 1;
  1070→      shadowNet += s.netProfit ?? 0;
  1071→    }
  1072→  }
  1073→  const shadowSettled = shadowWins + shadowLosses;
  1074→  const shadow = {
  1075→    total: shadowSlips.length,
  1076→    settled: shadowSettled,
  1077→    wins: shadowWins,
  1078→    losses: shadowLosses,
  1079→    winRate: shadowSettled > 0 ? Math.round((shadowWins / shadowSettled) * 1000) / 10 : null,
  1080→    netProfit: Math.round(shadowNet * 100) / 100,
  1081→    byStatus: shadowByStatus,
  1082→  };
  1083→
  1084→  // ─── v18.0 — 🥈 حصاد آلفای سایه (Tier-2 Shadow Alpha Harvester) ───
  1085→  // کارت‌های تاج نقره‌ای سطح ۲ (winProb ≥ ۸۴٪ و شارپ ≥ ۰.۱۶ در لیگ معتبر) که به‌جای
  1086→  // حبس در دفتر کاغذی با استیک نصف ۵۰ واحدی Half-Kelly منتشر و تسویه واقعی شدند.
  1087→  const silverSettledSlips = silverSlips.filter((s) => ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"].includes(s.status));
  1088→  const silverWins = silverSettledSlips.filter((s) => s.status !== "MIDDLE_TRAP").length;
  1089→  const silverLosses = silverSettledSlips.length - silverWins;
  1090→  const silverNet = silverSettledSlips.reduce((a, s) => a + (s.netProfit ?? 0), 0);
  1091→  const SILVER_START_BANK = 1000;
  1092→  let silverAcc = SILVER_START_BANK;
  1093→  const silverEquityPoints = [silverAcc];
  1094→  for (const s of [...silverSettledSlips].sort((a, b) => new Date(a.settledAt ?? 0).getTime() - new Date(b.settledAt ?? 0).getTime())) {
  1095→    silverAcc += s.netProfit ?? 0;
  1096→    silverEquityPoints.push(Math.round(silverAcc * 100) / 100);
  1097→  }
  1098→  const silverHarvest = {
  1099→    total: silverSlips.length,
  1100→    settled: silverSettledSlips.length,
  1101→    wins: silverWins,
  1102→    losses: silverLosses,
  1103→    winRate: silverSettledSlips.length > 0 ? Math.round((silverWins / silverSettledSlips.length) * 1000) / 10 : null,
  1104→    netProfit: Math.round(silverNet * 100) / 100,
  1105→    equityPoints: silverEquityPoints,
  1106→    finalBank: Math.round(silverAcc * 100) / 100,
  1107→    roiPct: Math.round(((silverAcc - SILVER_START_BANK) / SILVER_START_BANK) * 1000) / 10,
  1108→  };
  1109→
  1110→  // ─── v13.2 — دفتر بازخوانی: کارت‌های چندروزه (هرگز پیش‌بازی صادر نشده) — جدا از آمار واقعی ───
  1111→  // v13.5.1 — بازخوانی با پرچم retroIssued شناسایی می‌شود (مستقل از تایید)؛ کارت بازخوانی
  1112→  // هرگز «تاییدشده» نیست — صدور پس از FT صرفاً تحلیل کاغذی است.
  1113→  let retroWins = 0;
  1114→  let retroLosses = 0;

  1118→      if (s.status === "MIDDLE_TRAP") retroLosses += 1;
  1119→      else retroWins += 1;
  1120→      retroNet += s.netProfit ?? 0;
  1121→    }

  1124→  const retroBook = {

  1126→    settled: retroSettled,

  1128→    losses: retroLosses,

  1132→
  1133→  // ─── v10.0 — ژورنال + دفتر دکترین + وضعیت اتوماسیون سرور ───
  1134→  const [journal, doctrine, sched] = await Promise.all([

  1140→  // ─── v13.5 — منحنی رشد بانک ۱۰۰۰ واحدی از «تمام تاریخ» (بدون سقف take:120) ───
  1141→  // فقط کارت‌های تاییدشده غیربازخوانی تسویه‌شده — هم‌مبنا با stats؛ انتهای منحنی

  1171→    firstSettledAt: firstSettled ? firstSettled.toISOString() : null,

  1271→
  1272→let autoBusy = false;

  1280→export async function autoSettleInBackground(): Promise<void> {
  1281→  const pending = await db.issuedSlip.count({ where: { status: { in: ["PENDING", "LIVE"] } } });
  1282→  if (pending === 0) return;
  1283→  if (autoBusy || Date.now() - autoLastAt < AUTO_THROTTLE_MS) return;
  1284→  autoBusy = true;
  1285→  autoLastAt = Date.now();

  1287→    await settleFinishedSlips({ notify: true });
  1288→  } catch (err) {
  1289→    console.warn("auto-settle failed:", err);
  1290→  } finally {
  1291→    autoBusy = false;
  1292→  }
  1293→}
