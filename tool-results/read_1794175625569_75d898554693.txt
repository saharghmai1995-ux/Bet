    37→/** عرض تقریبی کاراکتر به واحد em در DejaVu Sans */
    38→function charW(ch: string): number {
    39→  if (" iIljt.,:;!'|".includes(ch)) return 0.32;
    40→  if ("ftr-()[]".includes(ch)) return 0.42;
    41→  if ("mwMW@".includes(ch)) return 0.92;
    42→  if (ch >= "A" && ch <= "Z") return 0.72;
    43→  if (ch >= "0" && ch <= "9") return 0.64;
    44→  if (ch === " ") return 0.34;
    45→  if (ch >= "a" && ch <= "z") return 0.56;
    46→  return 0.62;
    47→}
    48→
    49→export function textWidth(text: string, fontSize: number, weight = 400): number {
    50→  const w = String(text ?? "").split("").reduce((acc, ch) => acc + charW(ch), 0) * fontSize;
    51→  return w * (weight >= 600 ? 1.06 : 1);
    52→}

    54→/** شکستن متن به خطوط با حداکثر عرض پیکسلی */
    55→export function wrapText(text: string, maxWidth: number, fontSize: number, weight = 400, maxLines = 3): string[] {
    56→  const words = String(text ?? "").split(/\s+/).filter(Boolean);
    57→  const lines: string[] = [];
    58→  let cur = "";
    59→  for (const word of words) {
    60→    const attempt = cur ? `${cur} ${word}` : word;
    61→    if (textWidth(attempt, fontSize, weight) <= maxWidth) {
    62→      cur = attempt;
    63→    } else {
    64→      if (cur) lines.push(cur);
    65→      cur = word;
    66→      if (lines.length === maxLines) break;
    67→    }
    68→  }
    69→  if (cur && lines.length < maxLines) lines.push(cur);
    70→  if (lines.length === maxLines && words.join(" ") !== lines.join(" ")) {
    71→    let last = lines[maxLines - 1];
    72→    while (textWidth(`${last}…`, fontSize, weight) > maxWidth && last.length > 4) last = last.slice(0, -1);
    73→    lines[maxLines - 1] = `${last}…`;
    74→  }
    75→  return lines.length > 0 ? lines : ["—"];
    76→}

    78→/** کوچک‌سازی خودکار فونت تا متن در عرض بگنجد */
    79→export function fitSize(text: string, maxWidth: number, baseSize: number, minSize: number, weight = 700): number {
    80→  let size = baseSize;
    81→  while (size > minSize && textWidth(text, size, weight) > maxWidth) size -= 2;
    82→  return size;
    83→}

    85→function esc(s: string): string {
    86→  return String(s ?? "")
    87→    .replace(/&/g, "&amp;")
    88→    .replace(/</g, "&lt;")
    89→    .replace(/>/g, "&gt;")
    90→    .replace(/"/g, "&quot;");
    91→}

    93→/** چند خط tspan با line-height مشخص */
    94→function tspans(lines: string[], x: number, y: number, lh: number): string {
    95→  return lines.map((l, i) => `<tspan x="${x}" y="${y + i * lh}">${esc(l)}</tspan>`).join("");
    96→}

   207→function fmtKickoff(iso: string): string {
   208→  try {
   209→    const d = new Date(iso);
   210→    const day = new Intl.DateTimeFormat("en-GB", { weekday: "short", day: "2-digit", month: "short", timeZone: "Asia/Tehran" }).format(d);
   211→    const time = new Intl.DateTimeFormat("en-GB", { hour: "2-digit", minute: "2-digit", hour12: false, timeZone: "Asia/Tehran" }).format(d);
   212→    return `${day} · ${time} Tehran`;
   213→  } catch {
   214→    return String(iso);
   215→  }
   216→}

   220→function crestInitials(name: string): string {
   221→  const parts = String(name ?? "").split(/\s+/).filter(Boolean);
   222→  if (parts.length >= 2) return (parts[0][0] + parts[1][0]).toUpperCase();
   223→  return name.slice(0, 2).toUpperCase();
