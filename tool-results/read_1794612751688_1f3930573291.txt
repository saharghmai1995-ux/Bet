    83→async function learnIdsFromUpdate(db: Database, cfg: BotConfig, update: Update): Promise<void> {
    84→  const notes: string[] = [];
    85→
    86→  // ۱) my_chat_member — بات به کانال/سوپرگروه اضافه، حذف یا ارتقا یافت
    87→  const mcm = update.my_chat_member;
    88→  if (mcm?.chat && (mcm.chat.type === "channel" || mcm.chat.type === "supergroup")) {
    89→    if (isVipPlaceholder(cfg.vipChannelId)) {
    90→      cfg.vipChannelId = String(mcm.chat.id);
    91→      notes.push(`کانال VIP ← ${mcm.chat.id} (${mcm.chat.title ?? "بدون عنوان"})`);
    92→    }
    93→    if (mcm.from?.id && isAdminPlaceholder(cfg.adminUserId)) {
    94→      cfg.adminUserId = mcm.from.id;
    95→      notes.push(`ادمین ← ${mcm.from.id}`);
    96→    }
    97→  }
    98→
    99→  // ۲) channel_post — پستی از کانالی که بات ادمین آن است
   100→  const post = update.channel_post ?? (update.message && update.message.chat.type === "channel" ? update.message : undefined);
   101→  if (post?.chat.type === "channel" && isVipPlaceholder(cfg.vipChannelId)) {
   102→    cfg.vipChannelId = String(post.chat.id);
   103→    notes.push(`کانال VIP ← ${post.chat.id} (channel_post)`);
   104→  }
   105→
   106→  // ۳) پیام خصوصی — فوروارد از کانال VIP یا /start مالک
   107→  const msg = update.message;
   108→  if (msg && msg.chat.type === "private") {
   109→    const fwd = msg.forward_from_chat ?? msg.sender_chat;
   110→    if (fwd?.type === "channel" && isVipPlaceholder(cfg.vipChannelId)) {
   111→      cfg.vipChannelId = String(fwd.id);
   112→      notes.push(`کانال VIP ← ${fwd.id} (فوروارد)`);
   113→    }
   114→    if (msg.text?.startsWith("/start") && msg.from?.id && isAdminPlaceholder(cfg.adminUserId)) {
   115→      cfg.adminUserId = msg.from.id;
   116→      notes.push(`ادمین ← ${msg.from.id} (/start مالک)`);
   117→    }
   118→  }
   119→
   120→  if (notes.length === 0) return;
   121→  saveConfig(db, { vipChannelId: cfg.vipChannelId, adminUserId: cfg.adminUserId });
   122→  console.log(`[LEARN] auto-configured: ${notes.join(" | ")}`);
   123→  if (!isAdminPlaceholder(cfg.adminUserId)) {
   124→    await tgApiCall(cfg, "sendMessage", {
   125→      chat_id: cfg.adminUserId,
   126→      text: (
   127→        "⚙️ <b>پیکربندی خودکار @aineobet_bot ثبت شد</b>\n\n" +
   128→        notes.map((n) => `▫️ ${n}`).join("\n") +
   129→        "\n\n🔐 مقادیر در دیتابیس قفل شدند. برای تکمیل درگاه‌ها، کیف پول‌های USDT (TRC20/Polygon) و کارت بانکی را از پنل وب (تب ربات اشتراک VIP) ثبت کنید."
   130→      ),
   131→      parse_mode: "HTML",
   132→    });
   133→  }
   134→}

   215→export interface AdminDecisionResult {
   216→  ok: boolean;
   217→  statusMessage: string;
   218→  info?: { user_id: number; expires_at: string; invite_link: string };
   219→}

   221→/**
   222→ * تصمیم ادمین روی فیش کارت‌به‌کارت — از تلگرام (دکمه) و از پنل وب فراخوانی می‌شود.
   223→ * تایید → فعال‌سازی اشتراک + لینک یک‌بارمصرف VIP + پیام به کاربر.
   224→ */
   225→export async function processAdminReceiptDecision(
   226→  db: Database,
   227→  cfg: BotConfig,
   228→  receiptId: number,
   229→  approve: boolean
   230→): Promise<AdminDecisionResult> {
   231→  const rec = getCardReceipt(db, receiptId);
   232→  if (!rec) return { ok: false, statusMessage: "رسید یافت نشد." };
   233→  if (rec.status !== "PENDING") {
   234→    return { ok: false, statusMessage: `این رسید قبلاً در وضعیت ${rec.status} قرار گرفته است.` };
   235→  }
   236→
   237→  setCardReceiptStatus(db, receiptId, approve ? "APPROVED" : "REJECTED");
   238→
   239→  const msgs = buildMessages(cfg.publicChannelUsername);
   240→  const user = getOrCreateUser(db, rec.user_id);
   241→  const lang: Lang = user.lang === "en" ? "en" : "fa";
   242→  const m = msgs[lang];
   243→  const plan = SUBSCRIPTION_PLANS[rec.plan_key] ?? SUBSCRIPTION_PLANS.pro_monthly;
   244→  const planTitle = lang === "fa" ? plan.title_fa : plan.title_en;
   245→
   246→  if (approve) {
   247→    const expStr = extendUserVip(db, rec.user_id, plan.days, rec.plan_key);
   248→    const inviteLink = await createSingleUseVipInviteLink(cfg, rec.user_id, plan.days);
   249→    const userMsg = fmt(m.card_approved_user, {
   250→      receipt_id: receiptId,
   251→      plan_title: planTitle,
   252→      expires_at: expiryLabel(rec.plan_key, expStr, lang),
   253→      invite_link: inviteLink,
   254→    });
   255→    await tgApiCall(cfg, "sendMessage", { chat_id: rec.user_id, text: userMsg, parse_mode: "HTML" });
   256→    return {
   257→      ok: true,
   258→      statusMessage: `✅ رسید #${receiptId} تایید شد و لینک یک‌بارمصرف VIP برای کاربر ${rec.user_id} ارسال گردید.`,
   259→      info: { user_id: rec.user_id, expires_at: expStr, invite_link: inviteLink },
   260→    };
   261→  }
   262→
   263→  const userMsg = fmt(m.card_rejected_user, { receipt_id: receiptId });
   264→  await tgApiCall(cfg, "sendMessage", { chat_id: rec.user_id, text: userMsg, parse_mode: "HTML" });
   265→  return { ok: true, statusMessage: `❌ رسید #${receiptId} رد شد و به کاربر اطلاع داده شد.` };
   266→}

   268→// ─── کنار گذاشتن اعضای منقضی از کانال VIP (ساعتی) ───────────────────────────
   269→
   270→export async function sweepExpiredVipMembers(db: Database, cfg: BotConfig): Promise<number> {
   271→  const nowIso = new Date().toISOString();
   272→  const expired = db.query("SELECT user_id FROM users WHERE vip_expires_at IS NOT NULL AND vip_expires_at < ?").all(nowIso) as { user_id: number }[];
   273→  let removed = 0;
   274→  for (const row of expired) {
   275→    await tgApiCall(cfg, "banChatMember", { chat_id: cfg.vipChannelId, user_id: row.user_id });
   276→    await tgApiCall(cfg, "unbanChatMember", { chat_id: cfg.vipChannelId, user_id: row.user_id, only_if_banned: true });
   277→    db.run("UPDATE users SET vip_expires_at = NULL WHERE user_id = ?", [row.user_id]);
   278→    removed += 1;
   279→  }
   280→  return removed;
   281→}

   283→// ─── پردازشگر آپدیت‌های تلگرام (بخش ۷ پایتون) ───────────────────────────────

   285→export async function handleUpdate(db: Database, cfg: BotConfig, update: Update): Promise<void> {
   286→  // خودآموزی شناسه‌های مالک/کانال VIP — قبل از هر پردازشی
   287→  await learnIdsFromUpdate(db, cfg, update);
   288→
   289→  const msgs = buildMessages(cfg.publicChannelUsername);
   290→
   291→  if (update.message) {
   292→    const msg = update.message;
   293→    const chatId = msg.chat.id;
   294→    const userId = msg.from?.id ?? chatId;
   295→    const username = msg.from?.username ?? "";
   296→    const text = (msg.text ?? "").trim();
   297→    const photos = msg.photo ?? [];
   298→
   299→    let referredBy: number | null = null;
   300→    if (text.startsWith("/start") && text.includes("ref_")) {
   301→      try {
   302→        const refId = parseInt(text.split("ref_")[1].trim(), 10);
   303→        if (Number.isFinite(refId) && refId !== userId) referredBy = refId;
   304→      } catch {
   305→        /* ref نامعتبر */
   306→      }
   307→    }
   308→
   309→    const user = getOrCreateUser(db, userId, username, referredBy);
   310→    const lang: Lang = user.lang === "en" ? "en" : "fa";
   311→    const m = msgs[lang];
   312→
   313→    if (text.startsWith("/start")) {
   314→      setUserPlanAndMode(db, userId, user.selected_plan, "NONE", user.pending_toman_amount);
   315→      await tgApiCall(cfg, "sendMessage", {
   316→        chat_id: chatId,
   317→        text: m.welcome,
   318→        parse_mode: "HTML",
   319→        reply_markup: mainMenuKeyboard(lang, msgs),
   320→      });
   321→      return;
   322→    }
   323→
   324→    // ─── دستورات منوی BotFather — به همان مسیر دکمه‌های شیشه‌ای هدایت می‌شوند ───
   325→    const cmd = text.toLowerCase().split(/[\s@]/)[0];
   326→    const CMD_TO_CALLBACK: Record<string, string> = {
   327→      "/plans": "menu_plans",
   328→      "/free": "menu_free_signal",
   329→      "/status": "menu_status",
   330→      "/referral": "menu_referral",
   331→      "/lang": "toggle_lang",
   332→      "/vip": "menu_plans",
   333→      "/pay": "menu_plans",
   334→    };
   335→    if (CMD_TO_CALLBACK[cmd]) {
   336→      await handleUpdate(db, cfg, {
   337→        callback_query: {
   338→          id: "cmd-dispatch",
   339→          from: msg.from ?? { id: userId, username },
   340→          message: { chat: msg.chat },
   341→          data: CMD_TO_CALLBACK[cmd],
   342→        },
   343→      });
   344→      return;
   345→    }
   346→
   347→    // الف) عکس فیش واریزی → ثبت فیش + ارسال به ادمین
   348→    if (photos.length > 0) {
   349→      const bestPhotoFileId = photos[photos.length - 1].file_id;
   350→      const planKey = user.selected_plan || "pro_monthly";
   351→      const expectedToman =
   352→        user.pending_toman_amount ||
   353→        (SUBSCRIPTION_PLANS[planKey] ?? SUBSCRIPTION_PLANS.pro_monthly).price_toman;
   354→      const receiptId = createCardReceipt(db, userId, username, planKey, expectedToman, "PHOTO", bestPhotoFileId);
   355→      await notifyAdminOfCardReceipt(db, cfg, receiptId);
   356→      await tgApiCall(cfg, "sendMessage", {
   357→        chat_id: chatId,
   358→        text: fmt(m.receipt_submitted_user, { receipt_id: receiptId }),
   359→        parse_mode: "HTML",
   360→      });
   361→      return;
   362→    }
   363→
   364→    // ب) متن رسید (کد پیگیری / ۴ رقم آخر کارت) در حالت انتظار فیش — رشته‌های شبیه TxID فیش محسوب نمی‌شوند
   365→    if (
   366→      user.payment_mode === "AWAITING_CARD_RECEIPT" &&
   367→      text.length >= 4 &&
   368→      text.length !== 64 &&
   369→      text.length !== 66 &&
   370→      !isSolanaTxid(text)
   371→    ) {
   372→      const planKey = user.selected_plan || "pro_monthly";
   373→      const expectedToman =
   374→        user.pending_toman_amount ||
   375→        (SUBSCRIPTION_PLANS[planKey] ?? SUBSCRIPTION_PLANS.pro_monthly).price_toman;
   376→      const receiptId = createCardReceipt(db, userId, username, planKey, expectedToman, "TEXT", text);
   377→      await notifyAdminOfCardReceipt(db, cfg, receiptId);
   378→      await tgApiCall(cfg, "sendMessage", {
   379→        chat_id: chatId,
   380→        text: fmt(m.receipt_submitted_user, { receipt_id: receiptId }),
   381→        parse_mode: "HTML",
   382→      });
   383→      return;
   384→    }
   385→
   386→    // ج) TxID تتر (ترون ۶۴ hex / EVM 66 با 0x / سولانا ~87 Base58) → استعلام خودکار بلاکچین
   387→    const isTronOrEvm =
   388→      (text.length === 64 || text.length === 66) &&
   389→      [...text].every((c) => "0123456789abcdefABCDEFxX".includes(c));
   390→    if (isTronOrEvm || isSolanaTxid(text)) {
   391→      await tgApiCall(cfg, "sendMessage", { chat_id: chatId, text: m.verifying_tx });
   392→      const planKey = user.selected_plan || "pro_monthly";
   393→      const plan = SUBSCRIPTION_PLANS[planKey] ?? SUBSCRIPTION_PLANS.pro_monthly;
   394→
   395→      if (txidAlreadyUsed(db, text.trim())) {
   396→        await tgApiCall(cfg, "sendMessage", { chat_id: chatId, text: m.tx_already_used, parse_mode: "HTML" });
   397→        return;
   398→      }
   399→
   400→      const vr = await verifyUsdtTxid(
   401→        text,
   402→        {
   403→          trc20: cfg.walletUsdtTrc20,
   404→          polygon: cfg.walletUsdtPolygon,
   405→          bsc: cfg.walletUsdtBsc,
   406→          solana: cfg.walletUsdtSolana,
   407→        },
   408→        plan.price_usdt
   409→      );
   410→      if (!vr.ok) {
   411→        await tgApiCall(cfg, "sendMessage", {
   412→          chat_id: chatId,
   413→          text: fmt(m.tx_failed, { reason: vr.message }),
   414→          parse_mode: "HTML",
   415→        });
   416→        return;
   417→      }
   418→
   419→      insertPayment(db, text.trim(), userId, vr.network, vr.amount, planKey);
   420→      const expStr = extendUserVip(db, userId, plan.days, planKey);
   421→      const inviteLink = await createSingleUseVipInviteLink(cfg, userId, plan.days);
