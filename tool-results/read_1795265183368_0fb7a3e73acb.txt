   241→   241→ * منبع واحد حقیقت گیت «ضریب همپوشانی مثبت» — مشترک بین مسیریاب، سوپروایزر و خط صدور.
   242→   242→ */
   243→   243→export function overlapMetricsOf(oddsA: number, oddsB: number, doubleWinProb: number, jointLossProb: number): {
   244→   244→  impliedSum: number;
   245→   245→  effectiveOverlapOdds: number;
   246→   246→  guaranteedOverlapMarginPct: number;
   247→   247→  doubleWinOverlapProb: number;
   248→   248→  jointLossRiskPct: number;
   249→   249→  hasPositiveOverlap: boolean;
   250→   250→  failReason: string | null;
   251→   251→} {
   252→   252→  const oa = Math.max(1.01, oddsA);
   253→   253→  const ob = Math.max(1.01, oddsB);
   254→   254→  const impliedSum = 1 / oa + 1 / ob;
   255→   255→  const effectiveOverlapOdds = 1 / impliedSum;
   256→   256→  const guaranteedOverlapMarginPct = (effectiveOverlapOdds - 1) * 100;
   257→   257→  const doubleWinOverlapProb = Math.max(0, doubleWinProb);
   258→   258→  const jointLossRiskPct = Math.max(0, jointLossProb) * 100;
   259→   259→  const checks: { ok: boolean; why: string }[] = [
   260→   260→    { ok: impliedSum <= OVERLAP_IMPLIED_SUM_MAX, why: `مجموع احتمال ضمنی ${impliedSum.toFixed(3)} بالای سقف ۰.۸۶۵ — ضریب موثر همپوشانی کافی نیست` },
   261→   261→    { ok: guaranteedOverlapMarginPct >= OVERLAP_MARGIN_MIN_PCT, why: `حاشیه سود تضمینی همپوشانی ${guaranteedOverlapMarginPct.toFixed(1)}٪ زیر کف ۱۵٪` },
   262→   262→    // v24.5 — سند بازیابی فوری: شرط برد دوبل ≥ ۱۲٪ حذف شد (زوج‌های متعامد مکمل ذاتاً ناحیه
   263→   263→    // برد دوبل صفر دارند ولی در عمل ۹۰.۷٪ وین‌ریت و بیشترین جکپات را ثبت کرده‌اند — شرط رسمی
   264→   264→    // تایید فقط سه مورد: ریسک ≤ ۱۴.۵٪ · λ_Fav ≥ ۱.۱۲ · impliedSum ≤ ۰.۸۶۵)
   265→   265→    { ok: jointLossRiskPct <= OVERLAP_JOINT_LOSS_MAX_PCT, why: `ریسک تله میانی ${jointLossRiskPct.toFixed(1)}٪ بالای سقف ۱۴.۵٪` },
   266→   266→  ];
   267→   267→  const failed = checks.filter((c) => !c.ok);
   268→   268→  return {
   269→   269→    impliedSum: Math.round(impliedSum * 1000) / 1000,
   270→   270→    effectiveOverlapOdds: Math.round(effectiveOverlapOdds * 1000) / 1000,
   271→   271→    guaranteedOverlapMarginPct: Math.round(guaranteedOverlapMarginPct * 10) / 10,
   272→   272→    doubleWinOverlapProb,
   273→   273→    jointLossRiskPct: Math.round(jointLossRiskPct * 10) / 10,
   274→   274→    hasPositiveOverlap: failed.length === 0,
   275→   275→    failReason: failed.length > 0 ? failed.map((f) => f.why).join(" · ") : null,
   276→   276→  };
   277→   277→}

   282→   282→ * آنتروپی شانون ماتریس ۶×۶ — شاخص تمرکز یا پراکندگی نتایج.
   283→   283→ * H = −Σ mat[i][j]·ln(mat[i][j]) روی زیرماتریس ۰..۵ × ۰..۵ (نرمال‌شده درون همان بازه).
   284→   284→ */
   285→   285→export function shannonEntropy(matrix: number[][]): number {
   286→   286→  const N = Math.min(6, matrix.length);
   287→   287→  let sum = 0;
   288→   288→  for (let h = 0; h < N; h++) {
   289→   289→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) sum += matrix[h][a];
   290→   290→  }
   291→   291→  if (sum <= 0) return Math.log(36);
   292→   292→  let H = 0;
   293→   293→  for (let h = 0; h < N; h++) {
   294→   294→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   295→   295→      const p = matrix[h][a] / sum;
   296→   296→      if (p > 1e-12) H -= p * Math.log(p);
   297→   297→    }
   298→   298→  }
   299→   299→  return H;
   300→   300→}

   302→   302→/** نسبت عدم تقارن قدرت: Rλ = max(λH, λA) / min(λH, λA) */
   303→   303→export function asymmetryRatio(lambdaHome: number, lambdaAway: number): number {
   304→   304→  const hi = Math.max(lambdaHome, lambdaAway);
   305→   305→  const lo = Math.min(lambdaHome, lambdaAway);
   306→   306→  if (lo <= 0.01) return 9.99;
   307→   307→  return Math.min(9.99, hi / lo);
   308→   308→}

   311→   311→export function topCells(matrix: number[][], n = 4): { score: string; p: number }[] {
   312→   312→  const N = Math.min(6, matrix.length);
   313→   313→  const cells: { score: string; p: number }[] = [];
   314→   314→  for (let h = 0; h < N; h++) {
   315→   315→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   316→   316→      cells.push({ score: `${h}-${a}`, p: matrix[h][a] });
   317→   317→    }
   318→   318→  }
   319→   319→  cells.sort((x, y) => y.p - x.p);
   320→   320→  return cells.slice(0, n);
   321→   321→}

   336→   336→function matrixProb(matrix: number[][], predicate: (h: number, a: number) => boolean): number {
   337→   337→  const N = Math.min(6, matrix.length);

   344→   344→    }
   345→   345→  }
   346→   346→  return inRegion > 0 ? hit / inRegion : 0;

   371→   371→export function calculateNonLinearSupervisorScore(candidate: NonLinearCandidateInput): {
   372→   372→  ev: number;
   373→   373→  sharpe: number;
   374→   374→  score: number;
   375→   375→} {
   376→   376→  const netReturnOnWin = candidate.effectiveOdds - 1;

   379→   379→  const variance =
   380→   380→    candidate.winProb * Math.pow(netReturnOnWin - expectedValue, 2) +
   381→   381→    lossProb * Math.pow(-1 - expectedValue, 2);
   382→   382→  const sharpeRatio = expectedValue / Math.sqrt(Math.max(0.01, variance));
   383→   383→
   384→   384→  // جریمه غیرخطی ریسک و پاداش آنتروپی پایین
   385→   385→  const nonLinearSafetyBonus = Math.pow(Math.max(0, candidate.winProb), 2.2) * 100;
   386→   386→  const entropyBonus = Math.max(0, (2.8 - candidate.entropy) * 12);
   387→   387→
   388→   388→  const score = Number(
   389→   389→    ((sharpeRatio * 45 + nonLinearSafetyBonus * 0.55 + entropyBonus) * candidate.leagueReliability).toFixed(1)
   390→   390→  );
   391→   391→  return { ev: expectedValue, sharpe: sharpeRatio, score };
   392→   392→}

   735→   735→export function routeBetTypes(input: {

   778→   778→  const candidates: BetTypeCandidate[] = [

   803→   803→

   805→   805→  const regimeGatePass = (c: BetTypeCandidate): { pass: boolean; why: string } => {

   808→   808→      case "CORRECT_SCORE":

   810→   810→        return { pass: false, why: "رژیم خوشه نتیجه دقیق غیرفعال است (رژیم زیان‌ده −۱۶u سند بازگشت v19.0)" };

   813→   813→        if (fav < COMBO_FAV_LAMBDA_MIN || dog > COMBO_DOG_LAMBDA_MAX) {

   816→   816→        if (c.winProb < COMBO_COVERAGE_MIN_V18 || c.sharpe < COMBO_SHARPE_MIN_V18) {
   817→   817→          return { pass: false, why: `پوشش/شارپ کمبو کافی نیست (پوشش=${(c.winProb * 100).toFixed(0)}٪ ≥ ۷۸٪؟ شارپ=${c.sharpe.toFixed(2)} ≥ 0.22؟)` };
   818→   818→        }
   819→   819→        return { pass: true, why: "کمبوی رسمی — فاوریت کنترل‌گر با پوشش ≥ ۷۸٪" };
   820→   820→      case "BANKER":
   821→   821→        // مجاز برای rLambda ≥ 1.65 و احتمال پایه ≥ ۸۰٪
   822→   822→        if (structure.rLambda < BANKER_R_LAMBDA_MIN_V18 || c.mainProb < BANKER_BASE_PROB_MIN_V18) {
   823→   823→          return { pass: false, why: `عدم تقارن/احتمال پایه کافی نیست (Rλ=${structure.rLambda.toFixed(2)} ≥ 1.65؟ P=${(c.mainProb * 100).toFixed(0)}٪ ≥ ۸۰٪؟)` };
   824→   824→        }
   825→   825→        return { pass: true, why: "بنکر امن — فاوریت قاطع با عدم تقارن بالا" };
   826→   826→      case "DUAL_HEDGE": {
   827→   827→        // ═══ v19.0-restore — فیلتر طلایی ضریب همپوشانی مثبت (Strict Positive Overlap Return Filter) ═══
   828→   828→        // قانون آهنین: فقط بازی‌هایی تایید شوند که ضریب همپوشانی مثبت و قوی دارند —
   829→   829→        // impliedSum ≤ ۰.۸۶ · حاشیه تضمینی ≥ ۱۵٪ · ناحیه برد دوبل ≥ ۱۲٪ · ریسک تله ≤ ۱۱.۵٪
   830→   830→        const m = overlapMetricsOf(c.mainOdds, c.coverOdds ?? 1.01, c.doubleWinProb, c.jointLossProb);
   831→   831→        if (!m.hasPositiveOverlap) {
   832→   832→          return { pass: false, why: `فیلتر ضریب همپوشانی مثبت v19.0: ${m.failReason ?? ""} — وتو و انتقال به دفتر سایه` };
   833→   833→        }
   834→   834→        if (c.sharpe < DUAL_HEDGE_SHARPE_MIN_V18) {
   835→   835→          return { pass: false, why: `شارپ همپوشانی دوقلو ${c.sharpe.toFixed(2)} زیر کف ۰.۱۴ — انتقال به دفتر سایه` };
   836→   836→        }
   837→   837→        const dwProfit = c.doubleWinProfitUnits ?? 0;
   838→   838→        if (dwProfit > 0 && (dwProfit < OVERLAP_DOUBLE_WIN_PROFIT_MIN || dwProfit > OVERLAP_DOUBLE_WIN_PROFIT_MAX + 60)) {
   839→   839→          return { pass: false, why: `سود برد دوبل +${dwProfit.toFixed(1)}u خارج بازه هدف +۱۲۰..+۱۸۰ واحد ناحیه همپوشانی دو برد` };
   840→   840→        }
   841→   841→        return { pass: true, why: `همپوشانی دوقلو با ضریب موثر ${m.effectiveOverlapOdds.toFixed(2)}x و حاشیه تضمینی +${m.guaranteedOverlapMarginPct.toFixed(1)}٪ ≥ ۱۵٪ — برد دوبل +${dwProfit.toFixed(1)}u (تخصیص ۶۵/۳۵ + سپر دقیقه ۶۵)` };
   842→   842→      }
   843→   843→      default:
   844→   844→        return { pass: false, why: "رژیم نامشخص" };
   845→   845→    }
   846→   846→  };
   847→   847→
   848→   848→  const riskCap = (c: BetTypeCandidate): boolean => riskPctOf(c) <= GOLD_RISK_PCT_MAX;
   849→   849→  const eliteValue = (c: BetTypeCandidate): boolean => c.ev > APPROVE_EV_MIN && c.sharpe > APPROVE_SHARPE_MIN;
   850→   850→
   851→   851→  /** 👑 سطح ۱ — تاج طلایی: گیت اختصاصی رژیم + سقف جهانی ریسک + کف شارپ ۰.۱۴
   852→   852→   *  v19.0-restore — رژیم DUAL_HEDGE از مسیر فرار eliteValue معاف است: قانون آهنین
   853→   853→   *  «فیلتر ضریب همپوشانی مثبت» برای زوج‌تیپ‌های دوبال مطلق است (هیچ escape hatch ندارد). */

   857→   857→    (c.key === "DUAL_HEDGE" ? regimeGatePass(c).pass : regimeGatePass(c).pass || eliteValue(c));
   858→   858→

   862→   862→    c.winProb >= SILVER_WIN_PROB_MIN &&
   863→   863→    c.sharpe >= SILVER_SHARPE_MIN;
   864→   864→
   865→   865→  // برچسب‌گذاری تاج همه کاندیدها (شفافیت کامل تصمیم سطح‌بندی‌شده)
   866→   866→  for (const c of candidates) {
   867→   867→    c.tier = goldPass(c) ? "GOLD" : silverPass(c) ? "SILVER" : null;
   868→   868→  }
   869→   869→
   870→   870→  const gold = candidates.filter((c) => c.tier === "GOLD");
   871→   871→  const silver = candidates.filter((c) => c.tier === "SILVER");
   872→   872→
   873→   873→  // 🚫 وتوی قرمز مطلق سطح ۳ — همه کاندیدها ریسک بالای ۱۵٪ دارند → فقط دفتر کاغذی
   874→   874→  if (candidates.length === 0 || (gold.length === 0 && silver.length === 0)) {
   875→   875→    const worstRisk = candidates.length > 0 ? Math.max(...candidates.map(riskPctOf)) : 0;
   876→   876→    const reason =
   877→   877→      candidates.length === 0
   878→   878→        ? "🛡️ گیت v19.0-restore: هیچ‌یک از رژیم‌های فعال (Banker / Combo / Dual-Hedge) کاندید ساختاری معتبری ساخت — بازی پرنوسان؛ انتقال به دفتر سایه"
   879→   879→        : worstRisk > HARD_RED_VETO_RISK_PCT
   880→   880→          ? `🚫 وتوی قرمز مطلق v18.0: کمترین سناریوی کیفیت ریسک ${worstRisk.toFixed(1)}٪ بالای سقف ۱۵٪ دارد — سطح ۳ (فقط تسویه کاغذی دفتر سایه)`
   881→   881→          : "🛡️ گیت v18.0: هیچ رژیمی گیت اختصاصی خود (جدول سند) و هیچ کاندیدی آستانه تاج نقره‌ای (winProb ≥ ۸۴٪ و شارپ ≥ ۰.۱۶) را پاس نکرد — انتقال به دفتر سایه";
   882→   882→    return {
   883→   883→      structure,
   884→   884→      candidates,
   885→   885→      winnerVariantId: null,
   886→   886→      winner: null,
   887→   887→      crownPairId: null,
   888→   888→      veto: true,
   889→   889→      vetoReason: reason,
   890→   890→      tier: null,

   892→   892→  }
   893→   893→
   894→   894→  // ═══ سطح‌بندی نهایی: تاج طلایی اولویت مطلق دارد؛ در نبود طلایی، بهترین نقره‌ای
   895→   895→  // به‌جای حبس در دفتر سایه با استیک نصف ۵۰ واحدی (Half-Kelly) منتشر می‌شود ═══
   896→   896→  const useSilver = gold.length === 0 && silver.length > 0;
   897→   897→
   898→   898→  // ═══ v19.0 — توزیع متوازن ۵ رژیم (Multi-Regime Balanced Routing — سند جراحی v19.0) ═══
   899→   899→  // «کارت‌های تاییدشده بر اساس ویژگی‌های بازی بین هر ۵ رژیم توزیع شوند تا هر ۵ باکس
   900→   900→  // ویترین دارای کارت تسویه‌شده واقعی و سود مثبت باشد». استخر توزیع = طلایی‌ها +
   901→   901→  // کاندیدهایی که گیت اختصاصی رژیم خود + سقف ریسک + EV مثبت را پاس کرده‌اند (دکترین
   902→   902→  // v18: هر رژیم با گیت خودش قضاوت می‌شود — شارپ فقط معیار سطح تاج است).
   903→   903→  const regimeCounts = input.regimeCounts ?? {};
   904→   904→  const countOf = (k: BetTypeKey): number => regimeCounts[k] ?? 0;
   905→   905→  // سقف ریسک فقط برای رژیم‌هایی معنا دارد که jointLossProb «باخت همزمان دو تیپ» است؛

   907→   907→  // است، نه ریسک ساختاری. کنترل کیفیت این رژیم گیت اختصاصی خودش (آنتروپی/Top4/λ) است.

   910→   910→  const gateQualified = candidates.filter((c) => c.tier === null && balanceEligible(c));

   912→   912→  const poolRegimes = [...new Set(pool.map((c) => c.key))];
   913→   913→  let best = pool.reduce((a, b) => (b.score > a.score ? b : a));

   915→   915→    // سهمیه هر رژیم کم‌نمایت: تا minCount+4 (نمونه‌سازی کافی برای رگرسیون به EV مثبت)
   916→   916→    const minCount = Math.min(...poolRegimes.map(countOf));
   917→   917→    const underRep = poolRegimes.filter((k) => countOf(k) <= minCount + 4 && countOf(k) < countOf(best.key));
   918→   918→    if (underRep.length > 0) {
   919→   919→      // کف امتیاز مقطعی بین رژیم‌ها معنا ندارد (داچینگ نتیجه دقیق ذاتاً واریانس و شارپ
   920→   920→      // کمتری دارد اما گیت اختصاصی رژیم + EV مثبت + سقف ریسک آن را تایید کرده‌اند —
   921→   921→      // دکترین v18: «هر رژیم با گیت خودش قضاوت می‌شود»). بین رژیم‌های کم‌نمایت،
   922→   922→      // بالاترین امتیاز آن رژیم انتخاب می‌شود.
   923→   923→      const eligible = underRep
   924→   924→        .map((k) => pool.filter((c) => c.key === k).reduce((a, b) => (b.score > a.score ? b : a)))
   925→   925→        .sort((a, b) => b.score - a.score);
   926→   926→      if (eligible.length > 0) best = eligible[0];
   927→   927→    }
   928→   928→  }
   929→   929→
   930→   930→  if (useSilver) {
   931→   931→    // 🥈 تاج نقره‌ای — استیک نصف‌شده ۵۰ واحدی به روش Half-Kelly
   932→   932→    const scale = 0.5;
   933→   933→    best.stakeMain = Math.round(best.stakeMain * scale * 10) / 10;
   934→   934→    best.stakeCover = Math.round(best.stakeCover * scale * 10) / 10;
   935→   935→    best.guaranteedProfitUnits = Math.round(best.guaranteedProfitUnits * scale * 10) / 10;
   936→   936→    best.rationale = `${best.rationale} — 🥈 تایید سطح ۲ (شکارچی آلفای سایه — استیک ایمن ۵۰ واحدی به روش Half-Kelly): winProb ${(best.winProb * 100).toFixed(1)}٪ و شارپ ${best.sharpe.toFixed(2)} در لیگ معتبر؛ به‌جای حبس در دفتر سایه با استیک نیم‌مقیاس منتشر می‌شود`;
   937→   937→    return {
   938→   938→      structure,
   939→   939→      candidates,
   940→   940→      winnerVariantId: best.variantId,
   941→   941→      winner: best,
   942→   942→      crownPairId: best.pairId ?? null,
   943→   943→      veto: false,
   944→   944→      vetoReason: null,
   945→   945→      tier: "SILVER",
   946→   946→    };
   947→   947→  }
   948→   948→
   949→   949→  // قانون تایید هوشمند v17 (حفظ‌شده): شارپ ضعیف در بهترین روش → بررسی امکان تایید با روش دیگر
   950→   950→  if (best.sharpe < APPROVE_SHARPE_MIN || best.ev < APPROVE_EV_MIN) {
   951→   951→    const eliteEligible = gold.filter((c) => eliteValue(c));
   952→   952→    if (eliteEligible.length === 0) {
   953→   953→      // تایید مبنای سند — گیت اختصاصی رژیم برنده است
   954→   954→      return {
   955→   955→        structure,
   956→   956→        candidates,
   957→   957→        winnerVariantId: best.variantId,
   958→   958→        winner: best,
   959→   959→        crownPairId: best.pairId ?? null,
   960→   960→        veto: false,
   961→   961→        vetoReason: null,
   962→   962→        tier: "GOLD",
   963→   963→      };
   964→   964→    }
   965→   965→    const bestEligible = eliteEligible.reduce((a, b) => (b.score > a.score ? b : a));
   966→   966→    return {
   967→   967→      structure,
   968→   968→      candidates,
   969→   969→      winnerVariantId: bestEligible.variantId,
   970→   970→      winner: bestEligible,
   971→   971→      crownPairId: bestEligible.pairId ?? null,
   972→   972→      veto: false,
   973→   973→      vetoReason: null,
   974→   974→      tier: "GOLD",
   975→   975→    };
   976→   976→  }
   977→   977→
   978→   978→  return {
   979→   979→    structure,
   980→   980→    candidates,
   981→   981→    winnerVariantId: best.variantId,
   982→   982→    winner: best,
   983→   983→    crownPairId: best.pairId ?? null,
   984→   984→    veto: false,
   985→   985→    vetoReason: null,
   986→   986→    tier: "GOLD",
   987→   987→  };

  1025→  1025→
  1026→  1026→/**
  1027→  1027→ * 🔗 سازنده سه برگه میکس زنجیره‌ای از بهترین بنکرها/کمبوهای تاییدشده روز:
  1028→  1028→ *  ۱) 2-Fold Banker Accumulator — ترکیب ۲ پایه Banker (ضریب کل ۲.۱۰..۲.۵۵، نرخ اصابت > ۸۰٪)
  1029→  1029→ *  ۲) 3-Fold Pro Combo Accumulator — ترکیب ۳ پایه قوی Banker/Combo (ضریب کل ۳.۶۰..۵.۴۰)
  1030→  1030→ *  ۳) Time-Staggered Hedged Accumulator — میکس ۳ بازی با ساعت شروع متفاوت +
  1031→  1031→ *     فرمول محاسبه مبلغ پوشش زنده پای آخر (Cash-Out / Final-Leg Hedge).
  1032→  1032→ * استیک بهینه با معیار کلِی همزمان (Simultaneous Kelly fraction × ۱۰۰).
  1033→  1033→ */
  1034→  1034→export function buildAccumulators(
  1035→  1035→  winners: { matchId: string; matchLabel: string; league: string; kickoff: string; candidate: BetTypeCandidate }[]
  1036→  1036→): AccumulatorSlip[] {
  1037→  1037→  const bankers = winners.filter((w) => w.candidate.key === "BANKER");
  1038→  1038→  const comboOrBanker = winners.filter((w) => w.candidate.key === "BANKER" || w.candidate.key === "COMBO");
  1039→  1039→  const out: AccumulatorSlip[] = [];
  1040→  1040→
  1041→  1041→  const legOf = (w: (typeof winners)[number]): AccumulatorLeg => ({
  1042→  1042→    matchId: w.matchId,
  1043→  1043→    matchLabel: w.matchLabel,
  1044→  1044→    league: w.league,
  1045→  1045→    kickoff: w.kickoff,
  1046→  1046→    // v17.0 — پای زوج‌تیپ: کل پوزیشن داچینگ‌شده با «ضریب موثر» و «پوشش کل» لحاظ می‌شود
  1047→  1047→    tip: w.candidate.key === "DUAL_HEDGE" ? `${w.candidate.mainLabel} ↔ ${w.candidate.coverLabel ?? ""}` : w.candidate.mainLabel,
  1048→  1048→    tipEn: w.candidate.key === "DUAL_HEDGE" ? "Dual-Hedge Effective Position" : w.candidate.mainLabelEn,
  1049→  1049→    odds: w.candidate.key === "DUAL_HEDGE" ? w.candidate.effectiveOdds : w.candidate.mainOdds,
  1050→  1050→    betType: w.candidate.key,
  1051→  1051→  });
  1052→  1052→
  1053→  1053→  // ۱) 2-Fold Banker Accumulator
  1054→  1054→  if (bankers.length >= 2) {
  1055→  1055→    const legs = [legOf(bankers[0]), legOf(bankers[1])];
  1056→  1056→    const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
  1057→  1057→    const hit = bankers[0].candidate.mainProb * bankers[1].candidate.mainProb;
  1058→  1058→    // کلِی همزمان: f* ≈ (p·b − q)/b × نیم‌کِلی محافظه‌کار برای میکس چندپایه
  1059→  1059→    const b = totalOdds - 1;
  1060→  1060→    const f = Math.max(0.015, Math.min(0.06, (hit * b - (1 - hit)) / b * 0.5));
  1061→  1061→    out.push({
  1062→  1062→      id: "ACC-2F",
  1063→  1063→      kind: "TWO_FOLD_BANKER",
  1064→  1064→      labelFa: "میکس زنجیره‌ای ۲ پایه بنکر امن (2-Fold Banker Accumulator)",
  1065→  1065→      labelEn: "2-Fold Banker Accumulator",
  1066→  1066→      icon: "🔗",
  1067→  1067→      legs,
  1068→  1068→      totalOdds,
  1069→  1069→      stake: Math.round(f * 1000) / 10,
  1070→  1070→      hitRatePct: Math.round(hit * 1000) / 10,
  1071→  1071→      potentialReturn: Math.round(f * totalOdds * 10) / 10,
  1072→  1072→    });
  1073→  1073→  } else if (winners.length >= 2) {
  1074→  1074→    // v17.0 — fallback: قوی‌ترین پایه‌های تاییدشده روز (هر Bet-Type)
  1075→  1075→    const legs = [legOf(winners[0]), legOf(winners[1])];
