   205→function poissonTail(k: number, lambda: number): number {
   206→  // P(X > k) = 1 − P(X ≤ k)
   207→  let cdf = 0;
   208→  for (let i = 0; i <= k; i++) cdf += poissonPmf(i, lambda);
   209→  return Math.max(0, Math.min(1, 1 - cdf));
   210→}

   212→/** ماتریس نیمه اول با λ×۰.۴۵ (سهم استاندارد گل‌های نیمه اول) + تصحیح دیکسون-کولز */
   213→function buildHtProbs(lambdaHome: number, lambdaAway: number, rho = -0.08): { over05: number; zero: number } {
   214→  const lh = lambdaHome * 0.45;
   215→  const la = lambdaAway * 0.45;
   216→  let zero = 0;
   217→  let any = 0;
   218→  for (let h = 0; h <= 4; h++) {
   219→    for (let a = 0; a <= 4; a++) {
   220→      const p = Math.max(0, poissonPmf(h, lh) * poissonPmf(a, la) * dixonColesTau(h, a, lh, la, rho));
   221→      if (h === 0 && a === 0) zero = p;
   222→      if (h + a >= 1) any += p;
   223→    }
   224→  }
   225→  return { over05: Math.min(1, any), zero: Math.min(1, zero) };
   226→}

   230→function matrixProb(covers: (h: number, a: number) => boolean, matrix: number[][]): number {
   231→  let p = 0;
   232→  for (let h = 0; h < matrix.length; h++) {
   233→    const row = matrix[h] ?? [];
   234→    for (let a = 0; a < row.length; a++) {
   235→      if (covers(h, a)) p += row[a] ?? 0;
   236→    }
   237→  }
   238→  return p;
   239→}

   299→/** v20.0 — بذر هش پایدار از شناسه بازی (برای ضریب یکتا) */
   300→export function matchSeedOf(id: string | null | undefined): number {
   301→  const s = String(id ?? "NO-MATCH");
   302→  let h = 5381;
   303→  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) | 0;

   307→/**
   308→ * v19.0 — ضریب لبه ارزش بر اساس خطای قیمت‌گذاری بازار نسبت به مدل:
   309→ * implied = احتمال ضمنی بازار (۱ ÷ ضریب بازار). اگر بازار مدل را کم‌قیمت کرده باشد

   312→export function edgeBoostForMispricing(modelProb: number, marketImpliedProb: number | null): number {
   313→  if (!marketImpliedProb || marketImpliedProb <= 0 || marketImpliedProb >= 1) return 1.18;

   315→  return Math.round(Math.min(1.26, Math.max(1.14, 1.14 + mispricing * 0.5)) * 1000) / 1000;
   316→}

   329→  // کم‌قیمت نکرده و قیمت‌گذاری ارزش‌محور موتور (فرمول برنده v19.0) مبنا است —
   330→  // این همان شرطی است که حاشیه تضمینی +۱۶٪ تا +۳۵٪ و impliedSum ≤ ۰.۸۶ را شدنی می‌کند.
   331→  if (typeof realPrice === "number" && realPrice > 1.01 && realPrice >= lo - 0.1 && realPrice <= hi + 0.1) {
   332→    const edgeFloor = (edgeBoostForMispricing(prob, 1 / realPrice) / Math.max(0.18, Math.min(0.68, prob))) * 0.98;
   333→    if (realPrice >= Math.min(hi, edgeFloor)) {
   334→      return Math.round(realPrice * 100) / 100;
   335→    }
   336→  }
   337→  // v19.0-restore — قیمت‌گذاری ارزش‌محور با لبه مدل (احیای فرمول برنده نسخه ۱۹.۰):
   338→  // boost از خطای قیمت‌گذاری بازار نسبت به مدل (۱.۱۴..۱.۲۶ — پیش‌فرض ۱.۱۸)
   339→  void matchSeed;

   342→  const clamped = Math.min(hi, Math.max(lo, raw));

   481→export function generateOrthogonalComplementarySlips(ctx: OrthoContext): OrthogonalResult {
   482→  const ht = buildHtProbs(ctx.lambdaHome, ctx.lambdaAway);
   483→  const lp: LegProbs = {
   484→    htOver05: ht.over05,
   485→    ht00: ht.zero,
   486→    cornersOver95: poissonTail(9, ctx.lambdaCorners),
   487→    cornersUnder95: 1 - poissonTail(9, ctx.lambdaCorners),
   488→    cardsOver35: poissonTail(3, ctx.lambdaCards),
   489→    cardsOver45: poissonTail(4, ctx.lambdaCards),
   490→    cardsUnder45: 1 - poissonTail(4, ctx.lambdaCards),
   491→  };
   492→
   493→  const archetypes: OrthoPair[] = [];
   494→  for (const def of ARCHETYPES) {

   496→    const slipB = buildAtomicSlip(def.legB(ctx), "B", ctx, lp);
   497→
   498→    const pA = slipA.prob;
   499→    const pB = slipB.prob;
   500→    // احتمال برد همزمان: پایه‌های ماتریسی مشترک سلول
   501→    const bothWin = jointWinProb(slipA.legs, slipB.legs, ctx, lp);
   502→    const aOnly = Math.max(0, pA - bothWin);
   503→    const bOnly = Math.max(0, pB - bothWin);
   504→    const bothLose = Math.max(0, 1 - pA - pB + bothWin);
   505→    // v16.0 — پوشش کل بازی = اجتماع دقیق خانه‌های پوشش‌داده‌شده در ماتریس ۶×۶
   506→    const probUnion = Math.min(1, pA + pB - bothWin);
   507→    const coveragePct = probUnion;
   508→    const impliedSum = 1 / Math.max(1.01, slipA.odds) + 1 / Math.max(1.01, slipB.odds);
   509→
   510→    archetypes.push({
   511→      code: def.code,
   512→      name: def.name,
   513→      short: def.short,
   514→      rationale: def.rationale,
   515→      orthogonalityPct: 100, // خانواده شرط‌های A و B در همه الگوها کاملاً تفکیک‌شده است
   516→      coveragePct,
   517→      overlapPct: bothWin,
   518→      probUnion,
   519→      trueJointLossRiskPct: Math.round(bothLose * 1000) / 10,
   520→      doubleWinProbPct: Math.round(bothWin * 1000) / 10,
   521→      slipA,
   522→      slipB,
   523→      jointStates: { bothWin, aOnly, bOnly, bothLose },
   524→      impliedSum,
   525→      legsA: slipA.legs,
   526→      legsB: slipB.legs,
   527→    });
   528→  }
   529→
   530→  // پیشنهاد رسمی: بالاترین امتیاز انطباق با گیت‌های λ + کمینه‌سازی ریسک باخت همزمان
   531→  let recommended: ArchetypeCode = "G1";
   532→  let bestScore = -Infinity;
   533→  for (const a of archetypes) {
   534→    const def = ARCHETYPES.find((d) => d.code === a.code)!;
   535→    // الگویی که هم انطباق آماری دارد و هم ریسک باخت همزمان معقول
   536→    const s = def.score(ctx) - Math.max(0, a.jointStates.bothLose - 0.105) * 2.0;
   537→    if (s > bestScore) {
   538→      bestScore = s;
   539→      recommended = a.code;
   540→    }
   541→  }

   546→/** P(برد همزمان A و B) — پایه‌های ماتریسی روی سلول مشترک */
   547→function jointWinProb(legsA: string[], legsB: string[], ctx: OrthoContext, lp: LegProbs): number {
   548→  const offA = legsA.filter(isOffMatrixLeg).map((k) => orthoLegProb(k, ctx, lp));
   549→  const offB = legsB.filter(isOffMatrixLeg).map((k) => orthoLegProb(k, ctx, lp));
   550→  const offProd = [...offA, ...offB].reduce((s, x) => s * x, 1);
   551→  if (offProd <= 1e-6) return 0;
   552→  const matA = legsA.filter((k) => !isOffMatrixLeg(k));
   553→  const matB = legsB.filter((k) => !isOffMatrixLeg(k));
   554→  let p = 0;
   555→  for (let h = 0; h < ctx.matrix.length; h++) {
   556→    const row = ctx.matrix[h] ?? [];
   557→    for (let a = 0; a < row.length; a++) {
   558→      const cell = row[a] ?? 0;
   559→      if (cell <= 0) continue;
   560→      let ok = true;
   561→      for (const k of matA) if (!orthoLegCoversCell(k, h, a)) { ok = false; break; }
   562→      if (!ok) continue;
   563→      for (const k of matB) if (!orthoLegCoversCell(k, h, a)) { ok = false; break; }
   564→      if (ok) p += cell * offProd;
   565→    }
   566→  }
   567→  return p;
   568→}

   572→export function isOffMatrixLeg(key: string): boolean {
   573→  return OFF_MATRIX_LEGS.has(key);

   576→/**
   577→ * v11.0 — jointShadowMetrics — بازتولید سنجه‌های لحظه تصمیم سوپروایزر از λ و پایه‌های
   578→ * ذخیره‌شده کارت (بدون هیچ عدد ساختگی): ٪ باخت همزمان + ٪ برد طلایی همزمان.
   579→ * دقیقاً همان ریاضیات generateOrthogonalComplementarySlips است تا backfill
   580→ * کارت‌های قدیمی با صدور جدید کاملاً هم‌مقیاس بماند.
   581→ */
   582→export function jointShadowMetrics(
   583→  lambdaHome: number,
   584→  lambdaAway: number,
   585→  lambdaCorners: number,
   586→  lambdaCards: number,
   587→  legsA: string[],
   588→  legsB: string[]
   589→): { jointLossRiskPct: number; doubleWinProbPct: number } {
   590→  const matrix = buildScoreMatrix(lambdaHome || 1.35, lambdaAway || 1.25);
   591→  const ctx: OrthoContext = { lambdaHome, lambdaAway, matrix, lambdaCorners, lambdaCards };
   592→  const ht = buildHtProbs(lambdaHome || 1.35, lambdaAway || 1.25);
   593→  const lp: LegProbs = {
   594→    htOver05: ht.over05,
   595→    ht00: ht.zero,
   596→    cornersOver95: poissonTail(9, ctx.lambdaCorners),
   597→    cornersUnder95: 1 - poissonTail(9, ctx.lambdaCorners),
   598→    cardsOver35: poissonTail(3, ctx.lambdaCards),
   599→    cardsOver45: poissonTail(4, ctx.lambdaCards),
   600→    cardsUnder45: 1 - poissonTail(4, ctx.lambdaCards),
   601→  };
   602→  const pA = atomicLegProb(legsA, ctx, lp);
   603→  const pB = atomicLegProb(legsB, ctx, lp);
   604→  const bothWin = jointWinProb(legsA, legsB, ctx, lp);
   605→  const bothLose = Math.max(0, 1 - pA - pB + bothWin);
   606→  const r1 = (v: number) => Math.round(v * 1000) / 10; // کسر → ٪ با یک رقم اعشار
   607→  return { jointLossRiskPct: r1(bothLose), doubleWinProbPct: r1(bothWin) };
   608→}

   610→/** احتمال یک تیپ (همه پایه‌ها ببرند) — پشتیبانی از کارت‌های قدیمی چندپایه */
   611→function atomicLegProb(legs: string[], ctx: OrthoContext, lp: LegProbs): number {
   612→  const offFactors = legs.filter(isOffMatrixLeg).map((k) => orthoLegProb(k, ctx, lp));
   613→  for (const f of offFactors) if (f <= 1e-6) return 0;
   614→  const offProd = offFactors.reduce((s, x) => s * x, 1);
   615→  const matrixLegs = legs.filter((k) => !isOffMatrixLeg(k));
   616→  let p = 0;
   617→  for (let h = 0; h < ctx.matrix.length; h++) {
   618→    const row = ctx.matrix[h] ?? [];
   619→    for (let a = 0; a < row.length; a++) {
   620→      const cell = row[a] ?? 0;
   621→      if (cell <= 0) continue;
   622→      let ok = true;
   623→      for (const k of matrixLegs) {
   624→        if (!orthoLegCoversCell(k, h, a)) {
   625→          ok = false;
   626→          break;
   627→        }
   628→      }
   629→      if (ok) p += cell * offProd;
   630→    }
   631→  }
   632→  return p;
   633→}

   645→/** استیک ثابت تیپ A (بال اصلی) — سند v19.0: ۶۵ واحد */
   646→export const ASYM_STAKE_A_UNITS = 65;
   647→/** استیک ثابت تیپ B (بال پوششی) — سند v19.0: ۳۵ واحد (در ~۳.۱۰x → برگشت ۱۰۸.۵ = اصل + ۸.۵٪) */
   648→export const ASYM_STAKE_B_UNITS = 35;
   649→/**
   650→ * v19.0 — تخصیص هوشمند استیک ۱۰۰ واحدی (پیش‌فرض فعال برای همه کارت‌های تایید سوپروایزر):
   651→ *   · بال پوششی (تیپ B ~۳.۱۰x): دقیقاً ۳۵ واحد → برد B = ۱۰۸.۵ واحد (اصل پول + ۸.۵٪ سود)
   652→ *   · بال اصلی (تیپ A ~۲.۱۰x): ۶۵ واحد → برد A = ۱۳۶.۵ واحد (+۳۶.۵ سود خالص)
   653→ *   · برد طلایی همزمان (Double-Win در ۱-۰ و ۲-۰): ۱۳۶.۵ + ۱۰۸.۵ = ۲۴۵.۰ → +۱۴۵.۰ واحد در یک مسابقه
   654→ * «سود تضمینی» کارت = کمینه سود دو بال (محافظه‌کارانه و صادقانه).
   655→ */
   656→export function asymmetricAlphaStakes(
   657→  oddsA: number,
   658→  oddsB: number
   659→): { active: boolean; stakeA: number; stakeB: number; profitOnA: number; profitOnB: number; guaranteed: number; profitOnDoubleWin: number; note: string } {
   660→  const total = ASYM_STAKE_A_UNITS + ASYM_STAKE_B_UNITS;
   661→  const stakeA = ASYM_STAKE_A_UNITS;
   662→  const stakeB = ASYM_STAKE_B_UNITS;
   663→  const profitOnA = Math.round((stakeA * oddsA - total) * 10) / 10;
   664→  const profitOnB = Math.round((stakeB * oddsB - total) * 10) / 10;
   665→  const guaranteed = Math.round(Math.min(profitOnA, profitOnB) * 10) / 10;
   666→  const profitOnDoubleWin = Math.round((stakeA * oddsA + stakeB * oddsB - total) * 10) / 10;
   667→  return {
   668→    active: true,
   669→    stakeA,
   670→    stakeB,
   671→    profitOnA,
   672→    profitOnB,
   673→    guaranteed,
   674→    profitOnDoubleWin,
   675→    note: `⚡ تخصیص نامتقارن آلفا — ${stakeA} واحد روی تیپ A در ضریب ${oddsA.toFixed(2)} (برد = +${profitOnA.toFixed(1)}u) + ${stakeB} واحد روی تیپ B در ضریب ${oddsB.toFixed(2)} (برد = ${profitOnB >= 0 ? "+" : ""}${profitOnB.toFixed(1)}u) — برد طلایی همزمان ۱-۰/۲-۰ = جک‌پات +${profitOnDoubleWin.toFixed(1)} واحد`,
   676→  };
   677→}

   702→  const accelerator = asymmetricAlphaStakes(a.slipA.odds, a.slipB.odds);
   703→  const stakeA = accelerator.stakeA;
   704→  const stakeB = accelerator.stakeB;

   713→    modelProb: a.slipA.prob,
   714→    bestOdds: a.slipA.odds,

   716→    // v10.0 — ارزش عددی همیشه دارد: (prob مدل × ضریب − ۱) × ۱۰۰ نسبت به خط منصفانه

   718→    evPct: a.slipA.prob * a.slipA.odds - 1,

   742→    legsB: a.legsB,

   745→    doubleWin: a.overlapPct > 0.02,
   746→  };
   747→  const kelly = simultaneousKelly(
   748→    [
   749→      { p: a.jointStates.bothWin, aWins: true, bWins: true },
   750→      { p: a.jointStates.aOnly, aWins: true, bWins: false },
   751→      { p: a.jointStates.bOnly, aWins: false, bWins: true },
   752→      { p: a.jointStates.bothLose, aWins: false, bWins: false },
   753→    ],
   754→    a.slipA.odds,
   755→    a.slipB.odds
   756→  );
   757→

   759→    id: `ortho-${a.code}-${match.id}`,

   782→      profitOnA: accelerator.profitOnA,
