    77→    77→/** ضریب فرم از ۵ بازی اخیر واقعی (WWDLW) — بازه ۰.۷۸ تا ۱.۲۶ */
    78→    78→function formFactor(form: string): number {
    79→    79→  if (!form || form === "N/A") return 1.0;
    80→    80→  const pts = [...form].reduce((a, ch) => a + (ch === "W" ? 3 : ch === "D" ? 1 : 0), 0);
    81→    81→  return 0.78 + (pts / (form.length * 3)) * 0.48;
    82→    82→}

   106→   106→interface DerivedStrength {
   107→   107→  lambdaHome: number;
   108→   108→  lambdaAway: number;
   109→   109→  homeAttack: number;
   110→   110→  awayAttack: number;

   120→   120→/**
   121→   121→ * محاسبه امید گل (λ) از فرم + رکورد فصل واقعی دو تیم و اعمال تعدیل جوی
   122→   122→ * λ_base = avgGoals × سهم میزبانی/میهمانی × قدرت نسبی (فرم ۵۵٪ + فصل ۴۵٪)
   123→   123→ * تعدیل جوی: بارش > ۲mm یا باد > ۳۰km/h → λ × ۰.۹۲ (−۸٪)
   124→   124→ */
   125→   125→function deriveStrength(ev: LiveMatchEvent, weather: LiveWeather | null): DerivedStrength {
   126→   126→  const homeStr = 0.55 * formFactor(ev.homeForm) + 0.45 * seasonFactor(ev.homeRecord);
   127→   127→  const awayStr = 0.55 * formFactor(ev.awayForm) + 0.45 * seasonFactor(ev.awayRecord);
   128→   128→
   129→   129→  const adverse = weather?.adverse ?? false;
   130→   130→  const wAdj = adverse ? 0.92 : 1.0;
   131→   131→
   132→   132→  let lambdaHome = ev.avgGoals * 0.55 * homeStr * wAdj;
   133→   133→  let lambdaAway = ev.avgGoals * 0.45 * awayStr * wAdj;
   134→   134→
   135→   135→  // همگرایی به محدوده معقول پواسون
   136→   136→  lambdaHome = clamp(lambdaHome, 0.3, 3.8);
   137→   137→  lambdaAway = clamp(lambdaAway, 0.25, 3.6);
   138→   138→
   139→   139→  // استخراج ضرایب حمله/دفاع سازگار با موتور دیکسون-کولز:
   140→   140→  // λh = homeAttack × awayDefense × 1.35 × 1.12 → دفاع‌ها = ۱، حمله = λ / ثابت
   141→   141→  const homeAttack = clamp(lambdaHome / (1.35 * 1.12), 0.35, 2.3);
   142→   142→  const awayAttack = clamp(lambdaAway / 1.35, 0.3, 2.2);
   143→   143→
   144→   144→  const homeElo = eloFromRecord(ev.homeRecord);
   145→   145→  const awayElo = eloFromRecord(ev.awayRecord);
   146→   146→
   147→   147→  const homeFieldTilt = Math.round(clamp(50 + (homeElo - 1500) / 12, 42, 63));
   148→   148→  const awayFieldTilt = Math.round(clamp(50 + (awayElo - 1500) / 12, 40, 62));
   149→   149→  const homePpda = r2(clamp(12 - (homeStr - 1) * 7, 8.5, 15));
   150→   150→  const awayPpda = r2(clamp(12.4 - (awayStr - 1) * 7, 8.5, 15.5));
   151→   151→
   152→   152→  return { lambdaHome: r2(lambdaHome), lambdaAway: r2(lambdaAway), homeAttack: r2(homeAttack), awayAttack: r2(awayAttack), homeElo, awayElo, homePpda, awayPpda, homeFieldTilt, awayFieldTilt, weatherAdjusted: adverse };
   153→   153→}

   182→   182→function matchOddsEvent(ev: LiveMatchEvent, oddsEvents: OddsApiEvent[]): OddsApiEvent | null {
   183→   183→  const direct = `${normTeam(ev.homeTeam)}::${normTeam(ev.awayTeam)}`;
   184→   184→  const kick = Date.parse(ev.kickoff);
   185→   185→  let best: OddsApiEvent | null = null;
   186→   186→  let bestDist = Infinity;
   187→   187→  for (const oe of oddsEvents) {
   188→   188→    const same = oe.matchKey === direct || `${normTeam(oe.homeTeam)}::${normTeam(oe.awayTeam)}` === `${normTeam(ev.awayTeam)}::${normTeam(ev.homeTeam)}`;
   189→   189→    if (!same) continue;
   190→   190→    const dist = Math.abs(Date.parse(oe.commenceTime) - kick);
   191→   191→    if (dist < bestDist) {
   192→   192→      bestDist = dist;
   193→   193→      best = oe;
   194→   194→    }
   195→   195→  }
   196→   196→  // تطبیق فقط برای بازی‌های نزدیک (±۲۴h) — جلوگیری از جفت‌شدن بازی‌های دور
   197→   197→  return best && bestDist < 24 * 3600 * 1000 ? best : null;
   198→   198→}

   212→   212→function resolveOdds(ev: LiveMatchEvent, oddsEv: OddsApiEvent | null, probs: ReturnType<typeof modelProbs>): ResolvedOdds {
   213→   213→  // ─── ۱) ضرایب زنده The Odds API — بهترین قیمت هر انتخاب + سه بوک نماینده ───
   214→   214→  if (oddsEv && oddsEv.books.length > 0) {
   215→   215→    const booksSorted = [...oddsEv.books].sort(
   216→   216→      (a, b) => 1 / a.line.home + 1 / a.line.draw + 1 / a.line.away - (1 / b.line.home + 1 / b.line.draw + 1 / b.line.away)
   217→   217→    );
   218→   218→    const canon = ["PINNACLE", "SOFT_A", "SOFT_B"] as const;
   219→   219→    const books = booksSorted.slice(0, 3).map((b, i) => ({
   220→   220→      bookmaker: canon[i],
   221→   221→      bookmakerLabel: `${b.label}${i === 0 ? " (Sharp)" : ""}`,
   222→   222→      home: b.line.home,
   223→   223→      draw: b.line.draw,
   224→   224→      away: b.line.away,
   225→   225→      over25: b.line.over25,
   226→   226→      under25: b.line.under25,
   227→   227→    }));
   228→   228→    const best = (sel: (b: ResolvedOdds["books"][number]) => number | null): number => {
   229→   229→      const vals = books.map(sel).filter((v): v is number => v !== null && v >= 1.01);
   230→   230→      return vals.length > 0 ? Math.max(...vals) : 0;
   231→   231→    };
   232→   232→    const oddsHome = best((b) => b.home);
   233→   233→    const oddsDraw = best((b) => b.draw);
   234→   234→    const oddsAway = best((b) => b.away);
   235→   235→    let oddsOver25 = best((b) => b.over25);
   236→   236→    let oddsUnder25 = best((b) => b.under25);
   237→   237→    if (oddsOver25 <= 1.01 || oddsUnder25 <= 1.01) {
   238→   238→      oddsOver25 = fairOdds(probs.pOver25, 1.06);
   239→   239→      oddsUnder25 = fairOdds(probs.pUnder25, 1.06);
   240→   240→    }
   241→   241→    return { oddsHome, oddsDraw, oddsAway, oddsOver25, oddsUnder25, odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06), source: "ODDS_API", books };
   242→   242→  }
   243→   243→
   244→   244→  // ─── ۲) خط پایه ESPN (معمولاً DraftKings) ───
   245→   245→  if (ev.espnOdds && ev.espnOdds.home && ev.espnOdds.draw && ev.espnOdds.away) {
   246→   246→    const o = ev.espnOdds;
   247→   247→    const h = o.home as number;
   248→   248→    const d2 = o.draw as number;
   249→   249→    const a2 = o.away as number;
   250→   250→    return {
   251→   251→      oddsHome: h,
   252→   252→      oddsDraw: d2,
   253→   253→      oddsAway: a2,
   254→   254→      oddsOver25: o.over25 ?? fairOdds(probs.pOver25, 1.06),
   255→   255→      oddsUnder25: o.under25 ?? fairOdds(probs.pUnder25, 1.06),
   256→   256→      odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06),
   257→   257→      source: "ESPN_DRAFTKINGS",
   258→   258→      books: [
   259→   259→        {
   260→   260→          bookmaker: "SOFT_A",
   261→   261→          bookmakerLabel: `${o.provider} (ESPN)`,
   262→   262→          home: h,
   263→   263→          draw: d2,
   264→   264→          away: a2,
   265→   265→          over25: o.over25 ?? fairOdds(probs.pOver25, 1.06),
   266→   266→          under25: o.under25 ?? fairOdds(probs.pUnder25, 1.06),
   267→   267→        },
   268→   268→      ],
   269→   269→    };
   270→   270→  }
   271→   271→
   272→   272→  // ─── ۳) Fallback مدل — ضرایب منصفانه با حاشیه ۶٪ ───
   273→   273→  return {
   274→   274→    oddsHome: fairOdds(probs.pHome, 1.06),
   275→   275→    oddsDraw: fairOdds(probs.pDraw, 1.06),
   276→   276→    oddsAway: fairOdds(probs.pAway, 1.06),
   277→   277→    oddsOver25: fairOdds(probs.pOver25, 1.06),
   278→   278→    oddsUnder25: fairOdds(probs.pUnder25, 1.06),
   279→   279→    odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06),
   280→   280→    source: "ENGINE",
   281→   281→    books: [],
   282→   282→  };
   283→   283→}

   287→   287→async function weatherFor(city: string | null): Promise<LiveWeather | null> {
   288→   288→  if (!city) return null;
   289→   289→  const coords = cityCoords(city);
   290→   290→  if (!coords) return null;
   291→   291→  const hit = cache.weather.get(city);
   292→   292→  if (hit && Date.now() - hit.at < WEATHER_TTL_MS) return hit.w;
   293→   293→  const w = await fetchWeather(coords[0], coords[1], city);
   294→   294→  cache.weather.set(city, { w, at: Date.now() });
   295→   295→  return w;
   296→   296→}

   298→   298→async function poolWeather(cities: (string | null)[], limit = 3): Promise<Map<string, LiveWeather | null>> {
   299→   299→  const unique = [...new Set(cities.filter((c): c is string => Boolean(c)))];
   300→   300→  const out = new Map<string, LiveWeather | null>();
   301→   301→  const queue = [...unique];
   302→   302→  const workers = Array.from({ length: Math.min(limit, queue.length) }, async () => {
   303→   303→    for (;;) {
   304→   304→      const c = queue.shift();
   305→   305→      if (c === undefined) return;
   306→   306→      out.set(c, await weatherFor(c));
   307→   307→    }
   308→   308→  });
   309→   309→  await Promise.all(workers);
   310→   310→  return out;
   311→   311→}
