    99→/**
   100→ * v18.4 — claim اتمی کلید Setting برای dedupe بین‌پردازه‌ای (Atomic Dedupe Claim).
   101→ * الگوی قدیمی «check → send → write» در دو اجرای هم‌زمان (زمان‌بند + مسیر دستی یا دو
   102→ * پردازه) هر دو استمپ کهنه می‌خواند و همان پست را دوبار می‌فرستد. حالا claim قبل از
   103→ * ارسال انجام می‌شود؛ فقط یک فراخوان برنده است (SQLite تک‌نویسنده).
   104→ * خروجی false = دیگری قبلاً مالک شده — بدون ارسال.
   105→ */
   106→export async function claimSetting(key: string, value: string): Promise<boolean> {
   107→  const existing = await db.setting.findUnique({ where: { key } }).catch(() => null);
   108→  if (!existing) {
   109→    try {
   110→      await db.setting.create({ data: { key, value } });
   111→      return true;
   112→    } catch {
   113→      return false; // مسیر موازی هم‌زمان ردیف را ساخت
   114→    }
   115→  }
   116→  if (existing.value === value) return false;
   117→  const res = await db.setting.updateMany({ where: { key, value: { not: value } }, data: { value } });
   118→  return res.count === 1;
   119→}

   275→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
   276→  const token = await botToken();
   277→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   278→  // v13.2 — سپر توکن ناقص: خطای 404 مبهم تلگرام را با پیام دقیق جایگزین می‌کند
   279→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   280→  try {
   281→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
   282→      method: "POST",
   283→      headers: { "Content-Type": "application/json" },
   284→      body: JSON.stringify(body),
   285→      cache: "no-store",
   286→    });
   287→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   288→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
   289→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   290→  } catch (err) {
   291→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   292→  }
   293→}

   295→async function tgSendPhotoTo(
   296→  chatId: string | null,
   297→  photo: Buffer,
   298→  caption: string,
   299→  filename = "slip-card.png",
   300→  captionless = false
   301→): Promise<{ ok: boolean; detail: string }> {
   302→  const token = await botToken();
   303→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   304→  if (!chatId) return { ok: false, detail: "NO_CHANNEL" };
   305→  // v13.2 — سپر توکن ناقص (بخش hash بدون bot_id) — ریشه باگ «انتشار انجام نمیشه»
   306→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   307→  // ═══ v20.0 — سپر «هیچ پیامی بدون تصویر کامل ارسال نشود» ═══
   308→  // امضای PNG + حداقل حجم ۲۰KB — بافر ناقص/خراب هرگز به تلگرام نمی‌رود.
   309→  if (!isValidPng(photo)) return { ok: false, detail: "PHOTO_INCOMPLETE" };
   310→  try {
   311→    // v20.0 — کپشن عکس همیشه «کاملاً HTML-معتبر و ≤۱۰۰۰ کاراکتر در مرز خط» —
   312→    // ریشه ریشه‌ای «پیام بدون تصویر»: caption.slice(0,1024) تگ HTML را نصف می‌کرد
   313→    // → تلگرام sendPhoto را با parse-error رد می‌کرد → فالبک متنی بی‌تصویر می‌رفت.
   314→    const cap = captionless ? "" : safePhotoCaption(caption);
   315→    let result = await tgSendPhotoAttempt(token, chatId, photo, cap, filename, captionless);
   316→    // v20.0 — یک ری‌تای ارسال (خطای شبکه/آپلود گذرا) — همان بافر معتبر، بدون متن جایگزین
   317→    if (!result.ok && !/parse|entities|caption/i.test(result.detail)) {
   318→      await new Promise((r) => setTimeout(r, 1500));
   319→      result = await tgSendPhotoAttempt(token, chatId, photo, cap, filename, captionless);
   320→    }
   321→    return result;
   322→  } catch (err) {
   323→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   324→  }
   325→}

   327→/** v20.0 — اعتبارسنجی سخت PNG: امضای ۸ بایتی + حداقل ۲۰KB */
   328→function isValidPng(buf: Buffer | null | undefined): boolean {
   329→  if (!buf || buf.length < 20480) return false;
   330→  const SIG = [0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a];
   331→  for (let i = 0; i < 8; i++) if (buf[i] !== SIG[i]) return false;
   332→  return true;
   333→}
   334→
   335→/**
   336→ * v20.0 — کپشن ایمن عکس: برش فقط در مرز خط + بستن تگ‌های باز + سقف ۱۰۰۰ کاراکتر.
   337→ * خروجی همیشه HTML-معتبر است (تلگرام هرگز parse-error نمی‌گیرد → پیام بی‌تصویر صفر).
   338→ */
   339→export function safePhotoCaption(text: string, limit = 1000): string {
   340→  let s = String(text ?? "");
   341→  if (s.length <= limit) return s;
   342→  // برش در مرز خط — هیچ تگی نصف نمی‌شود
   343→  s = s.slice(0, limit);
   344→  const lastNl = s.lastIndexOf("\n");
   345→  if (lastNl > 200) s = s.slice(0, lastNl);
   346→  // شمارش تگ‌های باز تلگرام و بستن به ترتیب معکوس (HTML تلگرام فقط ۵ تگ پشتیبانی می‌کند)
   347→  const TAGS = ["b", "i", "u", "s", "code", "pre", "a", "blockquote"];
   348→  const stack: string[] = [];
   349→  const re = /<\/?([a-z]+)[^>]*>/gi;
   350→  let m: RegExpExecArray | null;
   351→  while ((m = re.exec(s)) !== null) {
   352→    const tag = m[1].toLowerCase();
   353→    if (!TAGS.includes(tag)) continue;
   354→    if (m[0][1] === "/") {
   355→      const idx = stack.lastIndexOf(tag);
   356→      if (idx >= 0) stack.length = idx;
   357→    } else if (!m[0].endsWith("/>")) {
   358→      stack.push(tag);
   359→    }
   360→  }
   361→  for (const tag of stack.reverse()) s += `</${tag}>`;
   362→  return s;
   363→}

   431→/** قفل رشته‌ای funnel — اجرای هم‌زمان دو پست فانل (دو مسیر ورود) را سریال می‌کند */
   432→function withFunnelLock<F extends (...args: never[]) => Promise<unknown>>(fn: F): F {
   433→  return (async (...args: Parameters<F>) => {
   434→    const prev = gBroadcast.__fqFunnelLock ?? Promise.resolve();
   435→    let release!: () => void;
   436→    const gate = new Promise<void>((r) => (release = r));
   437→    gBroadcast.__fqFunnelLock = prev.then(() => gate);
   438→    await prev.catch(() => undefined);
   439→    try {
   440→      return await fn(...(args as Parameters<F>));
   441→    } finally {
   442→      release();
   443→      if (gBroadcast.__fqFunnelLock === gate) gBroadcast.__fqFunnelLock = null;
   444→    }
   445→  }) as F;
   446→}

   657→    await logOutbox(slip.id, mode, false, "VETOED_BY_SUPERVISOR");
   658→    return { ok: false, detail: "VETOED_BY_SUPERVISOR", preview };
   659→  }
   660→  // v13.2 — کارت بازخوانی (بازی هنگام صدور از قبل FT بود) هرگز به کانال نمی‌رود؛
   661→  // انتشار «کارت پیش‌بازی» پس از پایان بازی فاقد اعتبار است.
   662→  if (slip.retroIssued) {
   663→    await logOutbox(slip.id, mode, false, "RETRO_CARD_NOT_PUBLISHABLE");

   666→  const status = await telegramStatus();
   667→  if (!status.configured || !status.channelSet || !status.tokenFormat) {
   668→    await logOutbox(slip.id, mode, false, status.configured ? "TELEGRAM_NOT_CONFIGURED" : "INVALID_TOKEN_FORMAT");
   669→    return { ok: false, detail: "TELEGRAM_NOT_CONFIGURED", preview };

   673→  // اگر CARD/GIFT/TEASER قبلاً برای این بازی رفته باشد، هیچ مسیر دومی دوباره نمی‌فرستد.
   674→  const mKey = matchKeyOf(slip);
   675→  if (mode === "CARD" && !opts?.force && (await matchCardAlreadyPosted(mKey))) {
   676→    await logOutbox(slip.id, mode, true, "MATCH_ALREADY_POSTED");
   677→    await db.issuedSlip.update({ where: { id: slip.id }, data: { telegramCard: true } }).catch(() => {});

   696→      await logOutbox(slip.id, mode, true, "MATCH_ALREADY_POSTED");
   697→      await db.issuedSlip.update({ where: { id: slip.id }, data: { telegramCard: true } }).catch(() => {});
   698→      return { ok: false, detail: "MATCH_ALREADY_POSTED", preview };

   704→    // فالبک متنی حذف شد: رندر ناموفق → هیچ پستی نمی‌رود؛ ادعا آزاد و تیک بعدی

   746→      await releaseSlipFlag(slip.id, "telegramCard");
   747→      await releaseBroadcast(dedupeKeyOf.card(slip.id));

   855→async function channelWinRateStats() {
   856→  // v11.0 — فقط کارت‌های واقعی تاییدشده؛ تسویه کاغذی دفتر سایه در وین‌ریت کانال دخالت نمی‌کند
   857→  // v13.2 — فقط کارت‌های واقعی تاییدشده؛ کارت‌های بازخوانی (retroIssued) و دفتر سایه در وین‌ریت کانال دخالت نمی‌کنند
   858→  const settled = await db.issuedSlip.findMany({
   859→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, supervisorApproved: true, retroIssued: false },
   860→    select: { status: true, netProfit: true, stakeA: true, stakeB: true },
   861→  });
   862→  const st = { settled: settled.length, wonA: 0, wonB: 0, doubleWin: 0, middleTrap: 0, netProfit: 0, staked: 0 };
   863→  for (const s of settled) {
   864→    if (s.status === "WON_A") st.wonA += 1;
   865→    else if (s.status === "WON_B") st.wonB += 1;
   866→    else if (s.status === "DOUBLE_WIN") st.doubleWin += 1;
   867→    else if (s.status === "MIDDLE_TRAP") st.middleTrap += 1;
   868→    st.netProfit += s.netProfit ?? 0;
   869→    st.staked += s.stakeA + s.stakeB;
   870→  }
   871→  const winRatePct = st.settled > 0 ? Math.round(((st.settled - st.middleTrap) / st.settled) * 1000) / 10 : 0;
   872→  const netPct = st.staked > 0 ? Math.round((st.netProfit / st.staked) * 1000) / 10 : 0;
   873→  // ─── v11.0 — دفتر سایه: شفافیت counterfactual وتوها (خط جداگانه گزارش) ───
   874→  // v13.5.1 — سایه فقط وتوهای زنده است؛ کارت‌های بازخوانی (retroIssued) داخل سایه شمرده نمی‌شوند
   875→  const shadowRows = await db.issuedSlip.findMany({
   876→    where: { supervisorApproved: false, retroIssued: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   877→    select: { status: true, netProfit: true },
   878→  });
   879→  const shadowWins = shadowRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   880→  const shadowNet = Math.round(shadowRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100;
   881→  const shadowWinRate = shadowRows.length > 0 ? Math.round((shadowWins / shadowRows.length) * 1000) / 10 : null;
   882→  // ─── v13.2 — دفتر بازخوانی: کارت‌های چندروزه‌ای که هرگز پیش‌بازی صادر نشده‌اند (تسویه کاغذی تحلیلی) ───
   883→  // v13.5.1 — بازخوانی با پرچم retroIssued شناسایی می‌شود (مستقل از تایید — کارت بازخوانی هرگز تاییدشده نیست)
   884→  const retroRows = await db.issuedSlip.findMany({
   885→    where: { retroIssued: true, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   886→    select: { status: true, netProfit: true },
   887→  });
   888→  const retroWins = retroRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   889→  // ─── v18.0 — 🥈 حصاد آلفای سایه: کارت‌های تاج نقره‌ای سطح ۲ (استیک ۵۰u Half-Kelly) ───
   890→  const silverRows = await db.issuedSlip.findMany({
   891→    where: { crownTier: "SILVER_SHADOW", retroIssued: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   892→    select: { status: true, netProfit: true },
   893→  });
   894→  const silverWins = silverRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   895→  return {
   896→    ...st,
   897→    winRatePct,
   898→    netPct,
   899→    netProfit: Math.round(st.netProfit * 100) / 100,
   900→    shadow: { settled: shadowRows.length, wins: shadowWins, winRatePct: shadowWinRate, netProfit: shadowNet },
   901→    silver: {
   902→      settled: silverRows.length,
   903→      wins: silverWins,
   904→      winRatePct: silverRows.length > 0 ? Math.round((silverWins / silverRows.length) * 1000) / 10 : null,
   905→      netProfit: Math.round(silverRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100,
   906→    },
   907→    retro: {
   908→      settled: retroRows.length,
   909→      wins: retroWins,
   910→      winRatePct: retroRows.length > 0 ? Math.round((retroWins / retroRows.length) * 1000) / 10 : null,
   911→      netProfit: Math.round(retroRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100,
   912→    },
   913→  };
   914→}

  1473→  return { ok, detail: ok ? (isFortress ? "fortress-gift-sent" : "gift-sent") : "send-failed" };
