    54→    54→function bestEspnCoreLine(core: EspnCoreOdds | null): { provider: string; home: number; draw: number; away: number } | null {
    55→    55→  if (!core) return null;
    56→    56→  let best: { provider: string; home: number; draw: number; away: number } | null = null;
    57→    57→  const impliedSum = (l: { home: number; draw: number; away: number }) => 1 / l.home + 1 / l.draw + 1 / l.away;
    58→    58→  for (const b of core.bookmakers) {
    59→    59→    if (!b.home || !b.draw || !b.away) continue;
    60→    60→    if (b.home <= 1.01 || b.draw <= 1.01 || b.away <= 1.01) continue;
    61→    61→    const line = { provider: b.name, home: b.home, draw: b.draw, away: b.away };
    62→    62→    if (!best || impliedSum(line) < impliedSum(best)) best = line; // کم‌حاشیه‌ترین خط
    63→    63→  }
    64→    64→  return best;
    65→    65→}

   116→   116→export function marketLambdasFor(odds: { oddsHome: number; oddsDraw: number; oddsAway: number; oddsOver25: number; oddsUnder25: number }): { home: number | null; away: number | null; fitError: number | null } {

   121→   121→      key,

   127→   127→        oddsUnder25: odds.oddsUnder25,

   129→   129→    );
   130→   130→  }
   131→   131→  const solved = marketSolverCache.get(key) ?? null;
   132→   132→  return solved

   134→   134→    : { home: null, away: null, fitError: null };

   140→   140→function formFactor(form: string): number {
   141→   141→  if (!form || form === "N/A") return 1.0;
   142→   142→  const pts = [...form].reduce((a, ch) => a + (ch === "W" ? 3 : ch === "D" ? 1 : 0), 0);
   143→   143→  return 0.78 + (pts / (form.length * 3)) * 0.48;
   144→   144→}

   196→   196→/**
   197→   197→ * v6.0 — محاسبه امید گل (λ) از جدول رسمی رده‌بندی فصل جاری ESPN (Standings API بدون کلید)
   198→   198→ * مسیر اولویت‌دار (قابل راستی‌آزمایی ۱۰۰٪):
   199→   199→ *   α_i = GF_i / (GP_i × LeagueTeamAvg)   ← قدرت تهاجمی واقعی از گل زده فصل
   200→   200→ *   β_i = GA_i / (GP_i × LeagueTeamAvg)   ← ضعف دفاعی واقعی از گل خورده فصل
   201→   201→ *   λ_Home = α_Home × β_Away × LeagueTeamAvg × 1.12 (فرمول استاندارد دیکسون-کولز)
   202→   202→ *   λ_Away = α_Away × β_Home × LeagueTeamAvg
   203→   203→ * مسیر جایگزین شفاف (فقط وقتی جدول در دسترس نیست): رکورد واقعی W-D-L + فرم ۵ بازی.
   204→   204→ * تعدیل جوی: بارش > ۲mm یا باد > ۳۰km/h → λ × ۰.۹۲ (−۸٪)
   205→   205→ * v9.0 — export شد: مشترک با خط لوله ثبت خودکار بازی‌های گذشته (auto-issue)
   206→   206→ */
   207→   207→export function deriveStrength(ev: LiveMatchEvent, weather: LiveWeather | null, st: LeagueStandings | null): DerivedStrength {
   208→   208→  const adverse = weather?.adverse ?? false;
   209→   209→  const wAdj = adverse ? 0.92 : 1.0;
   210→   210→
   211→   211→  const homeEntry = findStandingsEntry(st, ev.homeTeamId, ev.homeTeam);
   212→   212→  const awayEntry = findStandingsEntry(st, ev.awayTeamId, ev.awayTeam);
   213→   213→  const lta = st && st.leagueTeamAvg > 0.5 ? st.leagueTeamAvg : null;
   214→   214→
   215→   215→  if (lta && homeEntry && awayEntry && homeEntry.gp >= 3 && awayEntry.gp >= 3) {
   216→   216→    // ═══ v6.0 — مسیر واقعی: α/β از جدول رسمی فصل ═══
   217→   217→    const alphaH = homeEntry.gf / (homeEntry.gp * lta);
   218→   218→    const betaH = homeEntry.ga / (homeEntry.gp * lta);
   219→   219→    const alphaA = awayEntry.gf / (awayEntry.gp * lta);
   220→   220→    const betaA = awayEntry.ga / (awayEntry.gp * lta);
   221→   221→
   222→   222→    let lambdaHome = alphaH * betaA * lta * 1.12 * wAdj;
   223→   223→    let lambdaAway = alphaA * betaH * lta * wAdj;
   224→   224→    lambdaHome = clamp(lambdaHome, 0.3, 3.8);
   225→   225→    lambdaAway = clamp(lambdaAway, 0.25, 3.6);
   226→   226→
   227→   227→    const homeElo = eloFromRecord(ev.homeRecord);
   228→   228→    const awayElo = eloFromRecord(ev.awayRecord);
   229→   229→
   230→   230→    const homePower = clamp(alphaH, 0.5, 2.0);
   231→   231→    const awayPower = clamp(alphaA, 0.5, 2.0);
   232→   232→
   233→   233→    return {
   234→   234→      lambdaHome: r2(lambdaHome),
   235→   235→      lambdaAway: r2(lambdaAway),
   236→   236→      // قدرت‌های واقعی دیکسون-کولز — λ = α × β × leagueTeamAvg × homeAdv
   237→   237→      homeAttack: r2(clamp(alphaH, 0.35, 2.3)),
   238→   238→      awayAttack: r2(clamp(alphaA, 0.3, 2.2)),
   239→   239→      homeDefense: r2(clamp(betaH, 0.45, 1.9)),
   240→   240→      awayDefense: r2(clamp(betaA, 0.45, 1.9)),
   241→   241→      homeElo,
   242→   242→      awayElo,
   243→   243→      homePpda: r2(clamp(12 - (homePower - 1) * 7, 8.5, 15)),
   244→   244→      awayPpda: r2(clamp(12.4 - (awayPower - 1) * 7, 8.5, 15.5)),
   245→   245→      homeFieldTilt: Math.round(clamp(50 + (homeElo - 1500) / 12, 42, 63)),
   246→   246→      awayFieldTilt: Math.round(clamp(50 + (awayElo - 1500) / 12, 40, 62)),
   247→   247→      weatherAdjusted: adverse,
   248→   248→      statsSource: "ESPN_STANDINGS",
   249→   249→      homeRank: homeEntry.rank || null,
   250→   250→      homeGF: homeEntry.gf,
   251→   251→      homeGA: homeEntry.ga,
   252→   252→      homeGP: homeEntry.gp,
   253→   253→      awayRank: awayEntry.rank || null,
   254→   254→      awayGF: awayEntry.gf,
   255→   255→      awayGA: awayEntry.ga,
   256→   256→      awayGP: awayEntry.gp,
   257→   257→      leagueTeamAvg: r2(lta),
   258→   258→    };
   259→   259→  }
   260→   260→
   261→   261→  // ═══ v21.0 — مسیر Elo جهانی: پایان λ کور ۱.۰۰–۰.۹۲ برای بازی‌های ملی ═══
   262→   262→  // وقتی جدول رده‌بندی ESPN در دسترس نیست، امید گل از «اختلاف واقعی Elo دو تیم»
   263→   263→  // (جدول GLOBAL_TEAM_ELO_AND_XG_RATINGS — ۱۲۰ تیم ملی + باشگاه‌های برتر) مشتق می‌شود.
   264→   264→  // ژاپن × پاراگوئه دیگر با قرقیزستان × ویتنام عدد یکسان نمی‌گیرد!
   265→   265→  const eloSolved = lambdaFromEloRatings(ev.homeTeam, ev.awayTeam, null, null);
   266→   266→  if (eloSolved) {
   267→   267→    // اولویت Elo جدول جهانی؛ تیم ناشناخته → Elo رکورد واقعی ESPN (۱۵۰۰ = متوسط)

   288→   288→      awayDefense: 1.0,
   289→   289→      homeElo: eloHome,
   290→   290→      awayElo: eloAway,
   291→   291→      homePpda,
   292→   292→      awayPpda,
   293→   293→      homeFieldTilt,
   294→   294→      awayFieldTilt,
   295→   295→      weatherAdjusted: adverse,
   296→   296→      statsSource: "GLOBAL_ELO_FALLBACK",
   297→   297→      homeRank: null,
   298→   298→      homeGF: null,
   299→   299→      homeGA: null,
   300→   300→      homeGP: null,
   301→   301→      awayRank: null,
   302→   302→      awayGF: null,
   303→   303→      awayGA: null,
   304→   304→      awayGP: null,
   305→   305→      leagueTeamAvg: null,
   306→   306→    };
   307→   307→  }
   308→   308→
   309→   309→  // ═══ مسیر شفاف جایگزین — فقط رکورد/فرم واقعی ESPN (جدول رده‌بندی و جدول Elo هم در دسترس نیستند) ═══
   310→   310→  // ⚠️ v21.0: اگر λ نهایی این مسیر در بازه کور ۰.۹۰–۱.۰۵ بماند و خط بازار هم غایب باشد،
   311→   311→  // گیت «No Blind Card» سوپروایزر این بازی را وتو می‌کند (کارت کور صادر نمی‌شود).
   312→   312→  const homeStr = 0.55 * formFactor(ev.homeForm) + 0.45 * seasonFactor(ev.homeRecord);
   313→   313→  const awayStr = 0.55 * formFactor(ev.awayForm) + 0.45 * seasonFactor(ev.awayRecord);
   314→   314→
   315→   315→  let lambdaHome = ev.avgGoals * 0.55 * homeStr * wAdj;
   316→   316→  let lambdaAway = ev.avgGoals * 0.45 * awayStr * wAdj;
   317→   317→
   318→   318→  // همگرایی به محدوده معقول پواسون
   319→   319→  lambdaHome = clamp(lambdaHome, 0.3, 3.8);
   320→   320→  lambdaAway = clamp(lambdaAway, 0.25, 3.6);
   321→   321→
   322→   322→  // استخراج ضرایب حمله/دفاع سازگار با موتور دیکسون-کولز:
   323→   323→  // λh = homeAttack × awayDefense × leagueAvg × 1.12 → دفاع‌ها = ۱، حمله = λ / ثابت
   324→   324→  const homeAttack = clamp(lambdaHome / (1.35 * 1.12), 0.35, 2.3);
   325→   325→  const awayAttack = clamp(lambdaAway / 1.35, 0.3, 2.2);

   389→   389→function matchOddsEvent(ev: LiveMatchEvent, oddsEvents: OddsApiEvent[]): OddsApiEvent | null {
   390→   390→  const direct = `${normTeam(ev.homeTeam)}::${normTeam(ev.awayTeam)}`;

   404→   404→  return best && bestDist < 24 * 3600 * 1000 ? best : null;

   434→   434→      (a, b) => 1 / a.line.home + 1 / a.line.draw + 1 / a.line.away - (1 / b.line.home + 1 / b.line.draw + 1 / b.line.away)

   564→   564→async function poolWeather(cities: (string | null)[], limit = 3): Promise<Map<string, LiveWeather | null>> {

   573→   573→    }
   574→   574→  });
   575→   575→  await Promise.all(workers);
   576→   576→  return out;
