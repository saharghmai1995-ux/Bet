   598→   598→  // خط صفر وسط اگر منفی و مثبت داریم

   600→   600→  const hasPos = data.some((d) => d.netProfit > 0);
   601→   601→  const midZero = hasNeg && hasPos;
   602→   602→  const zeroLine = midZero ? (H - padB + padT) / 2 : zeroY;
   603→   603→  const upScale = midZero ? (zeroLine - padT) / maxAbs : (H - padB - padT) / maxAbs;
   604→   604→  const downScale = midZero ? (H - padB - zeroLine) / maxAbs : upScale;

   610→   610→        const v = d.netProfit;

   612→   612→        const x = i * barW + barW * 0.18;

   614→   614→        const y = v >= 0 ? zeroLine - h : zeroLine;
   615→   615→        const tone = v > 0 ? "#34d399" : v < 0 ? "#fb7185" : "rgba(255,255,255,0.14)";
   616→   616→        const label = d.label.length > 9 ? d.label.slice(0, 9) : d.label;
   617→   617→        return (

   619→   619→            <title>{`${d.label}: ${v >= 0 ? "+" : "−"}${Math.abs(v).toFixed(1)}u · ${d.settled} بازی (${d.wins} برد / ${d.losses} باخت) · بانک پایان: ${d.bankAtEnd.toFixed(0)}u`}</title>

   621→   621→            {Math.abs(v) > 0 && (
   622→   622→              <text
