   145→   145→ * H = −Σ mat[i][j]·ln(mat[i][j]) روی زیرماتریس ۰..۵ × ۰..۵ (نرمال‌شده درون همان بازه).

   147→   147→export function shannonEntropy(matrix: number[][]): number {
   148→   148→  const N = Math.min(6, matrix.length);
   149→   149→  let sum = 0;
   150→   150→  for (let h = 0; h < N; h++) {
   151→   151→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) sum += matrix[h][a];
   152→   152→  }
   153→   153→  if (sum <= 0) return Math.log(36);
   154→   154→  let H = 0;
   155→   155→  for (let h = 0; h < N; h++) {
   156→   156→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   157→   157→      const p = matrix[h][a] / sum;
   158→   158→      if (p > 1e-12) H -= p * Math.log(p);
   159→   159→    }
   160→   160→  }
   161→   161→  return H;
   162→   162→}

   164→   164→/** نسبت عدم تقارن قدرت: Rλ = max(λH, λA) / min(λH, λA) */
   165→   165→export function asymmetryRatio(lambdaHome: number, lambdaAway: number): number {
   166→   166→  const hi = Math.max(lambdaHome, lambdaAway);
   167→   167→  const lo = Math.min(lambdaHome, lambdaAway);
   168→   168→  if (lo <= 0.01) return 9.99;
   169→   169→  return Math.min(9.99, hi / lo);
   170→   170→}

   173→   173→export function topCells(matrix: number[][], n = 4): { score: string; p: number }[] {
   174→   174→  const N = Math.min(6, matrix.length);
   175→   175→  const cells: { score: string; p: number }[] = [];
   176→   176→  for (let h = 0; h < N; h++) {
   177→   177→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   178→   178→      cells.push({ score: `${h}-${a}`, p: matrix[h][a] });
   179→   179→    }
   180→   180→  }
   181→   181→  cells.sort((x, y) => y.p - x.p);
   182→   182→  return cells.slice(0, n);
   183→   183→}

   206→   206→    }
   207→   207→  }
   208→   208→  return inRegion > 0 ? hit / inRegion : 0;

   233→   233→export function calculateNonLinearSupervisorScore(candidate: NonLinearCandidateInput): {
   234→   234→  ev: number;
   235→   235→  sharpe: number;
   236→   236→  score: number;
   237→   237→} {
   238→   238→  const netReturnOnWin = candidate.effectiveOdds - 1;
   239→   239→  const lossProb = Math.max(0, 1 - candidate.winProb);
   240→   240→  const expectedValue = candidate.winProb * netReturnOnWin + candidate.doubleWinProb * 0.45 - lossProb;
   241→   241→  const variance =
   242→   242→    candidate.winProb * Math.pow(netReturnOnWin - expectedValue, 2) +
   243→   243→    lossProb * Math.pow(-1 - expectedValue, 2);
   244→   244→  const sharpeRatio = expectedValue / Math.sqrt(Math.max(0.01, variance));
   245→   245→
   246→   246→  // جریمه غیرخطی ریسک و پاداش آنتروپی پایین
   247→   247→  const nonLinearSafetyBonus = Math.pow(Math.max(0, candidate.winProb), 2.2) * 100;
   248→   248→  const entropyBonus = Math.max(0, (2.8 - candidate.entropy) * 12);
   249→   249→
   250→   250→  const score = Number(
   251→   251→    ((sharpeRatio * 45 + nonLinearSafetyBonus * 0.55 + entropyBonus) * candidate.leagueReliability).toFixed(1)
   252→   252→  );
   253→   253→  return { ev: expectedValue, sharpe: sharpeRatio, score };
   254→   254→}

   618→   618→      coverProb: r3(p.selectionB.modelProb),
   619→   619→      coverLegs: [],
   620→   620→      winProb: r3(winProb),
   621→   621→      effectiveOdds,
   622→   622→      doubleWinProb: r3(p.jointStates.bothWin),
   623→   623→      jointLossProb: r3(p.jointStates.bothLose),
   624→   624→      matrixCoveragePct: Math.round(winProb * 1000) / 10,
   625→   625→      eligible: winProb >= 0.9 && p.jointStates.bothLose <= 0.1,
   626→   626→      ...sc,
   627→   627→      stakeMain: p.equalProfit.stakeA,
   628→   628→      stakeCover: p.equalProfit.stakeB,
   629→   629→      guaranteedProfitUnits: p.equalProfit.profit,
   630→   630→      pairId: p.id,
   631→   631→    });
   632→   632→  }
   633→   633→  return out.sort((x, y) => y.score - x.score);
   634→   634→}
   635→   635→
   636→   636→// ═══════════════════════════════ مسیریاب اصلی ═══════════════════════════════
   637→   637→
   638→   638→/**
   639→   639→ * مسیریاب غیرخطی ۵ رژیمی — برای هر مسابقه هر ۴ روش تک‌بازی را با تابع مطلوبیت
   640→   640→ * غیرخطی شارپ/کلِی امتیاز می‌دهد و بهترین Bet-Type را برمی‌گرداند.
   641→   641→ *
   642→   642→ * قانون تایید هوشمند سوپروایزر (v17.0):
   643→   643→ *   · بازی‌های لیگ سمّی (esp.2 / usa.1 / fifa.friendly) → وتو مطلق
   644→   644→ *   · اگر بهترین کاندید EV > +0.08 و Sharpe > 0.32 دارد → تایید با آن Bet-Type
   645→   645→ *   · اگر هیچ روشی شارپ مثبت معنادار ندارد → وتو (بازی واقعاً بی‌کیفیت و پرنوسان)
   646→   646→ */
   647→   647→export function routeBetTypes(input: {
   648→   648→  matrix: number[][];
   649→   649→  lambdaHome: number;
   650→   650→  lambdaAway: number;
   651→   651→  prices: MarketPrices;
   652→   652→  pairs: PortfolioPairDTO[];
   653→   653→  league: string;
   654→   654→  /** ضریب اعتبار لیگ از کالیبراسیون (۰.۸۵..۱.۱۵) */
   655→   655→  leagueReliability?: number;
   656→   656→}): BetTypeRouteResult {
   657→   657→  const structure = matrixStructure(input.matrix, input.lambdaHome, input.lambdaAway);
   658→   658→  const reliability = Math.min(1.15, Math.max(0.85, input.leagueReliability ?? 1));
   659→   659→
   660→   660→  const poison = POISON_LEAGUES.some((p) => input.league.toLowerCase().includes(p.toLowerCase()));
   661→   661→  if (poison) {
   662→   662→    return {
   663→   663→      structure,
   664→   664→      candidates: [],
   665→   665→      winnerVariantId: null,
   666→   666→      winner: null,
   667→   667→      crownPairId: null,
   668→   668→      veto: true,
   669→   669→      vetoReason: `🛡️ گیت v17.0: لیگ سمّی (${input.league}) — نوسان ساختاری بالا و غیرقابل مدل‌سازی؛ همه ۴ روش تک‌بازی وتو می‌شوند — انتقال به دفتر سایه`,
   670→   670→    };
   671→   671→  }
   672→   672→
   673→   673→  // اعمال ضریب اعتبار لیگ روی امتیاز کاندیدها (بازمحاسبه با reliability)
   674→   674→  const candidates: BetTypeCandidate[] = [
   675→   675→    ...buildBankerCandidates(input.matrix, input.prices, structure),
   676→   676→    ...buildComboCandidates(input.matrix, input.prices, structure),
   677→   677→    ...(buildCorrectScoreCandidate(input.matrix, structure) ? [buildCorrectScoreCandidate(input.matrix, structure)!] : []),
   678→   678→    ...buildDualHedgeCandidates(input.pairs),
   679→   679→  ];
   680→   680→  for (const c of candidates) {
   681→   681→    const rescored = calculateNonLinearSupervisorScore({
   682→   682→      winProb: c.winProb,
   683→   683→      effectiveOdds: c.effectiveOdds,
   684→   684→      doubleWinProb: c.doubleWinProb,
   685→   685→      entropy: structure.entropy,
   686→   686→      leagueReliability: reliability,
   687→   687→    });
   688→   688→    c.sharpe = r2(rescored.sharpe);
   689→   689→    c.ev = r3(rescored.ev);
   690→   690→    c.score = rescored.score;
   691→   691→  }
   692→   692→
   693→   693→  // ═══ گیت تایید غیرخطی v17.0 (قانون تایید هوشمند سند) ═══
   694→   694→  //  · کاندیدهای با شارپ واضحاً منفی (EV سوخت) کنار گذاشته می‌شوند
   695→   695→  //  · تایید مبنای سند: معیار اختصاصی هر Bet-Type (eligible)
   696→   696→  //  · تایید ارزش: EV > +۸٪ و Sharpe > ۰.۳۲ (قانون بازبینی مسابقات ریسک ۱۱-۱۳٪ سند)
   697→   697→  const qualifying = candidates.filter(
   698→   698→    (c) => c.sharpe >= -0.05 && (c.eligible || (c.ev > APPROVE_EV_MIN && c.sharpe > APPROVE_SHARPE_MIN))
   699→   699→  );
   700→   700→
   701→   701→  if (candidates.length === 0 || qualifying.length === 0) {
   702→   702→    return {
   703→   703→      structure,
   704→   704→      candidates: [],
   705→   705→      winnerVariantId: null,
   706→   706→      winner: null,
   707→   707→      crownPairId: null,
   708→   708→      veto: true,
   709→   709→      vetoReason: "🛡️ گیت v17.0: هیچ‌یک از ۴ روش تک‌بازی (Banker / Combo / Correct Score / Dual-Hedge) آستانه کیفیت غیرخطی را پاس نکرد — بازی پرنوسان؛ انتقال به دفتر سایه",
   710→   710→    };
   711→   711→  }
   712→   712→
   713→   713→  const best = qualifying.reduce((a, b) => (b.score > a.score ? b : a));
   714→   714→
   715→   715→  // قانون تایید هوشمند: شارپ ضعیف در بهترین روش → بررسی امکان تایید با روش دیگر
   716→   716→  if (best.sharpe < APPROVE_SHARPE_MIN || best.ev < APPROVE_EV_MIN) {
   717→   717→    const eliteEligible = qualifying.filter((c) => c.ev > APPROVE_EV_MIN && c.sharpe > APPROVE_SHARPE_MIN);
   718→   718→    if (eliteEligible.length === 0) {
   719→   719→      // تایید مبنای سند — معیار اختصاصی Bet-Type (پوشش/آنتروپی/همبستگی) برنده است
   720→   720→      return {
   721→   721→        structure,
   722→   722→        candidates,
   723→   723→        winnerVariantId: best.variantId,
   724→   724→        winner: best,
   725→   725→        crownPairId: best.pairId ?? null,
   726→   726→        veto: false,
   727→   727→        vetoReason: null,
   728→   728→      };
   729→   729→    }
   730→   730→    const bestEligible = eliteEligible.reduce((a, b) => (b.score > a.score ? b : a));
   731→   731→    return {
   732→   732→      structure,
   733→   733→      candidates,
   734→   734→      winnerVariantId: bestEligible.variantId,
   735→   735→      winner: bestEligible,
   736→   736→      crownPairId: bestEligible.pairId ?? null,
   737→   737→      veto: false,
   738→   738→      vetoReason: null,
   739→   739→    };
   740→   740→  }
   741→   741→
   742→   742→  return {
   743→   743→    structure,
   744→   744→    candidates,
   745→   745→    winnerVariantId: best.variantId,
   746→   746→    winner: best,
   747→   747→    crownPairId: best.pairId ?? null,
   748→   748→    veto: false,
   749→   749→    vetoReason: null,
   750→   750→  };
   751→   751→}
   752→   752→
   753→   753→// ═══════════════════════════════ 🔗 ACCUMULATOR ═══════════════════════════════
   754→   754→
   755→   755→/** یک پای Accumulator — از بنکرها/کمبوهای تاییدشده بهترین بازی‌های روز */
   756→   756→export interface AccumulatorLeg {
   757→   757→  matchId: string;
   758→   758→  matchLabel: string;
   759→   759→  league: string;
   760→   760→  kickoff: string;
   761→   761→  tip: string;
   762→   762→  tipEn: string;
   763→   763→  odds: number;
   764→   764→  betType: BetTypeKey;
   765→   765→}
   766→   766→
   767→   767→export type AccumulatorKind = "TWO_FOLD_BANKER" | "THREE_FOLD_COMBO" | "TIME_STAGGERED";
   768→   768→
   769→   769→export interface AccumulatorSlip {
   770→   770→  id: string;
   771→   771→  kind: AccumulatorKind;
   772→   772→  labelFa: string;
   773→   773→  labelEn: string;
   774→   774→  icon: string;
   775→   775→  legs: AccumulatorLeg[];
   776→   776→  totalOdds: number;
   777→   777→  /** استیک بهینه بر اساس معیار کلِی همزمان (واحد) */
   778→   778→  stake: number;
   779→   779→  hitRatePct: number;
   780→   780→  potentialReturn: number;
   781→   781→  /** فرمول محاسبه مبلغ پوشش زنده پای آخر (Time-Staggered) */
   782→   782→  hedgeNote?: string;
   783→   783→  hedgeTargetLabel?: string;
   784→   784→  hedgeTargetOdds?: number;
   785→   785→  hedgeStake?: number;
   786→   786→  lockedProfitIfHedged?: number;
   787→   787→}
   788→   788→
   789→   789→/**
   790→   790→ * 🔗 سازنده سه برگه میکس زنجیره‌ای از بهترین بنکرها/کمبوهای تاییدشده روز:
   791→   791→ *  ۱) 2-Fold Banker Accumulator — ترکیب ۲ پایه Banker (ضریب کل ۲.۱۰..۲.۵۵، نرخ اصابت > ۸۰٪)
   792→   792→ *  ۲) 3-Fold Pro Combo Accumulator — ترکیب ۳ پایه قوی Banker/Combo (ضریب کل ۳.۶۰..۵.۴۰)
   793→   793→ *  ۳) Time-Staggered Hedged Accumulator — میکس ۳ بازی با ساعت شروع متفاوت +
   794→   794→ *     فرمول محاسبه مبلغ پوشش زنده پای آخر (Cash-Out / Final-Leg Hedge).
   795→   795→ * استیک بهینه با معیار کلِی همزمان (Simultaneous Kelly fraction × ۱۰۰).
   796→   796→ */
   797→   797→export function buildAccumulators(
   798→   798→  winners: { matchId: string; matchLabel: string; league: string; kickoff: string; candidate: BetTypeCandidate }[]
   799→   799→): AccumulatorSlip[] {
   800→   800→  const bankers = winners.filter((w) => w.candidate.key === "BANKER");
   801→   801→  const comboOrBanker = winners.filter((w) => w.candidate.key === "BANKER" || w.candidate.key === "COMBO");
   802→   802→  const out: AccumulatorSlip[] = [];
   803→   803→
   804→   804→  const legOf = (w: (typeof winners)[number]): AccumulatorLeg => ({
   805→   805→    matchId: w.matchId,
   806→   806→    matchLabel: w.matchLabel,
   807→   807→    league: w.league,
   808→   808→    kickoff: w.kickoff,
   809→   809→    // v17.0 — پای زوج‌تیپ: کل پوزیشن داچینگ‌شده با «ضریب موثر» و «پوشش کل» لحاظ می‌شود
   810→   810→    tip: w.candidate.key === "DUAL_HEDGE" ? `${w.candidate.mainLabel} ↔ ${w.candidate.coverLabel ?? ""}` : w.candidate.mainLabel,
   811→   811→    tipEn: w.candidate.key === "DUAL_HEDGE" ? "Dual-Hedge Effective Position" : w.candidate.mainLabelEn,
   812→   812→    odds: w.candidate.key === "DUAL_HEDGE" ? w.candidate.effectiveOdds : w.candidate.mainOdds,
   813→   813→    betType: w.candidate.key,
   814→   814→  });
   815→   815→
   816→   816→  // ۱) 2-Fold Banker Accumulator
   817→   817→  if (bankers.length >= 2) {
   818→   818→    const legs = [legOf(bankers[0]), legOf(bankers[1])];
   819→   819→    const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
   820→   820→    const hit = bankers[0].candidate.mainProb * bankers[1].candidate.mainProb;
   821→   821→    // کلِی همزمان: f* ≈ (p·b − q)/b × نیم‌کِلی محافظه‌کار برای میکس چندپایه
   822→   822→    const b = totalOdds - 1;
   823→   823→    const f = Math.max(0.015, Math.min(0.06, (hit * b - (1 - hit)) / b * 0.5));
   824→   824→    out.push({
   825→   825→      id: "ACC-2F",
   826→   826→      kind: "TWO_FOLD_BANKER",
   827→   827→      labelFa: "میکس زنجیره‌ای ۲ پایه بنکر امن (2-Fold Banker Accumulator)",
   828→   828→      labelEn: "2-Fold Banker Accumulator",
   829→   829→      icon: "🔗",
   830→   830→      legs,
   831→   831→      totalOdds,
   832→   832→      stake: Math.round(f * 1000) / 10,
   833→   833→      hitRatePct: Math.round(hit * 1000) / 10,
   834→   834→      potentialReturn: Math.round(f * totalOdds * 10) / 10,
   835→   835→    });
   836→   836→  } else if (winners.length >= 2) {
   837→   837→    // v17.0 — fallback: قوی‌ترین پایه‌های تاییدشده روز (هر Bet-Type)
   838→   838→    const legs = [legOf(winners[0]), legOf(winners[1])];
   839→   839→    const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
   840→   840→    const legProb = (w: (typeof winners)[number]) => (w.candidate.key === "DUAL_HEDGE" ? w.candidate.winProb : w.candidate.mainProb);
   841→   841→    const hit = legProb(winners[0]) * legProb(winners[1]);
   842→   842→    const b = totalOdds - 1;
   843→   843→    const f = Math.max(0.01, Math.min(0.045, (hit * b - (1 - hit)) / b * 0.5));
   844→   844→    const kinds = [...new Set(winners.slice(0, 2).map((w) => w.candidate.icon + " " + w.candidate.key))].join(" + ");
   845→   845→    out.push({
   846→   846→      id: "ACC-2F",
   847→   847→      kind: "TWO_FOLD_BANKER",
   848→   848→      labelFa: `میکس زنجیره‌ای ۲ پایه تاییدشده روز (${kinds})`,
   849→   849→      labelEn: "2-Fold Smart Accumulator",
   850→   850→      icon: "🔗",
   851→   851→      legs,
   852→   852→      totalOdds,
   853→   853→      stake: Math.round(f * 1000) / 10,
   854→   854→      hitRatePct: Math.round(hit * 1000) / 10,
   855→   855→      potentialReturn: Math.round(f * totalOdds * 10) / 10,
   856→   856→    });
   857→   857→  }
   858→   858→
   859→   859→  // ۲) 3-Fold Pro Combo Accumulator
   860→   860→  if (comboOrBanker.length >= 3) {
   861→   861→    const legs = comboOrBanker.slice(0, 3).map(legOf);
   862→   862→    const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
   863→   863→    const hit = comboOrBanker.slice(0, 3).reduce((s, w) => s * w.candidate.mainProb, 1);
   864→   864→    const b = totalOdds - 1;
   865→   865→    const f = Math.max(0.008, Math.min(0.035, (hit * b - (1 - hit)) / b * 0.5));
   866→   866→    out.push({
   867→   867→      id: "ACC-3F",
   868→   868→      kind: "THREE_FOLD_COMBO",
   869→   869→      labelFa: "میکس حرفه‌ای ۳ پایه بنکر/کمبو (3-Fold Pro Combo Accumulator)",
   870→   870→      labelEn: "3-Fold Pro Combo Accumulator",
   871→   871→      icon: "🔗",
   872→   872→      legs,
   873→   873→      totalOdds,
   874→   874→      stake: Math.round(f * 1000) / 10,
   875→   875→      hitRatePct: Math.round(hit * 1000) / 10,
   876→   876→      potentialReturn: Math.round(f * totalOdds * 10) / 10,
   877→   877→    });
   878→   878→  } else if (winners.length >= 3) {
   879→   879→    const legs = winners.slice(0, 3).map(legOf);
   880→   880→    const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
   881→   881→    const hit = winners.slice(0, 3).reduce((s, w) => s * (w.candidate.key === "DUAL_HEDGE" ? w.candidate.winProb : w.candidate.mainProb), 1);
   882→   882→    const b = totalOdds - 1;
   883→   883→    const f = Math.max(0.006, Math.min(0.025, (hit * b - (1 - hit)) / b * 0.5));
   884→   884→    out.push({
   885→   885→      id: "ACC-3F",
   886→   886→      kind: "THREE_FOLD_COMBO",
   887→   887→      labelFa: "میکس حرفه‌ای ۳ پایه تاییدشده روز (3-Fold Pro Accumulator)",
   888→   888→      labelEn: "3-Fold Pro Accumulator",
   889→   889→      icon: "🔗",
   890→   890→      legs,
   891→   891→      totalOdds,
   892→   892→      stake: Math.round(f * 1000) / 10,
   893→   893→      hitRatePct: Math.round(hit * 1000) / 10,
   894→   894→      potentialReturn: Math.round(f * totalOdds * 10) / 10,
   895→   895→    });
   896→   896→  }
   897→   897→
   898→   898→  // ۳) Time-Staggered Hedged Accumulator — ۳ بازی با ساعت شروع متفاوت
   899→   899→  const staggerPool = comboOrBanker.length >= 3 ? comboOrBanker : winners;
   900→   900→  if (staggerPool.length >= 3) {
   901→   901→    const sorted = [...staggerPool].sort((a, b) => new Date(a.kickoff).getTime() - new Date(b.kickoff).getTime());
   902→   902→    // سه پای با بیشترین فاصله ساعت شروع
   903→   903→    const picked = [sorted[0], sorted[Math.floor(sorted.length / 2)], sorted[sorted.length - 1]];
   904→   904→    const distinctHours = new Set(picked.map((p) => new Date(p.kickoff).getHours())).size;
   905→   905→    if (distinctHours >= 2) {
   906→   906→      const legs = picked.map(legOf);
   907→   907→      const totalOdds = r2(legs.reduce((s, l) => s * l.odds, 1));
   908→   908→      const stake = 30;
   909→   909→      const payoutIfFirstTwoWin = stake * legs[0].odds * legs[1].odds;
   910→   910→      // فرمول پوشش پای آخر: مبلغ شرط مقابل = (پرداخت فعلی − سود قفل‌شده هدف) ÷ ضریب مقابل
   911→   911→      const hedgeTargetOdds = Math.max(1.4, Math.round(1 / (1 - 1 / legs[2].odds) * 100) / 100);
   912→   912→      const lockedProfit = Math.round(payoutIfFirstTwoWin * 0.35 * 10) / 10;
   913→   913→      const hedgeStake = Math.round(((payoutIfFirstTwoWin - lockedProfit) / hedgeTargetOdds) * 10) / 10;
   914→   914→      const hit = picked.reduce((s, w) => s * (w.candidate.key === "DUAL_HEDGE" ? w.candidate.winProb : w.candidate.mainProb), 1);
   915→   915→      out.push({
   916→   916→        id: "ACC-TS",
   917→   917→        kind: "TIME_STAGGERED",
   918→   918→        labelFa: "میکس پله‌زمانی با پوشش زنده پای آخر (Time-Staggered Hedged Accumulator)",
   919→   919→        labelEn: "Time-Staggered Hedged Accumulator",
   920→   920→        icon: "🔗",
   921→   921→        legs,
   922→   922→        totalOdds,
   923→   923→        stake,
   924→   924→        hitRatePct: Math.round(hit * 1000) / 10,
   925→   925→        potentialReturn: Math.round(stake * totalOdds * 10) / 10,
   926→   926→        hedgeNote: `اگر ۲ بازی اول بردند، پیش از شروع بازی سوم: شرط مقابل «${legs[2].tipEn}» به مبلغ ${hedgeStake} واحد با ضریب ~${hedgeTargetOdds.toFixed(2)}x ببندید → سود قفل‌شده ~${lockedProfit} واحد در هر دو سناریو`,
   927→   927→        hedgeTargetLabel: "مقابل تیپ پای آخر (مثلاً X2 یا زیر گل)",
   928→   928→        hedgeTargetOdds,
   929→   929→        hedgeStake,
   930→   930→        lockedProfitIfHedged: lockedProfit,
   931→   931→      });
   932→   932→    }
   933→   933→  }
   934→   934→
   935→   935→  return out;
   936→   936→}
   937→   937→
