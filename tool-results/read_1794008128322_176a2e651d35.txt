    77→/** ضریب فرم از ۵ بازی اخیر واقعی (WWDLW) — بازه ۰.۷۸ تا ۱.۲۶ */
    78→function formFactor(form: string): number {
    79→  if (!form || form === "N/A") return 1.0;
    80→  const pts = [...form].reduce((a, ch) => a + (ch === "W" ? 3 : ch === "D" ? 1 : 0), 0);
    81→  return 0.78 + (pts / (form.length * 3)) * 0.48;
    82→}
    83→
    84→/** ضریب فصل از رکورد واقعی (مثل 3-4-2) — بازه ۰.۸۲ تا ۱.۱۸ */
    85→function seasonFactor(record: string | null): number {
    86→  const pct = recordToPointsPct(record);
    87→  if (pct === null) return 1.0;
    88→  return 0.82 + pct * 0.36;
    89→}

    91→/** الو تقریبی از درصد امتیاز فصل — ۱۵۰۰ = میانگین */
    92→function eloFromRecord(record: string | null): number {
    93→  const pct = recordToPointsPct(record);
    94→  if (pct === null) return 1500;
    95→  return Math.round(Math.max(1400, Math.min(2100, 1500 + (pct - 0.5) * 800)));
    96→}

    98→function clamp(v: number, lo: number, hi: number): number {

   102→function r2(v: number): number {
   103→  return Math.round(v * 100) / 100;

   106→interface DerivedStrength {
   107→  lambdaHome: number;
   108→  lambdaAway: number;
   109→  homeAttack: number;
   110→  awayAttack: number;
   111→  homeElo: number;
   112→  awayElo: number;
   113→  homePpda: number;
   114→  awayPpda: number;
   115→  homeFieldTilt: number;
   116→  awayFieldTilt: number;
   117→  weatherAdjusted: boolean;
   118→}
   119→
   120→/**
   121→ * محاسبه امید گل (λ) از فرم + رکورد فصل واقعی دو تیم و اعمال تعدیل جوی
   122→ * λ_base = avgGoals × سهم میزبانی/میهمانی × قدرت نسبی (فرم ۵۵٪ + فصل ۴۵٪)
   123→ * تعدیل جوی: بارش > ۲mm یا باد > ۳۰km/h → λ × ۰.۹۲ (−۸٪)
   124→ */
   125→function deriveStrength(ev: LiveMatchEvent, weather: LiveWeather | null): DerivedStrength {
   126→  const homeStr = 0.55 * formFactor(ev.homeForm) + 0.45 * seasonFactor(ev.homeRecord);
   127→  const awayStr = 0.55 * formFactor(ev.awayForm) + 0.45 * seasonFactor(ev.awayRecord);
   128→
   129→  const adverse = weather?.adverse ?? false;
   130→  const wAdj = adverse ? 0.92 : 1.0;
   131→
   132→  let lambdaHome = ev.avgGoals * 0.55 * homeStr * wAdj;
   133→  let lambdaAway = ev.avgGoals * 0.45 * awayStr * wAdj;
   134→
   135→  // همگرایی به محدوده معقول پواسون
   136→  lambdaHome = clamp(lambdaHome, 0.3, 3.8);
   137→  lambdaAway = clamp(lambdaAway, 0.25, 3.6);
   138→
   139→  // استخراج ضرایب حمله/دفاع سازگار با موتور دیکسون-کولز:
   140→  // λh = homeAttack × awayDefense × 1.35 × 1.12 → دفاع‌ها = ۱، حمله = λ / ثابت
   141→  const homeAttack = clamp(lambdaHome / (1.35 * 1.12), 0.35, 2.3);
   142→  const awayAttack = clamp(lambdaAway / 1.35, 0.3, 2.2);
   143→
   144→  const homeElo = eloFromRecord(ev.homeRecord);
   145→  const awayElo = eloFromRecord(ev.awayRecord);
   146→
   147→  const homeFieldTilt = Math.round(clamp(50 + (homeElo - 1500) / 12, 42, 63));
   148→  const awayFieldTilt = Math.round(clamp(50 + (awayElo - 1500) / 12, 40, 62));
   149→  const homePpda = r2(clamp(12 - (homeStr - 1) * 7, 8.5, 15));
   150→  const awayPpda = r2(clamp(12.4 - (awayStr - 1) * 7, 8.5, 15.5));
   151→
   152→  return { lambdaHome: r2(lambdaHome), lambdaAway: r2(lambdaAway), homeAttack: r2(homeAttack), awayAttack: r2(awayAttack), homeElo, awayElo, homePpda, awayPpda, homeFieldTilt, awayFieldTilt, weatherAdjusted: adverse };
   153→}

   155→function weatherString(w: LiveWeather | null): string {
   156→  if (!w) return "نامشخص - دیتای جوی در دسترس نیست";
   157→  const cond = w.precipMm > 2 ? "باران" : w.precipMm > 0.3 ? "نیمه‌بارانی" : w.windKmh > 30 ? "باد شدید" : "پاک";
   158→  return `${cond} - ${Math.round(w.tempC)}° · باد ${Math.round(w.windKmh)}km/h${w.precipMm > 0.3 ? ` · بارش ${w.precipMm.toFixed(1)}mm` : ""}`;
   159→}

   161→/** احتمالات مدل دیکسون-کولز از روی استحاله مشتق‌شده (برای ضرایب fallback و T-60) */
   162→function modelProbs(d: DerivedStrength) {
   163→  return runDixonColes({
   164→    homeAttack: d.homeAttack,
   165→    awayAttack: d.awayAttack,
   166→    homeDefense: 1.0,
   167→    awayDefense: 1.0,
   168→    leagueAvg: 1.35,
   169→    homeAdvantage: 1.12,
   170→    rho: -0.08,
   171→    eloDiff: d.homeElo - d.awayElo,
   172→    eloWeight: 0.35,
   173→    homeAbsenceImpact: 0,
   174→    awayAbsenceImpact: 0,
   175→    timeDecayXi: 0.0065,
   176→    daysAgo: 21,
   177→  });
   178→}

   182→function matchOddsEvent(ev: LiveMatchEvent, oddsEvents: OddsApiEvent[]): OddsApiEvent | null {
   183→  const direct = `${normTeam(ev.homeTeam)}::${normTeam(ev.awayTeam)}`;
   184→  const kick = Date.parse(ev.kickoff);
   185→  let best: OddsApiEvent | null = null;
   186→  let bestDist = Infinity;
   187→  for (const oe of oddsEvents) {
   188→    const same = oe.matchKey === direct || `${normTeam(oe.homeTeam)}::${normTeam(oe.awayTeam)}` === `${normTeam(ev.awayTeam)}::${normTeam(ev.homeTeam)}`;
   189→    if (!same) continue;
   190→    const dist = Math.abs(Date.parse(oe.commenceTime) - kick);
   191→    if (dist < bestDist) {
   192→      bestDist = dist;
   193→      best = oe;
   194→    }
   195→  }
   196→  // تطبیق فقط برای بازی‌های نزدیک (±۲۴h) — جلوگیری از جفت‌شدن بازی‌های دور
   197→  return best && bestDist < 24 * 3600 * 1000 ? best : null;
   198→}

   200→interface ResolvedOdds {
   201→  oddsHome: number;
   202→  oddsDraw: number;
   203→  oddsAway: number;
   204→  oddsOver25: number;
   205→  oddsUnder25: number;
   206→  odds1x: number;
   207→  source: "ENGINE" | "ESPN_DRAFTKINGS" | "ODDS_API";
   208→  /** تا ۳ ردیف بوک‌میکر با کلیدهای کنونیکال (Pinnacle = کم‌حاشیه‌ترین) */
   209→  books: { bookmaker: "PINNACLE" | "SOFT_A" | "SOFT_B"; bookmakerLabel: string; home: number; draw: number; away: number; over25: number | null; under25: number | null }[];
   210→}
   211→
   212→function resolveOdds(ev: LiveMatchEvent, oddsEv: OddsApiEvent | null, probs: ReturnType<typeof modelProbs>): ResolvedOdds {
   213→  // ─── ۱) ضرایب زنده The Odds API — بهترین قیمت هر انتخاب + سه بوک نماینده ───
   214→  if (oddsEv && oddsEv.books.length > 0) {
   215→    const booksSorted = [...oddsEv.books].sort(
   216→      (a, b) => 1 / a.line.home + 1 / a.line.draw + 1 / a.line.away - (1 / b.line.home + 1 / b.line.draw + 1 / b.line.away)
   217→    );
   218→    const canon = ["PINNACLE", "SOFT_A", "SOFT_B"] as const;
   219→    const books = booksSorted.slice(0, 3).map((b, i) => ({
   220→      bookmaker: canon[i],
   221→      bookmakerLabel: `${b.label}${i === 0 ? " (Sharp)" : ""}`,
   222→      home: b.line.home,
   223→      draw: b.line.draw,
   224→      away: b.line.away,
   225→      over25: b.line.over25,
   226→      under25: b.line.under25,
   227→    }));
   228→    const best = (sel: (b: ResolvedOdds["books"][number]) => number | null): number => {
   229→      const vals = books.map(sel).filter((v): v is number => v !== null && v >= 1.01);
   230→      return vals.length > 0 ? Math.max(...vals) : 0;
   231→    };
   232→    const oddsHome = best((b) => b.home);
   233→    const oddsDraw = best((b) => b.draw);
   234→    const oddsAway = best((b) => b.away);
   235→    let oddsOver25 = best((b) => b.over25);
   236→    let oddsUnder25 = best((b) => b.under25);
   237→    if (oddsOver25 <= 1.01 || oddsUnder25 <= 1.01) {
   238→      oddsOver25 = fairOdds(probs.pOver25, 1.06);
   239→      oddsUnder25 = fairOdds(probs.pUnder25, 1.06);
   240→    }
   241→    return { oddsHome, oddsDraw, oddsAway, oddsOver25, oddsUnder25, odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06), source: "ODDS_API", books };
   242→  }
   243→
   244→  // ─── ۲) خط پایه ESPN (معمولاً DraftKings) ───
   245→  if (ev.espnOdds && ev.espnOdds.home && ev.espnOdds.draw && ev.espnOdds.away) {
   246→    const o = ev.espnOdds;
   247→    const h = o.home as number;
   248→    const d2 = o.draw as number;
   249→    const a2 = o.away as number;
   250→    return {
   251→      oddsHome: h,
   252→      oddsDraw: d2,
   253→      oddsAway: a2,
   254→      oddsOver25: o.over25 ?? fairOdds(probs.pOver25, 1.06),
   255→      oddsUnder25: o.under25 ?? fairOdds(probs.pUnder25, 1.06),
   256→      odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06),
   257→      source: "ESPN_DRAFTKINGS",
   258→      books: [
   259→        {
   260→          bookmaker: "SOFT_A",
   261→          bookmakerLabel: `${o.provider} (ESPN)`,
   262→          home: h,
   263→          draw: d2,
   264→          away: a2,
   265→          over25: o.over25 ?? fairOdds(probs.pOver25, 1.06),
   266→          under25: o.under25 ?? fairOdds(probs.pUnder25, 1.06),
   267→        },
   268→      ],
   269→    };
   270→  }
   271→
   272→  // ─── ۳) Fallback مدل — ضرایب منصفانه با حاشیه ۶٪ ───
   273→  return {
   274→    oddsHome: fairOdds(probs.pHome, 1.06),
   275→    oddsDraw: fairOdds(probs.pDraw, 1.06),
   276→    oddsAway: fairOdds(probs.pAway, 1.06),
   277→    oddsOver25: fairOdds(probs.pOver25, 1.06),
   278→    oddsUnder25: fairOdds(probs.pUnder25, 1.06),
   279→    odds1x: fairOdds(probs.pHome + probs.pDraw, 1.06),
   280→    source: "ENGINE",
   281→    books: [],
   282→  };
   283→}

   287→async function weatherFor(city: string | null): Promise<LiveWeather | null> {
   288→  if (!city) return null;
   289→  const coords = cityCoords(city);
   290→  if (!coords) return null;
   291→  const hit = cache.weather.get(city);
   292→  if (hit && Date.now() - hit.at < WEATHER_TTL_MS) return hit.w;
   293→  const w = await fetchWeather(coords[0], coords[1], city);
   294→  cache.weather.set(city, { w, at: Date.now() });
   295→  return w;
   296→}

   298→async function poolWeather(cities: (string | null)[], limit = 3): Promise<Map<string, LiveWeather | null>> {
   299→  const unique = [...new Set(cities.filter((c): c is string => Boolean(c)))];
   300→  const out = new Map<string, LiveWeather | null>();
   301→  const queue = [...unique];
   302→  const workers = Array.from({ length: Math.min(limit, queue.length) }, async () => {
   303→    for (;;) {
   304→      const c = queue.shift();
   305→      if (c === undefined) return;
   306→      out.set(c, await weatherFor(c));
   307→    }
   308→  });
   309→  await Promise.all(workers);
   310→  return out;
   311→}

   322→/** ثبت لاگ بریر از نتیجه واقعی بازی — فقط یک‌بار برای هر بازی */
   323→async function logCalibrationFor(matchId: string, label: string, league: string, hg: number, ag: number): Promise<void> {
   324→  try {
   325→    const dup = await db.calibrationLog.findFirst({ where: { matchLabel: label } });
   326→    if (dup) return;
   327→    const analysis = await db.agentAnalysis.findFirst({
   328→      where: { matchId },
   329→      orderBy: { createdAt: "desc" },
   330→    });
   331→    if (!analysis) return;
   332→
   333→    const cands: { key: string; p: number }[] = [
   334→      { key: "HOME", p: analysis.pHome },
   335→      { key: "DRAW", p: analysis.pDraw },
   336→      { key: "AWAY", p: analysis.pAway },
   337→      { key: "O25", p: analysis.pOver25 },
   338→    ];
   339→    cands.sort((a, b) => b.p - a.p);
   340→    const pick = cands[0];
   341→    const actual = pick.key === "HOME" ? hg > ag : pick.key === "DRAW" ? hg === ag : pick.key === "AWAY" ? ag > hg : hg + ag > 2.5;
   342→    await db.calibrationLog.create({
   343→      data: {
   344→        matchLabel: label,
   345→        league,
   346→        pick: PICK_LABELS[pick.key],
   347→        predictedProb: pick.p,
   348→        outcome: actual ? "WIN" : "LOSS",
   349→        brier: (pick.p - (actual ? 1 : 0)) ** 2,
   350→        lesson: null,
   351→      },
   352→    });
   353→  } catch (err) {
   354→    console.warn("calibration log failed:", err);
   355→  }
   356→}

   366→async function runSync(opts: SyncOptions): Promise<LiveFeedDTO> {
   367→  const t0 = Date.now();
   368→  const sources: LiveFeedSourceDTO[] = [];
   369→
   370→  // ─── ۱) منابع زنده به صورت موازی ───
   371→  const tEspn = Date.now();
   372→  const espnPromise = fetchEspnEvents(0, 7);
   373→  const newsPromise = fetchEspnNews().then((n) => {
   374→    cache.news = n;
   375→    cache.newsAt = Date.now();
   376→    return n;
   377→  });
   378→  const sdbPromise = Promise.allSettled(ESPN_LEAGUES.map((l) => fetchSportsDbNext(l.sportsDbId)));
   379→
   380→  // ─── ۲) Odds API فقط با اجازه (هزینه اعتبار پلن رایگان) ───
   381→  let oddsEvents: OddsApiEvent[] = [];
   382→  let oddsUsed = false;
   383→  let oddsThrottled = false;
   384→  if (opts.useOddsApi) {
   385→    const since = Date.now() - cache.oddsSyncAt;
   386→    if (cache.oddsSyncAt > 0 && since < ODDS_THROTTLE_MS) {
   387→      oddsThrottled = true;
   388→    } else {
   389→      const keys = ESPN_LEAGUES.map((l) => l.oddsKey);
   390→      const res = await fetchOddsApiEvents(keys, oddsApiKey(opts.oddsApiKey));
   391→      oddsEvents = res.events;
   392→      oddsUsed = true;
   393→      cache.oddsSyncAt = Date.now();
   394→      if (res.requestsRemaining !== null) cache.oddsCreditsRemaining = res.requestsRemaining;
   395→    }
   396→  }
   397→
   398→  const [espnEvents, , sdbResults] = await Promise.all([espnPromise, newsPromise, sdbPromise]);
   399→
   400→  sources.push({
   401→    key: "espn",
   402→    label: "ESPN Scoreboard (۶ لیگ × ۸ روز)",
   403→    ok: espnEvents.length > 0,
   404→    detail: `${espnEvents.length} مسابقه واقعی استخراج شد`,
   405→    latencyMs: Date.now() - tEspn,
   406→  });
   407→
   408→  // TheSportsDB — مکمل ورزشگاه برای بازی‌های پیش‌رو
   409→  const sdbVenue = new Map<string, string>();
   410→  let sdbCount = 0;
   411→  for (const r of sdbResults) {
   412→    if (r.status !== "fulfilled") continue;
   413→    for (const e of r.value) {
   414→      sdbCount += 1;
   415→      if (e.strVenue) {
   416→        sdbVenue.set(`${normTeam(e.homeTeam)}::${normTeam(e.awayTeam)}`, e.strVenue);
   417→      }
   418→    }
   419→  }
   420→  sources.push({
   421→    key: "sportsdb",
   422→    label: "TheSportsDB (کلید عمومی ۱۲۳)",
   423→    ok: sdbCount > 0,
   424→    detail: `${sdbCount} فیکسچر پیش‌رو · ورزشگاه: ${sdbVenue.size}`,
   425→    latencyMs: 0,
   426→  });
   427→
   428→  // ─── ۳) نگاشت و ذخیره در دیتابیس ───
   429→  const now = Date.now();
   430→  const upcomingByLeague = new Map<string, number>();
   431→  const keepEvents: LiveMatchEvent[] = [];
   432→  for (const ev of espnEvents) {
   433→    const kick = Date.parse(ev.kickoff);
   434→    if (ev.statusState === "post") continue; // نتایج در فاز کالیبراسیون جدا پردازش می‌شوند
   435→    if (ev.statusState === "pre") {
   436→      if (kick < now - 3600_000) continue; // شروع‌شده بدون وضعیت زنده
   437→      const cnt = upcomingByLeague.get(ev.leagueSlug) ?? 0;
   438→      if (cnt >= MAX_UPCOMING_PER_LEAGUE) continue;
   439→      upcomingByLeague.set(ev.leagueSlug, cnt + 1);
   440→    }
   441→    keepEvents.push(ev);
   442→  }
   443→
   444→  // آب‌وهوا فقط برای ۴۸ ساعت آینده (کش ۳۰ دقیقه‌ای)
   445→  const weatherTargets = keepEvents.filter((e) => Date.parse(e.kickoff) - now < 48 * 3600_000);
   446→  const weatherMap = await poolWeather(weatherTargets.map((e) => e.venueCity));
   447→
   448→  let liveUpcoming = 0;
   449→  let liveInPlay = 0;
   450→  let weatherAdjustedCount = 0;
   451→  let oddsApiMatched = 0;
   452→
   453→  for (const ev of keepEvents) {
   454→    const kick = Date.parse(ev.kickoff);
   455→    const weather = weatherMap.get(ev.venueCity ?? "") ?? null;
   456→    const d = deriveStrength(ev, weather);
   457→    if (d.weatherAdjusted) weatherAdjustedCount += 1;
   458→    const probs = modelProbs(d);
   459→    const oddsEv = matchOddsEvent(ev, oddsEvents);
   460→    if (oddsEv) oddsApiMatched += 1;
   461→    const odds = resolveOdds(ev, oddsEv, probs);
   462→
   463→    // v4.0: حفظ ضرایب زنده معتبر قبلی — سینک بدون Odds API نباید ضرایب بوک‌میکرها
   464→    // را با ضرایب مدل جایگزین کند (فقط ضرایب کهنه‌تر از ۶ ساعت بازمحاسبه می‌شوند)
   465→    const existing = await db.match.findUnique({
   466→      where: { externalId: ev.externalId },
   467→      select: { id: true, status: true, oddsSource: true, oddsHome: true, oddsDraw: true, oddsAway: true, oddsOver25: true, oddsUnder25: true, odds1x: true, oddsFetchedAt: true },
   468→    });
   469→    const keepLiveOdds =
   470→      odds.source === "ENGINE" &&
   471→      existing &&
   472→      existing.oddsSource !== "ENGINE" &&
   473→      existing.oddsFetchedAt !== null &&
   474→      Date.now() - existing.oddsFetchedAt.getTime() < 6 * 3600_000;
   475→    if (keepLiveOdds && existing) {
   476→      odds.oddsHome = existing.oddsHome;
   477→      odds.oddsDraw = existing.oddsDraw;
   478→      odds.oddsAway = existing.oddsAway;
   479→      odds.oddsOver25 = existing.oddsOver25;
   480→      odds.oddsUnder25 = existing.oddsUnder25;
   481→      odds.odds1x = existing.odds1x;
   482→      odds.source = existing.oddsSource as ResolvedOdds["source"];
   483→    }
   484→
   485→    const isLive = ev.statusState === "in";
   486→    if (isLive) liveInPlay += 1;
   487→    else liveUpcoming += 1;
   488→
   489→    const base = {
   490→      league: ev.leagueName,
   491→      homeTeam: ev.homeTeam,
   492→      awayTeam: ev.awayTeam,
   493→      kickoff: new Date(kick),
   494→      homeAttack: d.homeAttack,
   495→      homeDefense: 1.0,
   496→      awayAttack: d.awayAttack,
   497→      awayDefense: 1.0,
   498→      homeElo: d.homeElo,
   499→      awayElo: d.awayElo,
   500→      homeXg: d.lambdaHome,
   501→      awayXg: d.lambdaAway,
   502→      homePpda: d.homePpda,
   503→      awayPpda: d.awayPpda,
   504→      homeFieldTilt: d.homeFieldTilt,
   505→      awayFieldTilt: d.awayFieldTilt,
   506→      homeForm: ev.homeForm,
   507→      awayForm: ev.awayForm,
   508→      weather: weatherString(weather),
   509→      homeAbsence: null,
   510→      awayAbsence: null,
   511→      homeMotivation: null,
   512→      awayMotivation: null,
   513→      tacticalNote: null,
   514→      oddsHome: odds.oddsHome,
   515→      oddsDraw: odds.oddsDraw,
   516→      oddsAway: odds.oddsAway,
   517→      oddsOver25: odds.oddsOver25,
   518→      oddsUnder25: odds.oddsUnder25,
   519→      odds1x: odds.odds1x,
   520→      homeNpxgLevel: d.lambdaHome,
   521→      homeNpxgLeading: r2(d.lambdaHome * 0.72),
   522→      homeNpxgTrailing: r2(d.lambdaHome * 1.25),
   523→      awayNpxgLevel: d.lambdaAway,
   524→      awayNpxgLeading: r2(d.lambdaAway * 0.72),
   525→      awayNpxgTrailing: r2(d.lambdaAway * 1.25),
   526→      homeCornersFor: r2(4.9 + (d.lambdaHome - 1.35) * 0.9),
   527→      homeCornersAgainst: 4.5,
   528→      awayCornersFor: r2(4.6 + (d.lambdaAway - 1.35) * 0.9),
   529→      awayCornersAgainst: 4.7,
   530→      awayTravelKm: 350,
   531→      homeRestDays: 6,
   532→      awayRestDays: 6,
   533→      // v4.0
   534→      dataSource: "ESPN",
   535→      homeLogo: ev.homeLogo,
   536→      awayLogo: ev.awayLogo,
   537→      venueName: ev.venueName ?? sdbVenue.get(`${normTeam(ev.homeTeam)}::${normTeam(ev.awayTeam)}`) ?? null,
   538→      venueCity: ev.venueCity,
   539→      weatherTemp: weather?.tempC ?? null,
   540→      weatherWind: weather?.windKmh ?? null,
   541→      weatherPrecip: weather?.precipMm ?? null,
   542→      weatherAdjusted: d.weatherAdjusted,
   543→      oddsSource: odds.source,
   544→      oddsFetchedAt: odds.source === "ENGINE" ? null : existing?.oddsFetchedAt && keepLiveOdds ? existing.oddsFetchedAt : new Date(),
   545→      homeGoals: isLive ? ev.homeScore : null,
   546→      awayGoals: isLive ? ev.awayScore : null,
   547→      // وضعیت ANALYZED قبلی حفظ می‌شود (فقط گذار LIVE/UPCOMING از منبع واقعی)
   548→      status: isLive ? "LIVE" : existing?.status === "ANALYZED" ? "ANALYZED" : "UPCOMING",
   549→    };
   550→
   551→    const row = await db.match.upsert({
   552→      where: { externalId: ev.externalId },
   553→      create: { ...base, externalId: ev.externalId },
   554→      update: base,
   555→    });
   556→
   557→    // بازنویسی ردیف‌های بوک‌میکر (فقط وقتی دیتای واقعی داریم؛ با حفظی‌ها دست نمی‌زنیم)
   558→    if (odds.books.length > 0) {
   559→      await db.oddsRecord.deleteMany({ where: { matchId: row.id } });
   560→      for (const b of odds.books) {
   561→        await db.oddsRecord.create({
   562→          data: {
   563→            matchId: row.id,
   564→            bookmaker: b.bookmaker,
   565→            bookmakerLabel: b.bookmakerLabel,
   566→            oddsHome: b.home,
   567→            oddsDraw: b.draw,
   568→            oddsAway: b.away,
   569→            oddsOver25: b.over25 ?? odds.oddsOver25,
   570→            oddsUnder25: b.under25 ?? odds.oddsUnder25,
   571→            oddsCornersOver95: fairOdds(0.45, 1.08),
   572→            oddsCornersUnder95: fairOdds(0.55, 1.08),
   573→          },
   574→        });
   575→      }
   576→    } else if (!keepLiveOdds && (await db.oddsRecord.count({ where: { matchId: row.id } })) === 0) {
   577→      // بدون دیتای زنده → ردیف نماینده از خط نهایی
   578→      await db.oddsRecord.create({
   579→        data: {
   580→          matchId: row.id,
   581→          bookmaker: "SOFT_A",
   582→          bookmakerLabel: odds.source === "ESPN_DRAFTKINGS" ? "خط ESPN" : "مدل داخلی",
   583→          oddsHome: odds.oddsHome,
   584→          oddsDraw: odds.oddsDraw,
   585→          oddsAway: odds.oddsAway,
   586→          oddsOver25: odds.oddsOver25,
   587→          oddsUnder25: odds.oddsUnder25,
   588→          oddsCornersOver95: fairOdds(0.45, 1.08),
   589→          oddsCornersUnder95: fairOdds(0.55, 1.08),
   590→        },
   591→      });
   592→    }
   593→  }
   594→
   595→  sources.push({
   596→    key: "odds",
   597→    label: "The Odds API v4",
   598→    ok: oddsUsed ? oddsApiMatched > 0 : cache.oddsSyncAt > 0,
   599→    detail: oddsUsed
   600→      ? `${oddsApiMatched} بازی با ضرایب زنده بوک‌میکرها جایگزین شد${cache.oddsCreditsRemaining !== null ? ` · اعتبار باقی‌مانده: ${cache.oddsCreditsRemaining}` : ""}`
   601→      : oddsThrottled
   602→        ? "بدلیل محدودیت پلن رایگان در بازه ۱۲ دقیقه — از ضرایب قبلی استفاده شد"
   603→        : "کلید وارد نشده — خط پایه ESPN/مدل استفاده می‌شود",
   604→    latencyMs: 0,
   605→  });
   606→
   607→  const newsList = cache.news;
   608→  sources.push({
   609→    key: "news",
   610→    label: "ESPN Live News",
   611→    ok: newsList.length > 0,
   612→    detail: `${newsList.length} تیتر زنده`,
   613→    latencyMs: 0,
   614→  });
   615→  sources.push({
   616→    key: "weather",
   617→    label: "Open-Meteo",
   618→    ok: weatherAdjustedCount > 0 || weatherMap.size > 0,
   619→    detail: `${weatherMap.size} ورزشگاه · تعدیل جوی روی ${weatherAdjustedCount} بازی`,
   620→    latencyMs: 0,
   621→  });
   622→
   623→  // ─── ۴) کالیبراسیون با نتایج واقعی دیروز و امروز ───
   624→  let calibratedCount = 0;
   625→  try {
   626→    const yesterday = await fetchEspnEvents(-2, 0);
   627→    for (const ev of yesterday) {
   628→      if (ev.statusState !== "post" || ev.homeScore === null || ev.awayScore === null) continue;
   629→      const existing = await db.match.findUnique({ where: { externalId: ev.externalId } });
   630→      if (!existing) continue;
   631→      await db.match.update({
   632→        where: { id: existing.id },
   633→        data: { homeGoals: ev.homeScore, awayGoals: ev.awayScore, status: "FINISHED" },
   634→      });
   635→      const label = `${ev.homeTeam} ${ev.homeScore}−${ev.awayScore} ${ev.awayTeam}`;
   636→      const before = await db.calibrationLog.count({ where: { matchLabel: label } });
   637→      await logCalibrationFor(existing.id, label, existing.league, ev.homeScore, ev.awayScore);
   638→      const after = await db.calibrationLog.count({ where: { matchLabel: label } });
   639→      if (after > before) calibratedCount += 1;
   640→    }
   641→  } catch (err) {
   642→    console.warn("real-result calibration failed:", err);
   643→  }
   644→
   645→  // ─── ۵) آرشیو: بازی‌های گذشته زنده + دیتای mock وقتی فید زنده پوشش می‌دهد ───
   646→  const cutoff = new Date(now - 3 * 3600_000);
   647→  const stale = await db.match.updateMany({
   648→    where: { dataSource: "ESPN", kickoff: { lt: cutoff }, status: { in: ["UPCOMING", "ANALYZED", "LIVE"] } },
   649→    data: { status: "ARCHIVED" },
   650→  });
   651→  // FINISHED های قدیمی هم آرشیو شوند (نتیجه و لاگ بریر ثبت شده است)
   652→  const finCutoff = new Date(now - 26 * 3600_000);
   653→  await db.match.updateMany({
   654→    where: { status: "FINISHED", kickoff: { lt: finCutoff } },
   655→    data: { status: "ARCHIVED" },
   656→  });
   657→
   658→  const liveTotal = liveUpcoming + liveInPlay;
   659→  if (liveTotal >= 4) {
   660→    await db.match.updateMany({ where: { dataSource: "SEED" }, data: { status: "ARCHIVED" } });
   661→  }
   662→
   663→  const payload: LiveFeedDTO = {
   664→    success: espnEvents.length > 0 || liveTotal > 0,
   665→    mode: liveTotal > 0 ? "LIVE" : "FALLBACK",
   666→    lastSyncAt: Date.now(),
   667→    matchesLive: liveTotal,
   668→    matchesUpcoming: liveUpcoming,
   669→    matchesInPlay: liveInPlay,
   670→    news: newsList.slice(0, 14),
   671→    sources,
   672→    odds: {
   673→      used: oddsUsed,
   674→      creditsRemaining: cache.oddsCreditsRemaining,
   675→      lastOddsSyncAt: cache.oddsSyncAt || null,
   676→      throttled: oddsThrottled,
   677→    },
   678→    weatherAdjusted: weatherAdjustedCount,
   679→    calibrated: calibratedCount,
   680→    archived: stale.count,
   681→    elapsedMs: Date.now() - t0,
   682→  };
   683→
   684→  cache.payload = payload;
   685→  cache.fetchedAt = Date.now();
   686→  return payload;
   687→}

   692→export async function getLiveFeed(): Promise<LiveFeedDTO> {
   693→  if (cache.payload && Date.now() - cache.fetchedAt < FEED_TTL_MS) {
   694→    return cache.payload;
   695→  }
   696→  if (cache.inFlight) return cache.inFlight;
   697→  cache.inFlight = runSync({}).finally(() => {
   698→    cache.inFlight = null;
   699→  });
   700→  return cache.inFlight;

   709→ * اخبار زنده مرتبط با دو تیم یک بازی — برای الصاق به استدلال ایجنت گاو/خرس.
   710→ * اگر کش خبری خالی/کهنه باشد، یک‌بار واکشی سبک انجام می‌شود.
   711→ */
   712→export async function getNewsForTeams(homeTeam: string, awayTeam: string): Promise<LiveNewsItem[]> {
   713→  if (!cache.news || Date.now() - cache.newsAt > NEWS_TTL_MS) {
   714→    try {
   715→      const n = await fetchEspnNews();
   716→      cache.news = n;
   717→      cache.newsAt = Date.now();
   718→    } catch {
   719→      // بی‌خیال — اخبار اختیاری است
   720→    }
   721→  }
   722→  const h = normTeam(homeTeam);
   723→  const a = normTeam(awayTeam);
   724→  const tokensH = h.split(" ").filter((t) => t.length > 2);
   725→  const tokensA = a.split(" ").filter((t) => t.length > 2);
   726→  return cache.news
   727→    .filter((n) => {
   728→      const text = `${n.headline} ${n.description}`.toLowerCase();
   729→      const hitH = tokensH.some((t) => text.includes(t));
   730→      const hitA = tokensA.some((t) => text.includes(t));
   731→      return hitH || hitA;
   732→    })
   733→    .slice(0, 4);
   734→}
