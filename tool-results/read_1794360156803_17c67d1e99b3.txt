   489→/** نقطه‌های فرم تیم (W/D/L) — ۵ بازی آخر */
   490→function formDots(form: string | null, cx: number, cy: number): string {
   491→  const seq = String(form ?? "")
   492→    .toUpperCase()
   493→    .replace(/[^WDL]/g, "")
   494→    .slice(-5)
   495→    .split("");
   496→  if (seq.length === 0) return "";
   497→  const gap = 21;
   498→  const x0 = cx - ((seq.length - 1) * gap) / 2;
   499→  return seq
   500→    .map((ch, i) => {
   501→      const color = ch === "W" ? C.emerald : ch === "L" ? C.rose : "#93a7c4";
   502→      return `<circle cx="${(x0 + i * gap).toFixed(1)}" cy="${cy}" r="7.5" fill="${color}" fill-opacity="0.92"/>`;
   503→    })
   504→    .join("");
   505→}

   508→function rankChip(cx: number, cy: number, rank: number | null): string {
   509→  if (!rank || rank <= 0) return "";

   511→  <rect x="${cx - 44}" y="${cy - 16}" width="88" height="30" rx="15" fill="rgba(6,10,20,0.7)" stroke="rgba(255,255,255,0.22)" stroke-width="1.2"/>

   513→}

   515→/** نوار احتمال ۱X۲ مدل — سه‌بخشی تجاری با برچسب درصد */
   516→function probBar1x2(pHome: number, pDraw: number, pAway: number, x: number, y: number, w: number, h: number): string {
   517→  const total = Math.max(0.0001, pHome + pDraw + pAway);
   518→  const hW = (pHome / total) * w;
   519→  const dW = (pDraw / total) * w;
   520→  const aW = (pAway / total) * w;
   521→  const seg = (sx: number, sw: number, color: string, fillOp: string) =>
   522→    `<rect x="${sx.toFixed(1)}" y="${y}" width="${sw.toFixed(1)}" height="${h}" fill="${color}" fill-opacity="${fillOp}"/>`;
   523→  return `
   524→  <g>
   525→    ${seg(x, hW, C.cyan, "0.55")}
   526→    ${seg(x + hW, dW, "#93a7c4", "0.4")}
   527→    ${seg(x + hW + dW, aW, C.emerald, "0.55")}
   528→    <rect x="${x}" y="${y}" width="${w}" height="${h}" rx="${h / 2}" fill="none" stroke="rgba(255,255,255,0.25)" stroke-width="1.6"/>
   529→    ${
   530→      hW > 62
   531→        ? `<text x="${x + hW / 2}" y="${y + h / 2 + 7}" text-anchor="middle" font-family="${C.mono}" font-size="19" font-weight="bold" fill="${C.ink}">${Math.round(pHome * 100)}%</text><text x="${x + hW / 2}" y="${y + h + 24}" text-anchor="middle" font-family="${C.sans}" font-size="13.5" fill="${C.cyan}" letter-spacing="1.5">HOME</text>`
   532→        : `<text x="${x + hW / 2}" y="${y - 10}" text-anchor="middle" font-family="${C.mono}" font-size="15" font-weight="bold" fill="${C.cyan}">${Math.round(pHome * 100)}%</text>`
   533→    }
   534→    ${
   535→      dW > 62
   536→        ? `<text x="${x + hW + dW / 2}" y="${y + h / 2 + 7}" text-anchor="middle" font-family="${C.mono}" font-size="19" font-weight="bold" fill="${C.ink}">${Math.round(pDraw * 100)}%</text><text x="${x + hW + dW / 2}" y="${y + h + 24}" text-anchor="middle" font-family="${C.sans}" font-size="13.5" fill="${C.inkDim}" letter-spacing="1.5">DRAW</text>`
   537→        : ""
   538→    }
   539→    ${
   540→      aW > 62
   541→        ? `<text x="${x + hW + dW + aW / 2}" y="${y + h / 2 + 7}" text-anchor="middle" font-family="${C.mono}" font-size="19" font-weight="bold" fill="${C.ink}">${Math.round(pAway * 100)}%</text><text x="${x + hW + dW + aW / 2}" y="${y + h + 24}" text-anchor="middle" font-family="${C.sans}" font-size="13.5" fill="${C.emerald}" letter-spacing="1.5">AWAY</text>`
   542→        : `<text x="${x + w - 4}" y="${y - 10}" text-anchor="end" font-family="${C.mono}" font-size="15" font-weight="bold" fill="${C.emerald}">${Math.round(pAway * 100)}%</text>`
   543→    }
   544→  </g>`;
   545→}
