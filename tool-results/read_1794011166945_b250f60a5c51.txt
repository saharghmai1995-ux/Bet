    63→function pitchNodes(topPlayers: LineupShockDTO["topPlayers"], side: "HOME" | "AWAY"): PitchNode[] {
    64→  const HOME_FORMATION: PitchNode[] = [
    65→    { x: 7, y: 50 }, { x: 22, y: 18 }, { x: 20, y: 39 }, { x: 20, y: 61 }, { x: 22, y: 82 },
    66→    { x: 40, y: 30 }, { x: 37, y: 50 }, { x: 40, y: 70 },
    67→    { x: 62, y: 22 }, { x: 66, y: 50 }, { x: 62, y: 78 },
    68→  ];
    69→  const AWAY_FORMATION: PitchNode[] = [
    70→    { x: 93, y: 50 }, { x: 78, y: 18 }, { x: 80, y: 39 }, { x: 80, y: 61 }, { x: 78, y: 82 },
    71→    { x: 60, y: 30 }, { x: 63, y: 50 }, { x: 60, y: 70 },
    72→    { x: 38, y: 22 }, { x: 34, y: 50 }, { x: 38, y: 78 },
    73→  ];
    74→  const players = topPlayers.filter((p) => p.side === side);
    75→  const formation = side === "HOME" ? HOME_FORMATION : AWAY_FORMATION;
    76→  return formation.map((slot, i) => {
    77→    const p = players[i];
    78→    if (!p) return slot;
    79→    const status: PitchNode["status"] =
    80→      p.status === "OUT_OF_XI" || p.status === "INJURED" || p.status === "SUSPENDED"
    81→        ? "MISSING"
    82→        : p.status === "NO_STATS"
    83→          ? "DOUBTFUL"
    84→          : "FIT";
    85→    return { ...slot, status, label: `${p.name} (${posLabel(p.position)}) — ${statusBadge(p.status).label}` };
    86→  });
    87→}

   195→  const setShock = useQuantStore((s) => s.setShock);
   196→
   197→  const [matchId, setMatchId] = useState("");
   198→  const [running, setRunning] = useState(false);
   199→  const [shock, setShockLocal] = useState<LineupShockDTO | null>(null);
   200→  const [absent, setAbsent] = useState<{ name: string; side: "HOME" | "AWAY" } | null>(null);
   201→
   202→  // بازی فعال از هدر → پیش‌فرض این کارت
   203→  useEffect(() => {
   204→    if (activeMatchId) {
   205→      setMatchId((cur) => {
   206→        if (cur !== activeMatchId) {
   207→          setShockLocal(null);
   208→          setAbsent(null);
   209→        }
   210→        return activeMatchId;
   211→      });
   212→    }
   213→  }, [activeMatchId]);
   214→
   215→  /** واکشی ترکیب/آمار واقعی از ESPN — Zero-Mock */
   216→  const load = useCallback(
   217→    async (mid: string, opts?: { absentName?: string; absentSide?: "HOME" | "AWAY"; silent?: boolean }) => {
   218→      if (!mid) return;
   219→      setRunning(true);
   220→      try {
   221→        const res = await fetch("/api/lineup", {
   222→          method: "POST",
   223→          headers: { "Content-Type": "application/json" },
   224→          body: JSON.stringify({ matchId: mid, depth: "FULL", absentName: opts?.absentName, absentSide: opts?.absentSide }),
   225→        });
   226→        const data = (await res.json()) as LineupShockDTO & { error?: string };
   227→        if (data.success) {
   228→          setShockLocal(data);
   229→          if (data.mode === "OFFICIAL") {
   230→            // زنجیره واکنشی تب ۴ → تب ۶: دلتای VAEP واقعی در استور منتشر می‌شود
   231→            setShock(mid, data.home.shockDelta, data.away.shockDelta);
   232→          }
   233→          if (!opts?.silent) {
   234→            toast.success(data.mode === "OFFICIAL" ? "ترکیب رسمی واقعی ESPN دریافت شد" : "آمار واقعی فصل بازیکنان ESPN دریافت شد", {

   236→            });
   237→          }
   238→        } else {
   239→          setShockLocal(null);
   240→          if (!opts?.silent) toast.warning(data.error ?? "دیتای واقعی ESPN در دسترس نیست");
   241→        }
   242→      } catch {
   243→        if (!opts?.silent) toast.error("ارتباط با سرور برقرار نشد");
   244→      } finally {
   245→        setRunning(false);
   246→      }

   270→    },
   271→    [absent, matchId, running, load]

   273→
   274→  const activeMatch = matches.find((m) => m.id === matchId) ?? null;

   277→    return [
