    52→    52→  offBoard: boolean;

    64→    64→ * هر شرط ممکن است روی بازی دیگری باشد (میکس دو بازی) → امتیاز همان بازی از نقشه اسکوربورد.
    65→    65→ */
    66→    66→function gradeLegs(
    67→    67→  legs: SettlementLegDTO[],
    68→    68→  scoreOf: (externalId: string) => { hg: number; ag: number; state: string } | undefined,
    69→    69→  fallbackExternalId: string | null
    70→    70→): { graded: GradedLeg[]; allWon: boolean; anyLost: boolean; hasOffBoard: boolean; allFinal: boolean } {
    71→    71→  const graded: GradedLeg[] = [];
    72→    72→  let allWon = true;
    73→    73→  let anyLost = false;
    74→    74→  let hasOffBoard = false;
    75→    75→  let allFinal = true;
    76→    76→
    77→    77→  for (const leg of legs) {
    78→    78→    const ext = leg.matchExternalId || fallbackExternalId;
    79→    79→    const label = leg.matchLabel || ext || "بازی نامشخص";
    80→    80→    if (!ext || isOffMatrixKey(leg.key)) {
    81→    81→      hasOffBoard = true;
    82→    82→      graded.push({ key: leg.key, won: false, offBoard: true, notFinal: false, matchLabel: label });
    83→    83→      continue;
    84→    84→    }
    85→    85→    const score = scoreOf(ext);
    86→    86→    if (!score) {
    87→    87→      // بازیِ این شرط در پنجره امروز/دیروز ESPN پیدا نشد
    88→    88→      allFinal = false;
    89→    89→      graded.push({ key: leg.key, won: false, offBoard: false, notFinal: true, matchLabel: label });
    90→    90→      continue;
    91→    91→    }
    92→    92→    if (score.state !== "post") {
    93→    93→      allFinal = false;
    94→    94→      graded.push({ key: leg.key, won: false, offBoard: false, notFinal: true, matchLabel: label });
    95→    95→      continue;
    96→    96→    }
    97→    97→    const won = legCovers(leg.key, score.hg, score.ag);
    98→    98→    if (!won) {
    99→    99→      allWon = false;
   100→   100→      anyLost = true;
   101→   101→    }
   102→   102→    graded.push({ key: leg.key, won, offBoard: false, notFinal: false, matchLabel: label });
   103→   103→  }
   104→   104→
   105→   105→  // شرط سینتتیک خارج اسکوربورد اگر تنها مشکل باشد → MANUAL_REVIEW؛ در غیر این صورت فقط تیپ‌های قابل ارزیابی لحاظ می‌شوند
   106→   106→  const evaluable = graded.filter((g) => !g.offBoard);
   107→   107→  const evaluableAllFinal = evaluable.length > 0 && evaluable.every((g) => !g.notFinal);
   108→   108→  const evaluableAllWon = evaluable.length > 0 && evaluable.every((g) => g.won);
   109→   109→  const evaluableAnyLost = evaluable.some((g) => !g.won);
   110→   110→
   111→   111→  return {
   112→   112→    graded,
   113→   113→    allWon: evaluableAllWon,
   114→   114→    anyLost: evaluableAnyLost,
   115→   115→    hasOffBoard,
   116→   116→    allFinal: evaluableAllFinal,
   117→   117→  };
   118→   118→}

   163→   163→/** فرمول سود مستند در سند ارتقا — مجموع استیک ۱۰۰ واحد */
   164→   164→function settlementProfit(
   165→   165→  status: Exclude<SlipStatus, "PENDING" | "LIVE" | "MANUAL_REVIEW" | "VOID">,
   166→   166→  stakeA: number,
   167→   167→  stakeB: number,
   168→   168→  oddsA: number,
   169→   169→  oddsB: number
   170→   170→): { netProfit: number; profitA: number; profitB: number } {
   171→   171→  const r1 = (v: number) => Math.round(v * 100) / 100;
   172→   172→  switch (status) {
   173→   173→    case "WON_A":
   174→   174→      // 🟢 برد تیپ A: سود خالص = stakeA × oddsA − ۱۰۰
   175→   175→      return { netProfit: r1(stakeA * oddsA - DUTCH_TOTAL), profitA: r1(stakeA * oddsA - DUTCH_TOTAL), profitB: r1(-stakeB) };
   176→   176→    case "WON_B":
   177→   177→      // 🟢 برد تیپ B: سود خالص = stakeB × oddsB − ۱۰۰
   178→   178→      return { netProfit: r1(stakeB * oddsB - DUTCH_TOTAL), profitA: r1(-stakeA), profitB: r1(stakeB * oddsB - DUTCH_TOTAL) };
   179→   179→    case "DOUBLE_WIN":
   180→   180→      // 🏆 برد طلایی هر دو تیپ: جمع سود هر دو
   181→   181→      return {
   182→   182→        netProfit: r1(stakeA * oddsA + stakeB * oddsB - DUTCH_TOTAL),
   183→   183→        profitA: r1(stakeA * oddsA - stakeA),
   184→   184→        profitB: r1(stakeB * oddsB - stakeB),
   185→   185→      };
   186→   186→    case "MIDDLE_TRAP":
   187→   187→      // 🔴 باخت همزمان — ثبت صادقانه
   188→   188→      return { netProfit: r1(-DUTCH_TOTAL), profitA: r1(-stakeA), profitB: r1(-stakeB) };
   189→   189→  }
   190→   190→}
