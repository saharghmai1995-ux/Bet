    31→/** بهترین قیمت یک انتخاب بین بوک‌میکرها */
    32→function bestPrice(key: SelectionKey, odds: OddsRec[]): { odds: number; book: string } {
    33→  const pick = (fn: (o: OddsRec) => number) => {
    34→    let best = { odds: 0, book: "-" };
    35→    for (const o of odds) {
    36→      const v = fn(o);
    37→      if (v > best.odds) best = { odds: v, book: o.bookmaker === "PINNACLE" ? "پیناکل" : o.bookmaker === "SOFT_A" ? "بت‌مستر" : "پرسیان‌بت" };
    38→    }
    39→    return best;
    40→  };
    41→  switch (key) {
    42→    case "HOME": return pick((o) => o.oddsHome);
    43→    case "DRAW": return pick((o) => o.oddsDraw);
    44→    case "AWAY": return pick((o) => o.oddsAway);
    45→    case "DC_1X":
    46→    case "DC_X2": {
    47→      // شانس مجدد: معکوس مجموع احتمالات ضمنی دو نتیجه
    48→      let best = { odds: 0, book: "-" };
    49→      for (const o of odds) {
    50→        const v = key === "DC_1X" ? 1 / (1 / o.oddsHome + 1 / o.oddsDraw) : 1 / (1 / o.oddsAway + 1 / o.oddsDraw);
    51→        if (v > best.odds) best = { odds: v, book: o.bookmaker === "PINNACLE" ? "پیناکل" : o.bookmaker === "SOFT_A" ? "بت‌مستر" : "پرسیان‌بت" };
    52→      }
    53→      return best;
    54→    }
    55→    case "O25": return pick((o) => o.oddsOver25);
    56→    case "U25": return pick((o) => o.oddsUnder25);
    57→    case "C_O95": return pick((o) => o.oddsCornersOver95);
    58→    case "C_U95": return pick((o) => o.oddsCornersUnder95);
    59→    default: return { odds: 0, book: "-" };
    60→  }
    61→}

    63→function seedFromId(id: string): number {
    64→  let h = 2166136261;
    65→  for (let i = 0; i < id.length; i++) {
    66→    h ^= id.charCodeAt(i);
    67→    h = Math.imul(h, 16777619);
    68→  }
    69→  return (h >>> 0) % 2 ** 31;
    70→}

    72→/** مونت‌کارلو + جدول انتخاب‌های یک بازی (کش per-request — بدون رقابت بین درخواست‌های موازی) */
    73→function analyzeMatch(mcInput: { matchId: string; homeTeam: string; awayTeam: string; matchLabel: string }, odds: OddsRec[], cache: Map<string, unknown>): { mc: MonteCarloResult; selections: PortfolioSelectionDTO[]; lambdas: { home: number; away: number }; breakdown: LambdaBreakdown } {
    74→  const match = cache.get(mcInput.matchId);
    75→  if (!match) throw new Error("match not cached");
    76→  const engine = computeMatchEngine(match as never);
    77→  const mc = runMonteCarlo({
    78→    matrix: engine.dc.matrix,
    79→    lambdaCorners: engine.lambdaCorners,
    80→    lambdaCards: engine.lambdaCards,
    81→    sims: 10000,
    82→    seed: seedFromId(mcInput.matchId),
    83→  });
    84→
    85→  const selections: PortfolioSelectionDTO[] = MC_SELECTIONS.map((key) => {
    86→    const price = bestPrice(key, odds);
    87→    const prob = mc.flags[key] ? mc.flags[key].reduce((s: number, v: number) => s + v, 0) / mc.flags[key].length : 0;
    88→    const sharpPin = odds.find((o) => o.bookmaker === "PINNACLE");
    89→    const sharpOdds = sharpPin
    90→      ? key === "HOME" ? sharpPin.oddsHome : key === "DRAW" ? sharpPin.oddsDraw : key === "AWAY" ? sharpPin.oddsAway
    91→      : key === "O25" ? sharpPin.oddsOver25 : key === "U25" ? sharpPin.oddsUnder25
    92→      : key === "C_O95" ? sharpPin.oddsCornersOver95 : key === "C_U95" ? sharpPin.oddsCornersUnder95 : null
    93→      : null;
    94→    return {
    95→      key,
    96→      label: selectionLabel(key),
    97→      matchId: mcInput.matchId,
    98→      matchLabel: mcInput.matchLabel,
    99→      modelProb: prob,
   100→      bestOdds: price.odds,
   101→      bestBook: price.book,
   102→      sharpOdds,
   103→      evPct: prob * price.odds - 1,
   104→    };
   105→  });
   106→
   107→  return { mc, selections, lambdas: { home: engine.dc.lambdaHome, away: engine.dc.lambdaAway }, breakdown: engine.lambdaBreakdown };
   108→}

   363→          varAKey,
   364→          varBKey,
   365→          varAMatchId: c.varA.matchId,
   366→          varBMatchId: c.varB.matchId,
   367→          band: c.band,
   368→          synthetic: c.synthetic,
   369→          doubleWin: c.doubleWin && joint.bothWin > 0.02,
   370→        },
   371→        selectionA: {
   372→          ...c.base,
   373→          label: `${c.base.label} × ${c.varA.label}`,
   374→          modelProb: pA,
   375→          bestOdds: oddsA,
   376→          bestBook: "ترکیبی",
   377→          evPct: pA * oddsA - 1,
   378→        },
   379→        selectionB: {
   380→          ...c.varB,
   381→          matchId: c.varB.matchId,
   382→          matchLabel: c.varB.matchLabel,
   383→          label: `${c.base.label} × ${c.varB.label}`,
   384→          modelProb: pB,
   385→          bestOdds: oddsB,
   386→          bestBook: "ترکیبی",
   387→          evPct: pB * oddsB - 1,
   388→        },
   389→        totalOddsA: oddsA,
   390→        totalOddsB: oddsB,
   391→        jointStates: joint,
   392→        impliedSum,
   393→        guaranteedMarginPct,
   394→        equalProfit: { stakeA: Math.round(stakeA * 10) / 10, stakeB: Math.round(stakeB * 10) / 10, profit: Math.round(profit * 10) / 10 },
   395→        // v19.0 — accelerator همیشه فعال با ۶۵/۳۵
   396→        accelerator: {
   397→          active: true,
   398→          stakeA: alphaStakes.stakeA,
   399→          stakeB: alphaStakes.stakeB,
   400→          profitOnA: alphaStakes.profitOnA,
   401→          profitOnB: alphaStakes.profitOnB,
   402→          guaranteed: alphaStakes.guaranteed,
   403→          profitOnDoubleWin: alphaStakes.profitOnDoubleWin,
   404→          note: alphaStakes.note,
   405→        },
   406→        kelly: { fA: Math.round(kelly.fA * 1000) / 1000, fB: Math.round(kelly.fB * 1000) / 1000, growth: kelly.growth },
   407→        sharpEdgePct,
   408→        trap,
   409→        verdict:
   410→          guaranteedMarginPct > 4 && joint.bothLose < 0.05
   411→            ? "پوشش ممتاز — سود تضمینی بالا با ریسک باخت همزمان ناچیز"
   412→            : joint.bothLose < 0.2
   413→              ? "پوشش معتبر — پایه امن، سود تضمینی در سناریوی پایه"
   414→              : joint.bothLose < 0.38
   415→                ? "پوشش نسبی — مشروط به برد پایه؛ ریسک باخت همزمان متوسط"
   416→                : "پوشش پرریسک — پایه ضعیف؛ صرفاً برای کاهش واریانس",
   417→      });
   418→    }
   419→
   420→    // رتبه‌بندی: امتیاز = حاشیه تضمینی − جریمه باخت همزمان + پاداش رشد کلی + پاداش دو-برد
   421→    const score = (p: PortfolioPairDTO) =>
   422→      p.guaranteedMarginPct - p.jointStates.bothLose * 80 + p.kelly.growth * 100 + (p.structure.doubleWin ? 1.5 : 0);
   423→    pairs.sort((x, y) => score(y) - score(x));
   424→
   425→    // حذف تکرار آینه‌ای (O25→U25 همان زوج U25→O25 است)
   426→    const sigOf = (p: PortfolioPairDTO) => {
   427→      const [x, y] = [p.structure.varAKey, p.structure.varBKey].sort();
   428→      return `${p.structure.baseKey}|${x}|${y}|${p.kind}`;
   429→    };
   430→    const seen = new Set<string>();
   431→    const deduped = pairs.filter((p) => {
   432→      const sig = sigOf(p);
   433→      if (seen.has(sig)) return false;
   434→      seen.add(sig);
   435→      return true;
   436→    });
   437→
   438→    // تضمین تنوع: بهترین زوج هر خانواده (باندی / دو-برد / بیلدر / میکس) حتماً حضور دارد
   439→    const bucketOf = (p: PortfolioPairDTO) => (p.structure.band ? "BAND" : p.structure.doubleWin ? "DW" : p.kind);
   440→    const bestOf = new Map<string, PortfolioPairDTO>();
   441→    for (const p of deduped) {
   442→      const b = bucketOf(p);
   443→      const cur = bestOf.get(b);
   444→      if (!cur || score(p) > score(cur)) bestOf.set(b, p);
   445→    }
   446→    const picked = [...bestOf.values()];
   447→    const rest = deduped.filter((p) => !picked.includes(p));
   448→    const topPairs = [...picked, ...rest].slice(0, 6);
   449→
   450→    // ─── v5.0: ادغام زوج متعامد پیشنهادی (۱۵ بازار، بدون شرط تکراری) ───
   451→    // موتور ارتوگونال برای همین بازی اجرا و بهترین الگو به صدر فهرست تیپ‌ها می‌رود
   452→    try {
   453→      const primCached = mcInputCache.get(primary.id) as Parameters<typeof computeMatchEngine>[0] | undefined;
   454→      if (primCached) {
   455→        const eng = computeMatchEngine(primCached);
   456→        const pinOdds = primaryOdds.find((o) => o.bookmaker === "PINNACLE") ?? primaryOdds[0];
   457→        const orthoCtx = {
   458→          lambdaHome: eng.dc.lambdaHome,
   459→          lambdaAway: eng.dc.lambdaAway,
   460→          matrix: eng.dc.matrix,
   461→          lambdaCorners: eng.lambdaCorners,
   462→          lambdaCards: eng.lambdaCards,
   463→          prices: pinOdds
   464→            ? {
   465→                HOME: pinOdds.oddsHome,
   466→                DRAW: pinOdds.oddsDraw,
   467→                AWAY: pinOdds.oddsAway,
   468→                DC_1X: 1 / (1 / pinOdds.oddsHome + 1 / pinOdds.oddsDraw),
   469→                DC_X2: 1 / (1 / pinOdds.oddsAway + 1 / pinOdds.oddsDraw),
   470→                O25: pinOdds.oddsOver25,
   471→                U25: pinOdds.oddsUnder25,
   472→                C_O95: pinOdds.oddsCornersOver95,
   473→                C_U95: pinOdds.oddsCornersUnder95,
   474→              }
   475→            : undefined,
   476→        };
   477→        const ortho = generateOrthogonalComplementarySlips(orthoCtx);
   478→        const rec = ortho.archetypes.find((a) => a.code === ortho.recommended) ?? ortho.archetypes[0];
   479→        if (rec) {
   480→          const orthoPair = buildOrthoPortfolioPair(rec, primary, bankroll);
   481→          if (!topPairs.some((p) => p.id === orthoPair.id)) {
   482→            topPairs.unshift(orthoPair);
   483→            if (topPairs.length > 7) topPairs.length = 7;
   484→          }
   485→        }
   486→      }
   487→    } catch (orthoErr) {
   488→      console.error("orthogonal merge failed:", orthoErr);
   489→    }
   490→
   491→    // ─── v17.0 — مسیریاب غیرخطی ۵ رژیمی Bet-Type (Non-Linear Multi-Regime Router) ───
   492→    // برای هر مسابقه آنتروپی شانون ماتریس ۶×۶، نسبت عدم تقارن قدرت Rλ و مجموع
   493→    // ۴ خانه برتر محاسبه و بازی به مناسب‌ترین استراتژی از ۵ روش تخصصی هدایت می‌شود:
   494→    // BANKER / COMBO / CORRECT SCORE CLUSTER / DUAL-HEDGE (+ ACCUMULATOR سطح پرتفوی روز)
   495→    let betTypeRouter: import("@/lib/engine/betrouter").BetTypeRouteResult | null = null;
   496→    try {
   497→      const priceOf = (key: "HOME" | "DRAW" | "AWAY" | "O25" | "U25"): number => {
   498→        let best = 0;
   499→        for (const o of primaryOdds) {
   500→          const v = key === "HOME" ? o.oddsHome : key === "DRAW" ? o.oddsDraw : key === "AWAY" ? o.oddsAway : key === "O25" ? o.oddsOver25 : o.oddsUnder25;
   501→          if (v > best) best = v;
   502→        }
   503→        if (best > 1.01) return best;
   504→        // fallback: خط اجماعی خود بازی
   505→        return key === "HOME" ? primary.oddsHome : key === "DRAW" ? primary.oddsDraw : key === "AWAY" ? primary.oddsAway : key === "O25" ? primary.oddsOver25 : primary.oddsUnder25;
   506→      };
   507→      const prices: MarketPrices = {
   508→        oddsHome: priceOf("HOME"),
   509→        oddsDraw: priceOf("DRAW"),
   510→        oddsAway: priceOf("AWAY"),
   511→        oddsOver25: priceOf("O25"),
   512→        oddsUnder25: priceOf("U25"),
   513→      };
   514→      // ضریب اعتبار لیگ از بردار قانون کالیبراسیون (انحراف λ → اعتماد معکوس)
   515→      const leagueReliability = activeRule?.lambdaMultiplier
   516→        ? Math.min(1.15, Math.max(0.85, activeRule.lambdaMultiplier))
   517→        : 1;
   518→      betTypeRouter = routeBetTypes({
   519→        matrix: buildScoreMatrix(prim.lambdas.home, prim.lambdas.away),
   520→        lambdaHome: prim.lambdas.home,
   521→        lambdaAway: prim.lambdas.away,
   522→        prices,
   523→        pairs: topPairs,
   524→        league: primary.league,
   525→        leagueReliability,
   526→      });
   527→    } catch (routerErr) {
   528→      console.error("bet-type router failed:", routerErr);
   529→    }
   530→
   531→    // ─── ذخیره TipSet برای زوج برتر ───
   532→    const recommendedId = topPairs[0]?.id ?? null;
   533→    if (topPairs.length > 0) {
   534→      for (const [i, p] of topPairs.entries()) {
   535→        await db.tipSet.create({
   536→          data: {
   537→            label: `${p.kind === "TWO_MATCH_MIX" ? "میکس دو بازی" : "بت‌بیلدر تک‌بازی"}: ${p.labelA} ↔ ${p.labelB}`,
   538→            strategy: "EQUAL_PROFIT",
   539→            mode: p.kind === "TWO_MATCH_MIX" ? "TWO_MATCH" : "SINGLE_MATCH",
   540→            legs: JSON.stringify([p.selectionA, p.selectionB]),
   541→            totalOdds: p.totalOddsA,
   542→            stake: p.equalProfit.stakeA,
   543→            potentialReturn: p.equalProfit.stakeA * p.totalOddsA,
   544→            guaranteedProfit: p.equalProfit.profit,
   545→            coverageProb: 1 - p.jointStates.bothLose,
   546→            coverageOk: p.impliedSum < 1 && p.jointStates.bothLose < 0.1,
   547→            trapRisk: p.trap.verdict,
   548→            jointLossProb: p.jointStates.bothLose,
   549→            meta: JSON.stringify({
   550→              kelly: p.kelly,
   551→              oddsB: p.totalOddsB,
   552→              stakeB: p.equalProfit.stakeB,
   553→              jointStates: p.jointStates,
   554→              mc: { sims: prim.mc.sims, corr: prim.mc.corrGoalsCorners },
   555→              structure: p.structure,
   556→              sharpEdgePct: p.sharpEdgePct,
   557→              rank: i,
   558→            }),
   559→            batchId,
   560→            notes: p.verdict,
   561→          },
   562→        });
   563→      }
   564→    }
   565→
   566→    const result: PortfolioResultDTO = {
   567→      success: true,
   568→      primary: {
   569→        matchId: primary.id,
   570→        matchLabel: label(primary),
   571→        lambdaHome: prim.lambdas.home,
   572→        lambdaAway: prim.lambdas.away,
   573→        // ─── v9.0 — شفافیت کامل λ (بازار / آمار فصل / ترکیبی نهایی) ───
   574→        lambdaMarketHome: prim.breakdown.marketHome,
   575→        lambdaMarketAway: prim.breakdown.marketAway,
   576→        lambdaStatsHome: prim.breakdown.statsHome,
   577→        lambdaStatsAway: prim.breakdown.statsAway,
   578→        lambdaFitError: prim.breakdown.fitError,
   579→        lambdaCalibrationHome: prim.breakdown.calibrationHome,
   580→        lambdaCalibrationAway: prim.breakdown.calibrationAway,
   581→      },
   582→      calibrationRule: activeRule?.rules ?? [],
   583→      monteCarlo: {
   584→        sims: prim.mc.sims,
   585→        pHome: prim.mc.pHome,
   586→        pDraw: prim.mc.pDraw,
   587→        pAway: prim.mc.pAway,
   588→        pOver25: prim.mc.pOver25,
   589→        pBtts: prim.mc.pBtts,
   590→        cornersOver95: prim.mc.cornersOver95,
   591→        cardsOver35: prim.mc.cardsOver35,
   592→        avgCorners: prim.mc.avgCorners,
   593→        avgCards: prim.mc.avgCards,
   594→        corrGoalsCorners: prim.mc.corrGoalsCorners,
   595→      },
   596→      selections: allSelections.filter((s) => s.bestOdds > 1.01),
   597→      pairs: topPairs,
   598→      recommendedId,
   599→      batchId,
   600→      bankroll,
   601→      betTypeRouter,
   602→    };
   603→
   604→    pollCacheSet(portfolioCacheKey, result);
   605→    return NextResponse.json(result);
   606→  } catch (error) {
   607→    console.error("POST /api/agent/portfolio error:", error);
   608→    return NextResponse.json({ success: false, error: "خطا در اجرای موتور پرتفوی" }, { status: 500 });
   609→  }
   610→}

   612→/** فلگ per-sim برای ترکیب پایه × متغیر */
   613→function combineFlags(flagsBase: Uint8Array, flagsVar: Uint8Array): Uint8Array {
   614→  const n = Math.min(flagsBase.length, flagsVar.length);
   615→  const out = new Uint8Array(n);
   616→  for (let i = 0; i < n; i++) {
   617→    out[i] = flagsBase[i] && flagsVar[i] ? 1 : 0;
   618→  }
   619→  return out;
   620→}
