    51→function equalProfitStakes(oddsA: number, oddsB: number): { stakeA: number; stakeB: number } {
    52→  const invA = 1 / Math.max(1.01, oddsA);
    53→  const invB = 1 / Math.max(1.01, oddsB);
    54→  const sum = invA + invB;
    55→  const r1 = (v: number) => Math.round(v * 10) / 10;
    56→  return { stakeA: r1((100 * invA) / sum), stakeB: r1((100 * invB) / sum) };
    57→}

   717→        {slip.archetype && <p className="mt-0.5 truncate text-[9px] text-fuchsia-300">📐 {slip.archetype}</p>}
   718→      </TableCell>
   719→
   720→      <TableCell className="py-2.5">
   721→        <Badge variant="outline" className={cn("gap-1 whitespace-nowrap text-[9px]", meta.cls)}>
   722→          <StatusIcon className={cn("size-2.5", slip.status === "LIVE" && "animate-spin")} />
   723→          {meta.label}
   724→        </Badge>
   725→        {slip.settleNote && (
   726→          <p className="mt-1 max-w-[160px] truncate text-[9px] text-muted-foreground" title={slip.settleNote}>
   727→            {slip.settleNote}
   728→          </p>

   730→      </TableCell>
   731→
   732→      <TableCell className="max-w-[170px] py-2.5">
   733→        <p className="line-clamp-2 text-[10px] leading-4">{slip.labelA}</p>
   734→        <p className="mt-0.5 text-[9px] text-muted-foreground">
   735→          ضریب <b className="font-num text-sky-300" dir="ltr">{slip.oddsA.toFixed(2)}</b> · استیک{" "}
   736→          <b className="font-num" dir="ltr">{faNum(slip.stakeA, 1)}</b>

   768→              </p>

   772→            )}
   773→          </>

   776→        )}
   777→      </TableCell>
   778→
   779→      <TableCell className="py-2.5">
   780→        <button type="button" className="flex items-center gap-1 font-mono text-[9px] text-cyan-300/90 hover:text-cyan-200" onClick={copyHash} title={`کپی هش کامل: ${slip.hash}`} aria-label="کپی هش SHA-256 کارت">
   781→          <span dir="ltr">{slip.hash.slice(0, 14)}…</span>
   782→          {copied ? <Check className="size-2.5 text-emerald-400" /> : <Copy className="size-2.5" />}
   783→        </button>
   784→        {slip.espnUrl && (
   785→          <a href={slip.espnUrl} target="_blank" rel="noreferrer" className="mt-1 flex items-center gap-0.5 text-[9px] text-cyan-400 hover:underline">
   786→            راستی‌آزمایی ESPN <ExternalLink className="size-2.5" />
   787→          </a>
   788→        )}
   789→      </TableCell>
   790→
   791→      <TableCell className="py-2.5">

   796→        </div>
   797→      </TableCell>
   798→    </TableRow>
   799→  );
   800→}
