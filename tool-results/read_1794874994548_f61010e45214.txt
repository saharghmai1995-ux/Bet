   178→function downsampleEvenly(arr: number[], maxPts: number): number[] {
   179→  if (arr.length <= maxPts) return arr;
   180→  const out: number[] = [];
   181→  for (let i = 0; i < maxPts - 1; i++) {
   182→    out.push(arr[Math.round((i * (arr.length - 1)) / (maxPts - 1))]);
   183→  }
   184→  out.push(arr[arr.length - 1]);
   185→  return out;
   186→}

   543→    const days = Math.round(hours / 24);
   544→    return `${faNum(days, 0)} روز پیش`;

   562→        {chip("آخرین تسویه رسمی", lastSettled ?? "—", "text-foreground/90")}
   563→        {chip("تسویه ۷ روز اخیر", faNum(pulse.settled7d, 0), pulse.settled7d > 0 ? "text-emerald-300" : "text-amber-300")}
