    47→function equalProfitStakes(oddsA: number, oddsB: number): { stakeA: number; stakeB: number } {
    48→  const invA = 1 / Math.max(1.01, oddsA);
    49→  const invB = 1 / Math.max(1.01, oddsB);
    50→  const sum = invA + invB;
    51→  const r1 = (v: number) => Math.round(v * 10) / 10;
    52→  return { stakeA: r1((100 * invA) / sum), stakeB: r1((100 * invB) / sum) };
    53→}
