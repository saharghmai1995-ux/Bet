    67→async function enforceDailyBroadcastQuorum(): Promise<number> {
    68→  const shifted = Date.now() + TEHRAN_OFFSET_MS;
    69→  const dayStart = new Date(Math.floor(shifted / 86_400_000) * 86_400_000 - TEHRAN_OFFSET_MS);
    70→  const dayEnd = new Date(dayStart.getTime() + 86_400_000);
    71→
    72→  const approvedToday = await db.issuedSlip.count({
    73→    where: { supervisorApproved: true, retroIssued: false, kickoff: { gte: dayStart, lt: dayEnd } },
    74→  });
    75→  if (approvedToday >= DAILY_QUORUM_MIN) return 0;
    76→
    77→  const need = DAILY_QUORUM_MIN - approvedToday;
    78→  const candidates = await db.issuedSlip.findMany({
    79→    where: {
    80→      supervisorApproved: false,
    81→      retroIssued: false,
    82→      status: "PENDING",
    83→      kickoff: { gte: new Date(Date.now() + QUORUM_MIN_LEAD_MS), lt: dayEnd },
    84→      vetoReason: { startsWith: "🛡️" }, // فقط گیت ریسک/فاوریت/سقف — لیگ سمّی (🚫) هرگز ترویج نمی‌شود
    85→      jointLossRiskPct: { lte: QUORUM_MAX_RISK_PCT },
    86→    },
    87→    orderBy: { supervisorScore: "desc" },
    88→    take: need,
    89→  });
    90→  if (candidates.length === 0) return 0;
    91→
    92→  let promoted = 0;
    93→  for (const c of candidates) {
    94→    try {
    95→      // v18.0 — ترویج به «تاج نقره‌ای سطح ۲» (شکارچی آلفای سایه) نه تاج طلایی قلابی:
    96→      // استیک نصف ۵۰ واحدی Half-Kelly + نشان crownTier=SILVER_SHADOW — صادقانه‌تر و
    97→      // هم‌مبنا با سیستم دو‌سطحی تاج سند v18.0.
    98→      const halfStake = (v: number): number => Math.round(v * 0.5 * 10) / 10;
    99→      await db.issuedSlip.update({
   100→        where: { id: c.id },
   101→        data: {
   102→          supervisorApproved: true,
   103→          crownTier: "SILVER_SHADOW",
   104→          stakeA: halfStake(c.stakeA),
   105→          stakeB: halfStake(c.stakeB),
   106→          guaranteedProfit: Math.max(0, halfStake(c.guaranteedProfit)),
   107→          vetoReason: `🥈 ترویج سطح ۲ v18.0 (شکارچی آلفای سایه): بهترین کاندیدای باقی‌مانده امروز (امتیاز ${(c.supervisorScore ?? 0).toFixed(1)}، ریسک باخت همزمان ${(c.jointLossRiskPct ?? 0).toFixed(1)}٪) با استیک ایمن ۵۰ واحدی Half-Kelly منتشر می‌شود — حفظ تقویم پخش در روز فاقد کارت تراز اول`,
   108→        },
   109→      });
   110→      promoted += 1;
   111→    } catch (err) {
   112→      console.warn("quorum promotion failed:", err);
   113→    }
   114→  }
   115→  return promoted;
   116→}

   121→// هر روز یک کارت میکس ۲-۳ پایه واقعی صادر می‌شود (externalId = acc-<dayKey> idempotent،
   122→// احتمال مدل = حاصل‌ضرب احتمال پای‌ها، ضریب کل = حاصل‌ضرب ضرایب) و با فید ESPN

   124→const ACC_MIN_LEGS = 2;
   125→const ACC_MAX_LEGS = 3;
   126→const ACC_LEG_LINES = ["BANKER", "COMBO"];
   127→/** v20.0 — نگاشت خط تولید کارت پای به دسته کالیبراسیون (برای فیلتر دسته‌های تعلیق‌شده) */
   128→const ACC_LEG_LINE_MAP: Record<string, "BANKER" | "COMBO"> = { BANKER: "BANKER", COMBO: "COMBO" };
   129→const ACC_LOOKBACK_DAYS = 7;

   131→async function issueDailyAccumulators(cleanSlate: boolean): Promise<number> {
   132→  const nowMs = Date.now();
   133→  const from = new Date(nowMs - ACC_LOOKBACK_DAYS * 86_400_000);
   134→  const eligible = await db.issuedSlip.findMany({
   135→    where: {
   136→      supervisorApproved: true,
   137→      retroIssued: false,
   138→      productLine: { in: ACC_LEG_LINES },
   139→      kickoff: { gte: from },
   140→      status: { in: ["PENDING", "LIVE", "WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] },
   141→    },
   142→    orderBy: { supervisorScore: "desc" },
   143→    take: 60,
   144→  });
   145→  if (eligible.length < ACC_MIN_LEGS) return 0;
   146→
   147→  // گروه‌بندی پای‌ها بر اساس روز تهران کیک‌آف
   148→  const byDay = new Map<string, typeof eligible>();
   149→  for (const s of eligible) {
   150→    if (!s.externalId) continue;
   151→    const dayKey = new Date(new Date(s.kickoff).getTime() + TEHRAN_OFFSET_MS).toISOString().slice(0, 10);
   152→    const arr = byDay.get(dayKey) ?? [];
   153→    arr.push(s);
   154→    byDay.set(dayKey, arr);
   155→  }
   156→
   157→  let created = 0;
   158→  for (const [dayKey, slips] of byDay) {
   159→    const externalId = `acc-${dayKey}`;
   160→    // Idempotency — آکومولاتور هر روز فقط یک‌بار
   161→    const existingAcc = await db.issuedSlip.findFirst({ where: { externalId, autoIssued: true }, select: { id: true } });
   162→    if (existingAcc) continue;
   163→
   164→    const usedExt = new Set<string>();
   165→    const legs: { key: string; matchExternalId: string; matchLabel: string; odds: number; prob: number; label: string; home: string; away: string }[] = [];
   166→    // v20.0 — پای‌های دسته‌های تعلیق‌شده کالیبراسیون در میکس روز نمی‌آیند
   167→    const suspendedSet = await getSuspendedCategories();
   168→    for (const s of slips) {
   169→      if (legs.length >= ACC_MAX_LEGS) break;
   170→      if (usedExt.has(s.externalId as string)) continue;
   171→      if (suspendedSet.size > 0) {
   172→        const pl = (s.productLine ?? "SINGLE") as keyof typeof ACC_LEG_LINE_MAP;
   173→        if (pl && suspendedSet.has(ACC_LEG_LINE_MAP[pl])) continue;
   174→      }
   175→      let leg: { key?: string; matchExternalId?: string; matchLabel?: string } | undefined;
   176→      try {
   177→        leg = (JSON.parse(s.legsA || "[]") as { key?: string; matchExternalId?: string; matchLabel?: string }[])[0];
   178→      } catch {
   179→        leg = undefined;
   180→      }
   181→      if (!leg?.key || !leg?.matchExternalId) continue;
   182→      usedExt.add(s.externalId as string);
   183→      legs.push({
   184→        key: leg.key,
   185→        matchExternalId: leg.matchExternalId,
   186→        matchLabel: leg.matchLabel ?? `${s.homeTeam} × ${s.awayTeam}`,
   187→        odds: s.oddsA,
   188→        prob: s.modelProbA ?? 0.5,
   189→        label: s.labelA ?? leg.key,
   190→        home: s.homeTeam,
   191→        away: s.awayTeam,
   192→      });
   193→    }
   194→    if (legs.length < ACC_MIN_LEGS) continue;
   195→
   196→    const totalOdds = Math.round(legs.reduce((a, l) => a * l.odds, 1) * 100) / 100;
   197→    const totalProb = Math.max(0.02, Math.min(0.99, legs.reduce((a, l) => a * Math.max(0.05, Math.min(0.97, l.prob)), 1)));
   198→    const firstKo = [...slips.map((s) => new Date(s.kickoff).getTime())].sort((a, b) => a - b)[0];
   199→    const isFinishedAcc = firstKo < nowMs;
   200→    const item: IssueSlipItemDTO = {
   201→      match: {
   202→        id: externalId,
   203→        externalId,
   204→        league: "Smart Accumulator",
   205→        // ═══ v20.0 — نام دقیق تیم‌های واقعی هر پای (حذف ردیف ساختگی ACC x-Leg Multi-Match Chain) ═══
   206→        homeTeam: legs.map((l) => l.home).join(" + ").slice(0, 90),
   207→        awayTeam: legs.map((l) => l.away).join(" + ").slice(0, 90),
   208→        kickoff: new Date(firstKo).toISOString(),
   209→        espnUrl: null,
   210→      },
   211→      labelA: `🔗 ${legs.length}-Leg Smart Accumulator — ${legs.map((l) => l.label.slice(0, 34)).join(" + ")}`,
   212→      labelB: "—",
   213→      legsA: legs.map((l) => ({ key: l.key, matchExternalId: l.matchExternalId, matchLabel: l.matchLabel })),
   214→      legsB: [],
   215→      oddsA: totalOdds,
   216→      oddsB: 1.01,
   217→      stakeA: 100,
   218→      stakeB: 0,
   219→      guaranteedProfit: 0,
   220→      marginPct: 0,
   221→      archetype: "Smart Accumulator — best approved banker/combo chain",
   222→      archetypeCode: "ACC",
   223→      productLine: "ACCUMULATOR",
   224→      modelProbAOverride: totalProb,
   225→      modelProbBOverride: null,
   226→      supervisorApproved: true,
   227→      retroIssued: isFinishedAcc && !cleanSlate,
   228→      crownTier: "GOLD",
   229→      supervisorScore: Math.round((slips.reduce((a, s) => a + (s.supervisorScore ?? 0), 0) / Math.max(1, slips.length)) * 10) / 10,
   230→      vetoReason: null,
   231→    };
   232→    try {
   233→      const issued = await issueSlip(item, { autoIssued: true });
   234→      if (issued.created) {
   235→        // کارت‌های تاریخی میکس هرگز به کانال اعلام نمی‌شوند (تاریخ‌اند، نه خبر تازه)
   236→        if (isFinishedAcc) {
   237→          await db.issuedSlip.update({
   238→            where: { id: issued.slip.id },
   239→            data: { telegramCard: true, telegramResult: true, settleNote: "v19.0 smart accumulator — legs settled on official ESPN FT" },
   240→          });
   241→        }
   242→        created += 1;
   243→        console.log(`[auto-issue] v19.0 accumulator ${dayKey}: ${legs.length} legs @ ${totalOdds.toFixed(2)}x issued`);
   244→      }
   245→    } catch (err) {
   246→      console.warn(`accumulator issuance failed for ${dayKey}:`, err);
   247→    }
   248→  }
   249→  return created;
   250→}

   281→  const probs = modelProbs(d);
   282→  const isPost = ev.statusState === "post";
   283→  const isLive = ev.statusState === "in";
   284→
   285→  const row = await db.match.upsert({
   286→    where: { externalId: ev.externalId },
   287→    create: {
   288→      externalId: ev.externalId,
   289→      league: ev.leagueName,
   290→      homeTeam: ev.homeTeam,
   291→      awayTeam: ev.awayTeam,
   292→      kickoff: new Date(Date.parse(ev.kickoff)),
   293→      homeAttack: d.homeAttack,
   294→      homeDefense: d.homeDefense,
   295→      awayAttack: d.awayAttack,
   296→      awayDefense: d.awayDefense,
   297→      homeElo: d.homeElo,
   298→      awayElo: d.awayElo,
   299→      homeXg: d.lambdaHome,

   309→      oddsDraw: fairOdds(probs.pDraw, 1.06),
