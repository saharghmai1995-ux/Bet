   435→def verify_trc20_usdt_transaction(
   436→    txid: str, expected_wallet: str, min_amount_usdt: float
   437→) -> Tuple[bool, float, str]:
   438→    clean_tx = txid.strip()
   439→    if len(clean_tx) != 64:
   440→        return False, 0.0, "Invalid TRC20 TxID length (must be 64 hex characters)."
   441→
   442→    url = f"https://apilist.tronscanapi.com/api/transaction-info?hash={clean_tx}"
   443→    try:
   444→        req = urllib.request.Request(url, headers={"User-Agent": "NeoBetQuantVerifier/1.0"})
   445→        with urllib.request.urlopen(req, timeout=12) as resp:
   446→            data = json.loads(resp.read().decode("utf-8"))
   447→    except Exception as exc:
   448→        return False, 0.0, f"Blockchain node connection error: {exc}"
   449→
   450→    if not data or data.get("contractRet") != "SUCCESS":
   451→        return False, 0.0, "Transaction not found or status is not SUCCESS on Tron blockchain."
   452→
   453→    transfers = data.get("trc20TransferInfo") or []
   454→    for tr in transfers:
   455→        contract_addr = tr.get("contract_address", "")
   456→        to_addr = tr.get("to_address", "")
   457→        decimals = int(tr.get("decimals", 6))
   458→        raw_amount = float(tr.get("amount_str", "0")) / (10**decimals)
   459→
   460→        if contract_addr == TRON_USDT_CONTRACT and to_addr.lower() == expected_wallet.lower():
   461→            if raw_amount + 0.05 >= min_amount_usdt:
   462→                return True, round(raw_amount, 2), "Confirmed on Tron (TRC-20)"
   463→            return (
   464→                False,
   465→                raw_amount,
   466→                f"Transferred amount ({raw_amount:.2f} USDT) is less than required ({min_amount_usdt:.2f} USDT).",
   467→            )
   468→
   469→    return False, 0.0, "No matching USDT-TRC20 transfer to the official NeoBet wallet was found in this TxID."
   470→

   472→def verify_polygon_usdt_transaction(
   473→    txid: str, expected_wallet: str, min_amount_usdt: float
   474→) -> Tuple[bool, float, str]:
   475→    clean_tx = txid.strip().lower()
   476→    if not clean_tx.startswith("0x") or len(clean_tx) != 66:
   477→        return False, 0.0, "Invalid Polygon TxID format (must start with 0x and be 66 characters)."
   478→
   479→    rpc_payload = json.dumps(
   480→        {
   481→            "jsonrpc": "2.0",
   482→            "method": "eth_getTransactionReceipt",
   483→            "params": [clean_tx],
   484→            "id": 1,
   485→        }
   486→    ).encode("utf-8")
   487→
   488→    try:
   489→        req = urllib.request.Request(
   490→            "https://polygon-rpc.com",
   491→            data=rpc_payload,
   492→            headers={"Content-Type": "application/json", "User-Agent": "NeoBetQuantVerifier/1.0"},
   493→        )
   494→        with urllib.request.urlopen(req, timeout=12) as resp:
   495→            res_json = json.loads(resp.read().decode("utf-8"))
   496→    except Exception as exc:
   497→        return False, 0.0, f"Polygon RPC connection error: {exc}"
   498→
   499→    receipt = res_json.get("result")
   500→    if not receipt or receipt.get("status") != "0x1":
   501→        return False, 0.0, "Transaction not found or failed on Polygon network."
   502→
   503→    clean_wallet_padded = "0x" + expected_wallet.lower().replace("0x", "").rjust(64, "0")
   504→    for log in receipt.get("logs", []):
   505→        if log.get("address", "").lower() == POLYGON_USDT_CONTRACT:
   506→            topics = [t.lower() for t in log.get("topics", [])]
   507→            if len(topics) >= 3 and topics[0] == ERC20_TRANSFER_TOPIC and topics[2] == clean_wallet_padded:
   508→                raw_hex = log.get("data", "0x0")
   509→                amount_usdt = int(raw_hex, 16) / 1e6
   510→                if amount_usdt + 0.05 >= min_amount_usdt:
   511→                    return True, round(amount_usdt, 2), "Confirmed on Polygon POS"
   512→                return (
   513→                    False,
   514→                    amount_usdt,
   515→                    f"Transferred amount ({amount_usdt:.2f} USDT) is less than required ({min_amount_usdt:.2f} USDT).",
   516→                )
   517→
   518→    return False, 0.0, "No matching Polygon USDT transfer to the official NeoBet wallet was found."
   519→

  1041→def sweep_expired_vip_members(conn: sqlite3.Connection) -> int:
  1042→    now_iso = datetime.now(timezone.utc).isoformat()
  1043→    expired_rows = conn.execute(
  1044→        "SELECT user_id FROM users WHERE vip_expires_at IS NOT NULL AND vip_expires_at < ?",
  1045→        (now_iso,),
  1046→    ).fetchall()
  1047→    removed_count = 0
  1048→    for row in expired_rows:
  1049→        uid = row["user_id"]
  1050→        tg_api_call("banChatMember", {"chat_id": VIP_CHANNEL_ID, "user_id": uid})
  1051→        tg_api_call("unbanChatMember", {"chat_id": VIP_CHANNEL_ID, "user_id": uid, "only_if_banned": True})
  1052→        conn.execute("UPDATE users SET vip_expires_at = NULL WHERE user_id = ?", (uid,))
  1053→        removed_count += 1
  1054→    if removed_count > 0:
  1055→        conn.commit()
  1056→    return removed_count
