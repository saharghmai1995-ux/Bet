    52→export async function verifyBotToken(token: string): Promise<{ ok: boolean; username: string | null; detail: string }> {
    53→  try {
    54→    const res = await fetch(`https://api.telegram.org/bot${token.trim()}/getMe`, { cache: "no-store" });
    55→    const json = (await res.json()) as { ok?: boolean; result?: { username?: string }; description?: string };
    56→    if (json.ok && json.result?.username) return { ok: true, username: json.result.username, detail: `@${json.result.username}` };
    57→    return { ok: false, username: null, detail: json.description ?? `HTTP ${res.status}` };
    58→  } catch (err) {
    59→    return { ok: false, username: null, detail: err instanceof Error ? err.message : "network error" };
    60→  }
    61→}

   124→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
   125→  const token = await botToken();
   126→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   127→  // v13.2 — سپر توکن ناقص: خطای 404 مبهم تلگرام را با پیام دقیق جایگزین می‌کند
   128→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   129→  try {
   130→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
   131→      method: "POST",
   132→      headers: { "Content-Type": "application/json" },
   133→      body: JSON.stringify(body),
   134→      cache: "no-store",
   135→    });
   136→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   137→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
   138→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   139→  } catch (err) {
   140→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   141→  }
   142→}

   144→async function tgSendPhotoTo(
   145→  chatId: string | null,
   146→  photo: Buffer,
   147→  caption: string,
   148→  filename = "slip-card.png",
   149→  captionless = false
   150→): Promise<{ ok: boolean; detail: string }> {
   151→  const token = await botToken();
   152→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   153→  if (!chatId) return { ok: false, detail: "NO_CHANNEL" };
   154→  // v13.2 — سپر توکن ناقص (بخش hash بدون bot_id) — ریشه باگ «انتشار انجام نمیشه»
   155→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   156→  try {
   157→    const form = new FormData();
   158→    form.append("chat_id", chatId);
   159→    if (!captionless) {
   160→      form.append("caption", caption.slice(0, 1024));
   161→      form.append("parse_mode", "HTML");
   162→    }
   163→    form.append("photo", new Blob([new Uint8Array(photo)], { type: "image/png" }), filename);
   164→    const res = await fetch(`https://api.telegram.org/bot${token}/sendPhoto`, { method: "POST", body: form });
   165→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   166→    if (json.ok) return { ok: true, detail: `photo:${json.result?.message_id ?? "?"}${captionless ? ":broadcast" : ""}` };
   167→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   168→  } catch (err) {
   169→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   170→  }
   171→}

   210→/** پیام HTML فلش‌کارت کامل — تیپ A + تیپ B + داچینگ + SHA-256 + لینک ESPN */
   211→export function buildCardMessage(slip: IssuedSlip): string {
   212→  const kickoff = new Date(slip.kickoff).toLocaleString("fa-IR", { dateStyle: "short", timeStyle: "short" });
   213→  const lines: string[] = [];
   214→  lines.push(`⚽️ <b>فوتبال‌کوانت — فلش‌کارت تیپ مکمل</b>`);
   215→  lines.push(`🏆 <b>${esc(slip.homeTeam)}</b> × <b>${esc(slip.awayTeam)}</b>`);
   216→  lines.push(`📡 ${esc(slip.league)} · ⏰ ${kickoff}`);
   217→  if (slip.archetype) lines.push(`📐 الگو: ${esc(slip.archetype)}`);
   218→  lines.push("");
   219→  lines.push(`🔹 <b>تیپ A</b> (ضریب ${slip.oddsA.toFixed(2)}): ${esc(slip.labelA)}`);
   220→  lines.push(`💵 مبلغ داچینگ A: <b>${slip.stakeA.toFixed(1)}</b> واحد`);
   221→  lines.push("");
   222→  lines.push(`🟢 <b>تیپ B مکمل</b> (ضریب ${slip.oddsB.toFixed(2)}): ${esc(slip.labelB)}`);
   223→  lines.push(`💵 مبلغ داچینگ B: <b>${slip.stakeB.toFixed(1)}</b> واحد`);
   224→  lines.push("");
   225→  // ─── v10.0 — ارزش عددی نسبت به خط مرجع (همیشه عدد دارد) ───
   226→  if (slip.edgeA !== null || slip.edgeB !== null) {
   227→    const isPin = slip.edgeSourceA === "PINNACLE" || slip.edgeSourceB === "PINNACLE";
   228→    const src = isPin ? "خط شارپ پیناکل" : "خط منصفانه بازار";
   229→    const fa = (v: number) => `${v >= 0 ? "+" : ""}${v.toFixed(1)}%`;
   230→    const a = slip.edgeA !== null ? fa(slip.edgeA) : "—";
   231→    const b = slip.edgeB !== null ? fa(slip.edgeB) : "—";
   232→    lines.push(`📈 ارزش عددی نسبت به ${src}: تیپ A <b>${a}</b> · تیپ B <b>${b}</b>`);
   233→    if (isPin) {
   234→      const cmp = (odds: number, pin: number | null): string | null =>
   235→        pin && pin > 1.01 ? `${odds.toFixed(2)} در برابر خط ${pin.toFixed(2)}` : null;
   236→      const pa = slip.edgeSourceA === "PINNACLE" ? cmp(slip.oddsA, slip.pinnacleOddsA) : null;
   237→      const pb = slip.edgeSourceB === "PINNACLE" ? cmp(slip.oddsB, slip.pinnacleOddsB) : null;
   238→      const parts = [pa ? `تیپ A: ${pa}` : null, pb ? `تیپ B: ${pb}` : null].filter((x): x is string => Boolean(x));
   239→      if (parts.length > 0) lines.push(`📌 مقایسه قیمت با خط پیناکل: ${parts.join(" · ")}`);
   240→    }
   241→  }
   242→  // ─── v10.0 — بهترین ساعت ثبت ضریب (Optimal Odds Timing Window) ───
   243→  if (slip.timingWindowA || slip.timingWindowB) {
   244→    const winName = timingWindowLabel(slip.timingWindowA ?? slip.timingWindowB);
   245→    const start = slip.timingStartA ?? slip.timingStartB;
   246→    const end = slip.timingEndA ?? slip.timingEndB;
   247→    const fmt = (d: Date) => new Intl.DateTimeFormat("fa-IR", { hour: "2-digit", minute: "2-digit", timeZone: "Asia/Tehran" }).format(d);
   248→    const range = start && end ? ` | ساعت دقیق پیشنهادی ثبت: ${fmt(new Date(start))} تا ${fmt(new Date(end))} (تهران)` : "";
   249→    lines.push(`⏱️ بهترین زمان ثبت این تیپ: <b>${esc(winName)}</b>${range}`);
   250→  }
   251→  lines.push(`💰 سود تضمینی پوشش: <b>+${slip.guaranteedProfit.toFixed(1)}%</b> (مجموع استیک ۱۰۰ واحد)`);
   252→  lines.push(`🔐 هش راستی‌آزمایی SHA-256:\n<code>${slip.hash.slice(0, 32)}</code>`);
   253→  if (slip.espnUrl) lines.push(`🔎 <a href="${slip.espnUrl}">لینک راستی‌آزمایی رسمی ESPN</a>`);
   254→  lines.push("");
   255→  lines.push(`⚖️ ابزار تحلیل کمّی — مسئولیت استفاده با کاربر است.`);
   256→  return lines.join("\n");
   257→}

   313→export function buildResultMessage(slip: IssuedSlip): string {

   327→  } else {

   329→    lines.push(`🧮 فرمول: ${(slip.status === "WON_A" ? `stakeA×oddsA−۱۰۰ = ${slip.stakeA.toFixed(1)}×${slip.oddsA.toFixed(2)}−۱۰۰` : `stakeB×oddsB−۱۰۰ = ${slip.stakeB.toFixed(1)}×${slip.oddsB.toFixed(2)}−۱۰۰`)}`);

   442→/** آمار وین‌ریت کانال — مستقیم از دیتابیس (بدون وابستگی چرخه‌ای) */
   443→async function channelWinRateStats() {
   444→  // v11.0 — فقط کارت‌های واقعی تاییدشده؛ تسویه کاغذی دفتر سایه در وین‌ریت کانال دخالت نمی‌کند
   445→  // v13.2 — فقط کارت‌های واقعی تاییدشده؛ کارت‌های بازخوانی (retroIssued) و دفتر سایه در وین‌ریت کانال دخالت نمی‌کنند
   446→  const settled = await db.issuedSlip.findMany({
   447→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, supervisorApproved: true, retroIssued: false },
   448→    select: { status: true, netProfit: true, stakeA: true, stakeB: true },
   449→  });
   450→  const st = { settled: settled.length, wonA: 0, wonB: 0, doubleWin: 0, middleTrap: 0, netProfit: 0, staked: 0 };
   451→  for (const s of settled) {
   452→    if (s.status === "WON_A") st.wonA += 1;
   453→    else if (s.status === "WON_B") st.wonB += 1;
   454→    else if (s.status === "DOUBLE_WIN") st.doubleWin += 1;
   455→    else if (s.status === "MIDDLE_TRAP") st.middleTrap += 1;
   456→    st.netProfit += s.netProfit ?? 0;
   457→    st.staked += s.stakeA + s.stakeB;
   458→  }
   459→  const winRatePct = st.settled > 0 ? Math.round(((st.settled - st.middleTrap) / st.settled) * 1000) / 10 : 0;
   460→  const netPct = st.staked > 0 ? Math.round((st.netProfit / st.staked) * 1000) / 10 : 0;
   461→  // ─── v11.0 — دفتر سایه: شفافیت counterfactual وتوها (خط جداگانه گزارش) ───
   462→  const shadowRows = await db.issuedSlip.findMany({
   463→    where: { supervisorApproved: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   464→    select: { status: true, netProfit: true },
   465→  });
   466→  const shadowWins = shadowRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   467→  const shadowNet = Math.round(shadowRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100;
   468→  const shadowWinRate = shadowRows.length > 0 ? Math.round((shadowWins / shadowRows.length) * 1000) / 10 : null;
   469→  // ─── v13.2 — دفتر بازخوانی: کارت‌های چندروزه‌ای که هرگز پیش‌بازی صادر نشده‌اند (تسویه کاغذی تحلیلی) ───
   470→  const retroRows = await db.issuedSlip.findMany({
   471→    where: { retroIssued: true, supervisorApproved: true, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   472→    select: { status: true, netProfit: true },
   473→  });
   474→  const retroWins = retroRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   475→  return {
   476→    ...st,
   477→    winRatePct,
   478→    netPct,
   479→    netProfit: Math.round(st.netProfit * 100) / 100,
   480→    shadow: { settled: shadowRows.length, wins: shadowWins, winRatePct: shadowWinRate, netProfit: shadowNet },
   481→    retro: {
   482→      settled: retroRows.length,
   483→      wins: retroWins,
   484→      winRatePct: retroRows.length > 0 ? Math.round((retroWins / retroRows.length) * 1000) / 10 : null,
   485→      netProfit: Math.round(retroRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100,
   486→    },
   487→  };
   488→}

   523→/** صف انتشار بدون ارسال — پیش‌نمایش زنده فید کانال (Live Telegram Channel Simulator) */
   524→export async function previewDispatchQueue(): Promise<{ queue: DispatchQueueItem[]; autoDispatch: boolean; configured: boolean }> {
   525→  // v13.2 — فقط کارت‌های بازی‌های «امروز تهران» به کانال ارسال می‌شوند؛ بازی‌های روزهای آینده
   526→  // در دک فلش‌کارت می‌مانند و نوبت انتشارشان روز بازی خودش می‌رسد.
   527→  const day = tehranDayRange();
   528→  const [pendingCards, pendingResults, autoDispatch, tg] = await Promise.all([
   529→    db.issuedSlip.findMany({
   530→      where: { status: { in: ["PENDING", "LIVE"] }, telegramCard: false, supervisorApproved: true, retroIssued: false, kickoff: { gte: day.start, lt: day.end } },
   531→      orderBy: { kickoff: "asc" },
   532→      take: 8,
   533→    }),
   534→    db.issuedSlip.findMany({
   535→      where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, telegramResult: false, supervisorApproved: true, retroIssued: false },
   536→      orderBy: { settledAt: "asc" },
   537→      take: 8,
   538→    }),
   539→    getAutoDispatch(),
   540→    telegramStatus(),
   541→  ]);
   542→
   543→  const queue: DispatchQueueItem[] = [];
   544→  for (const s of pendingCards) {
   545→    queue.push({
   546→      kind: "CARD",
   547→      slipId: s.id,
   548→      slipHash: s.hash,
   549→      matchLabel: `${s.homeTeam} × ${s.awayTeam}`,
   550→      message: buildCardMessage(s),
   551→      status: s.status,
   552→    });
   553→  }
   554→  for (const s of pendingResults) {
   555→    queue.push({
   556→      kind: "RESULT",
   557→      slipId: s.id,
   558→      slipHash: s.hash,
   559→      matchLabel: `${s.homeTeam} ${s.homeGoals ?? "?"}−${s.awayGoals ?? "?"} ${s.awayTeam}`,
   560→      message: buildResultMessage(s),
   561→      status: s.status,
   562→    });
   563→  }
   564→  const st = await channelWinRateStats();
   565→  if (pendingResults.length > 0) {
   566→    queue.push({ kind: "WINRATE", slipId: null, slipHash: null, matchLabel: "گزارش وین‌ریت کانال", message: buildWinRateReportMessage(st) });
   567→  }
   568→  return { queue, autoDispatch, configured: tg.configured };
   569→}

   571→/**
   572→ * صف ارسال خودکار دو مرحله‌ای (Auto-Dispatch Queue):
   573→ *   · پیام نوع اول (پیش از بازی): فقط کارت‌های تاییدشده سوپروایزر (وتوشده‌ها اسکیپ)
   574→ *   · پیام نوع دوم (پس از FT): برگه نتیجه + به‌روزرسانی وین‌ریت کل کانال
   575→ * Dedupe سمت سرور با پرچم‌های telegramCard/telegramResult — ارسال تکراری صفر.
   576→ * v9.0 — سقف هر پاس قابل تنظیم است: پس‌زمینه ۳/۳ (حفظ تأخیر سینک) و دکمه فوری ۸/۸.
   577→ */
   578→export async function autoDispatchApprovedSlips(options?: {
   579→  maxCards?: number;
   580→  maxResults?: number;
   581→}): Promise<{
   582→  ok: boolean;
   583→  sentCards: number;
   584→  sentResults: number;
   585→  winRateSent: boolean;
   586→  skippedVetoed: number;
   587→  detail: string;
   588→}> {
   589→  const maxCards = Math.max(1, Math.min(12, options?.maxCards ?? 3));
   590→  const maxResults = Math.max(1, Math.min(12, options?.maxResults ?? 3));
   591→  const [tg, autoOn] = await Promise.all([telegramStatus(), getAutoDispatch()]);
   592→  const skippedVetoed = await db.issuedSlip.count({ where: { supervisorApproved: false, telegramCard: false } });
   593→
   594→  if (!tg.configured || !tg.channelSet) {
   595→    return { ok: false, sentCards: 0, sentResults: 0, winRateSent: false, skippedVetoed, detail: "TELEGRAM_NOT_CONFIGURED" };
   596→  }
   597→  if (!tg.tokenFormat) {
   598→    return { ok: false, sentCards: 0, sentResults: 0, winRateSent: false, skippedVetoed, detail: "INVALID_TOKEN_FORMAT" };
   599→  }
   600→  if (!autoOn) {
   601→    return { ok: false, sentCards: 0, sentResults: 0, winRateSent: false, skippedVetoed, detail: "AUTO_DISPATCH_OFF" };
   602→  }
   603→
   604→  // v13.2 — «فقط کارت‌های روزانه»: پنجره انتشار کارت = بازی‌های امروز تهران
   605→  const day = tehranDayRange();
   606→
   607→  let sentCards = 0;
   608→  let sentResults = 0;
   609→
   610→  // ─── مرحله ۱: پست پیش از بازی — فقط کارت‌های 👑 تاییدشده سوپروایزر «بازی‌های امروز تهران» ───
   611→  const cards = await db.issuedSlip.findMany({
   612→    where: { status: { in: ["PENDING", "LIVE"] }, telegramCard: false, supervisorApproved: true, retroIssued: false, kickoff: { gte: day.start, lt: day.end } },
   613→    orderBy: { kickoff: "asc" },
   614→    take: maxCards,
   615→  });
   616→  for (const s of cards) {
   617→    try {
   618→      const res = await publishSlipToTelegram(s, "CARD");
   619→      if (res.ok) sentCards += 1;
   620→    } catch (err) {
   621→      console.warn("auto dispatch card failed:", err);
   622→    }
   623→  }
   624→
   625→  // ─── مرحله ۲: پست تسویه‌حساب پس از بازی — برگه نتیجه FT (کارت‌های بازخوانی مستثنا) ───
   626→  const results = await db.issuedSlip.findMany({
   627→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, telegramResult: false, supervisorApproved: true, retroIssued: false },
   628→    orderBy: { settledAt: "asc" },
   629→    take: maxResults,
   630→  });
   631→  for (const s of results) {
   632→    try {
   633→      const ok = await sendSettlementNotification(s);
   634→      if (ok) sentResults += 1;
   635→    } catch (err) {
   636→      console.warn("auto dispatch result failed:", err);
   637→    }
   638→  }
   639→
   640→  // ─── به‌روزرسانی وین‌ریت کل کانال — فقط وقتی نتیجه تازه‌ای مخابره شد ───
   641→  let winRateSent = false;
   642→  if (sentResults > 0) {
   643→    winRateSent = await sendWinRateReport();
   644→  }
   645→
   646→  return {
   647→    ok: sentCards + sentResults > 0,
   648→    sentCards,
   649→    sentResults,
   650→    winRateSent,
   651→    skippedVetoed,
   652→    detail: sentCards + sentResults > 0 ? "dispatched" : "QUEUE_EMPTY",
   653→  };
   654→}
   655→
   656→/** throttle گزارش وین‌ریت — حداکثر هر ۱۰ دقیقه یک‌بار */
   657→let lastWinRateAt = 0;
   658→
   659→/**
   660→ * v13.2 — بازه «امروز تهران» به UTC (تهران = UTC+۳:۳۰ ثابت، بدون DST).
   661→ * مبنای فیلتر «فقط کارت‌های روزانه به کانال ارسال شود».
   662→ */
   663→export function tehranDayRange(d = new Date()): { start: Date; end: Date } {
   664→  const TEHRAN_OFFSET_MS = 3.5 * 3600_000;
   665→  const shifted = d.getTime() + TEHRAN_OFFSET_MS;
   666→  const dayStartShifted = Math.floor(shifted / 86_400_000) * 86_400_000;
   667→  const start = new Date(dayStartShifted - TEHRAN_OFFSET_MS);
   668→  return { start, end: new Date(start.getTime() + 86_400_000) };
   669→}
   670→
   671→/**
   672→ * پیام نوع دوم (بخش وین‌ریت): به‌روزرسانی وین‌ریت کل کانال پس از تسویه.
   673→ * از موتور تسویه (settleFinishedSlips) و صف خودکار فراخوانی می‌شود — throttle ۱۰ دقیقه.
   674→ */
   675→export async function sendWinRateReport(force = false): Promise<boolean> {
   676→  if (!force && Date.now() - lastWinRateAt < 10 * 60_000) return false;
   677→  const status = await telegramStatus();
   678→  if (!status.configured || !status.channelSet) return false;
