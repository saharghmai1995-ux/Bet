    64→  try {
    65→    const res = await fetch(`https://api.telegram.org/bot${token.trim()}/getMe`, { cache: "no-store" });
    66→    const json = (await res.json()) as { ok?: boolean; result?: { username?: string }; description?: string };
    67→    if (json.ok && json.result?.username) return { ok: true, username: json.result.username, detail: `@${json.result.username}` };
    68→    return { ok: false, username: null, detail: json.description ?? `HTTP ${res.status}` };
    69→  } catch (err) {
    70→    return { ok: false, username: null, detail: err instanceof Error ? err.message : "network error" };
    71→  }

   135→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
   136→  const token = await botToken();
   137→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   138→  // v13.2 — سپر توکن ناقص: خطای 404 مبهم تلگرام را با پیام دقیق جایگزین می‌کند
   139→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   140→  try {
   141→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
   142→      method: "POST",
   143→      headers: { "Content-Type": "application/json" },
   144→      body: JSON.stringify(body),
   145→      cache: "no-store",
   146→    });
   147→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   148→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
   149→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   150→  } catch (err) {
   151→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   152→  }
   153→}

   155→async function tgSendPhotoTo(
   156→  chatId: string | null,
   157→  photo: Buffer,
   158→  caption: string,
   159→  filename = "slip-card.png",
   160→  captionless = false
   161→): Promise<{ ok: boolean; detail: string }> {
   162→  const token = await botToken();
   163→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   164→  if (!chatId) return { ok: false, detail: "NO_CHANNEL" };
   165→  // v13.2 — سپر توکن ناقص (بخش hash بدون bot_id) — ریشه باگ «انتشار انجام نمیشه»
   166→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   167→  try {
   168→    const form = new FormData();
   169→    form.append("chat_id", chatId);
   170→    if (!captionless) {
   171→      form.append("caption", caption.slice(0, 1024));
   172→      form.append("parse_mode", "HTML");
   173→    }
   174→    form.append("photo", new Blob([new Uint8Array(photo)], { type: "image/png" }), filename);
   175→    const res = await fetch(`https://api.telegram.org/bot${token}/sendPhoto`, { method: "POST", body: form });
   176→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   177→    if (json.ok) return { ok: true, detail: `photo:${json.result?.message_id ?? "?"}${captionless ? ":broadcast" : ""}` };
   178→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   179→  } catch (err) {
   180→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   181→  }
   182→}

   221→/** پیام HTML فلش‌کارت کامل — تیپ A + تیپ B + داچینگ + SHA-256 + لینک ESPN */
   222→export function buildCardMessage(slip: IssuedSlip): string {
   223→  const kickoff = new Date(slip.kickoff).toLocaleString("fa-IR", { dateStyle: "short", timeStyle: "short" });
   224→  // v15.0 — نام رسمی بین‌المللی مارکت هر تیپ (Institutional Execution Metadata)
   225→  const legKeys = (json: string): string[] => {
   226→    try {
   227→      const arr = JSON.parse(json);
   228→      return Array.isArray(arr) ? arr.map((l: { key?: string }) => String(l?.key ?? l)) : [];
   229→    } catch {
   230→      return [];
   231→    }
   232→  };
   233→  const marketAEn = marketEnLabel(legKeys(slip.legsA)[0] ?? "");
   234→  const marketBEn = marketEnLabel(legKeys(slip.legsB)[0] ?? "");
   235→  const lines: string[] = [];
   236→  lines.push(`⚽️ <b>فوتبال‌کوانت — فلش‌کارت الگوی طلایی اتمیک v16.0</b>`);
   237→  lines.push(`🏆 <b>${esc(slip.homeTeam)}</b> × <b>${esc(slip.awayTeam)}</b>`);
   238→  lines.push(`📡 ${esc(slip.league)} · ⏰ ${kickoff}`);
   239→  if (slip.archetype) lines.push(`📐 الگو: ${esc(slip.archetype)}`);
   240→  lines.push("");
   241→  lines.push(`🔹 <b>تیپ A</b> (ضریب ${slip.oddsA.toFixed(2)}): ${esc(slip.labelA)}`);
   242→  lines.push(`🏷 Market (EN): <i>${esc(marketAEn)}</i>`);
   243→  lines.push(`💵 مبلغ داچینگ A: <b>${slip.stakeA.toFixed(1)}</b> واحد`);
   244→  lines.push("");
   245→  lines.push(`🟢 <b>تیپ B مکمل</b> (ضریب ${slip.oddsB.toFixed(2)}): ${esc(slip.labelB)}`);
   246→  lines.push(`🏷 Market (EN): <i>${esc(marketBEn)}</i>`);
   247→  lines.push(`💵 مبلغ داچینگ B: <b>${slip.stakeB.toFixed(1)}</b> واحد`);
   248→  lines.push("");
   249→  lines.push(`${esc(SETTLEMENT_RULE_FA)}`);
   250→  // ─── v10.0 — ارزش عددی نسبت به خط مرجع (همیشه عدد دارد) ───
   251→  if (slip.edgeA !== null || slip.edgeB !== null) {
   252→    const isPin = slip.edgeSourceA === "PINNACLE" || slip.edgeSourceB === "PINNACLE";
   253→    const src = isPin ? "خط شارپ پیناکل" : "خط منصفانه بازار";
   254→    const fa = (v: number) => `${v >= 0 ? "+" : ""}${v.toFixed(1)}%`;
   255→    const a = slip.edgeA !== null ? fa(slip.edgeA) : "—";
   256→    const b = slip.edgeB !== null ? fa(slip.edgeB) : "—";
   257→    lines.push(`📈 ارزش عددی نسبت به ${src}: تیپ A <b>${a}</b> · تیپ B <b>${b}</b>`);
   258→    if (isPin) {
   259→      const cmp = (odds: number, pin: number | null): string | null =>
   260→        pin && pin > 1.01 ? `${odds.toFixed(2)} در برابر خط ${pin.toFixed(2)}` : null;
   261→      const pa = slip.edgeSourceA === "PINNACLE" ? cmp(slip.oddsA, slip.pinnacleOddsA) : null;
   262→      const pb = slip.edgeSourceB === "PINNACLE" ? cmp(slip.oddsB, slip.pinnacleOddsB) : null;
   263→      const parts = [pa ? `تیپ A: ${pa}` : null, pb ? `تیپ B: ${pb}` : null].filter((x): x is string => Boolean(x));
   264→      if (parts.length > 0) lines.push(`📌 مقایسه قیمت با خط پیناکل: ${parts.join(" · ")}`);
   265→    }
   266→  }
   267→  // ─── v10.0 — بهترین ساعت ثبت ضریب (Optimal Odds Timing Window) ───
   268→  if (slip.timingWindowA || slip.timingWindowB) {
   269→    const winName = timingWindowLabel(slip.timingWindowA ?? slip.timingWindowB);
   270→    const start = slip.timingStartA ?? slip.timingStartB;
   271→    const end = slip.timingEndA ?? slip.timingEndB;
   272→    const fmt = (d: Date) => new Intl.DateTimeFormat("fa-IR", { hour: "2-digit", minute: "2-digit", timeZone: "Asia/Tehran" }).format(d);
   273→    const range = start && end ? ` | ساعت دقیق پیشنهادی ثبت: ${fmt(new Date(start))} تا ${fmt(new Date(end))} (تهران)` : "";
   274→    lines.push(`⏱️ بهترین زمان ثبت این تیپ: <b>${esc(winName)}</b>${range}`);
   275→  }
   276→  lines.push(`💰 سود تضمینی پوشش: <b>+${slip.guaranteedProfit.toFixed(1)}%</b> (مجموع استیک ۱۰۰ واحد)`);
   277→  lines.push(`🔐 هش راستی‌آزمایی SHA-256:\n<code>${slip.hash.slice(0, 32)}</code>`);
   278→  if (slip.espnUrl) lines.push(`🔎 <a href="${slip.espnUrl}">لینک راستی‌آزمایی رسمی ESPN</a>`);
   279→  lines.push("");
   280→  lines.push(`⚖️ ابزار تحلیل کمّی — مسئولیت استفاده با کاربر است.`);
   281→  return lines.join("\n");
   282→}

   471→/** آمار وین‌ریت کانال — مستقیم از دیتابیس (بدون وابستگی چرخه‌ای) */
   472→async function channelWinRateStats() {
   473→  // v11.0 — فقط کارت‌های واقعی تاییدشده؛ تسویه کاغذی دفتر سایه در وین‌ریت کانال دخالت نمی‌کند
   474→  // v13.2 — فقط کارت‌های واقعی تاییدشده؛ کارت‌های بازخوانی (retroIssued) و دفتر سایه در وین‌ریت کانال دخالت نمی‌کنند
   475→  const settled = await db.issuedSlip.findMany({
   476→    where: { status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] }, supervisorApproved: true, retroIssued: false },
   477→    select: { status: true, netProfit: true, stakeA: true, stakeB: true },
   478→  });
   479→  const st = { settled: settled.length, wonA: 0, wonB: 0, doubleWin: 0, middleTrap: 0, netProfit: 0, staked: 0 };
   480→  for (const s of settled) {
   481→    if (s.status === "WON_A") st.wonA += 1;
   482→    else if (s.status === "WON_B") st.wonB += 1;
   483→    else if (s.status === "DOUBLE_WIN") st.doubleWin += 1;
   484→    else if (s.status === "MIDDLE_TRAP") st.middleTrap += 1;
   485→    st.netProfit += s.netProfit ?? 0;
   486→    st.staked += s.stakeA + s.stakeB;
   487→  }
   488→  const winRatePct = st.settled > 0 ? Math.round(((st.settled - st.middleTrap) / st.settled) * 1000) / 10 : 0;
   489→  const netPct = st.staked > 0 ? Math.round((st.netProfit / st.staked) * 1000) / 10 : 0;
   490→  // ─── v11.0 — دفتر سایه: شفافیت counterfactual وتوها (خط جداگانه گزارش) ───
   491→  // v13.5.1 — سایه فقط وتوهای زنده است؛ کارت‌های بازخوانی (retroIssued) داخل سایه شمرده نمی‌شوند
   492→  const shadowRows = await db.issuedSlip.findMany({
   493→    where: { supervisorApproved: false, retroIssued: false, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   494→    select: { status: true, netProfit: true },
   495→  });
   496→  const shadowWins = shadowRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   497→  const shadowNet = Math.round(shadowRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100;
   498→  const shadowWinRate = shadowRows.length > 0 ? Math.round((shadowWins / shadowRows.length) * 1000) / 10 : null;
   499→  // ─── v13.2 — دفتر بازخوانی: کارت‌های چندروزه‌ای که هرگز پیش‌بازی صادر نشده‌اند (تسویه کاغذی تحلیلی) ───
   500→  // v13.5.1 — بازخوانی با پرچم retroIssued شناسایی می‌شود (مستقل از تایید — کارت بازخوانی هرگز تاییدشده نیست)
   501→  const retroRows = await db.issuedSlip.findMany({
   502→    where: { retroIssued: true, status: { in: ["WON_A", "WON_B", "DOUBLE_WIN", "MIDDLE_TRAP"] } },
   503→    select: { status: true, netProfit: true },
   504→  });
   505→  const retroWins = retroRows.filter((s) => s.status !== "MIDDLE_TRAP").length;
   506→  return {
   507→    ...st,
   508→    winRatePct,
   509→    netPct,
   510→    netProfit: Math.round(st.netProfit * 100) / 100,
   511→    shadow: { settled: shadowRows.length, wins: shadowWins, winRatePct: shadowWinRate, netProfit: shadowNet },
   512→    retro: {
   513→      settled: retroRows.length,
   514→      wins: retroWins,
   515→      winRatePct: retroRows.length > 0 ? Math.round((retroWins / retroRows.length) * 1000) / 10 : null,
   516→      netProfit: Math.round(retroRows.reduce((a, b) => a + (b.netProfit ?? 0), 0) * 100) / 100,
   517→    },
   518→  };
   519→}

   522→export function buildWinRateReportMessage(st: {

   602→/**
   603→ * v15.0 — تعمیر پرچم‌های انتشار پیش‌ماتره (Premature Broadcast Repair).
   604→ * ریشه باگ «فلش‌کارت‌های بازی‌های روز به کانال VIP ارسال نمی‌شود»:
   605→ * کارت‌های بازی‌های فردا/آینده قبل از فعال‌شدن قانون «فقط کارت‌های روزانه» به کانال
   606→ * رفته و پرچم telegramCard=true گرفته‌اند؛ پس از آن صف روز بازی، کارت را «ارسال‌شده»
   607→ * می‌بیند و هرگز دوباره ارسال نمی‌کند. این تابع پرچم کارت‌هایی که ارسال موفقشان
   608→ * بیرون از «روز تهرانِ کیک‌آف» انجام شده را بازنشانی می‌کند تا روز بازی واقعاً منتشر شوند.
   609→ */
   610→async function repairPrematureCardFlags(): Promise<number> {
   611→  try {
   612→    const flagged = await db.issuedSlip.findMany({
   613→      where: { status: { in: ["PENDING", "LIVE"] }, telegramCard: true, supervisorApproved: true, retroIssued: false },
   614→      select: { id: true, kickoff: true },
   615→      take: 40,
   616→    });
   617→    if (flagged.length === 0) return 0;
   618→    const sends = await db.telegramOutbox.findMany({
   619→      where: { kind: { in: ["CARD", "FUNNEL_GIFT", "FUNNEL_TEASER"] }, slipId: { not: null }, ok: true },
   620→      orderBy: { createdAt: "desc" },
   621→      take: 200,
   622→      select: { slipId: true, createdAt: true },
   623→    });
   624→    const lastSend = new Map<string, Date>();
   625→    for (const s of sends) {
   626→      if (s.slipId && !lastSend.has(s.slipId)) lastSend.set(s.slipId, s.createdAt);
   627→    }
   628→    let repaired = 0;
   629→    for (const slip of flagged) {
   630→      const sentAt = lastSend.get(slip.id);
   631→      if (!sentAt) {
   632→        // پرچم خورده ولی هیچ ارسال موفقی ثبت نشده — قطعاً پیش‌ماتره یا خطای نیمه‌کاره
   633→        await db.issuedSlip.update({ where: { id: slip.id }, data: { telegramCard: false } });
   634→        repaired += 1;
   635→        continue;
   636→      }
   637→      const day = tehranDayRange(slip.kickoff);
   638→      if (sentAt.getTime() < day.start.getTime() || sentAt.getTime() >= day.end.getTime()) {
   639→        await db.issuedSlip.update({ where: { id: slip.id }, data: { telegramCard: false } });
   640→        repaired += 1;
   641→      }
   642→    }
   643→    if (repaired > 0) console.log(`[telegram] v16 premature-card repair: ${repaired} flags reset for match-day re-broadcast`);
   644→    return repaired;
   645→  } catch (err) {
   646→    console.warn("premature card repair failed:", err instanceof Error ? err.message : err);
   647→    return 0;
   648→  }
   649→}
