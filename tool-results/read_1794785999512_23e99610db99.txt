    99→/**
   100→ * v18.4 — claim اتمی کلید Setting برای dedupe بین‌پردازه‌ای (Atomic Dedupe Claim).
   101→ * الگوی قدیمی «check → send → write» در دو اجرای هم‌زمان (زمان‌بند + مسیر دستی یا دو
   102→ * پردازه) هر دو استمپ کهنه می‌خواند و همان پست را دوبار می‌فرستد. حالا claim قبل از
   103→ * ارسال انجام می‌شود؛ فقط یک فراخوان برنده است (SQLite تک‌نویسنده).
   104→ * خروجی false = دیگری قبلاً مالک شده — بدون ارسال.
   105→ */
   106→export async function claimSetting(key: string, value: string): Promise<boolean> {
   107→  const existing = await db.setting.findUnique({ where: { key } }).catch(() => null);
   108→  if (!existing) {
   109→    try {
   110→      await db.setting.create({ data: { key, value } });
   111→      return true;
   112→    } catch {
   113→      return false; // مسیر موازی هم‌زمان ردیف را ساخت
   114→    }
   115→  }
   116→  if (existing.value === value) return false;
   117→  const res = await db.setting.updateMany({ where: { key, value: { not: value } }, data: { value } });
   118→  return res.count === 1;
   119→}

   275→async function tgCall(method: string, body: Record<string, unknown>): Promise<{ ok: boolean; detail: string }> {
   276→  const token = await botToken();
   277→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   278→  // v13.2 — سپر توکن ناقص: خطای 404 مبهم تلگرام را با پیام دقیق جایگزین می‌کند
   279→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   280→  try {
   281→    const res = await fetch(`https://api.telegram.org/bot${token}/${method}`, {
   282→      method: "POST",
   283→      headers: { "Content-Type": "application/json" },
   284→      body: JSON.stringify(body),
   285→      cache: "no-store",
   286→    });
   287→    const json = (await res.json()) as { ok?: boolean; description?: string; result?: { message_id?: number } };
   288→    if (json.ok) return { ok: true, detail: `message:${json.result?.message_id ?? "?"}` };
   289→    return { ok: false, detail: json.description ?? `HTTP ${res.status}` };
   290→  } catch (err) {
   291→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   292→  }
   293→}

   295→async function tgSendPhotoTo(
   296→  chatId: string | null,
   297→  photo: Buffer,
   298→  caption: string,
   299→  filename = "slip-card.png",
   300→  captionless = false
   301→): Promise<{ ok: boolean; detail: string }> {
   302→  const token = await botToken();
   303→  if (!token) return { ok: false, detail: "NO_TOKEN" };
   304→  if (!chatId) return { ok: false, detail: "NO_CHANNEL" };
   305→  // v13.2 — سپر توکن ناقص (بخش hash بدون bot_id) — ریشه باگ «انتشار انجام نمیشه»
   306→  if (!isValidBotTokenFormat(token)) return { ok: false, detail: "INVALID_TOKEN_FORMAT" };
   307→  // ═══ v20.0 — سپر «هیچ پیامی بدون تصویر کامل ارسال نشود» ═══
   308→  // امضای PNG + حداقل حجم ۲۰KB — بافر ناقص/خراب هرگز به تلگرام نمی‌رود.
   309→  if (!isValidPng(photo)) return { ok: false, detail: "PHOTO_INCOMPLETE" };
   310→  try {
   311→    // v20.0 — کپشن عکس همیشه «کاملاً HTML-معتبر و ≤۱۰۰۰ کاراکتر در مرز خط» —
   312→    // ریشه ریشه‌ای «پیام بدون تصویر»: caption.slice(0,1024) تگ HTML را نصف می‌کرد
   313→    // → تلگرام sendPhoto را با parse-error رد می‌کرد → فالبک متنی بی‌تصویر می‌رفت.
   314→    const cap = captionless ? "" : safePhotoCaption(caption);
   315→    let result = await tgSendPhotoAttempt(token, chatId, photo, cap, filename, captionless);
   316→    // v20.0 — یک ری‌تای ارسال (خطای شبکه/آپلود گذرا) — همان بافر معتبر، بدون متن جایگزین
   317→    if (!result.ok && !/parse|entities|caption/i.test(result.detail)) {
   318→      await new Promise((r) => setTimeout(r, 1500));
   319→      result = await tgSendPhotoAttempt(token, chatId, photo, cap, filename, captionless);
   320→    }
   321→    return result;
   322→  } catch (err) {
   323→    return { ok: false, detail: err instanceof Error ? err.message : "network error" };
   324→  }
   325→}

   327→/** v20.0 — اعتبارسنجی سخت PNG: امضای ۸ بایتی + حداقل ۲۰KB */
   328→function isValidPng(buf: Buffer | null | undefined): boolean {
   329→  if (!buf || buf.length < 20480) return false;
   330→  const SIG = [0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a];
   331→  for (let i = 0; i < 8; i++) if (buf[i] !== SIG[i]) return false;
   332→  return true;
   333→}
   334→
   335→/**
   336→ * v20.0 — کپشن ایمن عکس: برش فقط در مرز خط + بستن تگ‌های باز + سقف ۱۰۰۰ کاراکتر.
   337→ * خروجی همیشه HTML-معتبر است (تلگرام هرگز parse-error نمی‌گیرد → پیام بی‌تصویر صفر).
   338→ */
   339→export function safePhotoCaption(text: string, limit = 1000): string {
   340→  let s = String(text ?? "");
   341→  if (s.length <= limit) return s;
   342→  // برش در مرز خط — هیچ تگی نصف نمی‌شود
   343→  s = s.slice(0, limit);
   344→  const lastNl = s.lastIndexOf("\n");
   345→  if (lastNl > 200) s = s.slice(0, lastNl);
