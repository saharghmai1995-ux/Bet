    83→async function learnIdsFromUpdate(db: Database, cfg: BotConfig, update: Update): Promise<void> {
    84→  const notes: string[] = [];
    85→
    86→  // ۱) my_chat_member — بات به کانال/سوپرگروه اضافه، حذف یا ارتقا یافت
    87→  const mcm = update.my_chat_member;
    88→  if (mcm?.chat && (mcm.chat.type === "channel" || mcm.chat.type === "supergroup")) {
    89→    if (isVipPlaceholder(cfg.vipChannelId)) {
    90→      cfg.vipChannelId = String(mcm.chat.id);
    91→      notes.push(`کانال VIP ← ${mcm.chat.id} (${mcm.chat.title ?? "بدون عنوان"})`);
    92→    }
    93→    if (mcm.from?.id && isAdminPlaceholder(cfg.adminUserId)) {
    94→      cfg.adminUserId = mcm.from.id;
    95→      notes.push(`ادمین ← ${mcm.from.id}`);
    96→    }
    97→  }
    98→
    99→  // ۲) channel_post — پستی از کانالی که بات ادمین آن است
   100→  const post = update.channel_post ?? (update.message && update.message.chat.type === "channel" ? update.message : undefined);
   101→  if (post?.chat.type === "channel" && isVipPlaceholder(cfg.vipChannelId)) {
   102→    cfg.vipChannelId = String(post.chat.id);
   103→    notes.push(`کانال VIP ← ${post.chat.id} (channel_post)`);
   104→  }
   105→
   106→  // ۳) پیام خصوصی — فوروارد از کانال VIP یا /start مالک
   107→  const msg = update.message;
   108→  if (msg && msg.chat.type === "private") {
   109→    const fwd = msg.forward_from_chat ?? msg.sender_chat;
   110→    if (fwd?.type === "channel" && isVipPlaceholder(cfg.vipChannelId)) {
   111→      cfg.vipChannelId = String(fwd.id);
   112→      notes.push(`کانال VIP ← ${fwd.id} (فوروارد)`);
   113→    }
   114→    if (msg.text?.startsWith("/start") && msg.from?.id && isAdminPlaceholder(cfg.adminUserId)) {
   115→      cfg.adminUserId = msg.from.id;
   116→      notes.push(`ادمین ← ${msg.from.id} (/start مالک)`);
   117→    }
   118→  }
   119→
   120→  if (notes.length === 0) return;
   121→  saveConfig(db, { vipChannelId: cfg.vipChannelId, adminUserId: cfg.adminUserId });
   122→  console.log(`[LEARN] auto-configured: ${notes.join(" | ")}`);
   123→  if (!isAdminPlaceholder(cfg.adminUserId)) {
   124→    await tgApiCall(cfg, "sendMessage", {
   125→      chat_id: cfg.adminUserId,
   126→      text: (
   127→        "⚙️ <b>پیکربندی خودکار @aineobet_bot ثبت شد</b>\n\n" +
   128→        notes.map((n) => `▫️ ${n}`).join("\n") +
   129→        "\n\n🔐 مقادیر در دیتابیس قفل شدند. برای تکمیل درگاه‌ها، کیف پول‌های USDT (TRC20/Polygon) و کارت بانکی را از پنل وب (تب ربات اشتراک VIP) ثبت کنید."
   130→      ),
   131→      parse_mode: "HTML",
   132→    });
   133→  }
   134→}

   230→/**
   231→ * تصمیم ادمین روی فیش کارت‌به‌کارت — از تلگرام (دکمه) و از پنل وب فراخوانی می‌شود.
   232→ * تایید → فعال‌سازی اشتراک + لینک یک‌بارمصرف VIP + پیام به کاربر.
   233→ */
   234→export async function processAdminReceiptDecision(
   235→  db: Database,
   236→  cfg: BotConfig,
   237→  receiptId: number,
   238→  approve: boolean
   239→): Promise<AdminDecisionResult> {
   240→  const rec = getCardReceipt(db, receiptId);
   241→  if (!rec) return { ok: false, statusMessage: "رسید یافت نشد." };
   242→  if (rec.status !== "PENDING") {
   243→    return { ok: false, statusMessage: `این رسید قبلاً در وضعیت ${rec.status} قرار گرفته است.` };
   244→  }
   245→
   246→  setCardReceiptStatus(db, receiptId, approve ? "APPROVED" : "REJECTED");
   247→
   248→  const msgs = buildMessages(cfg.publicChannelUsername);
   249→  const user = getOrCreateUser(db, rec.user_id);
   250→  const lang: Lang = user.lang === "en" ? "en" : "fa";
   251→  const m = msgs[lang];
   252→  const plan = SUBSCRIPTION_PLANS[rec.plan_key] ?? SUBSCRIPTION_PLANS.pro_monthly;
   253→  const planTitle = lang === "fa" ? plan.title_fa : plan.title_en;
   254→
   255→  if (approve) {
   256→    const expStr = extendUserVip(db, rec.user_id, plan.days, rec.plan_key);
   257→    const inviteLink = await createSingleUseVipInviteLink(cfg, rec.user_id, plan.days);
   258→    const userMsg = fmt(m.card_approved_user, {
   259→      receipt_id: receiptId,
   260→      plan_title: planTitle,
   261→      expires_at: expiryLabel(rec.plan_key, expStr, lang),
   262→      invite_link: inviteLink,
   263→    });
   264→    await tgApiCall(cfg, "sendMessage", { chat_id: rec.user_id, text: userMsg, parse_mode: "HTML" });
   265→    return {
   266→      ok: true,
   267→      statusMessage: `✅ رسید #${receiptId} تایید شد و لینک یک‌بارمصرف VIP برای کاربر ${rec.user_id} ارسال گردید.`,
   268→      info: { user_id: rec.user_id, expires_at: expStr, invite_link: inviteLink },
   269→    };
   270→  }
   271→
   272→  const userMsg = fmt(m.card_rejected_user, { receipt_id: receiptId });
   273→  await tgApiCall(cfg, "sendMessage", { chat_id: rec.user_id, text: userMsg, parse_mode: "HTML" });
   274→  return { ok: true, statusMessage: `❌ رسید #${receiptId} رد شد و به کاربر اطلاع داده شد.` };
   275→}

   277→// ─── کنار گذاشتن اعضای منقضی از کانال VIP (ساعتی) ───────────────────────────
   278→
   279→export async function sweepExpiredVipMembers(db: Database, cfg: BotConfig): Promise<number> {
   280→  const nowIso = new Date().toISOString();
   281→  const expired = db.query("SELECT user_id FROM users WHERE vip_expires_at IS NOT NULL AND vip_expires_at < ?").all(nowIso) as { user_id: number }[];
   282→  let removed = 0;
   283→  for (const row of expired) {
   284→    await tgApiCall(cfg, "banChatMember", { chat_id: cfg.vipChannelId, user_id: row.user_id });
   285→    await tgApiCall(cfg, "unbanChatMember", { chat_id: cfg.vipChannelId, user_id: row.user_id, only_if_banned: true });
   286→    db.run("UPDATE users SET vip_expires_at = NULL WHERE user_id = ?", [row.user_id]);
   287→    removed += 1;
   288→  }
   289→  return removed;
   290→}

   294→export async function handleUpdate(db: Database, cfg: BotConfig, update: Update): Promise<void> {
   295→  // خودآموزی شناسه‌های مالک/کانال VIP — قبل از هر پردازشی
   296→  await learnIdsFromUpdate(db, cfg, update);
   297→
   298→  const msgs = buildMessages(cfg.publicChannelUsername);
   299→
   300→  if (update.message) {
   301→    const msg = update.message;
   302→    const chatId = msg.chat.id;
   303→    const userId = msg.from?.id ?? chatId;
   304→    const username = msg.from?.username ?? "";
   305→    const text = (msg.text ?? "").trim();
   306→    const photos = msg.photo ?? [];
   307→
   308→    let referredBy: number | null = null;
   309→    if (text.startsWith("/start") && text.includes("ref_")) {
   310→      try {
   311→        const refId = parseInt(text.split("ref_")[1].trim(), 10);
   312→        if (Number.isFinite(refId) && refId !== userId) referredBy = refId;
   313→      } catch {
   314→        /* ref نامعتبر */
   315→      }
   316→    }
   317→
   318→    const user = getOrCreateUser(db, userId, username, referredBy);
   319→    const lang: Lang = user.lang === "en" ? "en" : "fa";
   320→    const m = msgs[lang];
   321→
   322→    if (text.startsWith("/start")) {
   323→      setUserPlanAndMode(db, userId, user.selected_plan, "NONE", user.pending_toman_amount);
   324→      await tgApiCall(cfg, "sendMessage", {
   325→        chat_id: chatId,
   326→        text: m.welcome,
   327→        parse_mode: "HTML",
   328→        reply_markup: mainMenuKeyboard(lang, msgs),
   329→      });
   330→      return;
   331→    }
   332→
   333→    // ─── دستورات منوی BotFather — به همان مسیر دکمه‌های شیشه‌ای هدایت می‌شوند ───
   334→    const cmd = text.toLowerCase().split(/[\s@]/)[0];
   335→    const CMD_TO_CALLBACK: Record<string, string> = {
   336→      "/plans": "menu_plans",
   337→      "/free": "menu_free_signal",
   338→      "/status": "menu_status",
   339→      "/referral": "menu_referral",
   340→      "/lang": "toggle_lang",
   341→      "/vip": "menu_plans",
   342→      "/pay": "menu_plans",
   343→    };
   344→    if (CMD_TO_CALLBACK[cmd]) {
   345→      await handleUpdate(db, cfg, {
   346→        callback_query: {
   347→          id: "cmd-dispatch",
   348→          from: msg.from ?? { id: userId, username },
   349→          message: { chat: msg.chat },
   350→          data: CMD_TO_CALLBACK[cmd],
   351→        },
   352→      });
   353→      return;
   354→    }
   355→
   356→    // الف) عکس فیش واریزی → ثبت فیش + ارسال به ادمین
   357→    if (photos.length > 0) {
   358→      const bestPhotoFileId = photos[photos.length - 1].file_id;
   359→      const planKey = user.selected_plan || "pro_monthly";
   360→      const expectedToman =
   361→        user.pending_toman_amount ||
   362→        (SUBSCRIPTION_PLANS[planKey] ?? SUBSCRIPTION_PLANS.pro_monthly).price_toman;
   363→      const receiptId = createCardReceipt(db, userId, username, planKey, expectedToman, "PHOTO", bestPhotoFileId);
   364→      await notifyAdminOfCardReceipt(db, cfg, receiptId);
   365→      await tgApiCall(cfg, "sendMessage", {
   366→        chat_id: chatId,
   367→        text: fmt(m.receipt_submitted_user, { receipt_id: receiptId }),
   368→        parse_mode: "HTML",
   369→      });
   370→      return;
   371→    }
   372→
   373→    // ب) متن رسید (کد پیگیری / ۴ رقم آخر کارت) در حالت انتظار فیش — رشته‌های شبیه TxID فیش محسوب نمی‌شوند
   374→    if (
   375→      user.payment_mode === "AWAITING_CARD_RECEIPT" &&
   376→      text.length >= 4 &&
   377→      text.length !== 64 &&
   378→      text.length !== 66 &&
   379→      !isSolanaTxid(text)
   380→    ) {
   381→      const planKey = user.selected_plan || "pro_monthly";
   382→      const expectedToman =
   383→        user.pending_toman_amount ||
   384→        (SUBSCRIPTION_PLANS[planKey] ?? SUBSCRIPTION_PLANS.pro_monthly).price_toman;
   385→      const receiptId = createCardReceipt(db, userId, username, planKey, expectedToman, "TEXT", text);
   386→      await notifyAdminOfCardReceipt(db, cfg, receiptId);
   387→      await tgApiCall(cfg, "sendMessage", {
   388→        chat_id: chatId,
   389→        text: fmt(m.receipt_submitted_user, { receipt_id: receiptId }),
   390→        parse_mode: "HTML",
   391→      });
   392→      return;
   393→    }
   394→
   395→    // ج) TxID تتر (ترون ۶۴ hex / EVM 66 با 0x / سولانا ~87 Base58) → استعلام خودکار بلاکچین
   396→    const isTronOrEvm =
   397→      (text.length === 64 || text.length === 66) &&
   398→      [...text].every((c) => "0123456789abcdefABCDEFxX".includes(c));
   399→    if (isTronOrEvm || isSolanaTxid(text)) {
   400→      await tgApiCall(cfg, "sendMessage", { chat_id: chatId, text: m.verifying_tx });
   401→      const planKey = user.selected_plan || "pro_monthly";
   402→      const plan = SUBSCRIPTION_PLANS[planKey] ?? SUBSCRIPTION_PLANS.pro_monthly;
   403→
   404→      if (txidAlreadyUsed(db, text.trim())) {
   405→        await tgApiCall(cfg, "sendMessage", { chat_id: chatId, text: m.tx_already_used, parse_mode: "HTML" });
   406→        return;
   407→      }
   408→
   409→      const vr = await verifyUsdtTxid(
   410→        text,
   411→        {
   412→          trc20: cfg.walletUsdtTrc20,
   413→          polygon: cfg.walletUsdtPolygon,
   414→          bsc: cfg.walletUsdtBsc,
   415→          solana: cfg.walletUsdtSolana,
   416→        },
   417→        plan.price_usdt
   418→      );
   419→      if (!vr.ok) {
   420→        await tgApiCall(cfg, "sendMessage", {
   421→          chat_id: chatId,
   422→          text: fmt(m.tx_failed, { reason: vr.message }),
   423→          parse_mode: "HTML",
   424→        });
   425→        return;
   426→      }
   427→
   428→      insertPayment(db, text.trim(), userId, vr.network, vr.amount, planKey);

   447→
   448→  if (update.callback_query) {

   468→
   469→    const user = getOrCreateUser(db, userId, cb.from.username ?? "");

   472→    if (data === "toggle_lang") {

   474→      setUserLang(db, userId, lang);

   476→

   484→        reply_markup: mainMenuKeyboard(lang, msgs),

   621→    if (cbId !== "cmd-dispatch") {

   624→  }
