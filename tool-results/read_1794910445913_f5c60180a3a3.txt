   227→/**
   228→ * آنتروپی شانون ماتریس ۶×۶ — شاخص تمرکز یا پراکندگی نتایج.
   229→ * H = −Σ mat[i][j]·ln(mat[i][j]) روی زیرماتریس ۰..۵ × ۰..۵ (نرمال‌شده درون همان بازه).
   230→ */
   231→export function shannonEntropy(matrix: number[][]): number {
   232→  const N = Math.min(6, matrix.length);
   233→  let sum = 0;
   234→  for (let h = 0; h < N; h++) {
   235→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) sum += matrix[h][a];
   236→  }
   237→  if (sum <= 0) return Math.log(36);
   238→  let H = 0;
   239→  for (let h = 0; h < N; h++) {
   240→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   241→      const p = matrix[h][a] / sum;
   242→      if (p > 1e-12) H -= p * Math.log(p);
   243→    }
   244→  }
   245→  return H;
   246→}

   248→/** نسبت عدم تقارن قدرت: Rλ = max(λH, λA) / min(λH, λA) */
   249→export function asymmetryRatio(lambdaHome: number, lambdaAway: number): number {
   250→  const hi = Math.max(lambdaHome, lambdaAway);
   251→  const lo = Math.min(lambdaHome, lambdaAway);
   252→  if (lo <= 0.01) return 9.99;
   253→  return Math.min(9.99, hi / lo);
   254→}

   256→/** مجموع n خانه برتر ماتریس ۶×۶ + فهرست آن‌ها */
   257→export function topCells(matrix: number[][], n = 4): { score: string; p: number }[] {
   258→  const N = Math.min(6, matrix.length);
   259→  const cells: { score: string; p: number }[] = [];
   260→  for (let h = 0; h < N; h++) {
   261→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   262→      cells.push({ score: `${h}-${a}`, p: matrix[h][a] });
   263→    }
   264→  }
   265→  cells.sort((x, y) => y.p - x.p);
   266→  return cells.slice(0, n);
   267→}

   269→export function matrixStructure(matrix: number[][], lambdaHome: number, lambdaAway: number): MatrixStructure {
   270→  const top4 = topCells(matrix, 4);
   271→  return {
   272→    entropy: Math.round(shannonEntropy(matrix) * 1000) / 1000,
   273→    rLambda: Math.round(asymmetryRatio(lambdaHome, lambdaAway) * 100) / 100,
   274→    top4Sum: Math.round(top4.reduce((s, c) => s + c.p, 0) * 1000) / 1000,
   275→    top4,
   276→  };
   277→}

   281→/** جمع احتمال خانه‌های ماتریس ۶×۶ که گزاره را ارضا می‌کنند (نرمال‌شده درون ۶×۶) */
   282→function matrixProb(matrix: number[][], predicate: (h: number, a: number) => boolean): number {
   283→  const N = Math.min(6, matrix.length);
   284→  let inRegion = 0;
   285→  let hit = 0;
   286→  for (let h = 0; h < N; h++) {
   287→    for (let a = 0; a < Math.min(6, matrix[h].length); a++) {
   288→      inRegion += matrix[h][a];
   289→      if (predicate(h, a)) hit += matrix[h][a];
   290→    }
   291→  }
   292→  return inRegion > 0 ? hit / inRegion : 0;
   293→}
   294→
   295→const r2 = (v: number) => Math.round(v * 100) / 100;
   296→const r3 = (v: number) => Math.round(v * 1000) / 1000;

   300→export interface NonLinearCandidateInput {
   301→  /** احتمال برد کل استراتژی (یا مجموع پوشش خوشه) */
   302→  winProb: number;
   303→  /** ضریب موثر کل پوزیشن (پس از داچینگ یا تک‌شرط) */
   304→  effectiveOdds: number;
   305→  /** احتمال برد دوبل همزمان در نتایج 1-0 / 2-0 */
   306→  doubleWinProb: number;
   307→  /** آنتروپی شانون ماتریس 6x6 */
   308→  entropy: number;
   309→  /** ضریب اعتبار لیگ از کالیبراسیون (0.85 تا 1.15) */
   310→  leagueReliability: number;
   311→}

   313→/**
   314→ * تابع غیرخطی امتیازدهی سوپروایزر برای انتخاب بهترین Bet-Type در هر مسابقه
   315→ * (Non-Linear Sharpe & Kelly Utility — عیناً مطابق سند ارتقا v17.0)
   316→ */
   317→export function calculateNonLinearSupervisorScore(candidate: NonLinearCandidateInput): {
   318→  ev: number;
   319→  sharpe: number;
   320→  score: number;
   321→} {
   322→  const netReturnOnWin = candidate.effectiveOdds - 1;
   323→  const lossProb = Math.max(0, 1 - candidate.winProb);
   324→  const expectedValue = candidate.winProb * netReturnOnWin + candidate.doubleWinProb * 0.45 - lossProb;
   325→  const variance =
   326→    candidate.winProb * Math.pow(netReturnOnWin - expectedValue, 2) +
   327→    lossProb * Math.pow(-1 - expectedValue, 2);
   328→  const sharpeRatio = expectedValue / Math.sqrt(Math.max(0.01, variance));
   329→
   330→  // جریمه غیرخطی ریسک و پاداش آنتروپی پایین
   331→  const nonLinearSafetyBonus = Math.pow(Math.max(0, candidate.winProb), 2.2) * 100;
   332→  const entropyBonus = Math.max(0, (2.8 - candidate.entropy) * 12);
   333→
   334→  const score = Number(
   335→    ((sharpeRatio * 45 + nonLinearSafetyBonus * 0.55 + entropyBonus) * candidate.leagueReliability).toFixed(1)
   336→  );
   337→  return { ev: expectedValue, sharpe: sharpeRatio, score };
   338→}

   527→export function routeBetTypes(input: {

   537→  matchSeed?: number;
   538→  /**

   605→        if (structure.rLambda < V22_BANKER_RLAMBDA_MIN || c.mainProb < V22_BANKER_PROB_MIN) {
   606→          return { pass: false, why: `عدم تقارن/احتمال پایه کافی نیست (Rλ=${structure.rLambda.toFixed(2)} ≥ ۲.۲؟ P=${(c.mainProb * 100).toFixed(0)}٪ ≥ ۷۸٪؟)` };
   607→        }
   608→        if (c.mainOdds < V22_BANKER_ODDS_MIN || c.mainOdds > V22_BANKER_ODDS_MAX) {
   609→          return { pass: false, why: `ضریب زنده خارج بازه بنکر v22 (odds=${c.mainOdds.toFixed(2)} ∉ [1.34, 1.68])` };
   610→        }
   611→        return { pass: true, why: "تک‌ستون بنکر خط ۳ — فاوریت قاطع با عدم تقارن ≥ ۲.۲" };
   612→      case "DEEP_VALUE":
   613→        // خط ۲ — تنها ۱ مارکت با EV ≥ +۶٪ و ضریب زنده 1.72..2.15
   614→        if (c.ev < V22_DEEP_EV_MIN) {
   615→          return { pass: false, why: `برتری ارزش انتظاری زیر کف سند (EV=${(c.ev * 100).toFixed(1)}٪ < +۶٪)` };
   616→        }
   617→        if (c.mainOdds < V22_DEEP_ODDS_MIN || c.mainOdds > V22_DEEP_ODDS_MAX) {
   618→          return { pass: false, why: `ضریب زنده خارج بازه تک‌گزینه عمیق v22 (odds=${c.mainOdds.toFixed(2)} ∉ [1.72, 2.15])` };
   619→        }
   620→        return { pass: true, why: "تک‌گزینه عمیق خط ۲ — +EV ≥ ۶٪ نسبت به خط زنده بوک‌میکر" };
   621→      default:
   622→        // 🚫 v22.0 — COMBO / CORRECT_SCORE / DUAL_HEDGE دیگر تولید نمی‌شوند
   623→        return { pass: false, why: `خط تولید ${c.key} در سند v22.0 حذف شده است (همپوشانی دو گزینه = کارمزد دوبل)` };
   624→    }
   625→  };
   626→
   627→  const gold = candidates.filter((c) => regimeGatePass(c).pass);
   628→
   629→  if (candidates.length === 0 || gold.length === 0) {
   630→    const reason =
   631→      candidates.length === 0
   632→        ? "🛡️ گیت v22.0: هیچ‌یک از دو خط تک‌بازی (BANKER / DEEP_VALUE) کاندید ساختاری معتبری ساخت — انتقال به دفتر سایه"
   633→        : `🛡️ گیت v22.0: کاندیدها گیت اختصاصی خط تولید خود را پاس نکردند (${gold.length === 0 ? "هیچ خطی واجد شرایط نیست" : ""}) — انتقال به دفتر سایه`;
   634→    return {
   635→      structure,
   636→      candidates,
   637→      winnerVariantId: null,
   638→      winner: null,
   639→      crownPairId: null,
   640→      veto: true,
   641→      vetoReason: reason,
   642→      tier: null,
   643→    };
   644→  }
   645→
   646→  // ═══ v22.0 — انتخاب برنده: بالاترین امتیاز غیرخطی بین خطوط واجد شرایط +
   647→  // توزیع متوازن بین BANKER و DEEP_VALUE (رژیم کم‌نمایت اولویت می‌گیرد) ═══
   648→  const regimeCounts = input.regimeCounts ?? {};
   649→  const countOf = (k: BetTypeKey): number => regimeCounts[k] ?? 0;
   650→  const pool = gold;
   651→  const poolRegimes = [...new Set(pool.map((c) => c.key))];
   652→  let best = pool.reduce((a, b) => (b.score > a.score ? b : a));
   653→  if (poolRegimes.length > 1) {
   654→    const minCount = Math.min(...poolRegimes.map(countOf));
   655→    const underRep = poolRegimes.filter((k) => countOf(k) <= minCount + 4 && countOf(k) < countOf(best.key));
   656→    if (underRep.length > 0) {
   657→      const eligible = underRep
   658→        .map((k) => pool.filter((c) => c.key === k).reduce((a, b) => (b.score > a.score ? b : a)))
   659→        .sort((a, b) => b.score - a.score);
   660→      if (eligible.length > 0) best = eligible[0];
   661→    }
   662→  }
   663→
   664→  // 🏆 v22.0 — تاج طلایی واحد: استیک استاندارد ۱۰۰ واحدی برای هر سه خط تولید
   665→  // (سطح‌بندی نقره‌ای نیم‌استیک سند v18 با معماری جدید ناسازگار است — حذف شد)
   666→  best.tier = "GOLD";
   667→  const favRatio = fav > 0 && dog > 0 ? fav / dog : 0;
   668→  best.rationale = `${best.rationale} | زمینه مسیریاب: λ_Fav/λ_Dog = ${favRatio.toFixed(2)}, Rλ = ${structure.rLambda.toFixed(2)}, H = ${structure.entropy.toFixed(2)}`;
   669→
   670→  return {
   671→    structure,
   672→    candidates,
   673→    winnerVariantId: best.variantId,
   674→    winner: best,
   675→    crownPairId: null, // v22.0 — زوج‌تیپ حذف شد؛ میکس ۲ تایی از پایه‌های بنکر دو بازی مستقل ساخته می‌شود
   676→    veto: false,
   677→    vetoReason: null,
   678→    tier: "GOLD",
   679→  };
   680→}

   684→/** یک پای Accumulator — از بنکرها/کمبوهای تاییدشده بهترین بازی‌های روز */
   685→export interface AccumulatorLeg {
   686→  matchId: string;
   687→  matchLabel: string;
   688→  league: string;
   689→  kickoff: string;
   690→  tip: string;
   691→  tipEn: string;
   692→  odds: number;
   693→  betType: BetTypeKey;
   694→}
   695→
   696→export type AccumulatorKind = "TWO_FOLD_BANKER" | "THREE_FOLD_COMBO" | "TIME_STAGGERED";
   697→
   698→export interface AccumulatorSlip {
   699→  id: string;
   700→  kind: AccumulatorKind;
   701→  labelFa: string;
   702→  labelEn: string;
   703→  icon: string;
   704→  legs: AccumulatorLeg[];
   705→  totalOdds: number;
   706→  /** استیک بهینه بر اساس معیار کلِی همزمان (واحد) */
   707→  stake: number;
   708→  hitRatePct: number;
   709→  potentialReturn: number;
   710→  /** فرمول محاسبه مبلغ پوشش زنده پای آخر (Time-Staggered) */
   711→  hedgeNote?: string;
   712→  hedgeTargetLabel?: string;
   713→  hedgeTargetOdds?: number;
   714→  hedgeStake?: number;
   715→  lockedProfitIfHedged?: number;
   716→}
   717→
   718→/**
   719→ * 🔗 خط ۱ سند v22.0 — میکس ۲ تایی هوشمند (2-Leg Smart Value Accumulator):
   720→ * انتخاب ۲ مسابقه «کاملاً مستقل» که در هر کدام یک پای امن (Anchor Leg) با
   721→ * احتمال مدل ۷۸..۸۴٪ و ضریب زنده واقعی 1.34x..1.50x وجود دارد؛
   722→ * ضریب کل میکس = حاصل‌ضرب ۲ پایه در بازه 1.80x..2.25x — استیک استاندارد ۱۰۰ واحد.
   723→ * هر برد: +80..+125 واحد سود خالص | نقطه سر‌به‌سر ۵۰٪ در برابر وین‌ریت واقعی ~۶۷..۷۲٪.
   724→ * (خط‌های ۳-Fold و Time-Staggered نسخه‌های قبلی — ناسازگار با سند v22 — حذف شدند.)
   725→ */
   726→export function buildAccumulators(
   727→  winners: { matchId: string; matchLabel: string; league: string; kickoff: string; candidate: BetTypeCandidate }[]
   728→): AccumulatorSlip[] {
   729→  const out: AccumulatorSlip[] = [];
   730→
   731→  // استخر پایه‌های امن — فقط بنکرهای تاییدشده در بازه پای سند v22
   732→  const anchors = winners.filter(
   733→    (w) =>
   734→      w.candidate.key === "BANKER" &&
   735→      w.candidate.mainOdds >= V22_ACC_LEG_ODDS_MIN &&
   736→      w.candidate.mainOdds <= V22_ACC_LEG_ODDS_MAX &&
   737→      w.candidate.mainProb >= V22_ACC_LEG_PROB_MIN &&
   738→      w.candidate.mainProb <= V22_ACC_LEG_PROB_MAX
   739→  );
   740→
   741→  // بهترین جفت مستقل: بیشترین احتمال ترکیبی که ضریب کل داخل پنجره سند باشد
   742→  let bestPair: { a: (typeof anchors)[number]; b: (typeof anchors)[number]; hit: number; totalOdds: number } | null = null;
   743→  for (let i = 0; i < anchors.length; i++) {
   744→    for (let j = i + 1; j < anchors.length; j++) {
   745→      const a = anchors[i];
   746→      const b = anchors[j];
   747→      if (a.matchId === b.matchId) continue; // فقط ۲ بازی کاملاً مستقل
   748→      const totalOdds = r2(a.candidate.mainOdds * b.candidate.mainOdds);
   749→      if (totalOdds < V22_ACC_TOTAL_ODDS_MIN || totalOdds > V22_ACC_TOTAL_ODDS_MAX) continue;
   750→      const hit = a.candidate.mainProb * b.candidate.mainProb;
   751→      if (!bestPair || hit > bestPair.hit) bestPair = { a, b, hit, totalOdds };
   752→    }
   753→  }
   754→
   755→  if (bestPair) {
   756→    const { a, b, hit, totalOdds } = bestPair;
   757→    const legs: AccumulatorLeg[] = [
   758→      {
   759→        matchId: a.matchId,
   760→        matchLabel: a.matchLabel,
   761→        league: a.league,
   762→        kickoff: a.kickoff,
   763→        tip: a.candidate.mainLabel,
   764→        tipEn: a.candidate.mainLabelEn,
   765→        odds: a.candidate.mainOdds,
   766→        betType: "BANKER",
   767→      },
   768→      {
   769→        matchId: b.matchId,
   770→        matchLabel: b.matchLabel,
   771→        league: b.league,
   772→        kickoff: b.kickoff,
   773→        tip: b.candidate.mainLabel,
   774→        tipEn: b.candidate.mainLabelEn,
   775→        odds: b.candidate.mainOdds,
   776→        betType: "BANKER",
   777→      },
   778→    ];
   779→    out.push({
   780→      id: "ACC-2F",
   781→      kind: "TWO_FOLD_BANKER",
   782→      labelFa: `میکس ۲ تایی هوشمند (2-Leg Smart Value Accumulator) — ${a.candidate.mainOdds.toFixed(2)} × ${b.candidate.mainOdds.toFixed(2)} = ${totalOdds.toFixed(2)}x`,
   783→      labelEn: "2-Leg Smart Value Accumulator",
   784→      icon: "🔗",
   785→      legs,
   786→      totalOdds,
   787→      stake: 100,
   788→      hitRatePct: Math.round(hit * 1000) / 10,
   789→      potentialReturn: Math.round(100 * totalOdds * 10) / 10,
   790→    });
   791→  }
   792→
   793→  return out;
   794→}
