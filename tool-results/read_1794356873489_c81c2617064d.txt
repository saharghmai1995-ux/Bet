    64→function equalProfitStakes(oddsA: number, oddsB: number): { stakeA: number; stakeB: number } {
    65→  const invA = 1 / Math.max(1.01, oddsA);
    66→  const invB = 1 / Math.max(1.01, oddsB);
    67→  const sum = invA + invB;
    68→  const r1 = (v: number) => Math.round(v * 10) / 10;
    69→  return { stakeA: r1((100 * invA) / sum), stakeB: r1((100 * invB) / sum) };
    70→}
